15use rustls::pki_types::CertificateDer;
16use rustls::pki_types::PrivateKeyDer;
17use rustls::server::WebPkiClientVerifier;

19pub(crate) const SERVER_NAME: &str = "monarch-mini";
20
21pub(crate) struct Config {
22    pub(crate) server: rustls::ServerConfig,
23    pub(crate) client: rustls::ClientConfig,
24}
25
26impl Config {
27    pub(crate) fn load() -> anyhow::Result<Self> {
28        let provider = selected_provider();
29        let certs = load_certs(&required_env("MM_QUIC_CERT")?)?;
30        let key = load_key(&required_env("MM_QUIC_KEY")?)?;
31        let mut roots = RootCertStore::empty();
32        for ca in load_certs(&required_env("MM_QUIC_CA")?)? {
33            roots.add(ca)?;
34        }
35        let roots = Arc::new(roots);
36
37        let client_verifier =
38            WebPkiClientVerifier::builder_with_provider(roots.clone(), provider.clone()).build()?;
39        let server = rustls::ServerConfig::builder_with_provider(provider.clone())
40            .with_protocol_versions(&[&rustls::version::TLS13])?
41            .with_client_cert_verifier(client_verifier)
42            .with_single_cert(certs.clone(), key.clone_key())?;
43        let client = rustls::ClientConfig::builder_with_provider(provider)
44            .with_protocol_versions(&[&rustls::version::TLS13])?
45            .with_root_certificates(roots)
46            .with_client_auth_cert(certs, key)?;
47        Ok(Self { server, client })
48    }
49}

51/// The ring crypto provider, optionally restricted to a single TLS 1.3 AEAD by
52/// `MM_QUIC_CIPHER` (`aes128` | `aes256` | `chacha20`) for both network transports.
53/// Built once per process and shared, so the `MM_QUIC_CIPHER` notice is logged only
54/// once even though both the quic and tcp transports call [`Config::load`].
55fn selected_provider() -> Arc<CryptoProvider> {
56    static PROVIDER: OnceLock<Arc<CryptoProvider>> = OnceLock::new();
57    PROVIDER.get_or_init(|| Arc::new(build_provider())).clone()
58}
59
60fn build_provider() -> CryptoProvider {
61    let base = rustls::crypto::ring::default_provider();
62    let suite = match std::env::var("MM_QUIC_CIPHER").ok().as_deref() {
63        Some("aes128") => cipher_suite::TLS13_AES_128_GCM_SHA256,
64        Some("aes256") => cipher_suite::TLS13_AES_256_GCM_SHA384,
65        Some("chacha20") => cipher_suite::TLS13_CHACHA20_POLY1305_SHA256,
66        _ => return base,
67    };
68    eprintln!("MM_QUIC_CIPHER: restricting TLS to a single cipher suite");
69    CryptoProvider {
70        cipher_suites: vec![suite],
71        ..base
72    }
73}

75fn required_env(name: &str) -> anyhow::Result<String> {
76    std::env::var(name).map_err(|_| anyhow::anyhow!("{name} not set"))
77}
78
79fn load_certs(path: &str) -> anyhow::Result<Vec<CertificateDer<'static>>> {
80    let data = std::fs::read(path).map_err(|err| anyhow::anyhow!("reading {path}: {err}"))?;
81    let certs = rustls_pemfile::certs(&mut &data[..]).collect::<Result<Vec<_>, _>>()?;
82    anyhow::ensure!(!certs.is_empty(), "no certificates in {path}");
83    Ok(certs)
84}
85
86fn load_key(path: &str) -> anyhow::Result<PrivateKeyDer<'static>> {
87    let data = std::fs::read(path).map_err(|err| anyhow::anyhow!("reading {path}: {err}"))?;
88    rustls_pemfile::private_key(&mut &data[..])?
89        .ok_or_else(|| anyhow::anyhow!("no private key in {path}"))
90}
