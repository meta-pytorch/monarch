9//! In-process transport: connects two actors living in the same context.
10//!
11//! The "pipe" is the context's own command channel. A serve and a join on the
12//! same url are paired by a [`Matcher`]; on a match each side gets an
13//! [`InprocConnectionTransport`] pointed at the other, announced to the command
14//! loop via `Command::TransportConnected`. From there establishment is the same
15//! generic flow the command loop runs for every transport — the command loop
16//! sends each side's `Establish` along its transport, and the peer receives it.
17//!
18//! There are no coroutines and no extra hops: `send` forwards a command straight
19//! to the peer connection through the command loop, and dropping the transport
20//! pushes a `Severed` to the peer (the in-process analogue of a socket closing).
21
22use std::collections::HashMap;
23
24use tokio::sync::mpsc;
25
26use crate::connection::ConnectionCommand;
27use crate::connection::ConnectionRef;
28use crate::connection::ConnectionTransport;
29use crate::ctx::Command;
30use crate::matcher::Matcher;
31use crate::shm::ShmClientSlot;
32use crate::transport::Transport;
33
34pub(crate) struct InprocTransport {
35    loop_tx: mpsc::UnboundedSender<Command>,
36    matchers: HashMap<String, Matcher<ConnectionRef, ConnectionRef>>,
37}
38
39impl InprocTransport {
40    pub(crate) fn new(loop_tx: mpsc::UnboundedSender<Command>) -> Self {
41        Self {
42            loop_tx,
43            matchers: HashMap::new(),
44        }
45    }
46}
47
48impl Transport for InprocTransport {
49    fn serve(&mut self, url: String, connection: ConnectionRef, _shm_client: ShmClientSlot) {
50        let loop_tx = self.loop_tx.clone();
51        let mut matcher = self.matchers.remove(&url).unwrap_or_else(Matcher::new);
52        let _ = matcher.push_left(connection, |serve, join| pair(&loop_tx, serve, join));
53        self.matchers.insert(url, matcher);
54    }
55
56    fn join(&mut self, url: String, connection: ConnectionRef, _shm_client: ShmClientSlot) {
57        let loop_tx = self.loop_tx.clone();
58        let mut matcher = self.matchers.remove(&url).unwrap_or_else(Matcher::new);
59        let _ = matcher.push_right(connection, |serve, join| pair(&loop_tx, serve, join));
60        self.matchers.insert(url, matcher);
61    }
62}
63
64/// A serve and a join matched: give each side a transport pointed at the other
65/// and announce both to the command loop.
66fn pair(loop_tx: &mpsc::UnboundedSender<Command>, serve: ConnectionRef, join: ConnectionRef) {
67    connect(loop_tx, serve, join);
68    connect(loop_tx, join, serve);
69}
70
71fn connect(loop_tx: &mpsc::UnboundedSender<Command>, local: ConnectionRef, peer: ConnectionRef) {
72    let transport = Box::new(InprocConnectionTransport {
73        tx: loop_tx.clone(),
74        peer,
75    });
76    let _ = loop_tx.send(Command::TransportConnected {
77        connection: local,
78        transport,
79    });
80}
81
82/// Transport for one end of an inproc connection. `send` delivers a command to
83/// the peer connection by re-entering the command loop.
84struct InprocConnectionTransport {
85    tx: mpsc::UnboundedSender<Command>,
86    peer: ConnectionRef,
87}
88
89impl ConnectionTransport for InprocConnectionTransport {
90    fn send(&self, action: ConnectionCommand) -> bool {
91        // Same process: every command — gateway state (raw fds and all) included —
92        // is delivered straight to the peer connection through the command loop.
93        self.tx
94            .send(Command::ConnectionAction {
95                connection: self.peer,
96                action,
97            })
98            .is_ok()
99    }
100}
101
102impl Drop for InprocConnectionTransport {
103    fn drop(&mut self) {
104        // Dropping the send half is the in-process analogue of a socket closing:
105        // tell the peer its side of the pipe is gone. (A no-op if the peer is
106        // already failed.)
107        let _ = self.tx.send(Command::ConnectionAction {
108            connection: self.peer,
109            action: ConnectionCommand::Severed {
110                reason: b"peer connection closed".to_vec(),
111            },
112        });
113    }
114}
