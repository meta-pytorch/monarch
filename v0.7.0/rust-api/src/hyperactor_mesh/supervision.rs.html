9//! Messages used in supervision of actor meshes.
10//!
11//! ## Mesh-name propagation
12//!
13//! When a `MeshFailure` is constructed for a supervision event
14//! whose constructing site has the mesh name locally in scope, the
15//! mesh name is carried on `MeshFailure.actor_mesh_name`. The
16//! constructing site does not perform a lookup to obtain the mesh
17//! name; if the mesh name is not locally available at the site,
18//! `None` is correct. `MeshFailure::Display` surfaces the mesh
19//! name as an `on mesh "{name}"` segment when `actor_mesh_name`
20//! is populated; stable identifiers continue to appear in detail
21//! segments where the renderer already includes them.
22//! A data actor mesh labels each member's supervision events with
23//! its mesh id, so the owner's supervision handler names the failed
24//! mesh from the event (see `ACTOR_MESH_ID` in `actor_mesh.rs` and
25//! `PythonActor::handle_supervision_event` in
26//! `monarch_hyperactor/src/actor.rs`).
27//!
28//! ## Mesh failure controller-attribution invariants (MFCA-*)
29//!
30//! `MeshFailure.reporting_controller` names the mesh controller that
31//! observed and reported the failure, distinct from `event.actor_id`
32//! (the actor the event concerns).
33//!
34//! - **MFCA-1 (complete controller reports):** every `MeshFailure` a
35//!   controller constructs stamps `reporting_controller =
36//!   Some(controller_id)`; `send_state_change` is the sole controller-to-owner
37//!   failure post and stamps before posting.
38//! - **MFCA-2 (same identity):** the stamped id is
39//!   `cx.instance().self_addr().id()`, which equals the id a consumer reads
40//!   from the mesh's controller ref (`ActorMesh::controller`, or the proc-mesh
41//!   controller).
42//! - **MFCA-3 (subject/source separation):** `event.actor_id` (the event
43//!   subject/root cause) and `reporting_controller` (the reporter) are separate
44//!   roles; their values may coincide on a controller-originated event, but
45//!   neither is overloaded to mean the other.
46//! - **MFCA-4 (non-controller absence):** `None` is reserved for construction
47//!   paths with no reporting mesh controller; such a failure carries no
48//!   controller attribution.

59/// Message about a supervision failure on a mesh of actors instead of a single
60/// actor.
61#[derive(Clone, Debug, Serialize, Deserialize, Named, PartialEq)]
62pub struct MeshFailure {
63    /// Mesh name carried by the `MeshFailure` construction site,
64    /// when locally available. On the direct actor-handled path
65    /// this is the failed member's recorded data mesh id. On
66    /// controller-owned paths this is the monitored mesh name
67    /// supplied by the controller path.
68    pub actor_mesh_name: Option<String>,
69    /// The supervision event on an actor located at mesh + rank.
70    pub event: ActorSupervisionEvent,
71    /// The set of crashed ranks in the mesh. Empty means the event
72    /// applies to the whole mesh (e.g. mesh stop, controller timeout).
73    pub crashed_ranks: Vec<usize>,
74    /// Identity of the mesh controller that reported this failure, when
75    /// one did (MFCA-3, MFCA-4). A separate role from `event.actor_id`
76    /// (the event subject); the two may hold the same `ActorId` on a
77    /// controller-originated event. `None` on construction paths with no
78    /// reporting controller, which carry no controller attribution.
79    pub reporting_controller: Option<ActorId>,
80}
81wirevalue::register_type!(MeshFailure);
82
83impl MeshFailure {
84    /// Returns true if the given rank is part of this failure.
85    /// A whole-mesh event (empty crashed_ranks) contains every rank.
86    pub fn contains_rank(&self, rank: usize) -> bool {
87        self.crashed_ranks.is_empty() || self.crashed_ranks.contains(&rank)
88    }
89
90    /// Helper function to handle a message to an actor that just wants to forward
91    /// it to the next owner.
92    pub fn default_handler(&self, cx: &impl context::Actor) -> Result<(), anyhow::Error> {
93        // If an actor spawned by this one fails, we can't handle it. We fail
94        // ourselves with a chained error and bubble up to the next owner. This
95        // converts the mesh failure into a supervision event, keeping only
96        // `event`; the mesh-level fields (`actor_mesh_name`, `crashed_ranks`,
97        // `reporting_controller`) are dropped and the next controller
98        // re-attributes (MFCA-1). Only a direct clone/re-post of a `MeshFailure`
99        // preserves `reporting_controller`; this conversion is not one.
100        let err = ActorErrorKind::UnhandledSupervisionEvent(Box::new(ActorSupervisionEvent::new(
101            cx.instance().self_addr().clone(),
102            None,
103            ActorStatus::Failed(ActorErrorKind::UnhandledSupervisionEvent(Box::new(
104                self.event.clone(),
105            ))),
106            None,
107        )));
108        Err(anyhow::Error::new(err))
109    }
110}
111
112impl std::fmt::Display for MeshFailure {
113    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
114        let actor_mesh_name = self
115            .actor_mesh_name
116            .as_ref()
117            .map(|m| format!(" on mesh \"{}\"", m))
118            .unwrap_or("".to_string());
119        let ranks = if self.crashed_ranks.is_empty() {
120            String::new()
121        } else {
122            format!(" at ranks {:?}", self.crashed_ranks)
123        };
124        write!(
125            f,
126            "failure{}{} with event: {}",
127            actor_mesh_name, ranks, self.event
128        )
129    }
130}

401    // MFCA-4 + wire format: `reporting_controller` round-trips through bincode
402    // for both `Some` and `None`.
403    #[test]
404    fn reporting_controller_serde_round_trips() {
405        for reporting_controller in [Some(mk_actor_id("ctrl_proc", "controller")), None] {
406            let failure = MeshFailure {
407                actor_mesh_name: Some("training".to_string()),

412            let bytes = bincode::serde::encode_to_vec(&failure, bincode::config::legacy()).unwrap();
413            let (decoded, _): (MeshFailure, usize) =
414                bincode::serde::decode_from_slice(&bytes, bincode::config::legacy()).unwrap();
415            assert_eq!(decoded, failure);
416            assert_eq!(decoded.reporting_controller, reporting_controller);
417        }

420    // MFCA-3: the reporter and the event subject are separate roles carried
421    // independently. Here they hold different ids (a ProcAgent subject, a
422    // distinct controller reporter); on a controller-originated event they may
423    // instead hold the same id.

425    fn reporting_controller_and_event_subject_are_independent() {
426        let controller = mk_actor_id("ctrl_proc", "controller");
427        let subject = ResourceId::proc_addr_from_name(ChannelAddr::Local(0), "agent_proc")
428            .actor_addr("proc_agent");
429        let failure = MeshFailure {
430            actor_mesh_name: Some("training".to_string()),
431            event: ActorSupervisionEvent::new(
432                subject.clone(),
433                None,
434                ActorStatus::Failed(ActorErrorKind::Generic("boom".to_string())),
435                None,
436            ),
437            crashed_ranks: Vec::new(),
438            reporting_controller: Some(controller.clone()),
439        };
440        assert_eq!(failure.reporting_controller, Some(controller));
441        assert_eq!(failure.event.actor_id, subject);
