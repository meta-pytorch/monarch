91    let kilobytes = hyperactor_config::global::get(crate::config::RDMA_MIN_STRIPE_SIZE_KB).get();

155/// Shared state for selecting a compatible registration pair and dispatching
156/// operations directly to initialized queue-pair workers.
157///
158/// The manager installs a route after creating a queue pair and removes all
159/// routes before tearing its queue pairs down. Each [`IbvBackend`] shares this
160/// router with the manager so warm submissions can bypass the manager mailbox;
161/// operations without local registrations or a matching route fall back to
162/// [`SubmitOps`].
163#[derive(Debug)]
164struct QueuePairRouter {
165    peer_device_affinity: PeerDeviceAffinityPolicy,
166    qps_per_peer: NonZeroUsize,
167    next_qp_indices: DashMap<QpRoute, usize>,
168    queue_pairs: DashMap<QpKey, mpsc::UnboundedSender<ProcessOps>>,
169}
170
171impl QueuePairRouter {
172    fn new(peer_device_affinity: PeerDeviceAffinityPolicy, qps_per_peer: NonZeroUsize) -> Self {
173        Self {
174            peer_device_affinity,
175            qps_per_peer,
176            next_qp_indices: DashMap::new(),
177            queue_pairs: DashMap::new(),
178        }
179    }
180
181    fn next_qp_index(&self, route: QpRoute) -> usize {
182        let mut next = self.next_qp_indices.entry(route).or_insert(0);
183        let current = *next;
184        *next = (current + 1) % self.qps_per_peer.get();
185        current
186    }
187
188    /// Returns a disjoint list of registration pairings allowed by the affinity policy.
189    fn pair_registrations<'a>(
190        &self,
191        local: &'a [IbvMemoryRegionView],
192        remote: &'a [IbvRemoteMemoryRegionView],
193    ) -> Result<Vec<(&'a IbvMemoryRegionView, &'a IbvRemoteMemoryRegionView)>, anyhow::Error> {
194        // `PeerDeviceAffinityPolicy::pairs` is sensitive to input order. Sort
195        // both sides so selection depends only on the unordered registration
196        // sets and is consistent across processes.
197        let mut local: Vec<&IbvMemoryRegionView> = local.iter().collect();
198        local.sort_by(|a, b| a.device_name.cmp(&b.device_name));
199        let mut remote: Vec<&IbvRemoteMemoryRegionView> = remote.iter().collect();
200        remote.sort_by(|a, b| a.device_name.cmp(&b.device_name));
201
202        let local_names: Vec<String> = local.iter().map(|mr| mr.device_name.clone()).collect();
203        let remote_names: Vec<String> = remote.iter().map(|mr| mr.device_name.clone()).collect();
204        let pairs = self
205            .peer_device_affinity
206            .pairs(&local_names, &remote_names)
207            .into_iter()
208            .enumerate()
209            .filter_map(|(i, peer)| peer.map(|j| (local[i], remote[j])))
210            .collect::<Vec<_>>();
211        anyhow::ensure!(
212            !pairs.is_empty(),
213            "no NIC of {local_names:?} pairs with a peer NIC of {remote_names:?} under {:?}",
214            self.peer_device_affinity,
215        );
216        Ok(pairs)
217    }
218
219    /// Splits an operation into stripes over its compatible NIC pairs.
220    ///
221    /// An operation gets at least `qps_per_peer` stripes, rounded up to a
222    /// multiple of the pair count, as far as
223    /// [`RDMA_MIN_STRIPE_SIZE_KB`](crate::config::RDMA_MIN_STRIPE_SIZE_KB)
224    /// allows. One that is too small for a stripe on every pair uses a random
225    /// subset of the pairs, one stripe each. Otherwise the pairs take stripes
226    /// in turn, in device-name order. Each stripe uses the next queue pair on
227    /// its route.
228    fn plan_ops<I: IbvDeviceImpl>(
229        &self,
230        op_idx: usize,
231        op: &IbvOp<IbvManagerActor<I>>,
232    ) -> Result<Vec<QueuePairOp>, anyhow::Error> {
233        let local_mrs = op.local_memory.registered_mrs::<I>();
234        let mut pairs = self.pair_registrations(&local_mrs, &op.remote_buffers)?;
235        let other_id = op.remote_manager.actor_addr().id().clone();
236
237        let size = match op.op_type {
238            RdmaOpType::WriteFromLocal => op.local_memory.size(),
239            RdmaOpType::ReadIntoLocal => op
240                .remote_buffers
241                .first()
242                .map(|remote| remote.size)
243                .unwrap_or(0),
244        };
245        let max_stripes = (size / configured_min_stripe_size().get()).max(1);
246        let stripe_count = if max_stripes < pairs.len() {
247            pairs.shuffle(&mut rand::rng());
248            max_stripes
249        } else {
250            // Every pair takes the same number of stripes, so no NIC carries
251            // more than its share.
252            let wanted_per_pair = self.qps_per_peer.get().div_ceil(pairs.len());
253            let allowed_per_pair = max_stripes / pairs.len();
254            pairs.len() * wanted_per_pair.min(allowed_per_pair)
255        };
256
257        let stripe_size = size / stripe_count / STRIPE_ALIGNMENT * STRIPE_ALIGNMENT;
258        let mut offset = 0;
259        (0..stripe_count)
260            .map(|stripe_idx| {
261                let (local, remote) = pairs[stripe_idx % pairs.len()];
262                let size = if stripe_idx + 1 == stripe_count {
263                    size - offset
264                } else {
265                    stripe_size
266                };
267                let route = QpRoute {
268                    self_device: local.device_name.clone(),
269                    other_id: other_id.clone(),
270                    other_device: remote.device_name.clone(),
271                };
272                let op = QueuePairOp {
273                    stripe_id: StripeId {
274                        op_idx,
275                        stripe_idx,
276                        stripe_count,
277                    },
278                    qp_index: self.next_qp_index(route),
279                    op_type: op.op_type,
280                    local_memory: op.local_memory.clone(),
281                    local: local.try_slice(offset, size)?,
282                    remote: remote.try_slice(offset, size)?,
283                };
284                offset += size;
285                Ok(op)
286            })
287            .collect()
288    }
289
290    fn try_dispatch<I: IbvDeviceImpl>(
291        &self,
292        op_idx: usize,
293        op: IbvOp<IbvManagerActor<I>>,
294        reply: &mpsc::UnboundedSender<StripeResult>,
295    ) -> Option<ManagerOp<I>> {
296        if op.local_memory.registered_mrs::<I>().is_empty() {
297            return Some(ManagerOp::NeedsMemReg { op_idx, op });
298        }
299        let planned = match self.plan_ops(op_idx, &op) {
300            Ok(planned) => planned,
301            Err(error) => {
302                let _ = reply.send(StripeResult {
303                    stripe_id: StripeId {
304                        op_idx,
305                        stripe_idx: 0,
306                        stripe_count: 1,
307                    },
308                    result: Err(error.to_string()),
309                });
310                return None;
311            }
312        };
313        let senders = planned
314            .iter()
315            .map(|planned| {
316                let key = QpKey {
317                    self_device: planned.local.device_name.clone(),
318                    other_id: op.remote_manager.actor_addr().id().clone(),
319                    other_device: planned.remote.device_name.clone(),
320                    qp_index: planned.qp_index,
321                };
322                self.queue_pairs
323                    .get(&key)
324                    .map(|sender| sender.value().clone())
325            })
326            .collect::<Option<Vec<_>>>();
327        let Some(senders) = senders else {
328            return Some(ManagerOp::NeedsQueuePair {
329                peer_manager: op.remote_manager,
330                ops: planned,
331            });
332        };
333
334        for (sender, planned) in senders.into_iter().zip(planned) {
335            let stripe_id = planned.stripe_id;
336            if sender
337                .send(ProcessOps {
338                    items: vec![planned],
339                    reply: reply.clone(),
340                })
341                .is_err()
342            {
343                let _ = reply.send(StripeResult {
344                    stripe_id,
345                    result: Err("queue pair worker stopped".to_owned()),
346                });
347            }
348        }
349        None
350    }
351}

468impl<I: IbvDeviceImpl> Drop for IbvManagerActor<I> {
469    fn drop(&mut self) {
470        self.queue_pair_router.queue_pairs.clear();
471
472        // Stop each supervising QP actor; its cleanup cancels and joins the
473        // data-plane worker that owns the QP.
474        for (_key, handle) in self.qp_handles.drain() {
475            let _ = handle.drain_and_stop("IbvManagerActor dropped");
476        }
477
478        for (_key, poller) in self.cq_pollers.drain() {
479            let _ = poller.drain_and_stop("IbvManagerActor dropped");
480        }
481
482        // The remaining fields (`peer_created_qps`, `devices`) free their FFI
483        // resources through their elements' `Drop`s when this struct is dropped.
484    }
485}

488    /// Create a new IbvManagerActor with the given configuration.
489    pub async fn new(params: Option<IbvConfig>) -> Result<Self, anyhow::Error> {
490        if !ibverbs_supported() {
491            return Err(anyhow::anyhow!(
492                "Cannot create IbvManagerActor because RDMA is not supported on this machine"
493            ));
494        }
495
496        // Use the caller's config; when none is given, start from the
497        // defaults and let the backend seed its own.
498        let mut config = match params {
499            Some(config) => config,
500            None => {
501                let mut config = IbvConfig::default();
502                I::apply_config_defaults(&mut config);
503                config
504            }
505        };
506
507        // Preserve an explicit per-manager target. Otherwise parse the process
508        // default from `RDMA_IBVERBS_TARGET`; empty preserves automatic selection.
509        if config.target.is_none() {
510            config.target = super::device_selection::configured_ibverbs_target()?;
511        }
512        tracing::debug!("rdma is enabled, config target: {:?}", config.target);
513
514        // check config and hardware support align
515        if config.use_gpu_direct {
516            match validate_execution_context().await {
517                Ok(_) => {
518                    tracing::info!("GPU Direct RDMA execution context validated successfully");
519                }
520                Err(e) => {
521                    tracing::warn!(
522                        "GPU Direct RDMA execution context validation failed: {}. Downgrading to standard ibverbs mode.",
523                        e
524                    );
525                    config.use_gpu_direct = false;
526                }
527            }
528        }
529
530        let peer_device_affinity = configured_peer_device_affinity()?;
531        let actor = Self {
532            owner: OnceLock::new(),
533            qp_handles: HashMap::new(),
534            peer_created_qps: HashMap::new(),
535            devices: HashMap::new(),
536            peer_device_affinity: peer_device_affinity.clone(),
537            queue_pair_router: Arc::new(QueuePairRouter::new(
538                peer_device_affinity,
539                hyperactor_config::global::get(crate::config::RDMA_QPS_PER_PEER).into_std(),
540            )),
541            cq_pollers: HashMap::new(),
542            config,
543        };
544
545        Ok(actor)
546    }

562    /// Get the named RDMA device, opening it on first use.
563    fn get_or_create_device(
564        &mut self,
565        device_name: &str,
566    ) -> Result<&mut IbvDevice<I>, anyhow::Error> {
567        if !self.devices.contains_key(device_name) {

579    /// Chooses a set of NICs on which to register `mem`, then registers
580    /// `mem` on those NICs (caching the registrations inside the handle)
581    /// and returns the relevant [`IbvMemoryRegionView`]s. The call fails
582    /// only when there are no successful registrations.
583    ///
584    /// If an explicit `config.target` is set, then that is the NIC that is
585    /// chosen. Otherwise, NICs are chosen based on `pick_optimal_devices`.
586    ///
587    /// A region already registered on this backend keeps the NICs it has: the
588    /// set is chosen once, on the first call.
589    fn resolve_local_mrs(
590        &mut self,
591        mem: &KeepaliveLocalMemory,
592    ) -> Result<Vec<IbvMemoryRegionView>, anyhow::Error> {
593        let already_serving = mem.registered_mrs::<I>();
594        if !already_serving.is_empty() {
595            return Ok(already_serving);
596        }
597
598        let device_names = match &self.config.target {
599            Some(target) => vec![
600                resolve_target::<I>(target)?
601                    .ok_or_else(|| {
602                        anyhow::anyhow!("configured device target {:?} not found", target)
603                    })?
604                    .name()
605                    .clone(),
606            ],
607            None => self.pick_optimal_devices(mem)?,
608        };
609
610        let mut views = Vec::with_capacity(device_names.len());
611        let mut failure: Option<anyhow::Error> = None;
612        for device_name in &device_names {
613            match self.resolve_local_mr_on(mem, device_name) {
614                Ok(view) => views.push(view),
615                Err(error) => {
616                    tracing::warn!(
617                        "not serving [{:#x}, {:#x}) from {device_name}: {error:#}",
618                        mem.addr(),
619                        mem.addr() + mem.size(),
620                    );
621                    failure.get_or_insert(error);
622                }
623            }
624        }
625        if views.is_empty() {
626            return Err(failure.expect("`device_names` is never empty, so one of them failed"));
627        }
628        Ok(views)
629    }

631    /// Resolve `mem` to its [`IbvMemoryRegionView`] on `device_name`,
632    /// registering the region there on first use.
633    ///
634    /// The registration lives on `mem` itself, shared by every clone of that
635    /// handle, so one handle registers a given device at most once. Nothing
636    /// deduplicates separate handles over the same region, though: each
637    /// carries its own registrations, so a region covered by two handles is
638    /// registered once per handle. A failure is recorded on `mem` too and
639    /// returned to later callers rather than retried.
640    fn resolve_local_mr_on(
641        &mut self,
642        mem: &KeepaliveLocalMemory,
643        device_name: &str,
644    ) -> Result<IbvMemoryRegionView, anyhow::Error> {
645        if let Some(mrv) = mem.registered_mr::<I>(device_name)? {
646            return Ok(mrv);
647        }
648        tracing::debug!(
649            "Using RDMA device: {} for memory at 0x{:x}",
650            device_name,
651            mem.addr()
652        );
653
654        // The backend strategy handles host vs. device memory (standard MR,
655        // dmabuf MR, or a device-specific segment binding).
656        let registered = self
657            .get_or_create_device_domain(device_name)
658            .and_then(|domain| domain.register_mr(mem));
659        match registered {
660            Ok(mrv) => mem.install_mr::<I>(mrv),
661            Err(error) => {
662                mem.record_mr_failure::<I>(device_name, &error);
663                Err(error)
664            }
665        }
666    }
667
668    /// The NICs of backend `I` to serve `mem` from: up to
669    /// [`RDMA_MAX_NICS_PER_BUFFER`](crate::config::RDMA_MAX_NICS_PER_BUFFER)
670    /// of the optimal NICs returned by [`select_optimal_ibv_devices`], chosen
671    /// using [`PeerDeviceAffinityPolicy::choose`].
672    fn pick_optimal_devices(
673        &self,
674        mem: &KeepaliveLocalMemory,
675    ) -> Result<Vec<String>, anyhow::Error> {
676        let location = mem.location();
677        let devices = select_optimal_ibv_devices::<I>(location)?;
678        anyhow::ensure!(
679            !devices.is_empty(),
680            "no {} RDMA device has a path to {location:?}",
681            I::backend_name(),
682        );
683        let names: Vec<String> = devices.iter().map(|device| device.name().clone()).collect();
684        let max = hyperactor_config::global::get(crate::config::RDMA_MAX_NICS_PER_BUFFER);
685        // `names` is passed in the order returned by `select_optimal_ibv_devices`, which is
686        // lexicographic. `PeerDeviceAffinityPolicy::choose` is sensitive to input order,
687        // so ensuring `names` has the same order across procs is important for consistency.
688        Ok(self
689            .peer_device_affinity
690            .choose(&names, max.map(hyperactor_config::NonZeroUsize::into_std)))
691    }

693    /// Build a passive-side mirror QP for `qp_key`, connect it to
694    /// `sender_info`, and store it in [`Self::peer_created_qps`].
695    /// Returns the local endpoint the active side needs to finish
696    /// its own `connect`. Called from
697    /// [`Handler<CreatePeerQueuePair>`].
698    fn create_peer_qp(
699        &mut self,
700        qp_key: &QpKey,
701        sender_info: &IbvQpInfo,
702    ) -> Result<IbvQpInfo, anyhow::Error> {
703        if self.peer_created_qps.contains_key(qp_key) {
704            anyhow::bail!("peer queue pair already exists for {qp_key:?}");
705        }
706        let self_device = qp_key.self_device.clone();
707        let config = self.config.clone();
708        let mut qp = self
709            .get_or_create_device(&self_device)?
710            .create_non_posting_queue_pair(DEFAULT_DOMAIN, &config)
711            .map_err(|e| anyhow::anyhow!("could not create peer IbvQueuePair: {}", e))?;
712        let local_info = qp
713            .get_qp_info()
714            .map_err(|e| anyhow::anyhow!("could not extract peer QP info: {}", e))?;
715        qp.connect(sender_info)
716            .map_err(|e| anyhow::anyhow!("could not connect peer QP: {}", e))?;
717        self.peer_created_qps.insert(qp_key.clone(), qp);
718        Ok(local_info)
719    }

721    /// Returns the poller for `device`, spawning it on first use.
722    fn ensure_cq_poller(
723        &mut self,
724        cx: &Context<'_, Self>,
725        device: &str,
726    ) -> ActorHandle<CompletionQueueActor<IbvCq>> {
727        let key = if hyperactor_config::global::get(crate::config::RDMA_CQ_POLLER_PER_DEVICE) {
728            device.to_owned()
729        } else {
730            String::new()
731        };
732        self.cq_pollers
733            .entry(key)
734            .or_insert_with(|| cx.spawn(CompletionQueueActor::new()))
735            .clone()
736    }

738    /// Lazy active-side QP worker: if `qp_key` is absent from
739    /// [`Self::qp_handles`], create an [`IbvQueuePair`] on the
740    /// requested device and spawn a [`QueuePairActor`] to drive its handshake
741    /// and supervise its data-plane task. Returns a handle to the QP actor.
742    fn ensure_qp_actor(
743        &mut self,
744        cx: &Context<'_, Self>,
745        qp_key: &QpKey,
746        peer_manager: ActorRef<Self>,
747    ) -> Result<QueuePairHandle<Self, <I::Domain as IbvDomainImpl>::QueuePair>, anyhow::Error> {
748        if let Some(h) = self.qp_handles.get(qp_key) {
749            return Ok(h.clone());
750        }
751        let self_device = qp_key.self_device.clone();
752        let config = self.config.clone();
753        let cq_poller = self.ensure_cq_poller(cx, &self_device);
754        let (qp, cq_lease) = self
755            .get_or_create_device(&self_device)?
756            .create_queue_pair(DEFAULT_DOMAIN, &config)
757            .map_err(|e| anyhow::anyhow!("could not create IbvQueuePair for {qp_key:?}: {}", e))?;
758        let local_manager: ActorRef<Self> = cx.bind();
759        let is_loopback = local_manager.actor_addr() == peer_manager.actor_addr()
760            && qp_key.self_device == qp_key.other_device;
761        let (actor, sender) = QueuePairActor::new(
762            qp_key.clone(),
763            local_manager,
764            peer_manager,
765            qp,
766            cq_poller,
767            cq_lease,
768            is_loopback,
769            config.max_send_wr,
770        );
771        let handle = QueuePairHandle::new(cx.spawn(actor), sender);
772        self.queue_pair_router
773            .queue_pairs
774            .insert(qp_key.clone(), handle.sender());
775        self.qp_handles.insert(qp_key.clone(), handle.clone());
776        Ok(handle)
777    }
778
779    fn dispatch_ops(
780        &mut self,
781        cx: &Context<'_, Self>,
782        peer_manager: ActorRef<Self>,
783        ops: Vec<QueuePairOp>,
784        reply: &mpsc::UnboundedSender<StripeResult>,
785    ) {
786        for op in ops {
787            let stripe_id = op.stripe_id;
788            let qp_key = QpKey {
789                self_device: op.local.device_name.clone(),
790                other_id: peer_manager.actor_addr().id().clone(),
791                other_device: op.remote.device_name.clone(),
792                qp_index: op.qp_index,
793            };
794            let handle = match self.ensure_qp_actor(cx, &qp_key, peer_manager.clone()) {
795                Ok(handle) => handle,
796                Err(error) => {
797                    let _ = reply.send(StripeResult {
798                        stripe_id,
799                        result: Err(error.to_string()),
800                    });
801                    continue;
802                }
803            };
804            if let Err(error) = handle.send(ProcessOps {
805                items: vec![op],
806                reply: reply.clone(),
807            }) {
808                let _ = reply.send(StripeResult {
809                    stripe_id,
810                    result: Err(error.to_string()),
811                });
812            }
813        }
814    }
815}

817#[async_trait]
818impl<I: IbvDeviceImpl> Handler<SubmitOps<I>> for IbvManagerActor<I> {
819    async fn handle(&mut self, cx: &Context<Self>, msg: SubmitOps<I>) -> Result<(), anyhow::Error> {
820        let SubmitOps { ops, reply } = msg;
821
822        // Interleave MR resolution with QP dispatch: as soon as an op is
823        // planned, send its stripes to their queue-pair workers. They can post
824        // the stripes while the manager resolves the next op's registrations.
825        for op in ops {
826            let (peer_manager, planned) = match op {
827                ManagerOp::NeedsQueuePair { peer_manager, ops } => (peer_manager, ops),
828                ManagerOp::NeedsMemReg { op_idx, op } => {
829                    let whole_op = StripeId {
830                        op_idx,
831                        stripe_idx: 0,
832                        stripe_count: 1,
833                    };
834                    if let Err(error) = self.resolve_local_mrs(&op.local_memory) {
835                        let _ = reply.send(StripeResult {
836                            stripe_id: whole_op,
837                            result: Err(error.to_string()),
838                        });
839                        continue;
840                    }
841                    let planned = match self.queue_pair_router.plan_ops(op_idx, &op) {
842                        Ok(planned) => planned,
843                        Err(error) => {
844                            let _ = reply.send(StripeResult {
845                                stripe_id: whole_op,
846                                result: Err(error.to_string()),
847                            });
848                            continue;
849                        }
850                    };
851                    (op.remote_manager, planned)
852                }
853            };
854            self.dispatch_ops(cx, peer_manager, planned, &reply);
855        }
856        Ok(())
857    }
858}

1063    /// Submit a batch of RDMA operations.
1064    ///
1065    /// Translates each op to an `IbvOp`, then forwards them directly to the
1066    /// relevant queue-pairs when possible; any ops that cannot be forwarded
1067    /// directly (memory registration needed, QPs don't exist yet) are shipped to
1068    /// [`IbvManagerActor`] via [`SubmitOps`]. The manager interleaves local-MR
1069    /// resolution with dispatch.
1070    ///
1071    /// Waits for every stripe of every operation. Every stripe failure is
1072    /// collected into a single error, grouped by operation.
1073    async fn submit(
1074        &self,
1075        cx: &(impl hyperactor::context::Actor + Send + Sync),
1076        ops: Vec<RdmaOp>,
1077        timeout: Duration,
1078    ) -> Result<(), anyhow::Error> {
1079        let mut ibv_ops = Vec::with_capacity(ops.len());
1080        for op in ops {
1081            let ctx = <RdmaRemoteBuffer as ResolveRemoteBackendContext<IbvBackend<I>>>::resolve(
1082                &op.remote,
1083            )
1084            .expect("op routed to incompatible backend");
1085            ibv_ops.push(IbvOp {
1086                op_type: op.op_type,
1087                local_memory: op.local.clone(),
1088                remote_buffers: ctx.buffers,
1089                remote_manager: ctx.manager,
1090            });
1091        }
1092        let n = ibv_ops.len();
1093        let (reply, mut reply_rx) = mpsc::unbounded_channel();
1094
1095        let mut manager_ops = Vec::new();
1096        for (op_idx, op) in ibv_ops.into_iter().enumerate() {
1097            if let Some(op) = self.queue_pair_router.try_dispatch::<I>(op_idx, op, &reply) {
1098                manager_ops.push(op);
1099            }
1100        }
1101        if !manager_ops.is_empty() {
1102            self.manager.try_post(
1103                cx,
1104                SubmitOps {
1105                    ops: manager_ops,
1106                    reply,
1107                },
1108            )?;
1109        }
1110
1111        let mut outstanding: HashMap<usize, usize> = HashMap::with_capacity(n);
1112        // Per operation, the index and failure message of each of its stripes
1113        // that failed.
1114        let mut failures: HashMap<usize, Vec<(usize, String)>> = HashMap::new();
1115        let mut completed = 0usize;
1116        let mut received_stripes = 0usize;
1117        let mut terminal: Option<String> = None;
1118        let deadline = tokio::time::Instant::now() + timeout;
1119        while completed < n {
1120            tokio::select! {
1121                () = tokio::time::sleep_until(deadline) => {
1122                    terminal = Some(format!(
1123                        "submit timed out after {completed}/{n} ops ({received_stripes} stripe replies) with {} failed ops",
1124                        failures.len()
1125                    ));
1126                    break;
1127                }
1128                recv = reply_rx.recv() => {
1129                    match recv {
1130                        Some(StripeResult { stripe_id, result }) => {
1131                            received_stripes += 1;
1132                            if let Err(error) = result {
1133                                failures
1134                                    .entry(stripe_id.op_idx)
1135                                    .or_default()
1136                                    .push((stripe_id.stripe_idx, error));
1137                            }
1138                            let remaining = outstanding
1139                                .entry(stripe_id.op_idx)
1140                                .or_insert(stripe_id.stripe_count);
1141                            *remaining = remaining
1142                                .checked_sub(1)
1143                                .expect("an operation reports no more than its stripe count");
1144                            if *remaining == 0 {
1145                                outstanding.remove(&stripe_id.op_idx);
1146                                completed += 1;
1147                            }
1148                        }
1149                        None => {
1150                            terminal = Some(format!(
1151                                "operation result channel closed after {completed}/{n} ops ({received_stripes} stripe replies) with {} failed ops: \
1152                                some built-in RDMA actor likely stopped or failed, see logs for more details",
1153                                failures.len()
1154                            ));
1155                            break;
1156                        }
1157                    }
1158                }
1159            }
1160        }
1161
1162        if terminal.is_none() && failures.is_empty() {
1163            return Ok(());
1164        }
1165
1166        let mut failures: Vec<_> = failures.into_iter().collect();
1167        failures.sort_by_key(|(idx, _)| *idx);
1168        let mut msg = terminal.unwrap_or_else(|| format!("{}/{n} ops failed", failures.len()));
1169        if !failures.is_empty() {
1170            msg.push(':');
1171            for (idx, op_failures) in &mut failures {
1172                op_failures.sort_by_key(|(stripe_idx, _)| *stripe_idx);
1173                write!(msg, "\n  op {idx}:").expect("infallible String write");
1174                for (stripe_idx, err) in op_failures {
1175                    write!(msg, "\n    stripe {stripe_idx}: {err}")
1176                        .expect("infallible String write");
1177                }
1178            }
1179        }
1180        Err(anyhow::anyhow!(msg))
1181    }
1182}

1320    pub enum BufferHelperMessage {
1321        /// Allocate `size` bytes on `device`, pre-fill with `pattern`,
1322        /// register with the local `RdmaManagerActor`, and reply with
1323        /// the resulting `RdmaRemoteBuffer`.
1324        Allocate {
1325            size: usize,
1326            device: BufferDevice,
1327            pattern: u8,
1328            #[reply]
1329            reply: OncePortRef<RdmaRemoteBuffer>,
1330        },
1331        /// Like `Allocate`, but fills byte `i` with [`positional_byte`]`(i)`.
1332        AllocatePositional {
1333            size: usize,
1334            device: BufferDevice,
1335            #[reply]
1336            reply: OncePortRef<RdmaRemoteBuffer>,
1337        },
1338        /// Look up the local memory behind `remote.id` and reply with
1339        /// the byte range `[offset, offset + len)`. Tests use this to
1340        /// sample buffers too large to ship over a single actor
1341        /// message in one piece.
1342        ReadContents {
1343            remote: Box<RdmaRemoteBuffer>,
1344            offset: usize,
1345            len: usize,
1346            #[reply]
1347            reply: OncePortRef<Vec<u8>>,
1348        },
1349        /// Drive a batch of RDMA ops through `IbvBackend::submit`.
1350        /// Each op's `local_buf` is resolved against this helper's
1351        /// `RdmaManagerActor`; `remote_buf` is shipped as-is to the
1352        /// peer.
1353        Submit {
1354            ops: Vec<BufferHelperOp>,
1355            timeout_secs: u64,
1356            #[reply]
1357            reply: OncePortRef<Result<(), String>>,
1358        },
1359    }

1361    impl BufferHelperActor {
1362        async fn allocate_impl(
1363            &mut self,
1364            cx: &Context<'_, Self>,
1365            device: BufferDevice,
1366            contents: Vec<u8>,
1367        ) -> Result<RdmaRemoteBuffer, anyhow::Error> {
1368            let size = contents.len();
1369            let local = match device {
1370                BufferDevice::Cpu => {
1371                    KeepaliveLocalMemory::try_new(Arc::new(contents.into_boxed_slice()))?
1372                }
1373                BufferDevice::Cuda(device_id) => {
1374                    let alloc = CudaAllocator::get().allocate(device_id, size, size);
1375                    let local = KeepaliveLocalMemory::try_new(Arc::new(alloc.clone()))?;
1376                    self.cuda_allocs.push(alloc);
1377                    // SAFETY: `local` is freshly constructed; no other
1378                    // holder touches this CUDA range yet.
1379                    unsafe { local.write_at(0, &contents) }?;
1380                    local
1381                }
1382            };
1383            let handle = self
1384                .rdma_manager
1385                .downcast_handle(cx)
1386                .ok_or_else(|| anyhow::anyhow!("rdma_manager not local to BufferHelperActor"))?;
1387            handle.request_buffer(cx, local).await
1388        }
1389
1390        async fn read_contents_impl(
1391            &mut self,
1392            cx: &Context<'_, Self>,
1393            remote: RdmaRemoteBuffer,
1394            offset: usize,
1395            len: usize,
1396        ) -> Result<Vec<u8>, anyhow::Error> {
1397            let handle = self
1398                .rdma_manager
1399                .downcast_handle(cx)
1400                .ok_or_else(|| anyhow::anyhow!("rdma_manager not local"))?;
1401            let local = handle
1402                .request_local_memory(cx, remote.id)
1403                .await?
1404                .ok_or_else(|| {
1405                    anyhow::anyhow!(
1406                        "no local memory registered on this side for remote_buf_id={}",
1407                        remote.id,
1408                    )
1409                })?;
1410            let mut out = vec![0u8; len];
1411            // SAFETY: by convention the caller has ensured all RDMA
1412            // ops against this buffer have completed before invoking
1413            // ReadContents.
1414            unsafe { local.read_at(offset, &mut out)? };
1415            Ok(out)
1416        }
1417
1418        async fn submit_impl(
1419            &mut self,
1420            cx: &Context<'_, Self>,
1421            ops: Vec<BufferHelperOp>,
1422            timeout_secs: u64,
1423        ) -> Result<Result<(), String>, anyhow::Error> {
1424            let handle = self
1425                .rdma_manager
1426                .downcast_handle(cx)
1427                .ok_or_else(|| anyhow::anyhow!("rdma_manager not local"))?;
1428            let mut rdma_ops = Vec::with_capacity(ops.len());
1429            for (i, op) in ops.into_iter().enumerate() {
1430                let local = handle
1431                    .request_local_memory(cx, op.local_buf.id)
1432                    .await?
1433                    .ok_or_else(|| {
1434                        anyhow::anyhow!(
1435                            "op {i}: no local memory registered for remote_buf_id={}",
1436                            op.local_buf.id,
1437                        )
1438                    })?;
1439                rdma_ops.push(RdmaOp {
1440                    op_type: op.op_type,
1441                    local,
1442                    remote: op.remote_buf,
1443                });
1444            }
1445            let nic = RdmaManagerActor::local_handle(cx)
1446                .get_backend_handles(cx)
1447                .await?
1448                .into_iter()
1449                .find(|h| !matches!(h, RdmaBackendHandle::Tcp(_)))
1450                .ok_or_else(|| anyhow::anyhow!("no NIC backend on this proc"))?;
1451            let result = nic
1452                .submit(cx, rdma_ops, Duration::from_secs(timeout_secs))
1453                .await;
1454            Ok(result.map_err(|e| format!("{e}")))
1455        }
1456    }

1612                "{what}: byte {offset} of {size} is {:#04x}, expected {:#04x}",

1614                positional_byte(offset),

1616        }
