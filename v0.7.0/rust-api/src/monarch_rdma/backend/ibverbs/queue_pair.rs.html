80/// A per-work-request completion failure: a work request completed with

194// =====================================================================
195// QueuePairActor
196// =====================================================================
197//
198// Owns one simplex queue pair to a peer. The active side spawns a
199// `QueuePairActor` that owns and operates the local QP; the peer's
200// manager eagerly creates its mirror QP during the handshake, stores
201// it as a passive endpoint, and never wraps it in an actor. If the
202// peer later wants to RDMA us, it spawns its own `QueuePairActor`
203// locally, which creates a fresh pair the same way.
204
205/// A NIC-backend queue pair: the unit of RDMA communication between two
206/// endpoints, and the operations a [`QueuePairActor`] performs on it. Each
207/// [`IbvDomainImpl`] names its concrete queue-pair type via
208/// [`IbvDomainImpl::QueuePair`](super::domain::IbvDomainImpl::QueuePair) and builds
209/// it through [`Self::new`].
210pub trait IbvQueuePair: std::fmt::Debug + Send + Sync + 'static + Sized {
211    /// Creates a queue pair against `domain` in the RESET state, reporting its
212    /// completions on `send_cq` and `recv_cq`; [`Self::connect`] transitions it
213    /// to RTS before use.
214    ///
215    /// The completion queues are created by the caller rather than here, so that
216    /// one queue can back many queue pairs. Implementers hold a clone of each,
217    /// which is what keeps them alive for the QP's lifetime.
218    ///
219    /// # Safety
220    ///
221    /// `domain`'s PD (`domain.as_ptr()`) must be null or a valid protection
222    /// domain. Callers must ensure the PD outlives the QP; the easiest way to
223    /// do this is for implementers to store a clone of `domain.pd()` (an
224    /// `Arc<IbvPd>`) inside the QP. `send_cq` and `recv_cq` must have been
225    /// created on `domain`'s device context.
226    unsafe fn new<I: IbvDomainImpl<QueuePair = Self>>(
227        domain: &IbvDomain<I>,
228        config: IbvConfig,
229        send_cq: Arc<IbvCq>,
230        recv_cq: Arc<IbvCq>,
231    ) -> Result<Self, anyhow::Error>;
232
233    /// Transitions the QP through `INIT -> RTR -> RTS`, connected to `info`.
234    fn connect(&mut self, info: &IbvQpInfo) -> Result<(), anyhow::Error>;
235
236    /// Returns the local endpoint other peers need in order to connect to this
237    /// QP.
238    fn get_qp_info(&mut self) -> Result<IbvQpInfo, anyhow::Error>;
239
240    /// Returns the current `ibv_qp_state` of the QP.
241    fn state(&mut self) -> Result<u32, anyhow::Error>;
242
243    /// Largest transfer this QP issues as a single work request; [`Self::put`]
244    /// and [`Self::get`] split anything larger into that many WRs. The owning
245    /// [`QueuePairActor`] budgets send-queue credits against this value, so an
246    /// implementation that chunks more finely must report it here or it will
247    /// over-commit the send queue.
248    fn max_msg_size(&self) -> usize {
249        MAX_RDMA_MSG_SIZE
250    }
251
252    /// Post an RDMA WRITE of `local_src` into `remote_dst`. The request may
253    /// be chunked into multiple WRs. This method returns the list of WR ids
254    /// that were posted.
255    fn put(
256        &mut self,
257        remote_dst: IbvRemoteMemoryRegionView,
258        local_src: IbvMemoryRegionView,
259    ) -> Result<Vec<u64>, anyhow::Error>;
260
261    /// Post an RDMA READ of `remote_src` into `local_dst`. The request may
262    /// be chunked into multiple WRs. This method returns the list of WR ids
263    /// that were posted.
264    fn get(
265        &mut self,
266        local_dst: IbvMemoryRegionView,
267        remote_src: IbvRemoteMemoryRegionView,
268    ) -> Result<Vec<u64>, anyhow::Error>;
269}

271/// Queries the local endpoint info for `qp`, whose device `context` and the QP
272/// `config` (port, PSN) describe the connection. `gid` is the port's source GID.
273///
274/// # Safety
275///
276/// `qp` must be a live `ibv_qp` (non-null) and `context` must be its live device
277/// context.
278pub(super) unsafe fn get_qp_info(
279    qp: *mut rdmaxcel_sys::ibv_qp,
280    context: *mut rdmaxcel_sys::ibv_context,
281    config: &IbvConfig,
282    gid: Gid,
283) -> Result<IbvQpInfo, anyhow::Error> {
284    let mut port_attr = rdmaxcel_sys::ibv_port_attr::default();
285    // SAFETY: `context` is a live device context (caller contract); the
286    // out-param is a writable, properly aligned `ibv_port_attr`. `ibv_query_port`
287    // returns the errno on failure.
288    let errno = unsafe {
289        rdmaxcel_sys::ibv_query_port(
290            context,
291            config.port_num,
292            &mut port_attr as *mut rdmaxcel_sys::ibv_port_attr as *mut _,
293        )
294    };
295    if errno != 0 {
296        return Err(anyhow::anyhow!(
297            "failed to query port attributes: {}",
298            Error::from_raw_os_error(errno)
299        ));
300    }
301
302    // SAFETY: `qp` is a live `ibv_qp` (caller contract).
303    let qp_num = unsafe { (*qp).qp_num };
304    Ok(IbvQpInfo {
305        qp_num,
306        lid: port_attr.lid,
307        gid: Some(gid),
308        psn: config.psn,
309    })
310}

312/// Returns the current `ibv_qp_state` of `qp`.
313///
314/// # Safety
315///
316/// `qp` must be a live `ibv_qp` (non-null).
317pub(super) unsafe fn state(qp: *mut rdmaxcel_sys::ibv_qp) -> Result<u32, anyhow::Error> {
318    let mut qp_attr = rdmaxcel_sys::ibv_qp_attr::default();
319    let mut qp_init_attr = rdmaxcel_sys::ibv_qp_init_attr::default();
320    let mask = rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_STATE;
321    // SAFETY: `qp` wraps a live `ibv_qp` (caller contract); the out-params are
322    // writable, properly aligned attr structs. `ibv_query_qp` returns the errno.
323    let errno =
324        unsafe { rdmaxcel_sys::ibv_query_qp(qp, &mut qp_attr, mask.0 as i32, &mut qp_init_attr) };
325    if errno != 0 {
326        return Err(anyhow::anyhow!(
327            "failed to query QP state: {}",
328            Error::from_raw_os_error(errno)
329        ));
330    }
331    Ok(qp_attr.qp_state)
332}

334/// Transitions `qp` through `INIT -> RTR -> RTS`, connected to `info`, granting
335/// remote peers `access_flags` and using the connection parameters in `config`.
336/// `sgid_index` is the local source-GID table index, used as the address
337/// handle's `sgid_index` when `info` carries a (remote) GID.
338///
339/// # Safety
340///
341/// `qp` must be a live `ibv_qp` (non-null).
342pub(super) unsafe fn connect(
343    qp: *mut rdmaxcel_sys::ibv_qp,
344    config: &IbvConfig,
345    access_flags: i32,
346    info: &IbvQpInfo,
347    sgid_index: u8,
348) -> Result<(), anyhow::Error> {
349    // Transition to INIT.
350    let mut qp_attr = rdmaxcel_sys::ibv_qp_attr {
351        qp_state: rdmaxcel_sys::ibv_qp_state::IBV_QPS_INIT,
352        qp_access_flags: access_flags as u32,
353        pkey_index: config.pkey_index,
354        port_num: config.port_num,
355        ..Default::default()
356    };
357    let mask = rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_STATE
358        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_PKEY_INDEX
359        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_PORT
360        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_ACCESS_FLAGS;
361    // SAFETY: `qp` is a live `ibv_qp` (caller contract); `qp_attr` is a valid
362    // `ibv_qp_attr` whose populated fields match `mask`. `ibv_modify_qp` returns
363    // the errno.
364    let errno = unsafe { rdmaxcel_sys::ibv_modify_qp(qp, &mut qp_attr, mask.0 as i32) };
365    if errno != 0 {
366        return Err(anyhow::anyhow!(
367            "failed to transition QP to INIT: {}",
368            Error::from_raw_os_error(errno)
369        ));
370    }
371
372    // Transition to RTR (Ready to Receive).
373    let mut qp_attr = rdmaxcel_sys::ibv_qp_attr {
374        qp_state: rdmaxcel_sys::ibv_qp_state::IBV_QPS_RTR,
375        path_mtu: config.path_mtu,
376        dest_qp_num: info.qp_num,
377        rq_psn: info.psn,
378        max_dest_rd_atomic: config.max_dest_rd_atomic,
379        min_rnr_timer: config.min_rnr_timer,
380        ah_attr: rdmaxcel_sys::ibv_ah_attr {
381            dlid: info.lid,
382            sl: 0,
383            src_path_bits: 0,
384            port_num: config.port_num,
385            grh: Default::default(),
386            ..Default::default()
387        },
388        ..Default::default()
389    };
390    if let Some(gid) = info.gid {
391        qp_attr.ah_attr.is_global = 1;
392        qp_attr.ah_attr.grh.dgid = rdmaxcel_sys::ibv_gid::from(gid);
393        qp_attr.ah_attr.grh.hop_limit = 0xff;
394        qp_attr.ah_attr.grh.sgid_index = sgid_index;
395    } else {
396        qp_attr.ah_attr.is_global = 0;
397    }
398    let mask = rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_STATE
399        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_AV
400        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_PATH_MTU
401        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_DEST_QPN
402        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_RQ_PSN
403        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_MAX_DEST_RD_ATOMIC
404        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_MIN_RNR_TIMER;
405    // SAFETY: as for the INIT transition above.
406    let errno = unsafe { rdmaxcel_sys::ibv_modify_qp(qp, &mut qp_attr, mask.0 as i32) };
407    if errno != 0 {
408        return Err(anyhow::anyhow!(
409            "failed to transition QP to RTR: {}",
410            Error::from_raw_os_error(errno)
411        ));
412    }
413
414    // Transition to RTS (Ready to Send).
415    let mut qp_attr = rdmaxcel_sys::ibv_qp_attr {
416        qp_state: rdmaxcel_sys::ibv_qp_state::IBV_QPS_RTS,
417        sq_psn: config.psn,
418        max_rd_atomic: config.max_rd_atomic,
419        retry_cnt: config.retry_cnt,
420        rnr_retry: config.rnr_retry,
421        timeout: config.qp_timeout,
422        ..Default::default()
423    };
424    let mask = rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_STATE
425        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_TIMEOUT
426        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_RETRY_CNT
427        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_SQ_PSN
428        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_RNR_RETRY
429        | rdmaxcel_sys::ibv_qp_attr_mask::IBV_QP_MAX_QP_RD_ATOMIC;
430    // SAFETY: as for the INIT transition above.
431    let errno = unsafe { rdmaxcel_sys::ibv_modify_qp(qp, &mut qp_attr, mask.0 as i32) };
432    if errno != 0 {
433        return Err(anyhow::anyhow!(
434            "failed to transition QP to RTS: {}",
435            Error::from_raw_os_error(errno)
436        ));
437    }
438    Ok(())
439}

441/// An RDMA reliable-connected (RC) queue pair built on plain ibverbs
442/// (`ibv_post_send`), independent of any device-specific verbs.
443///
444/// Single-owner: it owns the [`IbvQp`] — which in turn owns its two completion
445/// queues and the protection domain — and destroys them on drop, so the type is
446/// intentionally `!Clone`. The device context and completion queues are reached
447/// through the [`IbvQp`].
448#[derive(Debug)]
449pub struct RCQueuePair {
450    qp: IbvQp,
451    config: IbvConfig,
452    /// The source GID (carrying its table index), resolved from the owning
453    /// device's `IbvDeviceInfo` at construction: the first global RoCE v2 GID on
454    /// `config.port_num`.
455    gid: Gid,
456    /// Remote-access flags granted to peers at connect time, taken from the
457    /// owning domain at construction.
458    access_flags: i32,
459    /// Monotonic work-request id, handed out one per posted WR. Standard
460    /// ibverbs carries no internal counter, so the QP tracks its own.
461    next_wr_id: u64,
462}
463
464impl RCQueuePair {
465    /// Assembles an `RCQueuePair` that owns `qp` (and, through it, its
466    /// completion queues, protection domain, and device context). `access_flags`
467    /// is granted to peers at [`Self::connect`]; it is taken from the owning
468    /// domain by the caller.
469    ///
470    /// # Safety
471    ///
472    /// `qp` must wrap a live, non-null `ibv_qp`, and everything reached through
473    /// it — its send/recv completion queues, protection domain, and device
474    /// context — must likewise be non-null and valid: the data-path verbs invoke
475    /// them without re-checking.
476    pub(super) unsafe fn from_qp(
477        qp: IbvQp,
478        config: IbvConfig,
479        gid: Gid,
480        access_flags: i32,
481    ) -> Self {
482        RCQueuePair {
483            qp,
484            config,
485            gid,
486            access_flags,
487            next_wr_id: 0,
488        }
489    }
490
491    /// Posts `op` over `[laddr, laddr + total_size)` to `[raddr, ...)`,
492    /// splitting into `MAX_RDMA_MSG_SIZE`-bound chunks and returning one wr_id
493    /// per chunk.
494    fn post_chunked(
495        &mut self,
496        op: IbvOperation,
497        laddr: usize,
498        lkey: u32,
499        raddr: usize,
500        rkey: u32,
501        total_size: usize,
502    ) -> Result<Vec<u64>, anyhow::Error> {
503        let mut remaining = total_size;
504        let mut offset = 0;
505        let mut wr_ids = Vec::new();
506        while remaining > 0 {
507            let chunk = std::cmp::min(remaining, MAX_RDMA_MSG_SIZE);
508            let wr_id = self.next_wr_id;
509            self.next_wr_id += 1;
510            self.post_one(op, laddr + offset, lkey, chunk, raddr + offset, rkey, wr_id)?;
511            wr_ids.push(wr_id);
512            remaining -= chunk;
513            offset += chunk;
514        }
515        Ok(wr_ids)
516    }
517
518    /// Posts a single signaled RDMA `op` work request via `ibv_post_send`.
519    fn post_one(
520        &self,
521        op: IbvOperation,
522        laddr: usize,
523        lkey: u32,
524        length: usize,
525        raddr: usize,
526        rkey: u32,
527        wr_id: u64,
528    ) -> Result<(), anyhow::Error> {
529        let qp = self.qp.as_ptr();
530        let context = self.qp.context().as_ptr();
531        let mut sge = rdmaxcel_sys::ibv_sge {
532            addr: laddr as u64,
533            length: length as u32,
534            lkey,
535        };
536        let mut wr = rdmaxcel_sys::ibv_send_wr {
537            wr_id,
538            next: std::ptr::null_mut(),
539            sg_list: &mut sge as *mut _,
540            num_sge: 1,
541            opcode: op.into(),
542            send_flags: rdmaxcel_sys::ibv_send_flags::IBV_SEND_SIGNALED.0,
543            wr: Default::default(),
544            qp_type: Default::default(),
545            __bindgen_anon_1: Default::default(),
546            __bindgen_anon_2: Default::default(),
547        };
548        // Set the RDMA target. Writing through a union field is safe; the device
549        // reads the `rdma` member for the RDMA_WRITE/RDMA_READ opcodes set above.
550        wr.wr.rdma.remote_addr = raddr as u64;
551        wr.wr.rdma.rkey = rkey;
552        let mut bad_wr: *mut rdmaxcel_sys::ibv_send_wr = std::ptr::null_mut();
553        // SAFETY: `context` is the QP's live device context (read from the live
554        // `qp`); we invoke its `post_send` verb through the ops table. `qp` is
555        // live and `wr`/`sge`/`bad_wr` are valid for the duration of the call.
556        let errno = unsafe {
557            let post_send = (*context)
558                .ops
559                .post_send
560                .expect("post_send verb missing from ibv_context ops");
561            post_send(qp, &mut wr as *mut _, &mut bad_wr)
562        };
563        if errno != 0 {
564            return Err(anyhow::anyhow!(
565                "failed to post {:?} request: {}",
566                op,
567                Error::from_raw_os_error(errno)
568            ));
569        }
570        Ok(())
571    }
572}

574impl IbvQueuePair for RCQueuePair {
575    unsafe fn new<I: IbvDomainImpl<QueuePair = Self>>(
576        domain: &IbvDomain<I>,
577        config: IbvConfig,
578        send_cq: Arc<IbvCq>,
579        recv_cq: Arc<IbvCq>,
580    ) -> Result<Self, anyhow::Error> {
581        tracing::debug!("creating an RCQueuePair from config {}", config);
582        // `IbvDomain`'s `pd` accessor permits null (e.g. a test domain); a real
583        // QP needs one, so reject null up front.
584        let pd = domain.as_ptr();
585        if pd.is_null() {
586            anyhow::bail!("cannot create an RCQueuePair on a null protection domain");
587        }
588
589        // Resolve the source GID up front (before allocating any FFI resources),
590        // so a port without a global RoCE v2 GID fails cleanly here.
591        let gid = domain.device_info().select_gid(
592            config.port_num,
593            Some(GidScope::Global),
594            Some(GidType::RoCEv2),
595        )?;
596
597        // A standard RC QP with the caps from `config`.
598        let mut init_attr = rdmaxcel_sys::ibv_qp_init_attr {
599            send_cq: send_cq.as_ptr(),
600            recv_cq: recv_cq.as_ptr(),
601            cap: rdmaxcel_sys::ibv_qp_cap {
602                max_send_wr: config.max_send_wr,
603                max_recv_wr: config.max_recv_wr,
604                max_send_sge: config.max_send_sge,
605                max_recv_sge: config.max_recv_sge,
606                max_inline_data: 0,
607            },
608            qp_type: rdmaxcel_sys::ibv_qp_type::IBV_QPT_RC,
609            sq_sig_all: 0,
610            ..Default::default()
611        };
612        // SAFETY: `pd` is non-null (checked above) and live per `IbvDomain`'s
613        // construction contract; `init_attr` is a fully initialized
614        // `ibv_qp_init_attr`. `ibv_create_qp` returns null on failure.
615        let qp = unsafe { rdmaxcel_sys::ibv_create_qp(pd, &mut init_attr) };
616        if qp.is_null() {
617            anyhow::bail!(
618                "failed to create queue pair (QP): {}",
619                Error::last_os_error()
620            );
621        }
622        // SAFETY: `qp` is a live RC QP just created against `pd` with
623        // `send_cq`/`recv_cq`; `IbvQp` holds a clone of each, keeping them alive
624        // for at least as long as the QP it destroys on drop.
625        let qp = unsafe { IbvQp::from_raw(qp, send_cq, recv_cq, domain.pd().clone()) };
626        let access_flags = domain.access_flags();
627        // SAFETY: `qp` and its `send_cq`/`recv_cq`/PD/context were all created
628        // non-null.
629        Ok(unsafe { Self::from_qp(qp, config, gid, access_flags) })
630    }
631
632    fn connect(&mut self, info: &IbvQpInfo) -> Result<(), anyhow::Error> {
633        // SAFETY: `self.qp` is the live QP, kept alive for `self`'s lifetime.
634        unsafe {
635            connect(
636                self.qp.as_ptr(),
637                &self.config,
638                self.access_flags,
639                info,
640                self.gid.index(),
641            )
642        }
643    }
644
645    fn get_qp_info(&mut self) -> Result<IbvQpInfo, anyhow::Error> {
646        let context = self.qp.context().as_ptr();
647        // SAFETY: `self.qp` is the live QP and `context` its non-null device
648        // context (validated inside `new`), both valid for `self`'s lifetime.
649        unsafe { get_qp_info(self.qp.as_ptr(), context, &self.config, self.gid) }
650    }
651
652    fn state(&mut self) -> Result<u32, anyhow::Error> {
653        // SAFETY: `self.qp` is the live QP, kept alive for `self`'s lifetime.
654        unsafe { state(self.qp.as_ptr()) }
655    }
656
657    fn put(
658        &mut self,
659        remote_dst: IbvRemoteMemoryRegionView,
660        local_src: IbvMemoryRegionView,
661    ) -> Result<Vec<u64>, anyhow::Error> {
662        if remote_dst.size < local_src.size {
663            return Err(anyhow::anyhow!(
664                "remote buffer size ({}) is smaller than local buffer size ({})",
665                remote_dst.size,
666                local_src.size
667            ));
668        }
669        self.post_chunked(
670            IbvOperation::Write,
671            local_src.rdma_addr,
672            local_src.lkey,
673            remote_dst.addr,
674            remote_dst.rkey,
675            local_src.size,
676        )
677    }
678
679    fn get(
680        &mut self,
681        local_dst: IbvMemoryRegionView,
682        remote_src: IbvRemoteMemoryRegionView,
683    ) -> Result<Vec<u64>, anyhow::Error> {
684        if local_dst.size < remote_src.size {
685            return Err(anyhow::anyhow!(
686                "local buffer size ({}) is smaller than remote buffer size ({})",
687                local_dst.size,
688                remote_src.size
689            ));
690        }
691        self.post_chunked(
692            IbvOperation::Read,
693            local_dst.rdma_addr,
694            local_dst.lkey,
695            remote_src.addr,
696            remote_src.rkey,
697            remote_src.size,
698        )
699    }

969impl<Qp: IbvQueuePair> QueuePairState<Qp> {
970    fn new(qp_key: QpKey, qp: Qp, max_send_wr: u32, posted_wrs: Arc<AtomicU64>) -> Self {
971        Self {
972            qp_key,
973            qp,
974            max_send_wr,
975            queue: VecDeque::new(),
976            posted: HashMap::new(),
977            wr_to_op: HashMap::new(),
978            next_op_id: 0,
979            in_flight: 0,
980            posted_wrs,
981            poller_wake: None,
982        }
983    }
984
985    fn enqueue(&mut self, message: ProcessOps) {
986        let max_msg_size = self.qp.max_msg_size();
987        self.queue
988            .extend(message.items.into_iter().map(|op| PendingOp {
989                wrs: op.local.size.div_ceil(max_msg_size).max(1) as u32,
990                op,
991                reply: message.reply.clone(),
992            }));
993    }
994
995    /// Try to post the head of `queue`.
996    ///
997    /// * `Ok(true)` — head was either posted or rejected with a
998    ///   per-op error (e.g. op too large for this QP); caller
999    ///   should attempt the next head.
1000    /// * `Ok(false)` — head can't be posted because the QP is at
1001    ///   `max_send_wr`. The op stays at the head; caller should stop
1002    ///   walking.
1003    /// * `Err(_)` — `qp.put`/`qp.get` failed (e.g. the QP is in
1004    ///   error state). Fatal: the worker reports this to its supervising actor.
1005    fn try_post_head(&mut self) -> Result<bool, anyhow::Error> {
1006        let pending = self.queue.pop_front().expect("non-empty queue");
1007        let PendingOp { op, reply, wrs } = pending;
1008        let stripe_id = op.stripe_id;
1009
1010        // 1. Per-op fatal: op alone exceeds the QP's capacity.
1011        if wrs > self.max_send_wr {
1012            let err = format!(
1013                "op too large for this QP [{stripe_id}, qp_key={:?}, op_type={:?}, wrs={}, max_send_wr={}, local: {:?}, remote: {:?}]",
1014                self.qp_key, op.op_type, wrs, self.max_send_wr, op.local_memory, op.remote,
1015            );
1016            let _ = reply.send(StripeResult {
1017                stripe_id,
1018                result: Err(err),
1019            });
1020            return Ok(true);
1021        }
1022
1023        // 2. Credit gating. Every op consumes send-queue slots.
1024        //    A head op that does not fit stalls the queue behind
1025        //    it until enough WRs are reaped.
1026        if self.in_flight + wrs > self.max_send_wr {
1027            self.queue.push_front(PendingOp { op, reply, wrs });
1028            return Ok(false);
1029        }
1030
1031        // 3. Post.
1032        let local = op.local;
1033        let post_result = match op.op_type {
1034            RdmaOpType::WriteFromLocal => self.qp.put(op.remote.clone(), local.clone()),
1035            RdmaOpType::ReadIntoLocal => self.qp.get(local.clone(), op.remote.clone()),
1036        };
1037        let wr_ids = post_result.map_err(|e| {
1038            anyhow::anyhow!(
1039                "qp.{} failed [{stripe_id}, qp_key={:?}, local: {:?}, remote: {:?}]: {e}",
1040                if matches!(op.op_type, RdmaOpType::ReadIntoLocal) {
1041                    "get"
1042                } else {
1043                    "put"
1044                },
1045                self.qp_key,
1046                local,
1047                op.remote,
1048            )
1049        })?;
1050
1051        // 4. Track in-flight state.
1052        let op_id = self.next_op_id;
1053        self.next_op_id += 1;
1054        let mut pending_wrs = HashSet::with_capacity(wr_ids.len());
1055        for id in &wr_ids {
1056            self.wr_to_op.insert(*id, op_id);
1057            pending_wrs.insert(*id);
1058        }
1059        self.in_flight += wr_ids.len() as u32;
1060        self.posted.insert(
1061            op_id,
1062            PostedOpEntry {
1063                stripe_id,
1064                pending_wrs,
1065                _local_memory: op.local_memory,
1066                _local_mrv: local,
1067                reply,
1068                first_error: None,
1069            },
1070        );
1071        // This state is the counter's only writer. The following `unpark`
1072        // publishes the increment before a parked poller checks the count.
1073        self.posted_wrs
1074            .fetch_add(wr_ids.len() as u64, Ordering::Relaxed);
1075        self.poller_wake
1076            .as_ref()
1077            .expect("the queue pair attached before posting work")
1078            .notify()
1079            .map_err(|error| {
1080                anyhow::anyhow!(
1081                    "could not wake the completion queue poller [qp_key={:?}]: {error}",
1082                    self.qp_key,
1083                )
1084            })?;
1085        Ok(true)
1086    }
1087
1088    fn post_ready(&mut self) -> Result<(), anyhow::Error> {
1089        while !self.queue.is_empty() {
1090            if !self.try_post_head()? {
1091                break;
1092            }
1093        }
1094        Ok(())
1095    }
1096
1097    fn complete_wr(&mut self, completion: CompletionResult) -> Result<(), anyhow::Error> {
1098        let (wr_id, wr_error) = match completion {
1099            Ok(wc) => (wc.wr_id(), None),
1100            Err(per_wr) => (per_wr.wr_id, Some(per_wr.to_string())),
1101        };
1102        let op_id = self
1103            .wr_to_op
1104            .remove(&wr_id)
1105            .expect("completed wr_id missing from wr_to_op");
1106        let entry = self
1107            .posted
1108            .get_mut(&op_id)
1109            .expect("op_id missing from posted");
1110        entry.pending_wrs.remove(&wr_id);
1111        self.in_flight -= 1;
1112        if let Some(err) = wr_error
1113            && entry.first_error.is_none()
1114        {
1115            entry.first_error = Some(err);
1116        }
1117        if entry.pending_wrs.is_empty() {
1118            let entry = self.posted.remove(&op_id).expect("just verified");
1119            let result = match entry.first_error {
1120                Some(err) => Err(err),
1121                None => Ok(()),
1122            };
1123            let _ = entry.reply.send(StripeResult {
1124                stripe_id: entry.stripe_id,
1125                result,
1126            });
1127        }
1128        Ok(())
1129    }
1130
1131    fn into_detached(self) -> DetachedQueuePair {
1132        DetachedQueuePair::new(DetachedQueuePairResources {
1133            _qp: self.qp,
1134            // If we dropped these now, the user would get an error immediately
1135            // even though the operations are still being processed and it isn't
1136            // safe for them to access the relevant memory yet.
1137            _posted: self.posted,
1138        })
1139    }
1140}

1142impl<Qp: IbvQueuePair> QueuePairWorker<Qp> {
1143    fn state(&self) -> &QueuePairState<Qp> {
1144        self.state
1145            .as_ref()
1146            .expect("the queue pair has not been detached")
1147    }
1148
1149    fn state_mut(&mut self) -> &mut QueuePairState<Qp> {
1150        self.state
1151            .as_mut()
1152            .expect("the queue pair has not been detached")
1153    }
1154
1155    async fn run(&mut self, cancellation: CancellationToken) -> Result<(), anyhow::Error> {
1156        if cancellation.is_cancelled() {
1157            return Ok(());
1158        }
1159        let qp_num = self.state_mut().qp.get_qp_info()?.qp_num;
1160        let cq = Arc::clone(
1161            self.cq_lease
1162                .as_ref()
1163                .expect("the queue pair worker starts with a CQ lease")
1164                .cq(),
1165        );
1166        let cq_id = cq.cq_id();
1167        let route = self
1168            .completion_route
1169            .take()
1170            .expect("the queue pair attaches its completion route once");
1171        let posted = Arc::clone(&self.state().posted_wrs);
1172        let (reply, installed) = tokio::sync::oneshot::channel();
1173        self.cq_poller.try_post(
1174            Instance::<()>::self_client(),
1175            Attach {
1176                cq,
1177                qp_num,
1178                route,
1179                posted,
1180                reply,
1181            },
1182        )?;
1183        self.detach_target = Some((cq_id, qp_num));
1184        let poller_wake = tokio::select! {
1185            biased;
1186
1187            _ = cancellation.cancelled() => return Ok(()),
1188            result = installed => result.map_err(|error| {
1189                anyhow::anyhow!("CQ poller did not attach queue pair: {error}")
1190            })?,
1191        };
1192        self.state_mut().poller_wake = Some(poller_wake);
1193
1194        loop {
1195            tokio::select! {
1196                biased;
1197
1198                _ = cancellation.cancelled() => return Ok(()),
1199                // A closed completion channel is only a QP failure while work
1200                // is outstanding. An idle QP need not observe CQ teardown.
1201                completion = self.completions.recv(), if self.state().in_flight > 0 => {
1202                    let completion = completion.ok_or_else(|| {
1203                        anyhow::anyhow!(
1204                            "completion queue poller stopped [qp_key={:?}]",
1205                            self.state().qp_key,
1206                        )
1207                    })?;
1208                    self.state_mut().complete_wr(completion)?;
1209                    self.state_mut().post_ready()?;
1210                }
1211                message = self.process_ops.recv() => {
1212                    let message = message.expect(
1213                        "the queue pair worker retains a submission sender while it runs",
1214                    );
1215                    self.state_mut().enqueue(message);
1216                    self.state_mut().post_ready()?;
1217                }
1218            }
1219        }
1220    }
1221}

1225        let Some((cq_id, qp_num)) = self.detach_target else {

1227        };
1228        let state = self
1229            .state
1230            .take()
1231            .expect("a queue pair with a detach target has state");

1253        }

1257async fn run_queue_pair_worker<Qp: IbvQueuePair>(
1258    mut worker: QueuePairWorker<Qp>,
1259    cancellation: CancellationToken,
1260    failed: PortHandle<QueuePairWorkerFailed>,
1261) {
1262    let result = AssertUnwindSafe(worker.run(cancellation))
1263        .catch_unwind()
1264        .await;
1265    let error = match result {
1266        Ok(Ok(())) => return,
1267        Ok(Err(error)) => error.to_string(),
1268        Err(_) => "queue pair worker panicked".to_owned(),
1269    };
1270    let error_for_log = error.clone();
1271    if let Err(post_error) = failed.try_post(
1272        Instance::<()>::self_client(),
1273        QueuePairWorkerFailed { error },
1274    ) {
1275        tracing::error!(
1276            error = %error_for_log,
1277            %post_error,
1278            "reporting queue pair worker failure failed",
1279        );
1280    }

1283#[async_trait]
1284impl<M: Manager, Qp: IbvQueuePair> Actor for QueuePairActor<M, Qp> {
1285    async fn init(&mut self, this: &Instance<Self>) -> Result<(), anyhow::Error> {
1286        this.set_system();
1287        let qp_key = self.worker_mut().state().qp_key.clone();
1288        let local_info = self
1289            .worker_mut()
1290            .state_mut()
1291            .qp
1292            .get_qp_info()
1293            .map_err(|e| {
1294                tracing::error!(?qp_key, error = %e, "QueuePairActor init: get_qp_info failed");
1295                anyhow::anyhow!("could not extract local QP info: {e}")
1296            })?;
1297
1298        let peer_info = if self.is_loopback {
1299            // The "peer" is ourselves; skip the round-trip and
1300            // connect to our own endpoint.
1301            local_info.clone()
1302        } else {
1303            let (reply, rx) = this.mailbox().open_once_port::<Result<IbvQpInfo, String>>();
1304            self.peer_manager.post(
1305                this,
1306                CreatePeerQueuePair {
1307                    sender: self.local_manager.clone(),
1308                    sender_device: qp_key.self_device.clone(),
1309                    receiver_device: qp_key.other_device.clone(),
1310                    qp_index: qp_key.qp_index,
1311                    sender_info: local_info,
1312                    reply: reply.bind(),
1313                },
1314            );
1315            match tokio::time::timeout(self.init_timeout, rx.recv()).await {
1316                Ok(Ok(Ok(info))) => info,
1317                Ok(Ok(Err(e))) => {
1318                    tracing::error!(
1319                        ?qp_key,
1320                        peer_manager = ?self.peer_manager,
1321                        error = %e,
1322                        "QueuePairActor init: peer manager rejected CreatePeerQueuePair",
1323                    );
1324                    return Err(anyhow::anyhow!("peer manager rejected QP request: {e}"));
1325                }
1326                Ok(Err(e)) => {
1327                    tracing::error!(
1328                        ?qp_key,
1329                        peer_manager = ?self.peer_manager,
1330                        error = %e,
1331                        "QueuePairActor init: peer reply port closed",
1332                    );
1333                    return Err(anyhow::anyhow!("peer reply port closed: {e}"));
1334                }
1335                Err(_) => {
1336                    tracing::error!(
1337                        ?qp_key,
1338                        peer_manager = ?self.peer_manager,
1339                        timeout = ?self.init_timeout,
1340                        "QueuePairActor init: timed out waiting for peer reply",
1341                    );
1342                    return Err(anyhow::anyhow!(
1343                        "QP initialization timed out after {:?}",
1344                        self.init_timeout
1345                    ));
1346                }
1347            }
1348        };
1349
1350        self.worker_mut()
1351            .state_mut()
1352            .qp
1353            .connect(&peer_info)
1354            .map_err(|e| {
1355                tracing::error!(
1356                    ?qp_key,
1357                    peer_info = ?peer_info,
1358                    error = %e,
1359                    "QueuePairActor init: connect failed",
1360                );
1361                anyhow::anyhow!("could not connect QP to peer: {e}")
1362            })?;
1363
1364        let worker = self
1365            .worker
1366            .take()
1367            .expect("the queue pair worker starts only once");
1368        let failed = this.handle().port::<QueuePairWorkerFailed>();
1369        self.task = Some(crate::rdma_runtime::spawn_on_rdma_runtime(
1370            run_queue_pair_worker(worker, self.cancellation.clone(), failed),
1371        ));
1372        Ok(())
1373    }
1374
1375    async fn cleanup(
1376        &mut self,
1377        _this: &Instance<Self>,
1378        _err: Option<&ActorError>,
1379    ) -> Result<(), anyhow::Error> {
1380        self.cancellation.cancel();
1381        let Some(task) = self.task.take() else {
1382            return Ok(());
1383        };
1384        task.await
1385            .map_err(|error| anyhow::anyhow!("joining queue pair worker failed: {error}"))
1386    }

1394        F::Output: Send + 'static,
1395    {
1396        crate::rdma_runtime::spawn_on_rdma_runtime(future)
1397    }
