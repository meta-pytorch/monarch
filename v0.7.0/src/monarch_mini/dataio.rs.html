9//! Data-plane I/O for message parts, above the wire [`crate::framing`]: reading and
10//! writing each part in whichever form fits — a small part inline on the control stream,
11//! a large part read/written straight into a shared-memory slab (or a heap buffer with no
12//! slab), and a *large* part optionally **striped** across several parallel data streams
13//! of the connection to fill a fat cross-region pipe. [`crate::framing`] frames the
14//! header and drives these per part; the shared-memory and striping policy — *when* to
15//! stripe, opening the data streams, and holding a message back until its shards land —
16//! lives here, exposed as two objects framing/the transport drive directly:
17//!
18//! - [`PartWriter`] — [`write_message_parts`](PartWriter::write_message_parts) writes a
19//!   message's parts, each inline on the control stream or striped across the data streams.
20//! - [`PartReader`] — [`read_message_parts`](PartReader::read_message_parts) reads them
21//!   back, each part returned before its bytes land and its shards tracked for completion.
22//! - [`read_messages`] — the receive loop: reads frames, and delivers assembled messages
23//!   to the command loop **in order** (a straggling large message holds back only what
24//!   genuinely follows it) while reading ahead.
25//!
26//! A single stream is one throughput-bound flow (crypto core-bound, and cross-region
27//! cwnd/RTT-bound to one flow's share of the link), so striping a large message across
28//! several data streams gives the parallel flows that fill a fat pipe, while small
29//! messages stay inline on the control stream. See `PARALLEL_DATA_STREAMS_DESIGN.md`.
30//!
31//! ## Fixed stream indices (direction by parity)
32//!
33//! The control stream is index 0 and the heartbeat stream index 1 (see
34//! [`crate::net_transport`]). Data streams take higher indices, with **direction
35//! encoded by parity** so the two directions' index spaces never overlap regardless of
36//! how many streams each opens: `td` (toward-dialer) at even indices `2, 4, …` (the
37//! acceptor writes, the dialer reads) and `ta` (toward-acceptor) at odd indices
38//! `3, 5, …` (the dialer writes, the acceptor reads). Shard `k` of a direction is at
39//! `base + 2k`; both ends name the same fixed index, so their halves pair with no
40//! negotiation. Data streams carry **unframed** shard bytes — the control-stream
41//! descriptor ([`crate::framing::WirePart::Striped`]) fully determines the split.
42//!
43//! ## Streams and tasks
44//!
45//! Each object holds a clone of the (`Send`) connection handle and opens the data
46//! streams it needs itself, lazily on the first large message — blocking the control
47//! stream for that one setup is fine (unlike the heartbeat stream, the control stream
48//! has no deadline). Each open spawns one per-stream I/O task ([`shard_reader_task`] /
49//! [`shard_writer_task`]) via `tokio::spawn`; the halves are `Send`, so this composes
50//! with the data runtime the control coroutines may run on. There are no other tasks —
51//! no per-connection coordinator — and nothing exists until a connection carries a
52//! large message, and only for the direction(s) that do.
53//!
54//! ## The unsafe: shards write straight into the destination
55//!
56//! A striped part is read **zero-copy** into its final buffer: the reader allocates the
57//! destination once (a shared-memory slab block, or a plain heap `Vec` when there is no
58//! slab) and hands each per-stream reader a raw `*mut u8` into its shard's sub-range.
59//! This is a deliberately localized `unsafe` (we do not add a bespoke [`MsgPart`]
60//! constructor for it): the destination buffer is owned by the assembled [`MsgPart`],
61//! which [`read_messages`]'s buffer holds until *every* shard has landed, so the pointers
62//! stay valid for the whole of the (disjoint) writes.

90/// Send priority of a data stream (quic; tcp ignores it): the same low priority as the

94/// Default size at/above which a message part is striped, when `MM_NET_LARGE_MSG_BYTES`
95/// is unset. 1 MiB — comfortably above the shared-memory inline threshold
96/// ([`crate::shm::SHM_THRESHOLD`]), so mid-size slab parts stay inline and only bulk
97/// payloads take the parallel path.
98const DEFAULT_LARGE_MSG_BYTES: u64 = 1 << 20;

100/// Number of data streams per striping direction (sender-side width). `0` (unset)
101/// disables striping entirely: every part stays inline and the transport behaves
102/// exactly as before. A *sender* knob — the receiver follows the wire descriptor — so
103/// the two ends need not agree.
104pub(crate) fn n_data_streams() -> usize {
105    std::env::var("MM_NET_N_DATA_STREAMS")
106        .ok()
107        .and_then(|v| v.parse::<usize>().ok())
108        .unwrap_or(0)

111/// Size at/above which a message part is striped (see [`DEFAULT_LARGE_MSG_BYTES`]).
112fn large_msg_bytes() -> u64 {
113    std::env::var("MM_NET_LARGE_MSG_BYTES")
114        .ok()
115        .and_then(|v| v.parse::<u64>().ok())
116        .filter(|&n| n > 0)
117        .unwrap_or(DEFAULT_LARGE_MSG_BYTES)
118}

120/// The first data-stream index for a task that opened its connection by **dialing**
121/// (`dialed`) vs **accepting**, and in the given direction — `write` (toward the peer)
122/// or `read` (from the peer). Direction is encoded by parity: `ta` (odd) flows
123/// dialer→acceptor, `td` (even) acceptor→dialer, and shard `k` is at `base + 2k`, so the
124/// two ends of one stream name the same index. A one-directional gateway side channel
125/// (dialer writes, acceptor reads) needs no special case: both land on `ta` (base 3).
126fn stream_base(dialed: bool, write: bool) -> usize {
127    // toward-acceptor `ta` (odd, 3) when dialer-writing or acceptor-reading; else `td`.
128    if dialed == write { 3 } else { 2 }
129}

131/// Split `len` bytes evenly across `n` shards; the first `len % n` shards get one extra
132/// byte. For a striped part (`len >= large_msg_bytes`, `n` small) every shard is
133/// comfortably non-empty.
134fn even_split(len: u64, n: usize) -> Vec<u64> {
135    let n = n as u64;
136    let base = len / n;
137    let rem = len % n;
138    (0..n).map(|i| base + u64::from(i < rem)).collect()
139}

141/// A raw pointer to a shard's destination sub-range, made `Send` so it can ride to a
142/// per-stream reader task. Sound because the owning [`MsgPart`] keeps the buffer alive
143/// past every write and each pointer names a disjoint range (see the module docs).
144struct ShardDst(*mut u8);
145// SAFETY: see the module docs — the destination buffer outlives the writes and the
146// ranges are disjoint, so moving the raw pointer to the reader task is sound.
147unsafe impl Send for ShardDst {}

149/// One shard-read instruction to a per-stream reader: fill `[dst, dst+len)` from the
150/// stream, then acknowledge success or failure to the owning message's [`Batch`].
151struct ShardRead {
152    dst: ShardDst,
153    len: usize,
154    done: oneshot::Sender<io::Result<()>>,
155}

157/// A batch of in-flight shard reads and their completion acknowledgements. A
158/// [`PartReader`] accumulates a message's striped-part acknowledgements into the current
159/// batch; [`PartReader::sync`] then awaits all of them and rotates in a fresh batch.
160/// Waiting never stops at the first error because sibling readers may still be writing
161/// through raw pointers into the message's destination.
162struct Batch {
163    done: Vec<oneshot::Receiver<io::Result<()>>>,
164}
165
166impl Batch {
167    fn new() -> Self {
168        Self { done: Vec::new() }
169    }
170}

172/// Await every dispatched shard, retaining the first failure until all sibling reads
173/// have either completed or discarded their queued instruction.
174async fn wait_for_shards(done: Vec<oneshot::Receiver<io::Result<()>>>) -> io::Result<()> {
175    let mut first_error = None;
176    for completion in done {
177        let result = completion
178            .await
179            .unwrap_or_else(|_| Err(io::Error::other("data stream closed")));
180        match result {
181            Err(error) if first_error.is_none() => first_error = Some(error),
182            _ => {}
183        }
184    }
185    first_error.map_or(Ok(()), Err)
186}

188/// Keep a destination owner alive while already-dispatched shards finish after a
189/// higher-level operation has failed. The original operation's error takes precedence,
190/// so callers intentionally ignore the shard result after the lifetime barrier.
191async fn wait_for_shards_before_drop<T>(
192    keep_alive: T,
193    done: Vec<oneshot::Receiver<io::Result<()>>>,
194) -> io::Result<()> {
195    let result = wait_for_shards(done).await;
196    drop(keep_alive);
197    result
198}

200/// One shard-write job to a per-stream writer: write `part`'s `[offset, offset+len)`
201/// onto the stream. The whole part is shared (`Arc`) across the streams that carry its
202/// shards; no bytes are copied.
203struct ShardWrite {
204    part: Arc<MsgPart>,
205    offset: usize,
206    len: usize,
207}
208
209/// Read shards off one data stream in order, each straight into its instruction's
210/// destination sub-range, acknowledging success or failure as each finishes. A read
211/// error ends the task; dropping queued instructions closes their acknowledgement
212/// channels so their message waiters also observe failure.
213async fn shard_reader_task<N: Net>(
214    mut recv: ConnRecv<N>,
215    mut instrs: mpsc::UnboundedReceiver<ShardRead>,
216) {
217    while let Some(instr) = instrs.recv().await {
218        // SAFETY: `dst` points at `len` bytes of the destination buffer owned by the
219        // still-undelivered `MsgPart` (held in the delivery gate until every shard
220        // lands); this task is the sole writer of this disjoint sub-range. See the
221        // module docs on the localized unsafe.
222        let buf = unsafe { std::slice::from_raw_parts_mut(instr.dst.0, instr.len) };
223        let result = recv.read_exact(buf).await.map(|_| ());
224        let failed = result.is_err();
225        let _ = instr.done.send(result);
226        if failed {
227            return;
228        }
229    }
230}

232/// Write shards onto one data stream in the order the message writer enqueued them
233/// (per-stream FIFO by message). Ends when the striper drops its sender or a write
234/// fails (finishing the stream so the peer's reader sees EOF and severs).
235async fn shard_writer_task<N: Net>(
236    mut send: ConnSend<N>,
237    mut jobs: mpsc::UnboundedReceiver<ShardWrite>,
238) {
239    while let Some(job) = jobs.recv().await {
240        let bytes = &job.part.as_bytes()[job.offset..job.offset + job.len];
241        if send.write_all(bytes).await.is_err() || send.flush().await.is_err() {
242            return;
243        }
244    }
245    let _ = send.shutdown().await;
246}

248/// Writes a message's parts to the control stream (inline) or across the connection's
249/// data streams (striped), the send-side mirror of [`PartReader`]. It decides the split
250/// ([`Self::plan`], for the header framing writes) and writes each part
251/// ([`Self::write`]) — a small part inline, a large part striped across the data streams
252/// (opened lazily off the connection handle). When striping is disabled (`n == 0`)
253/// [`Self::plan`] marks every part inline and the wire is byte-for-byte the non-striped
254/// path. Striped writes are fire-and-forget (per-stream FIFO handles ordering), so —
255/// unlike [`PartReader`] — there is no completion to await.
256pub(crate) struct PartWriter<N: Net> {
257    conn: N::Conn,
258    base: usize,
259    n: usize,
260    threshold: u64,
261    writers: Vec<mpsc::UnboundedSender<ShardWrite>>,
262}
263
264impl<N: Net> PartWriter<N> {
265    /// A part writer over this side's outbound data streams. `dialed` (did we open this
266    /// connection by dialing?) picks the write direction's stream indices.
267    pub(crate) fn new(conn: N::Conn, dialed: bool) -> Self {
268        Self {
269            conn,
270            base: stream_base(dialed, true),
271            n: n_data_streams(),
272            threshold: large_msg_bytes(),
273            writers: Vec::new(),
274        }
275    }
276
277    /// Write a message's parts onto `control` after its routing header (written by
278    /// [`crate::framing`]): the per-part plan as a header, then each part — inline bytes
279    /// onto `control`, or striped across the data streams — and flush. The whole
280    /// message-part write path for both pathways (they differ only in the routing header
281    /// framing writes first). Striped shards are enqueued (they flow asynchronously)
282    /// before the flush; the peer reads the header, opens its data streams, and pairs
283    /// them, so nothing blocks here.
284    pub(crate) async fn write_message_parts<W: tokio::io::AsyncWrite + Unpin>(
285        &mut self,
286        control: &mut W,
287        parts: Vec<MsgPart>,
288    ) -> io::Result<()> {
289        let plan: Vec<WirePart> = parts
290            .iter()
291            .map(|part| self.plan(part.len() as u64))
292            .collect();
293        crate::framing::write_header(control, &plan).await?;
294        for (part, wire_part) in parts.into_iter().zip(&plan) {
295            self.write(control, part, wire_part).await?;
296        }
297        control.flush().await
298    }
299
300    /// The wire descriptor for a part of `len` bytes: inline, or striped with its
301    /// per-shard byte lengths. The plan goes in the header before any part bytes.
302    fn plan(&self, len: u64) -> WirePart {
303        if self.n > 0 && len >= self.threshold {
304            WirePart::Striped {
305                shard_lens: even_split(len, self.n),
306            }
307        } else {
308            WirePart::Inline { len }
309        }
310    }
311
312    /// Write one part per its `wire_part` descriptor: an inline part's bytes stream onto
313    /// `control`; a striped part's shards are handed to the data streams' per-stream
314    /// writers (opened lazily). Striped shards go out asynchronously, so this returns
315    /// once they are enqueued.
316    async fn write<W: tokio::io::AsyncWrite + Unpin>(
317        &mut self,
318        control: &mut W,
319        part: MsgPart,
320        wire_part: &WirePart,
321    ) -> io::Result<()> {
322        let WirePart::Striped { shard_lens } = wire_part else {
323            control.write_all(part.as_bytes()).await?;
324            return Ok(());
325        };
326        self.ensure(shard_lens.len()).await?;
327        let part = Arc::new(part);
328        let mut offset = 0usize;
329        for (i, &shard_len) in shard_lens.iter().enumerate() {
330            let len = shard_len as usize;
331            self.writers[i]
332                .send(ShardWrite {
333                    part: Arc::clone(&part),
334                    offset,
335                    len,
336                })
337                .map_err(|_| io::Error::other("data stream writer closed"))?;
338            offset += len;
339        }
340        Ok(())
341    }
342
343    /// Ensure `n` per-stream writers are open, opening stream `base + 2i` for each new
344    /// index and spawning its [`shard_writer_task`].
345    async fn ensure(&mut self, n: usize) -> io::Result<()> {
346        while self.writers.len() < n {
347            let index = self.base + 2 * self.writers.len();
348            let (send, _recv) = self.conn.stream(index, PRIORITY_DATA).await?;
349            let (tx, rx) = mpsc::unbounded_channel();
350            tokio::spawn(shard_writer_task::<N>(send, rx));
351            self.writers.push(tx);
352        }
353        Ok(())
354    }
355}

357/// Reads a message's parts off the control stream, returning each [`MsgPart`] before its
358/// bytes have landed. A small part is read inline (into the slab, or a heap buffer when
359/// there is no slab); a large part is striped across the connection's data streams
360/// (opened lazily off the handle), its shards read straight into the destination. It
361/// carries the shared-memory context and accumulates the in-flight shards into the
362/// current [`Batch`]; [`Self::sync`] hands back a future for that batch's completion and
363/// starts a fresh one, so the reader can deliver each message once its shards land.
364pub(crate) struct PartReader<N: Net> {
365    conn: N::Conn,
366    base: usize,
367    mapper: MapperHandle,
368    client: ShmClientSlot,
369    readers: Vec<mpsc::UnboundedSender<ShardRead>>,
370    batch: Batch,
371}

373impl<N: Net> PartReader<N> {
374    /// A part reader over this side's inbound data streams, landing striped parts in the
375    /// slab via `mapper`/`client`. `dialed` (did we open this connection by dialing?)
376    /// picks the read direction's stream indices.
377    pub(crate) fn new(
378        conn: N::Conn,
379        dialed: bool,
380        mapper: MapperHandle,
381        client: ShmClientSlot,
382    ) -> Self {
383        Self {
384            conn,
385            base: stream_base(dialed, false),
386            mapper,
387            client,
388            readers: Vec::new(),
389            batch: Batch::new(),
390        }
391    }
392
393    /// Read a message's parts off `control` after its routing header (read by
394    /// [`crate::framing`]): the per-part plan header, then each part — inline off
395    /// `control`, or striped from the data streams. The whole message-part read path for
396    /// both pathways. Striped parts come back before their bytes land, accumulated into
397    /// the current batch; the reader loop calls [`Self::sync`] afterward to gate delivery.
398    pub(crate) async fn read_message_parts<R: AsyncRead + Unpin>(
399        &mut self,
400        control: &mut R,
401    ) -> io::Result<Vec<MsgPart>> {
402        let plan: Vec<WirePart> = crate::framing::read_header(control).await?;
403        let mut out = Vec::with_capacity(plan.len());
404        for wire_part in plan {
405            match self.read(control, wire_part).await {
406                Ok(part) => out.push(part),
407                Err(error) => {
408                    let Batch { done } = std::mem::replace(&mut self.batch, Batch::new());
409                    let _ = wait_for_shards_before_drop(out, done).await;
410                    return Err(error);
411                }
412            }
413        }
414        Ok(out)
415    }
416
417    /// Read one part off `control`, per its wire descriptor. Returns before a striped
418    /// part's bytes land — they fill from the data streams into the returned part's
419    /// destination, tracked by the current batch (see [`Self::sync`]).
420    async fn read<R: AsyncRead + Unpin>(
421        &mut self,
422        control: &mut R,
423        part: WirePart,
424    ) -> io::Result<MsgPart> {
425        match part {
426            WirePart::Inline { len } => {
427                read_inline(control, len, self.mapper.clone(), self.client()).await
428            }
429            WirePart::Striped { shard_lens } => self.read_striped(&shard_lens).await,
430        }
431    }
432
433    /// A future that resolves once every shard read since the last `sync` has landed
434    /// (`Ok`) or a data stream died (`Err`), rotating in a fresh batch for the next
435    /// message's parts — or `None` for a message with no striped parts (the common case),
436    /// which is already complete and needs no waiting or allocation. Only a genuinely
437    /// striped message produces (and boxes) a future.
438    fn sync(&mut self) -> Option<Done> {
439        if self.batch.done.is_empty() {
440            return None;
441        }
442        let Batch { done } = std::mem::replace(&mut self.batch, Batch::new());
443        Some(Box::pin(wait_for_shards(done)))
444    }
445
446    /// Allocate the destination for a striped part and hand each shard's sub-range to its
447    /// per-stream reader, adding this part's shards to the current batch. Returns the
448    /// [`MsgPart`] (its bytes fill in as the shards land).
449    async fn read_striped(&mut self, shard_lens: &[u64]) -> io::Result<MsgPart> {
450        self.ensure(shard_lens.len()).await?;
451        let total: u64 = shard_lens.iter().sum();
452        let (part, base) = alloc_dst(self.mapper.clone(), self.client(), total).await?;
453
454        let mut done = Vec::with_capacity(shard_lens.len());
455        let mut offset = 0usize;
456        for (i, &shard_len) in shard_lens.iter().enumerate() {
457            let len = shard_len as usize;
458            // SAFETY: `base` is a `total`-byte destination and `offset + len <= total`,
459            // so `base + offset` starts a distinct in-bounds `len`-byte range.
460            let dst = ShardDst(unsafe { base.add(offset) });
461            let (done_tx, done_rx) = oneshot::channel();
462            if self.readers[i]
463                .send(ShardRead {
464                    dst,
465                    len,
466                    done: done_tx,
467                })
468                .is_err()
469            {
470                let error = io::Error::other("data stream reader closed");
471                let _ = wait_for_shards_before_drop(part, done).await;
472                return Err(error);
473            }
474            done.push(done_rx);
475            offset += len;
476        }
477        self.batch.done.extend(done);
478        Ok(part)
479    }
480
481    /// Snapshot the owning actor's gateway client (`None` until learned).
482    fn client(&self) -> Option<ShmClient> {
483        *self.client.lock().expect("shm client slot mutex poisoned")
484    }
485
486    /// Ensure `n` per-stream readers are open, opening stream `base + 2i` for each new
487    /// index and spawning its [`shard_reader_task`].
488    async fn ensure(&mut self, n: usize) -> io::Result<()> {
489        while self.readers.len() < n {
490            let index = self.base + 2 * self.readers.len();
491            let (_send, recv) = self.conn.stream(index, PRIORITY_DATA).await?;
492            let (tx, rx) = mpsc::unbounded_channel();
493            tokio::spawn(shard_reader_task::<N>(recv, rx));
494            self.readers.push(tx);
495        }
496        Ok(())
497    }
498}

500/// Why [`read_messages`] stopped.

502    /// The control stream ended or errored (the caller severs, folding in `io::Error`).

504    /// A data stream died mid-message, so a striped message can no longer complete and
505    /// order can't skip it (the caller severs).
506    DataStreamDied,
507    /// `on_message` returned `false` — its command sink is gone; stop quietly.
508    Delivered,
509}

511/// A striped message's shard-completion future ([`PartReader::sync`]), boxed so the
512/// buffer can hold it: `Ok` once every shard landed, `Err` if a data stream died. The
513/// buffer pairs each message with an `Option<Done>` — `None` for the common message with
514/// no striped parts (already complete), so only a striped one ever allocates a future.
515type Done = Pin<Box<dyn Future<Output = io::Result<()>> + Send>>;

517/// Reads and delivers a pathway's messages, driven by [`Self::read_messages`]. Each
518/// pathway (a join/serve connection, a gateway side channel) is one implementation,
519/// supplying how to decode a control frame ([`Self::read_frame`]) and where a ready
520/// message goes ([`Self::on_message`]); the loop and the striping/ordering are shared.
521///
522/// Making the pathway an implementation (rather than a generic closure) keeps
523/// [`Self::read_frame`]'s future *concrete per impl*, so its `Send`-ness — needed to run
524/// the loop on the data runtime — is checked structurally instead of as a higher-ranked
525/// bound the compiler can't yet discharge; that is why this is a trait and not two
526/// closures.
527pub(crate) trait MessageReader<N: Net>: Send {
528    /// The delivered message type (`ConnectionCommand` / `SideChannelMessage`).
529    type Item: Send;
530
531    /// Decode the next control frame off `control`, reading its parts through `parts`.
532    /// Not cancel-safe (a frame is several reads), so [`Self::read_messages`] drives one
533    /// to completion before starting the next.
534    fn read_frame(
535        control: &mut ConnRecv<N>,
536        parts: &mut PartReader<N>,
537    ) -> impl Future<Output = io::Result<Self::Item>> + Send;
538
539    /// Deliver one in-order message. Returns `false` if its sink is gone (stop the loop).
540    fn on_message(&mut self, item: Self::Item) -> bool;
541
542    /// The whole receive loop. Read a frame, then hand it to [`Self::on_message`] **in
543    /// order** — but read ahead while a large message's shards land, delivering a message
544    /// only once its batch ([`PartReader::sync`]) completes, so a straggler only delays
545    /// what genuinely follows it. A message with no striped parts completes at once, so
546    /// with nothing striped nothing is ever held.
547    fn read_messages(
548        &mut self,
549        mut control: ConnRecv<N>,
550        mut parts: PartReader<N>,
551    ) -> impl Future<Output = ReadStop> + Send {
552        async move {
553            // In-order, each with its shards' completion future — `None` when nothing was
554            // striped. Empty in the steady state; only a striped straggler and whatever
555            // arrives behind it ever land here.
556            let mut buffer: VecDeque<(Self::Item, Option<Done>)> = VecDeque::new();
557            loop {
558                // The latency path: nothing buffered ⇒ nothing to wait on, so just read
559                // the next frame directly — no completion race, no buffer round trip.
560                let frame = if buffer.is_empty() {
561                    Self::read_frame(&mut control, &mut parts).await
562                } else {
563                    // Something striped is still landing: read ahead while delivering any
564                    // already-complete leading messages. The read future is kept
565                    // un-cancelled (a frame is several reads) until it yields.
566                    let read = Self::read_frame(&mut control, &mut parts);
567                    tokio::pin!(read);
568                    loop {
569                        tokio::select! {
570                            frame = &mut read => break frame,
571                            done = await_front(&mut buffer) => match done {
572                                Err(_) => {
573                                    buffer.pop_front().expect("front just completed");
574                                    drain_buffer(&mut buffer).await;
575                                    return ReadStop::DataStreamDied;
576                                }
577                                Ok(()) => {
578                                    let (item, _) =
579                                        buffer.pop_front().expect("front just completed");
580                                    if !self.on_message(item) {
581                                        drain_buffer(&mut buffer).await;
582                                        return ReadStop::Delivered;
583                                    }
584                                }
585                            },
586                        }
587                    }
588                };
589                let item = match frame {
590                    Ok(item) => item,
591                    Err(err) => {
592                        drain_buffer(&mut buffer).await;
593                        return ReadStop::Ended(err);
594                    }
595                };
596                match parts.sync() {
597                    // The latency path continues: no shards to wait on and nothing ahead
598                    // of it — deliver straight through, never touching the buffer.
599                    None if buffer.is_empty() => {
600                        if !self.on_message(item) {
601                            return ReadStop::Delivered;
602                        }
603                    }
604                    // Striped (must wait), or something ahead of it is still landing
605                    // (deliver in order): buffer it behind the rest.
606                    done => buffer.push_back((item, done)),
607                }
608            }
609        }
610    }
611}
612
613/// Await the front buffered message's shard completion — pending forever when the buffer
614/// is empty, so the reader only reads. `Ok` ⇒ its shards all landed, or it had none
615/// (deliver it); `Err` ⇒ a data stream died.
616async fn await_front<Item>(buffer: &mut VecDeque<(Item, Option<Done>)>) -> io::Result<()> {
617    match buffer.front_mut() {
618        None => std::future::pending().await,
619        Some((_, None)) => Ok(()),
620        Some((_, Some(done))) => done.as_mut().await,
621    }
622}
623
624/// Wait out every buffered message's shard sync before dropping its destination parts.
625/// Errors are irrelevant once the reader has already decided to stop, but each `Done`
626/// must reach completion because its shard tasks may still hold raw destination pointers.
627async fn drain_buffer<Item>(buffer: &mut VecDeque<(Item, Option<Done>)>) {
628    while let Some((_, done)) = buffer.front_mut() {
629        if let Some(done) = done {
630            let _ = done.as_mut().await;
631        }
632        buffer.pop_front();
633    }
634}

636/// Read an inline part of `len` bytes off `control` into its destination. A part `>=
637/// SHM_THRESHOLD` whose actor has learned its gateway [`ShmClient`] lands straight in a
638/// freshly allocated slab block (the only copy is the unavoidable kernel-to-userspace
639/// one, and a later unix hop forwards the descriptor without recopying); everything else
640/// lands in an owned heap buffer. A free function (not a `&self` method) so the read
641/// future stays `Send` without requiring the connection handle to be `Sync`.
642async fn read_inline<R: AsyncRead + Unpin>(
643    control: &mut R,
644    len: u64,
645    mapper: MapperHandle,
646    client: Option<ShmClient>,
647) -> io::Result<MsgPart> {
648    // The slab only pays off for a large part, so filter the client out below the
649    // threshold; `alloc_dst` then gives a heap buffer for the small/no-slab case.
650    let (part, dst) = alloc_dst(mapper, client.filter(|_| len >= SHM_THRESHOLD), len).await?;
651    // SAFETY: `dst` is `len` writable bytes of the destination buffer owned by `part`
652    // (slab mapping or heap `Vec`), valid past this read; this is the sole writer.
653    let buf = unsafe { std::slice::from_raw_parts_mut(dst, len as usize) };
654    control.read_exact(buf).await?;
655    Ok(part)
656}
657
658/// Allocate a `total`-byte destination for a message part and return it plus a raw
659/// pointer to its start. With a gateway [`ShmClient`] it is a slab block (read
660/// zero-copy, ready to relay across a later unix hop by descriptor); without one it is
661/// a plain heap buffer. The pointer stays valid for the returned [`MsgPart`]'s lifetime
662/// (the slab mapping is context-lived; the heap `Vec`'s buffer never moves once boxed).
663async fn alloc_dst(
664    mapper: MapperHandle,
665    client: Option<ShmClient>,
666    total: u64,
667) -> io::Result<(MsgPart, *mut u8)> {
668    let Some(client) = client else {
669        // No slab: assemble into a plain heap buffer. Take the raw pointer from the
670        // freshly-allocated `Vec` before handing it to `MsgPart::from_bytes` — moving
671        // the `Vec` into the part does not move its heap allocation, so the pointer
672        // stays valid, and this side owns the part exclusively until delivery.
673        let mut buf: Vec<u8> = Vec::with_capacity(total as usize);
674        // SAFETY: the allocation above reserves capacity for `total` bytes, so growing
675        // the length to `total` names only those bytes. The buffer is exposed solely as
676        // a write destination (a raw `*mut u8`); every byte is filled by the
677        // `read_exact`/shard reads that follow before any is read, so its uninitialized
678        // contents are never observed. Skipping the zero-fill avoids wasted work.
679        // (`clippy::uninit_vec` flags this pattern generically; the read-before-use
680        // invariant above is exactly what makes it sound here.)
681        #[allow(clippy::uninit_vec)]
682        unsafe {
683            buf.set_len(total as usize)
684        };
685        let ptr = buf.as_mut_ptr();
686        return Ok((MsgPart::from_bytes(buf), ptr));
687    };
688    let (offset, token) = client.allocate(total).await?;
689    let ptr = {
690        let mut guard = mapper.lock().expect("shm mapper mutex poisoned");
691        // SAFETY: `offset` was just granted for `total` bytes against `client`'s slab,
692        // so the file covers it and the mapper can map the range.
693        unsafe { guard.map(client.slab_fd(), offset, total as usize)? }
694    };
695    Ok((
696        MsgPart::new_shm(mapper, client.slab_fd(), token, offset, total),
697        ptr,
698    ))
699}
