9//! QUIC implementation of the [`Net`] transport seam.
10//!
11//! This module supplies only the quic-specific networking: TLS material, the client
12//! endpoint pool, binding a server endpoint, dialing, and a [`QuicConn`] that opens
13//! and accepts bidirectional streams. Everything protocol-independent — establishment,
14//! heartbeats, matching serves to joins, side channels, retry/backoff, shutdown —
15//! lives in [`crate::net_transport`], generic over [`Net`]. The command loop drives
16//! quic as `NetTransport<Quic>` (aliased `QuicTransport` in [`crate::ctx`]).
17//!
18//! ## Two streams per connection
19//!
20//! Each connection carries two bidirectional streams (see `net_transport`): a
21//! data/control stream and a companion heartbeat stream. QUIC's streams are
22//! independently ordered and flow-controlled and interleave at the packet level, so a
23//! large message on the data stream cannot head-of-line-block a beat. The heartbeat
24//! stream is given a higher send priority (its stream index) so beats are packed into
25//! packets ahead of data even under a full congestion window. (Flow-control *windows*
26//! are left at their defaults: they are credit ceilings, not pre-allocated buffers,
27//! and beats are tiny.)
28//!
29//! ## Why QUIC needs heartbeats
30//!
31//! QUIC runs over UDP in userspace, so there is no file-descriptor close to signal a
32//! lost peer: a crashed, frozen, or partitioned peer simply stops sending. So instead
33//! of relying on EOF the transport runs an application-level bidirectional heartbeat
34//! on the heartbeat stream (see [`crate::net_transport`] and
35//! [`crate::heartbeat`]).
36//!
37//! ## Security
38//!
39//! TLS material is taken from the environment (the "we will provide it" hook):
40//! `MM_QUIC_CERT` / `MM_QUIC_KEY` (the cert chain + key this endpoint presents) and
41//! `MM_QUIC_CA` (the authority it trusts). The client verifies the server certificate
42//! against the CA for the fixed name [`SERVER_NAME`], while the server verifies the
43//! client's `clientAuth` certificate against the CA.

77/// Per-stream flow-control receive window (`MAX_STREAM_DATA`): how many bytes a peer
78/// may have in flight on one stream before it must wait for the receiver to extend
79/// the window. quinn's default is ~1.25 MB (sized for a 100 ms RTT), which caps a
80/// large message's throughput at window/RTT — a severe throttle on a fast, low-RTT
81/// link. We raise it so a big message can keep the pipe full.
82///
83/// This is a *ceiling* the receiver may buffer up to on a stream actively receiving
84/// unread data, not a preallocation: idle connections cost nothing. It does raise the
85/// worst-case memory a busy connection can hold, which matters for a root with very
86/// many connections all mid-large-transfer — so it is env-tunable
87/// (`MM_QUIC_STREAM_RECV_WINDOW_BYTES`). Only the data stream ever fills it; the
88/// companion heartbeat stream carries tiny frames. The connection-level
89/// `receive_window` is raised in lockstep (to `window * 8`, matching the send window)
90/// in [`load_tls`] so the connection does not re-throttle its streams below this
91/// per-stream window.
92const DEFAULT_STREAM_RECV_WINDOW_BYTES: u64 = 16 * 1024 * 1024;
93
94fn stream_recv_window_bytes() -> u64 {
95    std::env::var("MM_QUIC_STREAM_RECV_WINDOW_BYTES")
96        .ok()
97        .and_then(|v| v.parse::<u64>().ok())
98        .filter(|&n| n > 0)
99        .unwrap_or(DEFAULT_STREAM_RECV_WINDOW_BYTES)
100}

109fn load_tls() -> anyhow::Result<TlsConfig> {
110    let tls::Config {
111        mut server,
112        mut client,
113    } = tls::Config::load()?;
114
115    // Disable all periodic QUIC traffic so a delegated link is truly silent:
116    // `keep_alive_interval = None` (no PING keep-alives) and `max_idle_timeout = None`
117    // (QUIC must not reap an idle delegated connection — liveness is now the heartbeat
118    // subsystem's responsibility, not the transport's). Applied to both roles.
119    // Message-carrying links are unaffected; delegated links rely on the sibling
120    // fabric. See `HEARTBEAT_DELEGATION_DESIGN.md` §9.
121    let mut transport = quinn::TransportConfig::default();
122    transport.keep_alive_interval(None);
123    transport.max_idle_timeout(None);
124    // Raise the per-stream flow-control window (and the send window with it, keeping
125    // quinn's 8x send:stream ratio) so a large message is not throttled to window/RTT.
126    // See `stream_recv_window_bytes`.
127    let window = stream_recv_window_bytes();
128    transport.stream_receive_window(VarInt::from_u64(window).unwrap_or(VarInt::MAX));
129    transport.send_window(window.saturating_mul(8));
130    // Raise the connection-level receive window in lockstep with the send window so
131    // the aggregate of a connection's streams can fill the bandwidth-delay product on
132    // a high-RTT cross-region path, rather than being re-throttled below the
133    // per-stream window. Sized to match `send_window` (window * 8).
134    let conn_window = window.saturating_mul(8);
135    transport.receive_window(VarInt::from_u64(conn_window).unwrap_or(VarInt::MAX));
136    // Congestion control: quinn defaults to CUBIC, which on a high-RTT cross-region
137    // path with sporadic loss ramps slowly and backs off hard (quinn#2262: CUBIC
138    // stalls at ~1 MiB cwnd where iperf/BBR reach the BDP). Pin BBR unconditionally —
139    // it is delay-model-based and holds a higher steady rate across light loss, and on
140    // loopback / the lossless intra-cluster fabric it is ~neutral. quinn re-exports
141    // BbrConfig at quinn::congestion.
142    transport.congestion_controller_factory(Arc::new(quinn::congestion::BbrConfig::default()));
143    // Raise the MTU-discovery ceiling so QUIC can use the jumbo headroom of the
144    // intra-cluster fabric. On GPU hosts the frontend `eth0` carries a deliberate
145    // jumbo MTU of 5000 (chef sets `JUMBO_MTU=5000` for GPU_TC hosts), but quinn's
146    // default MTU discovery caps the UDP payload at 1452, so it never uses that
147    // headroom — leaving ~3x more datagrams and their per-packet kernel cost
148    // (`sendmsg` on TX, skb alloc/free on RX) on the table at the single-host receive
149    // ceiling. We only raise the *ceiling* (from `mtu_search_upper_bound`) and let
150    // quinn's DPLPMTUD *discover* the real path MTU — we never pin `initial_mtu` or
151    // disable discovery — so an over-large ceiling on a smaller path simply settles
152    // lower (a lost probe leaves the MTU where it was, it never stalls data or
153    // black-holes the handshake). The companion `endpoint_config()` advertises the
154    // same `max_udp_payload_size` on both roles; without that the peer would clamp our
155    // probes back to its 1472 default (quinn-proto `mtud`: `current_mtu.min(peer_max)`).
156    let mut mtud = quinn::MtuDiscoveryConfig::default();
157    mtud.upper_bound(mtu_search_upper_bound());
158    transport.mtu_discovery_config(Some(mtud));
159    let transport = Arc::new(transport);
160
161    server.max_early_data_size = u32::MAX;
162    let mut server = ServerConfig::with_crypto(Arc::new(QuicServerConfig::try_from(server)?));
163    server.transport_config(transport.clone());
164
165    client.enable_early_data = true;
166    let mut client = ClientConfig::new(Arc::new(QuicClientConfig::try_from(client)?));
167    client.transport_config(transport);
168
169    Ok(TlsConfig { server, client })
170}

172/// IP + UDP header overhead subtracted from a link MTU to get the largest QUIC UDP
173/// payload that fits. Cross-machine QUIC is IPv6 (40 B header) + UDP (8 B) = 48 B;
174/// using the IPv6 figure is the conservative choice (IPv4's 28 B is smaller, so this
175/// never over-estimates the payload).
176const IP_UDP_OVERHEAD: u16 = 48;
177
178/// Fallback UDP-payload discovery ceiling when the local link MTU can't be read (see
179/// [`eth0_udp_payload`]). Larger than quinn's 1452 default so discovery can still
180/// climb on a jumbo fabric; because it is only a *ceiling* for DPLPMTUD probing (never
181/// pinned), an over-large value on a smaller path just settles lower.
182const DEFAULT_MTU_SEARCH_UPPER_BOUND: u16 = 4800;
183
184/// The largest QUIC UDP payload that fits `eth0`'s link MTU (`mtu − IP_UDP_OVERHEAD`),
185/// or `None` if the sysfs file is absent/unparseable. `eth0` is the frontend fabric
186/// that on GPU hosts carries a jumbo MTU (chef sets `JUMBO_MTU=5000` for GPU_TC
187/// hosts). Deliberately reads only this one file — no interface scanning and no active
188/// probing.
189fn eth0_udp_payload() -> Option<u16> {
190    let mtu: u16 = std::fs::read_to_string("/sys/class/net/eth0/mtu")
191        .ok()?
192        .trim()
193        .parse()
194        .ok()?;
195    mtu.checked_sub(IP_UDP_OVERHEAD).filter(|&p| p >= 1200)
196}

198/// The UDP-payload ceiling for QUIC MTU discovery, read in one place so the
199/// [`quinn::TransportConfig`] (the DPLPMTUD `upper_bound`) and the
200/// [`quinn::EndpointConfig`] (the advertised `max_udp_payload_size`, which caps what
201/// the peer probes toward us) stay consistent. It is `MM_QUIC_MTU` if set (≥1200; an
202/// operator override, e.g. to reduce the ceiling), else `eth0`'s link MTU (see
203/// [`eth0_udp_payload`]), else [`DEFAULT_MTU_SEARCH_UPPER_BOUND`]. This is only a
204/// ceiling — quinn always *discovers* the real path MTU up to it (see [`load_tls`]),
205/// so a too-large value never breaks a smaller path.
206fn mtu_search_upper_bound() -> u16 {
207    std::env::var("MM_QUIC_MTU")
208        .ok()
209        .and_then(|m| m.parse::<u16>().ok())
210        .filter(|&m| m >= 1200)
211        .or_else(eth0_udp_payload)
212        .unwrap_or(DEFAULT_MTU_SEARCH_UPPER_BOUND)
213}

215/// The [`quinn::EndpointConfig`] for every client/server socket in this context.
216/// quinn's default advertises `max_udp_payload_size = 1472` as this endpoint's receive
217/// limit, and QUIC clamps the *peer's* probed MTU down to it (quinn-proto
218/// `mtud::on_peer_max_udp_payload_size_received`: `current_mtu.min(peer_max)`), so
219/// raising the discovery ceiling on the sender alone is a no-op — the receiver must
220/// advertise the larger size too. We set it to [`mtu_search_upper_bound`] on both
221/// roles; this value also sizes quinn's GRO recv buffer (`max_udp_payload_size ·
222/// gro_segments · BATCH_SIZE`).
223fn endpoint_config() -> quinn::EndpointConfig {
224    let mut cfg = quinn::EndpointConfig::default();
225    // Only errors if the value is < 1200, already excluded by `mtu_search_upper_bound`.
226    let _ = cfg.max_udp_payload_size(mtu_search_upper_bound());
227    cfg
228}

230/// Parse a `quic://host:port` (or bare `host:port`) url into a socket address.
231fn parse_addr(url: &str) -> anyhow::Result<SocketAddr> {
232    let authority = url.strip_prefix("quic://").unwrap_or(url);
233    authority
234        .parse::<SocketAddr>()
235        .map_err(|err| anyhow::anyhow!("invalid quic address {authority:?}: {err}"))
236}

238/// The wildcard client bind address matching `target`'s address family. A QUIC client
239/// endpoint binds a local UDP socket before dialing, and that socket's family must
240/// match the destination: an IPv4-bound socket cannot reach an IPv6 peer (and vice
241/// versa). Dialing across machines is normally over IPv6, so this must pick `[::]:0`
242/// for an IPv6 target rather than always binding `0.0.0.0:0`.
243fn client_bind_addr(target: &SocketAddr) -> SocketAddr {
244    match target {
245        SocketAddr::V6(_) => {
246            SocketAddr::new(std::net::IpAddr::V6(std::net::Ipv6Addr::UNSPECIFIED), 0)
247        }
248        SocketAddr::V4(_) => {
249            SocketAddr::new(std::net::IpAddr::V4(std::net::Ipv4Addr::UNSPECIFIED), 0)
250        }
251    }
252}

254/// Default *total* kernel UDP buffer budget across the whole client endpoint pool
255/// (64 MiB). It is divided evenly among the pool's sockets (see
256/// [`client_udp_buf_per_socket`](client_udp_buf_total_bytes)), so adding endpoints
257/// shrinks each socket's share while keeping the aggregate fixed. This matters for
258/// non-root deployments: a per-socket request is clamped to `net.core.rmem_max`, so to
259/// reach a large total without `CAP_NET_ADMIN` you spread it over more sockets.
260/// Override the total with `MM_QUIC_UDP_BUF_BYTES`.
261const DEFAULT_UDP_BUF_TOTAL_BYTES: usize = 64 * 1024 * 1024;
262
263fn client_udp_buf_total_bytes() -> usize {
264    std::env::var("MM_QUIC_UDP_BUF_BYTES")
265        .ok()
266        .and_then(|v| v.parse::<usize>().ok())
267        .filter(|&n| n > 0)
268        .unwrap_or(DEFAULT_UDP_BUF_TOTAL_BYTES)
269}

271/// Optional explicit override for the number of client endpoints. When unset, the pool
272/// size is chosen adaptively (see [`Quic::build_client_pool`]): one socket if it can
273/// hold the whole buffer budget, otherwise enough sockets to reach it. When set,
274/// exactly that many sockets are used, each requesting an even split of the budget.
275fn explicit_client_endpoints() -> Option<usize> {
276    std::env::var("MM_QUIC_CLIENT_ENDPOINTS")
277        .ok()
278        .and_then(|v| v.parse::<usize>().ok())
279        .filter(|&n| n > 0)
280}

282/// When `MM_QUIC_UDP_BUF_NO_FORCE` is set, skip the privileged `SO_*BUFFORCE` options
283/// so the buffer is exactly what an unprivileged process would get (clamped to
284/// `net.core.rmem_max`). Used to validate a non-root-deployable config even while the
285/// job happens to run as root.
286fn udp_buf_force_disabled() -> bool {
287    std::env::var("MM_QUIC_UDP_BUF_NO_FORCE").is_ok_and(|v| v != "0" && !v.is_empty())
288}

290/// Enlarge a UDP socket's kernel send/recv buffers, best-effort. Tries the privileged
291/// `SO_*BUFFORCE` options first — these bypass the `net.core.{r,w}mem_max` ceiling and
292/// work when the process has `CAP_NET_ADMIN` (e.g. running as root on MAST) — and
293/// falls back to the ordinary setters (which the kernel clamps to that ceiling) when
294/// not permitted.
295fn set_udp_buffers(socket: &socket2::Socket, bytes: usize) {
296    use std::os::fd::AsRawFd;
297
298    if udp_buf_force_disabled() {
299        // Non-root path only: the kernel clamps these to net.core.{r,w}mem_max.
300        let _ = socket.set_recv_buffer_size(bytes);
301        let _ = socket.set_send_buffer_size(bytes);
302        return;
303    }
304
305    let fd = socket.as_raw_fd();
306    let size = bytes.min(i32::MAX as usize) as libc::c_int;
307    let len = std::mem::size_of::<libc::c_int>() as libc::socklen_t;
308    // SAFETY: `fd` is a valid UDP socket for the call's duration; the option value is
309    // an `int` of length `len`, as required by SO_RCVBUFFORCE/SO_SNDBUFFORCE.
310    let (forced_recv, forced_send) = unsafe {
311        let value = std::ptr::from_ref(&size).cast::<libc::c_void>();
312        (
313            libc::setsockopt(fd, libc::SOL_SOCKET, libc::SO_RCVBUFFORCE, value, len),
314            libc::setsockopt(fd, libc::SOL_SOCKET, libc::SO_SNDBUFFORCE, value, len),
315        )
316    };
317    if forced_recv != 0 {
318        let _ = socket.set_recv_buffer_size(bytes);
319    }
320    if forced_send != 0 {
321        let _ = socket.set_send_buffer_size(bytes);
322    }
323}

325/// Build a client [`Endpoint`] on a UDP socket whose kernel buffers we request at
326/// `requested` bytes (see [`set_udp_buffers`]), bound to `bind`, with `client_config`
327/// as its default. Returns the endpoint and the *usable* recv buffer the kernel
328/// actually granted (it reports ~2× the usable size for bookkeeping, so we halve it),
329/// which the caller uses to decide whether one socket reached the target or a pool is
330/// needed.
331fn make_client_endpoint(
332    bind: SocketAddr,
333    client_config: ClientConfig,
334    requested: usize,
335) -> anyhow::Result<(Endpoint, usize)> {
336    let domain = if bind.is_ipv6() {
337        socket2::Domain::IPV6
338    } else {
339        socket2::Domain::IPV4
340    };
341    let socket = socket2::Socket::new(domain, socket2::Type::DGRAM, Some(socket2::Protocol::UDP))?;
342    set_udp_buffers(&socket, requested);
343    let granted = socket.recv_buffer_size().unwrap_or(0);
344    let usable = granted / 2;
345    if crate::ctx::connection_debug() {
346        eprintln!(
347            "MM_QUIC client endpoint: recv buf requested {} B, granted {} B (~{} B usable, force={})",
348            requested,
349            granted,
350            usable,
351            !udp_buf_force_disabled()
352        );
353    }
354    socket.bind(&bind.into())?;
355    let std_socket: std::net::UdpSocket = socket.into();
356    let runtime =
357        quinn::default_runtime().ok_or_else(|| anyhow::anyhow!("no quic async runtime"))?;
358    let mut endpoint = Endpoint::new(endpoint_config(), None, std_socket, runtime)?;
359    endpoint.set_default_client_config(client_config);
360    Ok((endpoint, usable))
361}

363/// Default kernel recv-buffer size for the *server* UDP socket (64 MiB). quinn never
364/// raises `SO_RCVBUF` itself, so a high-rate receiver drops datagrams whenever the
365/// socket backlog spikes (→ QUIC RTO stalls, seen as throughput variance at the
366/// single-host ceiling). We force it up by default — matching the total budget the
367/// client pool targets ([`DEFAULT_UDP_BUF_TOTAL_BYTES`]) and the value validated in
368/// the throughput sweep. Override (typically to *reduce* it on a small host) with
369/// `MM_QUIC_SERVER_UDP_BUF_BYTES`.
370const DEFAULT_SERVER_UDP_BUF_BYTES: usize = 64 * 1024 * 1024;
371
372fn server_udp_buf_bytes() -> usize {
373    std::env::var("MM_QUIC_SERVER_UDP_BUF_BYTES")
374        .ok()
375        .and_then(|v| v.parse::<usize>().ok())
376        .filter(|&n| n > 0)
377        .unwrap_or(DEFAULT_SERVER_UDP_BUF_BYTES)
378}

380/// Build a server [`Endpoint`] bound to `addr`. Like [`Endpoint::server`] but (a) it
381/// uses [`endpoint_config`] so the advertised `max_udp_payload_size` matches our MTU
382/// discovery ceiling (a plain `Endpoint::server` hardcodes quinn's 1472 default, which
383/// would clamp the peer's probed MTU back down), and (b) it enlarges the receive
384/// buffer to [`server_udp_buf_bytes`] (best-effort; see [`set_udp_buffers`]).
385fn make_server_endpoint(addr: SocketAddr, server_config: ServerConfig) -> anyhow::Result<Endpoint> {
386    let domain = if addr.is_ipv6() {
387        socket2::Domain::IPV6
388    } else {
389        socket2::Domain::IPV4
390    };
391    let socket = socket2::Socket::new(domain, socket2::Type::DGRAM, Some(socket2::Protocol::UDP))?;
392    set_udp_buffers(&socket, server_udp_buf_bytes());
393    socket.bind(&addr.into())?;
394    let std_socket: std::net::UdpSocket = socket.into();
395    let runtime =
396        quinn::default_runtime().ok_or_else(|| anyhow::anyhow!("no quic async runtime"))?;
397    Ok(Endpoint::new(
398        endpoint_config(),
399        Some(server_config),
400        std_socket,
401        runtime,
402    )?)
403}

405/// The shared per-context QUIC state: TLS configs (loaded lazily on first use) and a
406/// pool of client endpoints per address family, created lazily and assigned
407/// round-robin to joins/side-channels.
408///
409/// One endpoint per connection (the original behaviour) exhausted sockets and the
410/// event loop at high fan-out. Collapsing to a single shared endpoint fixed that but
411/// made the one UDP socket a throughput bottleneck: a burst of tens of thousands of
412/// sends/receives overflows its buffers, dropping packets and forcing multi-second
413/// QUIC retransmit timeouts. A small pool is the middle ground — connections (and
414/// their burst load) are spread across `K` sockets. `K` is `MM_QUIC_CLIENT_ENDPOINTS`
415/// (default adaptive).
416pub(crate) struct Quic {
417    tls: Option<Arc<TlsConfig>>,
418    client_endpoints_v4: Vec<Endpoint>,
419    client_endpoints_v6: Vec<Endpoint>,
420    // Round-robin cursor for assigning connections to pool endpoints.
421    client_rr: usize,
422    // When set, the multi-threaded runtime to build endpoints under, so quinn spawns
423    // each endpoint's driver (which runs the packet crypto) on the pool rather than on
424    // the command-loop thread. Entered around `Endpoint::new` in `bind`/`dialer`.
425    rt: Option<Handle>,
426}
427
428impl Quic {
429    /// Build (or return the cached) TLS configs from the environment.
430    fn tls(&mut self) -> anyhow::Result<Arc<TlsConfig>> {
431        if let Some(tls) = &self.tls {
432            return Ok(tls.clone());
433        }
434        let tls = Arc::new(load_tls()?);
435        self.tls = Some(tls.clone());
436        Ok(tls)
437    }
438
439    /// A client [`Endpoint`] for `target`'s address family, assigned round-robin from
440    /// a lazily-created pool. Returns a cheap clone (the endpoint is internally
441    /// reference-counted). Spreading connections across several UDP sockets keeps any
442    /// one socket's buffers from overflowing under a fan-out burst, which otherwise
443    /// causes packet loss and multi-second QUIC retransmit stalls.
444    fn client_endpoint(&mut self, target: &SocketAddr) -> anyhow::Result<Endpoint> {
445        let is_v6 = target.is_ipv6();
446        let empty = if is_v6 {
447            self.client_endpoints_v6.is_empty()
448        } else {
449            self.client_endpoints_v4.is_empty()
450        };
451        if empty {
452            let pool = self.build_client_pool(client_bind_addr(target))?;
453            if is_v6 {
454                self.client_endpoints_v6 = pool;
455            } else {
456                self.client_endpoints_v4 = pool;
457            }
458        }
459        let pool = if is_v6 {
460            &self.client_endpoints_v6
461        } else {
462            &self.client_endpoints_v4
463        };
464        let endpoint = pool[self.client_rr % pool.len()].clone();
465        self.client_rr = self.client_rr.wrapping_add(1);
466        Ok(endpoint)
467    }
468
469    /// Build the client endpoint pool for one address family. Aim for
470    /// [`client_udp_buf_total_bytes`] of total UDP buffer. First try to get it all on a
471    /// single socket; if the kernel caps a single socket below the target (e.g.
472    /// unprivileged, clamped to `net.core.rmem_max`), fall back to enough sockets —
473    /// each requesting the measured per-socket max — to reach the total.
474    /// `MM_QUIC_CLIENT_ENDPOINTS` overrides this with a fixed pool size.
475    fn build_client_pool(&mut self, bind: SocketAddr) -> anyhow::Result<Vec<Endpoint>> {
476        let total = client_udp_buf_total_bytes();
477        let client_config = self.tls()?.client.clone();
478
479        if let Some(n) = explicit_client_endpoints() {
480            let per = (total / n).max(1);
481            let mut pool = Vec::with_capacity(n);
482            for _ in 0..n {
483                pool.push(make_client_endpoint(bind, client_config.clone(), per)?.0);
484            }
485            if crate::ctx::connection_debug() {
486                eprintln!("MM_QUIC client pool: {n} endpoints (explicit), {per} B requested each");
487            }
488            return Ok(pool);
489        }
490
491        // Adaptive: try to put the whole budget on one socket.
492        let (first, usable) = make_client_endpoint(bind, client_config.clone(), total)?;
493        if usable >= total {
494            if crate::ctx::connection_debug() {
495                eprintln!("MM_QUIC client pool: 1 endpoint holds the full {total} B budget");
496            }
497            return Ok(vec![first]);
498        }
499
500        // Capped below target — spread across enough sockets of `usable` each.
501        let n = total.div_ceil(usable.max(1));
502        if crate::ctx::connection_debug() {
503            eprintln!(
504                "MM_QUIC client pool: single socket capped at {usable} B usable; using {n} endpoints to reach {total} B total"
505            );
506        }
507        let mut pool = Vec::with_capacity(n);
508        pool.push(first);
509        for _ in 1..n {
510            pool.push(make_client_endpoint(bind, client_config.clone(), usable)?.0);
511        }
512        Ok(pool)
513    }
514}

516/// A dialing handle: a pooled client endpoint plus the data runtime to enter when
517/// dialing, so quinn spawns the resulting connection's driver (which runs its packet
518/// crypto) on that runtime rather than on whatever thread calls [`Net::connect`] (the
519/// command loop). `None` ⇒ the connection driver follows the caller (single core).
520#[derive(Clone)]
521pub(crate) struct QuicDialer {
522    endpoint: Endpoint,
523    rt: Option<Handle>,
524}

526/// A bound server endpoint plus the data runtime to enter when accepting, for the same
527/// reason as [`QuicDialer`]: the accepted connection's driver spawns on that runtime.
528pub(crate) struct QuicListener {
529    endpoint: Endpoint,
530    rt: Option<Handle>,
531}
532
533impl Net for Quic {
534    // A pooled client endpoint (cheap clone; internally reference-counted) plus the
535    // runtime to place dialed connections' drivers on.
536    type Dialer = QuicDialer;
537    // A server endpoint plus that runtime. quic multiplexes all streams on one
538    // connection; each stream is opened/accepted on demand, addressed by index.
539    type Listener = QuicListener;
540    type Conn = QuicConn;
541
542    fn create(runtime: Option<Handle>) -> anyhow::Result<Self> {
543        Ok(Self {
544            tls: None,
545            client_endpoints_v4: Vec::new(),
546            client_endpoints_v6: Vec::new(),
547            client_rr: 0,
548            rt: runtime,
549        })
550    }
551
552    fn parse_addr(url: &str) -> anyhow::Result<SocketAddr> {
553        parse_addr(url)
554    }
555
556    fn dialer(&mut self, addr: SocketAddr) -> anyhow::Result<QuicDialer> {
557        // Enter the data runtime (if any) so quinn spawns the pool endpoints' *endpoint*
558        // drivers there. Clone the handle first: the guard borrows the clone, leaving
559        // `self` free for the `&mut self` pool build below. The handle also rides along
560        // on the QuicDialer so `connect` can place each connection's driver there too.
561        let rt = self.rt.clone();
562        let endpoint = {
563            let _guard = rt.as_ref().map(Handle::enter);
564            self.client_endpoint(&addr)?
565        };
566        Ok(QuicDialer { endpoint, rt })
567    }
568
569    fn bind(&mut self, addr: SocketAddr) -> anyhow::Result<QuicListener> {
570        let server = self.tls()?.server.clone();
571        // Enter the data runtime (if any) so quinn spawns this server endpoint's driver
572        // there; the handle also rides on the QuicListener for `accept` to use.
573        let rt = self.rt.clone();
574        let endpoint = {
575            let _guard = rt.as_ref().map(Handle::enter);
576            make_server_endpoint(addr, server)?
577        };
578        Ok(QuicListener { endpoint, rt })
579    }
580
581    async fn accept(listener: &QuicListener) -> Option<QuicConn> {
582        let incoming = listener.endpoint.accept().await?;
583        // `incoming.accept()` synchronously creates the Connecting and spawns its
584        // connection driver (via ambient `tokio::spawn`), so enter the data runtime
585        // around exactly that call to place the driver — and its crypto — on the pool.
586        let connecting = {
587            let _guard = listener.rt.as_ref().map(Handle::enter);
588            incoming.accept().ok()?
589        };
590        let conn = connecting.await.ok()?;
591        Some(QuicConn::new_acceptor(conn))
592    }
593
594    async fn connect(dialer: &QuicDialer, addr: SocketAddr) -> Option<QuicConn> {
595        // connect() fails synchronously on a bad config; the handshake (.await) fails
596        // until the server is up. Both surface as `None` and the caller retries. Enter
597        // the data runtime around the synchronous connect() — which spawns the
598        // connection driver — so that driver's crypto runs on the pool, not the caller.
599        let connecting = {
600            let _guard = dialer.rt.as_ref().map(Handle::enter);
601            dialer.endpoint.connect(addr, SERVER_NAME).ok()?
602        };
603        let conn = connecting.await.ok()?;
604        Some(QuicConn::new_dialer(conn))
605    }
606
607    /// quic multiplexes all streams on one endpoint-paced connection, so a large number
608    /// of simultaneous handshakes is cheap; 1024 has proven safe at 64k+ fan-out.
609    /// Overridable via `MM_QUIC_MAX_CONCURRENT_CONNECTS`.
610    fn default_connect_concurrency() -> Option<usize> {
611        Some(1024)
612    }
613}

615/// The two-byte little-endian index prefix the dialer writes at the head of every
616/// bidirectional stream it opens, so the acceptor can pair it with the peer's stream of
617/// the same index (a bare quic connection is already private to the two peers, so unlike
618/// tcp no connection id is needed — only the stream index). `open_bi`/`accept_bi` pair
619/// in open order per initiator, but only the dialer opens, and it opens streams in
620/// whatever order the transport requests them; the explicit index removes any dependence
621/// on that order and lets the acceptor demultiplex.
622async fn write_index_prefix(send: &mut SendStream, index: usize) -> io::Result<()> {
623    // quinn's inherent `write_all` returns its own `WriteError`; map it to io::Error.
624    send.write_all(&(index as u16).to_le_bytes())
625        .await
626        .map_err(io::Error::other)
627}
628
629async fn read_index_prefix(recv: &mut RecvStream) -> io::Result<usize> {
630    let mut buf = [0u8; 2];
631    // quinn's inherent `read_exact` returns its own `ReadExactError`; map it to io::Error.
632    recv.read_exact(&mut buf).await.map_err(io::Error::other)?;
633    Ok(u16::from_le_bytes(buf) as usize)
634}

636/// A [`NetConn::stream`] request handed to an acceptor connection's demux: the wanted
637/// index, its send priority, and the one-shot the demux fulfils with the paired stream.
638pub(crate) struct QuicStreamRequest {
639    index: usize,
640    priority: i32,
641    reply: oneshot::Sender<io::Result<(QuicSend, QuicRecv)>>,
642}
643
644/// Fulfil a request with its paired stream: set the send priority and send the halves
645/// back through the one-shot.
646fn fulfill_quic(conn: &Connection, req: QuicStreamRequest, stream: (SendStream, RecvStream)) {
647    let (send, recv) = stream;
648    let _ = send.set_priority(req.priority);
649    let _ = req.reply.send(Ok(quic_halves(conn.clone(), send, recv)));
650}
651
652/// The acceptor-side demux for one connection: the single owner of `accept_bi`. It reads
653/// each accepted stream's index prefix and pairs it with the matching [`QuicStreamRequest`]
654/// using a per-index [`Matcher`] (each index sees at most one stream and one request,
655/// whichever arrives first parks for the other). Being the only accept-driver means
656/// `stream` never touches `accept_bi` directly — it just sends a request and awaits the
657/// one-shot — so there is no mutex held across an await. Ends when the connection drops
658/// (its senders close) or `accept_bi` errors; dropping the parked requests then fails
659/// their awaiting `stream` calls.
660async fn quic_acceptor_demux(
661    conn: Connection,
662    mut requests: mpsc::UnboundedReceiver<QuicStreamRequest>,
663) {
664    let mut matchers: HashMap<usize, Matcher<QuicStreamRequest, (SendStream, RecvStream)>> =
665        HashMap::new();
666    loop {
667        tokio::select! {
668            req = requests.recv() => {
669                let Some(req) = req else {
670                    return; // every `stream` handle dropped: the connection is going away
671                };
672                matchers
673                    .entry(req.index)
674                    .or_insert_with(Matcher::new)
675                    .push_left(req, |req, stream| fulfill_quic(&conn, req, stream));
676            }
677            // `accept_bi` is cancel-safe, so racing it in `select!` never drops a stream.
678            accepted = conn.accept_bi() => {
679                let Ok((send, mut recv)) = accepted else {
680                    return; // connection lost: dropping the matchers fails parked requests
681                };
682                let Ok(index) = read_index_prefix(&mut recv).await else {
683                    continue; // peer opened a stream with a bad prefix — drop it
684                };
685                matchers
686                    .entry(index)
687                    .or_insert_with(Matcher::new)
688                    .push_right((send, recv), |req, stream| fulfill_quic(&conn, req, stream));
689            }
690        }
691    }
692}

694/// One QUIC connection, in one of two roles set at construction. A **dialer** connection
695/// (from [`Net::connect`]) opens each requested stream with `open_bi` and writes the
696/// index prefix. An **acceptor** connection (from [`Net::accept`]) delegates all
697/// `accept_bi` work to a per-connection [`quic_acceptor_demux`] coroutine; its `stream`
698/// just sends a request down `requests` and awaits the paired stream. The demux holds the
699/// `Connection` alive; the acceptor keeps `remote` only for [`fmt::Debug`].
700///
701/// Cloneable so the reader, writer, heartbeat task, and striping coordinator can
702/// each hold a handle: a dialer clone shares the `Connection` (internally
703/// reference-counted); an acceptor clone shares the request-sender to the one demux.
704#[derive(Clone)]
705pub(crate) enum QuicConn {
706    Dialer {
707        conn: Connection,
708    },
709    Acceptor {
710        remote: SocketAddr,
711        requests: mpsc::UnboundedSender<QuicStreamRequest>,
712    },
713}
714
715impl QuicConn {
716    fn new_dialer(conn: Connection) -> Self {
717        Self::Dialer { conn }
718    }
719
720    fn new_acceptor(conn: Connection) -> Self {
721        let remote = conn.remote_address();
722        let (requests, rx) = mpsc::unbounded_channel();
723        // Spawn on the command-loop LocalSet (`accept` runs there); the demux owns the
724        // connection and lives until every `stream` handle is dropped or it errors.
725        tokio::task::spawn_local(quic_acceptor_demux(conn, rx));
726        Self::Acceptor { remote, requests }
727    }
728}
729
730impl fmt::Debug for QuicConn {
731    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
732        let remote = match self {
733            QuicConn::Dialer { conn } => conn.remote_address(),
734            QuicConn::Acceptor { remote, .. } => *remote,
735        };
736        f.debug_tuple("QuicConn").field(&remote).finish()
737    }
738}

740fn quic_halves(conn: Connection, send: SendStream, recv: RecvStream) -> (QuicSend, QuicRecv) {
741    (
742        QuicSend {
743            send,
744            _conn: conn.clone(),
745        },
746        QuicRecv { recv, _conn: conn },
747    )
748}
749
750impl NetConn for QuicConn {
751    type Send = QuicSend;
752    type Recv = QuicRecv;
753    type Stream = Pin<Box<dyn Future<Output = io::Result<(QuicSend, QuicRecv)>> + Send>>;
754
755    fn stream(&self, index: usize, priority: i32) -> Self::Stream {
756        match self {
757            QuicConn::Dialer { conn } => {
758                let conn = conn.clone();
759                Box::pin(async move {
760                    let (mut send, recv) = conn.open_bi().await.map_err(io::Error::other)?;
761                    write_index_prefix(&mut send, index).await?;
762                    let _ = send.set_priority(priority);
763                    Ok(quic_halves(conn, send, recv))
764                })
765            }
766            QuicConn::Acceptor { requests, .. } => {
767                let requests = requests.clone();
768                Box::pin(async move {
769                    let (reply, rx) = oneshot::channel();
770                    requests
771                        .send(QuicStreamRequest {
772                            index,
773                            priority,
774                            reply,
775                        })
776                        .map_err(|_| io::Error::other("quic connection closed"))?;
777                    rx.await
778                        .map_err(|_| io::Error::other("quic connection closed"))?
779                })
780            }
781        }
782    }
783}
