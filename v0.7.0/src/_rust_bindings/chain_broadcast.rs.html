9//! Pipelined-chain bulk broadcast over hyperactor's channel transport.
10//!
11//! A leader streams a large block down an ordered chain of workers --
12//! `leader -> worker[0] -> worker[1] -> ... -> worker[N-1]` -- where each
13//! interior worker forwards every chunk to its successor the instant it lands
14//! (temporal pipelining), so the source egresses ~one copy regardless of N.
15//! This is the bandwidth-optimal large-block broadcast (NCCL's ring, MPI's
16//! pipelined chain); it replaces a single-leader O(N) fan-out.
17//!
18//! Unlike a bolt-on transport, this reuses [`hyperactor::channel`]: connections
19//! are persistent channel streams (the process default transport -- metatls in
20//! cluster, for Meta mutual-TLS identity with no cert paths to read -- or any
21//! transport `serve` is given), framing and reconnect live in the channel layer,
22//! and addresses are exchanged as plain strings. The only new machinery here is the
23//! chunk relay ([`forward`]) and a byte-counter recv-completion signal
24//! ([`new_ctx`]/[`ctx_wait_into`]).
25//!
26//! The Python surface mirrors the C transport it supersedes so the remotemount
27//! orchestration is unchanged: `serve` (bind a listener), `connect` (dial the
28//! head/successor), `send_block` (prime the pipeline at the source), `forward`
29//! (relay + deliver at each worker), and `new_ctx`/`ctx_wait_into`/`ctx_close`.

59/// Default number of parallel TLS streams per hop, to fill the NIC (as the C ctcp
60/// transport did). Delivery is offset-addressed, so the out-of-order arrival that
61/// multi-stream allows is fine. Tradeoff vs the single ordered stream: the
62/// unordered path does NOT retransmit on reconnect, so a mid-transfer connection
63/// drop can lose a chunk (surfacing as a `ctx_wait_into` timeout, then a re-fault);
64/// acceptable on a stable in-cluster fabric. `connect`/`forward` take the count as
65/// an argument (this is only the default) so the caller can tune it per run without
66/// a rebuild -- more streams fill a fatter cross-DC pipe, fewer reduce per-stream
67/// ramp overhead on a short chain.
68const NUM_CBC_STREAMS: usize = 32;

70/// One frame of a block on the wire: a slice of the block plus enough header to
71/// place it (`offset`) and to know when the block is complete (`total`). `tag`
72/// is the block id, carried through for debugging / future multiplexing. `data`
73/// is a `serde_multipart::Part` so the channel carries the payload out-of-band
74/// (zero-copy) rather than bincode-encoding it byte by byte -- and forwarding it
75/// (`Chunk::clone`) is a refcount bump, not a memcpy.
76#[derive(Debug, Clone, Serialize, Deserialize, Named)]
77struct Chunk {
78    tag: u64,
79    offset: u64,
80    total: u64,
81    data: Part,
82}
83
84/// Recv-completion state, mirroring the byte-counter ctx of the C transport it
85/// replaces. The relay stores each landed chunk as an `(offset, Bytes)` fragment
86/// (a refcount bump + a push -- NO memcpy on the forwarding hot path); the block
87/// is assembled from the fragments once, in `wait_into`, off that path. A running
88/// `received` counter lets `wait` know when the whole block has landed. Delivery
89/// is synchronous (one block in flight per mount), so the fragments are drained
90/// and reused across blocks.
91struct Ctx {
92    inner: Mutex<CtxInner>,
93    cond: Condvar,
94}
95
96struct CtxInner {
97    fragments: Vec<(usize, Bytes)>,
98    /// Offsets already delivered for the current block, so a duplicate chunk (a
99    /// reconnect re-send on the unordered multi-stream path) is dropped instead of
100    /// double-counting `received` and completing `wait` on a hole. Cleared when a
101    /// block is drained or abandoned.
102    seen: HashSet<usize>,
103    received: u64,
104    closed: bool,
105}
106
107impl Ctx {
108    fn new(_block_size: usize) -> Self {
109        Self {
110            inner: Mutex::new(CtxInner {
111                fragments: Vec::new(),
112                seen: HashSet::new(),
113                received: 0,
114                closed: false,
115            }),
116            cond: Condvar::new(),
117        }
118    }
119
120    /// Stash a landed chunk as an `(offset, Bytes)` fragment (zero-copy: a Bytes
121    /// refcount bump + a Vec push, no memcpy) and bump the received counter,
122    /// waking any parked `wait_into`. The actual assembly happens in `wait_into`,
123    /// so the relay can forward the next chunk immediately.
124    fn deliver(&self, chunk: &Chunk) {
125        let bytes = chunk.data.to_bytes();
126        let off = chunk.offset as usize;
127        let len = bytes.len();
128        let mut inner = self.inner.lock().expect("ctx mutex poisoned");
129        if !inner.seen.insert(off) {
130            // Duplicate offset (a reconnect re-sent a chunk on the unordered
131            // multi-stream path). Counting it again would push `received` past the
132            // block size while a real gap stays unfilled, so `wait` would complete
133            // on a hole -- drop it; the first copy already landed.
134            tracing::warn!(
135                offset = off,
136                "chain_broadcast: dropping duplicate chunk at already-received offset"
137            );
138            return;
139        }
140        inner.fragments.push((off, bytes));
141        inner.received += len as u64;
142        drop(inner);
143        self.cond.notify_all();
144    }
145
146    fn mark_closed(&self) {
147        self.inner.lock().expect("ctx mutex poisoned").closed = true;
148        self.cond.notify_all();
149    }
150
151    /// Block until `want` bytes of the current block have landed, then assemble
152    /// them directly into `dst` (draining the counter by `want`) and return
153    /// `true`. `false` means the predecessor closed before the block completed,
154    /// or the timeout elapsed. `dst` must be at least `want` bytes and is
155    /// expected pre-zeroed: a gap left by a malformed sender reads as zero
156    /// without this re-zeroing, keeping the assemble to one copy per chunk
157    /// straight into the caller's buffer, with no intermediate block.
158    fn wait_into(&self, dst: &mut [u8], want: u64, timeout_ms: i64) -> bool {
159        let mut inner = self.inner.lock().expect("ctx mutex poisoned");
160        let deadline =
161            (timeout_ms >= 0).then(|| Instant::now() + Duration::from_millis(timeout_ms as u64));
162        while inner.received < want && !inner.closed {
163            match deadline {
164                None => inner = self.cond.wait(inner).expect("ctx cond poisoned"),
165                Some(dl) => {
166                    let now = Instant::now();
167                    if now >= dl {
168                        break;
169                    }
170                    let (guard, _) = self
171                        .cond
172                        .wait_timeout(inner, dl - now)
173                        .expect("ctx cond poisoned");
174                    inner = guard;
175                }
176            }
177        }
178        if inner.received < want {
179            // Timed out, or the predecessor closed mid-block. Drop the partial
180            // fragments and reset the counter so the ctx (reused for the mount's life)
181            // is clean for the next block; the design does not retransmit, so the
182            // caller re-faults to trigger a fresh delivery.
183            inner.fragments.clear();
184            inner.seen.clear();
185            inner.received = 0;
186            return false;
187        }
188        // Assemble the block from its fragments -- one copy per chunk, done ONCE
189        // here (off the relay's forwarding path, not per hop) and straight into
190        // `dst`. Drain so the ctx is clean for the next block. A fragment that
191        // would overrun the buffer (a stray frame) is skipped rather than
192        // panicking.
193        debug_assert!(
194            want as usize <= dst.len(),
195            "wait_into: dst ({} bytes) is smaller than want ({want})",
196            dst.len()
197        );
198        let cap = (want as usize).min(dst.len());
199        let frags = std::mem::take(&mut inner.fragments);
200        inner.seen.clear();
201        for (off, b) in frags {
202            let end = off.saturating_add(b.len());
203            if end <= cap {
204                dst[off..end].copy_from_slice(&b);
205            } else {
206                tracing::warn!(
207                    offset = off,
208                    len = b.len(),
209                    cap = cap,
210                    "chain_broadcast: dropping out-of-range fragment that would overrun the block"
211                );
212            }
213        }
214        inner.received -= want;
215        true
216    }
217}

227/// Where a node's chunks arrive from.
228///
229/// Interior nodes are dialed by their predecessor -- one multi-stream `channel::serve`
230/// listener, the fast in-cluster path. The head cannot be: its predecessor is the client,
231/// which may reach the mesh only through a scheduler gateway, and a raw `channel::dial` is
232/// not gateway-routed. A mailbox port is.
233///
234/// Both yield the same `Chunk`, so [`forward`] and everything downstream of it -- the
235/// relay, the ctx, block assembly -- are identical either way.
236enum ChunkSource {
237    Channel(ChannelRx<Chunk>),
238    Port(PortReceiver<Chunk>),
239}
240
241impl ChunkSource {
242    /// Next chunk, or `None` at end-of-stream.
243    ///
244    /// The variants differ in a way that matters for the head. A `ChannelRx` ends as soon
245    /// as the predecessor's connection drops, so the relay exits and `mark_closed()` wakes
246    /// a parked `ctx_wait_into`. A bound `PortReceiver` ends only when every sender is
247    /// gone, and the binding keeps one alive for the actor's lifetime -- so a client that
248    /// disconnects, or simply stops sending, produces no end-of-stream at all.
249    ///
250    /// Accepted consequence: the head's relay runs until proc exit and holds its successor
251    /// session open, and a head-side `ctx_wait_into` falls back to its timeout instead of
252    /// being woken by predecessor loss. A port has no connection to lose, so "the
253    /// predecessor is gone" is not observable. Interior nodes keep the prompt behaviour.
254    async fn recv(&mut self) -> Option<Chunk> {
255        match self {
256            ChunkSource::Channel(rx) => rx.recv().await.ok(),
257            ChunkSource::Port(rx) => rx.recv().await.ok(),
258        }
259    }
260}

302/// Bind a listener for the predecessor to dial. Returns the concrete bound
303/// address (with its ephemeral port) as a string to distribute, plus the server
304/// handle that [`forward`] consumes.
305///
306/// `bind_addr` chooses the transport. `None` defers to monarch's process-wide
307/// default transport -- the same `default_transport` /
308/// `HYPERACTOR_MESH_DEFAULT_TRANSPORT` knob every other serve site uses -- so the
309/// chain listener binds whatever the rest of the worker's channels bind (metatls
310/// in cluster: mutual-TLS identity, no cert paths to distribute). A caller without
311/// Meta TLS infra (OSS, tests) flips that knob to e.g. tcp, or passes an explicit
312/// address string here, e.g. `"tcp![::]:0"`; `connect` and `forward` dial whatever
313/// transport the resulting address encodes, so no other knob is needed.
314#[pyfunction]
315#[pyo3(signature = (bind_addr=None))]
316fn serve(bind_addr: Option<String>) -> PyResult<PyChainServer> {
317    let bind = match bind_addr {
318        Some(a) => parse_addr(&a)?,
319        None => hyperactor_mesh::transport::default_bind_spec().binding_addr(),
320    };
321    // channel::serve spawns its listener task, so it must run inside the shared
322    // tokio runtime -- and on the SAME runtime as the relay task in `forward`, so
323    // the listener and its consumer are never split across runtimes.
324    let _guard = monarch_hyperactor::runtime::get_tokio_runtime().enter();
325    let (addr, rx) = channel::serve::<Chunk>(bind)
326        .map_err(|e| PyRuntimeError::new_err(format!("serve failed: {e}")))?;
327    Ok(PyChainServer {
328        addr: addr.to_string(),
329        rx: Some(ChunkSource::Channel(rx)),
330    })
331}

333/// Dial a successor (or the chain head). The connection is persistent and reused
334/// for every block; dialing is lazy, so the peer need not be serving yet.
335/// `num_streams` is the number of parallel streams to open (defaults to
336/// [`NUM_CBC_STREAMS`]); the caller picks it at runtime to tune the pipe.
337#[pyfunction]
338#[pyo3(signature = (addr, num_streams = NUM_CBC_STREAMS))]
339fn connect(addr: &str, num_streams: usize) -> PyResult<PyChainSession> {
340    // unordered::dial opens `num_streams` parallel streams to fill the NIC; it
341    // spawns sender tasks, so run it on the shared runtime.
342    let _guard = monarch_hyperactor::runtime::get_tokio_runtime().enter();
343    let tx = channel::unordered::dial::<Chunk>(parse_addr(addr)?, num_streams.max(1))
344        .map_err(|e| PyRuntimeError::new_err(format!("dial {addr} failed: {e}")))?;
345    Ok(PyChainSession { tx: Arc::new(tx) })
346}

348/// How many bytes of a `len`-byte block to send: the leading `size` if given,
349/// otherwise all of it.
350fn payload_len(len: usize, size: Option<usize>) -> PyResult<usize> {
351    match size {
352        None => Ok(len),
353        Some(n) if n <= len => Ok(n),
354        Some(n) => Err(PyValueError::new_err(format!(
355            "size {n} exceeds the {len}-byte block"
356        ))),
357    }
358}

360/// Stripe `buf` into `chunk_size`-byte frames and post them to the peer. The chain
361/// source calls this to prime the pipeline; interior nodes forward via the relay
362/// in [`forward`]. If `size` is given, only the block's leading `size` bytes are sent.
363/// Returns the number of bytes sent.
364#[pyfunction]
365#[pyo3(signature = (session, data, chunk_size, tag, size = None))]
366fn send_block(
367    py: Python<'_>,
368    session: &PyChainSession,
369    data: &Bound<'_, PyBytes>,
370    chunk_size: usize,
371    tag: u64,
372    size: Option<usize>,
373) -> PyResult<usize> {
374    // Wrap the Python bytes as a shared Bytes with no payload copy: the wrapper
375    // pins the PyBytes alive for the lifetime of every chunk sliced from it, so
376    // the source egresses the block without ever copying it. This reuses the
377    // buffers module's zero-copy PyBytes view (the same conversion
378    // `Buffer::take_part` uses); each chunk is then a zero-copy slice of it
379    // wrapped as a Part.
380    let owned = monarch_hyperactor::buffers::py_bytes_to_bytes(data.clone().unbind());
381    let total = payload_len(owned.len(), size)?;
382    let chunk_size = chunk_size.max(1);
383    let tx = session.tx.clone();
384    py.detach(|| {
385        let mut offset = 0usize;
386        while offset < total {
387            let end = (offset + chunk_size).min(total);
388            tx.post(Chunk {
389                tag,
390                offset: offset as u64,
391                total: total as u64,
392                data: Part::from(owned.slice(offset..end)),
393            });
394            offset = end;
395        }
396    });
397    Ok(total)
398}

400/// Run the chain relay for this node: consume chunks from the bound listener
401/// (`server`), forward each to the successor the instant it lands (if any), and
402/// deliver it into the local recv ctx. Spawns on the shared tokio runtime and
403/// returns immediately; the relay runs until the predecessor closes the
404/// connection. A node with no successor is the chain tail (recv + deliver only).
405/// `num_streams` (defaults to [`NUM_CBC_STREAMS`]) is how many parallel streams
406/// this node opens to its successor, tuned at runtime to match the pipe.
407#[pyfunction]
408#[pyo3(signature = (server, successor, ctx, num_streams = NUM_CBC_STREAMS))]
409fn forward(
410    server: &mut PyChainServer,
411    successor: Option<String>,
412    ctx: &PyCtx,
413    num_streams: usize,
414) -> PyResult<()> {
415    let mut rx = server
416        .rx
417        .take()
418        .ok_or_else(|| PyRuntimeError::new_err("forward: server already consumed"))?;
419    let ctx = ctx.ctx.clone();
420    let runtime = monarch_hyperactor::runtime::get_tokio_runtime();
421    // Dial the successor up front (if any) so a malformed address or a failed dial
422    // surfaces here as an error, instead of silently collapsing this node into the
423    // chain tail and stranding every node downstream. `None` = this node is the tail
424    // (recv + deliver only). Dialing is lazy, so the successor need not be serving yet.
425    let succ_tx: Option<ChannelTx<Chunk>> = match successor {
426        None => None,
427        Some(addr) => {
428            let parsed = parse_addr(&addr)?;
429            let _guard = runtime.enter();
430            let tx =
431                channel::unordered::dial::<Chunk>(parsed, num_streams.max(1)).map_err(|e| {
432                    PyRuntimeError::new_err(format!("forward: dial successor {addr} failed: {e}"))
433                })?;
434            Some(tx)
435        }
436    };
437    runtime.spawn(async move {
438        while let Some(chunk) = rx.recv().await {
439            // Forward first so the successor starts receiving while we write
440            // locally -- the temporal pipeline that makes the chain fast.
441            if let Some(tx) = &succ_tx {
442                tx.post(chunk.clone());
443            }
444            ctx.deliver(&chunk);
445        }
446        ctx.mark_closed();
447    });
448    Ok(())
449}

451/// Block until `want` bytes of the current block have landed locally, then
452/// assemble them straight into the caller's buffer at `dst_addr` (`dst_len`
453/// bytes) -- the mount's `block_ptr` staging buffer -- draining the counter by
454/// `want`. Writing directly into that buffer avoids materialising an
455/// intermediate `bytes` and the extra copy from it into the mount buffer that a
456/// return-by-value would cost; `receive_block` then freezes the buffer into the
457/// served block with no further copy.
458#[pyfunction]
459fn ctx_wait_into(
460    py: Python<'_>,
461    ctx: &PyCtx,
462    dst_addr: usize,
463    dst_len: usize,
464    want: u64,
465    timeout_ms: i64,
466) -> PyResult<()> {
467    let c = ctx.ctx.clone();
468    let ok = py.detach(|| {
469        // SAFETY: `dst_addr`/`dst_len` are the return of `block_ptr`, i.e. the
470        // mount's staging `BytesMut` for this block -- a live, non-null,
471        // `AVAILABILITY_BLOCK_SIZE`-byte allocation that is never grown (so the
472        // pointer stays valid) and that nothing else reads or writes between
473        // `block_ptr` and `receive_block`, because delivery is one block in
474        // flight per mount (block_ptr -> ctx_wait_into -> receive_block runs
475        // sequentially on one actor task). So this slice is the unique writer for
476        // the buffer over its whole lifetime here, is valid for `dst_len` bytes,
477        // and `u8` needs no alignment; `wait_into` never writes past `dst.len()`.
478        let dst = unsafe { std::slice::from_raw_parts_mut(dst_addr as *mut u8, dst_len) };
479        c.wait_into(dst, want, timeout_ms)
480    });
481    if ok {
482        Ok(())
483    } else {
484        Err(PyRuntimeError::new_err(format!(
485            "ctx_wait_into: did not receive {want} bytes (predecessor closed or timed out)"
486        )))
487    }
488}

490/// Wake any parked `wait_into` and mark the ctx closed (mount teardown).
491#[pyfunction]
492fn ctx_close(ctx: &PyCtx) {
493    ctx.ctx.mark_closed();

507/// Bind the chain HEAD's receive endpoint as a mailbox port, for a predecessor that
508/// cannot dial this worker directly.
509///
510/// A `channel::dial` is point-to-point, so a client reaching the mesh only through a
511/// scheduler gateway can never reach a `serve`d listener. A mailbox post is handed to the
512/// proc's forwarder, which is exactly the layer `channel::dial` bypasses.
513///
514/// Only the head needs this; interior hops stay on [`serve`]'s multi-stream listener,
515/// which is faster and routable in cluster. The returned `port_id` is picklable, and
516/// `Chunk` rides the mailbox unchanged -- multipart keeps the payload out of band, so
517/// forwarding is still a refcount bump.
518#[pyfunction]
519fn serve_port(instance: &PyInstance) -> PyResult<PyChainServer> {
520    let (handle, rx) = instance.open_port::<Chunk>();
521    let port_ref = handle.bind();
522    Ok(PyChainServer {
523        addr: port_ref.port_addr().to_string(),
524        rx: Some(ChunkSource::Port(rx)),
525    })
526}

528/// Check that `addr` names a well-formed port, and raise if not.
529///
530/// `send_block_to_port` posts fire-and-forget, so a malformed head address would otherwise
531/// go unnoticed until the receiver's completion timeout, one block later and far from the
532/// cause. Callers validate once at wiring time instead -- restoring the failure point the
533/// old `connect()` gave them.
534#[pyfunction]
535fn validate_port_addr(addr: &str) -> PyResult<()> {
536    let _: PortAddr = addr
537        .parse()
538        .map_err(|e| PyRuntimeError::new_err(format!("bad port addr {addr}: {e}")))?;
539    Ok(())
540}

542/// Stripe `buf` into `chunk_size`-byte frames and post them to the head's mailbox port.
543/// The [`send_block`] counterpart for a source that cannot dial the head.
544///
545/// Same zero-copy discipline as [`send_block`]: the block is wrapped once as a shared
546/// `Bytes` and each chunk is a slice of it, so the source never copies the payload.
547/// `size` limits the send to the block's leading bytes, as in [`send_block`].
548#[pyfunction]
549#[pyo3(signature = (instance, port, data, chunk_size, tag, size = None))]
550fn send_block_to_port(
551    py: Python<'_>,
552    instance: &PyInstance,
553    port: &str,
554    data: &Bound<'_, PyBytes>,
555    chunk_size: usize,
556    tag: u64,
557    size: Option<usize>,
558) -> PyResult<usize> {
559    // `attest` is unchecked: the receiving port must be one minted by `serve_port`, i.e.
560    // typed `Chunk`. Pointing this at any other port is a decode error on delivery.
561    //
562    // Posting is fire-and-forget, like `send_block` -- `Endpoint::post` returns `()`, so a
563    // bad address cannot be reported here and this always returns `total`. It is not
564    // swallowed, though: `attest` leaves `return_undeliverable` set, so an unroutable
565    // chunk comes back to this actor as a delivery failure. What the CALLER sees first is
566    // still the downstream `ctx_wait_into` timeout, so check the sender's delivery-failure
567    // log before suspecting the chain. There is likewise no flow control: a whole block's
568    // chunks are queued into the forwarder as fast as they are sliced.
569    let port_addr: PortAddr = port
570        .parse()
571        .map_err(|e| PyRuntimeError::new_err(format!("bad port addr {port}: {e}")))?;
572    let port_ref = PortRef::<Chunk>::attest(port_addr);
573    let owned = monarch_hyperactor::buffers::py_bytes_to_bytes(data.clone().unbind());
574    let total = payload_len(owned.len(), size)?;
575    let chunk_size = chunk_size.max(1);
576    let instance = instance.clone();
577    py.detach(|| {
578        let mut offset = 0usize;
579        while offset < total {
580            let end = (offset + chunk_size).min(total);
581            (&port_ref).post(
582                &*instance,
583                Chunk {
584                    tag,
585                    offset: offset as u64,
586                    total: total as u64,
587                    data: Part::from(owned.slice(offset..end)),
588                },
589            );
590            offset = end;
591        }
592    });
593    Ok(total)
594}

629    /// A two-hop chain over TCP (metatls needs Meta TLS infra not present in a
630    /// unit test): source -> relay -> tail. The relay forwards each chunk to the
631    /// tail as it lands; both the relay's and the tail's ctx must reconstruct the
632    /// exact block, proving the transport + byte-counter completion end to end.
633    #[test]
634    fn chain_relays_and_delivers_block() {
635        let rt = tokio::runtime::Runtime::new().expect("build tokio runtime");
636        rt.block_on(async {
637            let block: Vec<u8> = (0..(3 * 1024 * 1024u32)).map(|i| (i % 251) as u8).collect();
638            let block_size = block.len();
639            let any_tcp = || ChannelAddr::any(ChannelTransport::Tcp(TcpMode::Localhost));
640
641            // Tail: serve, relay into its ctx with no successor.
642            let (tail_addr, mut tail_rx) = channel::serve::<Chunk>(any_tcp()).expect("tail serve");
643            let tail_ctx = Arc::new(Ctx::new(block_size));
644            {
645                let tail_ctx = tail_ctx.clone();
646                tokio::spawn(async move {
647                    while let Ok(chunk) = tail_rx.recv().await {
648                        tail_ctx.deliver(&chunk);
649                    }
650                    tail_ctx.mark_closed();
651                });
652            }
653
654            // Relay: serve, forward each chunk to the tail as it lands.
655            let (relay_addr, mut relay_rx) =
656                channel::serve::<Chunk>(any_tcp()).expect("relay serve");
657            let relay_ctx = Arc::new(Ctx::new(block_size));
658            {
659                let relay_ctx = relay_ctx.clone();
660                let tail_tx = channel::dial::<Chunk>(tail_addr).expect("dial tail");
661                tokio::spawn(async move {
662                    while let Ok(chunk) = relay_rx.recv().await {
663                        tail_tx.post(chunk.clone());
664                        relay_ctx.deliver(&chunk);
665                    }
666                    relay_ctx.mark_closed();
667                });
668            }
669
670            // Source: stripe the block into 512 KiB chunks to the relay head.
671            let head_tx = channel::dial::<Chunk>(relay_addr).expect("dial relay");
672            let chunk = 512 * 1024usize;
673            let mut offset = 0usize;
674            while offset < block_size {
675                let end = (offset + chunk).min(block_size);
676                head_tx.post(Chunk {
677                    tag: 0,
678                    offset: offset as u64,
679                    total: block_size as u64,
680                    data: Part::from(block[offset..end].to_vec()),
681                });
682                offset = end;
683            }
684
685            let mut relay_got = vec![0u8; block_size];
686            assert!(
687                relay_ctx.wait_into(&mut relay_got, block_size as u64, 30_000),
688                "relay received the whole block"
689            );
690            let mut tail_got = vec![0u8; block_size];
691            assert!(
692                tail_ctx.wait_into(&mut tail_got, block_size as u64, 30_000),
693                "tail received the whole block via the relay"
694            );
695            assert_eq!(relay_got, block, "relay reconstructs the source block");
696            assert_eq!(
697                tail_got, block,
698                "tail reconstructs the block forwarded through the relay"
699            );
700        });
701    }

703    /// Many distinct blocks streamed over ONE persistent connection (the real
704    /// remotemount flow: `send_block` reuses one Tx, `ctx_wait` drains one block
705    /// per call), each verified byte-identical. This is the multi-block path the
706    /// single-block test above does not cover -- the case that failed on MAST.
707    #[test]
708    fn many_blocks_over_persistent_channel_are_byte_identical() {
709        let rt = tokio::runtime::Runtime::new().expect("build tokio runtime");
710        rt.block_on(async {
711            let block_size = 16 * 1024 * 1024usize;
712            let chunk = 4 * 1024 * 1024usize;
713            let any_tcp = ChannelAddr::any(ChannelTransport::Tcp(TcpMode::Localhost));
714
715            let (addr, mut rx) = channel::serve::<Chunk>(any_tcp).expect("serve");
716            let ctx = Arc::new(Ctx::new(block_size));
717            {
718                let ctx = ctx.clone();
719                tokio::spawn(async move {
720                    while let Ok(c) = rx.recv().await {
721                        ctx.deliver(&c);
722                    }
723                    ctx.mark_closed();
724                });
725            }
726
727            // ONE persistent Tx, reused for every block (as LeaderActor does).
728            let tx = channel::dial::<Chunk>(addr).expect("dial");
729            for b in 0..6u64 {
730                // A distinct pattern per block so a stale/mixed buffer is caught.
731                let block: Vec<u8> = (0..block_size)
732                    .map(|i| {
733                        ((i as u64).wrapping_mul(31).wrapping_add(b.wrapping_mul(97)) % 251) as u8
734                    })
735                    .collect();
736                let mut off = 0usize;
737                while off < block_size {
738                    let end = (off + chunk).min(block_size);
739                    tx.post(Chunk {
740                        tag: b,
741                        offset: off as u64,
742                        total: block_size as u64,
743                        data: Part::from(block[off..end].to_vec()),
744                    });
745                    off = end;
746                }
747                let mut got = vec![0u8; block_size];
748                assert!(
749                    ctx.wait_into(&mut got, block_size as u64, 30_000),
750                    "block {b} not received"
751                );
752                assert_eq!(
753                    got, block,
754                    "block {b} byte-identical over the persistent channel"
755                );
756            }
757        });
758    }

760    /// `wait` drains exactly one block's worth per call and returns a fresh owned
761    /// block each time, so a later block is never contaminated by an earlier one
762    /// -- the synchronous one-block-in-flight pattern remotemount uses (the leader
763    /// waits for every worker's `ctx_wait` of block N before sourcing block N+1).
764    #[test]
765    fn ctx_wait_drains_each_block_without_contamination() {
766        let ctx = Ctx::new(16);
767
768        ctx.deliver(&Chunk {
769            tag: 0,
770            offset: 0,
771            total: 16,
772            data: Part::from(vec![1u8; 16]),
773        });
774        let mut first = vec![0u8; 16];
775        assert!(ctx.wait_into(&mut first, 16, 0), "block 0 ready");
776        assert_eq!(first, vec![1u8; 16], "first wait drains block 0");
777
778        // Block 1 gets a fresh owned block after block 0 was drained (counter back to 0).
779        ctx.deliver(&Chunk {
780            tag: 1,
781            offset: 0,
782            total: 16,
783            data: Part::from(vec![2u8; 16]),
784        });
785        let mut second = vec![0u8; 16];
786        assert!(ctx.wait_into(&mut second, 16, 0), "block 1 ready");
787        assert_eq!(
788            second,
789            vec![2u8; 16],
790            "second wait drains block 1 independently of block 0"
791        );
792    }
