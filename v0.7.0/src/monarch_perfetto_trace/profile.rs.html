82struct EosTrackingReader<R> {
83    inner: R,
84    tail: [u8; ARROW_STREAM_EOS.len()],
85    tail_len: usize,
86}
87
88impl<R> EosTrackingReader<R> {
89    /// Wrap a reader and retain enough trailing bytes to validate stream completion.
90    fn new(inner: R) -> Self {
91        Self {
92            inner,
93            tail: [0; ARROW_STREAM_EOS.len()],
94            tail_len: 0,
95        }
96    }
97
98    /// Return whether the bytes consumed from the reader end with Arrow's EOS marker.
99    fn ended_with_eos(&self) -> bool {
100        self.tail_len == self.tail.len() && self.tail == ARROW_STREAM_EOS
101    }
102}
103
104impl<R: Read> Read for EosTrackingReader<R> {
105    /// Read from the wrapped stream while retaining its final EOS-sized suffix.
106    fn read(&mut self, buffer: &mut [u8]) -> std::io::Result<usize> {
107        let count = self.inner.read(buffer)?;
108        let bytes = &buffer[..count];
109        let tail_len = self.tail.len();
110
111        if bytes.len() >= tail_len {
112            self.tail.copy_from_slice(&bytes[bytes.len() - tail_len..]);
113            self.tail_len = tail_len;
114        } else if !bytes.is_empty() {
115            let retained = self.tail_len.min(tail_len - bytes.len());
116            self.tail
117                .copy_within(self.tail_len - retained..self.tail_len, 0);
118            self.tail[retained..retained + bytes.len()].copy_from_slice(bytes);
119            self.tail_len = retained + bytes.len();
120        }
121
122        Ok(count)
123    }
124}

239#[derive(Debug, Clone, Copy, PartialEq, Eq)]
240enum BoundaryKind {
241    Begin,
242    End,
243}
244
245#[derive(Debug, Clone, Copy, PartialEq, Eq)]
246struct Boundary {
247    timestamp_ns: u64,
248    span_index: usize,
249    kind: BoundaryKind,
250}
251
252impl Ord for Boundary {
253    fn cmp(&self, other: &Self) -> Ordering {
254        self.timestamp_ns
255            .cmp(&other.timestamp_ns)
256            .then_with(|| match (self.kind, other.kind) {
257                (BoundaryKind::Begin, BoundaryKind::End) => Ordering::Greater,
258                (BoundaryKind::End, BoundaryKind::Begin) => Ordering::Less,
259                (BoundaryKind::Begin, BoundaryKind::Begin) => {
260                    self.span_index.cmp(&other.span_index)
261                }
262                (BoundaryKind::End, BoundaryKind::End) => other.span_index.cmp(&self.span_index),
263            })
264    }
265}
266
267impl PartialOrd for Boundary {
268    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
269        Some(self.cmp(other))
270    }

273/// Export a completed telemetry interval to a Perfetto trace.
274///
275/// `output` selects the destination file. When it is absent, the trace is written
276/// under `/tmp/$USER/monarch_profiles`. The destination is never overwritten.
277pub fn export_profile(
278    telemetry_url: &str,
279    start_us: i64,
280    end_us: i64,
281    output: Option<PathBuf>,
282) -> Result<PathBuf> {
283    if start_us < 0 || end_us <= start_us {
284        bail!("profile interval must satisfy 0 <= start_us < end_us");
285    }
286
287    let output = resolve_output_path(output, start_us)?;
288    if output.exists() {
289        bail!("output already exists: {}", output.display());
290    }
291    export_profile_to_path(telemetry_url, start_us, end_us, output)
292}

294fn export_profile_to_path(
295    telemetry_url: &str,
296    start_us: i64,
297    end_us: i64,
298    output: PathBuf,
299) -> Result<PathBuf> {
300    std::thread::sleep(DRAIN_DELAY);
301    // Each span falls into 1 of 4 cases:
302    //
303    //         |-- profile window --|
304    // 1.             |----|
305    // 2.             |--------------
306    // 3. |-----------------|
307    // 4. |--------------------------
308    //
309    // 1. Starts inside, ends inside
310    // 2. Starts inside, ends after
311    // 3. Starts before, ends inside
312    // 4. Starts before, ends after
313    //
314    // `windows_spans` contains cases 1-3 because it is able to match on at least one boundary in the window
315    let window_spans = query_rows(telemetry_url, &profile_sql(start_us, end_us))?;
316    // `spans_active_at_start` is intended to contain case 4 but also case 3
317    let spans_active_at_start = query_spans_active_at_start(telemetry_url, start_us)?;
318
319    // `stitch_spans` deduplicates starts from case 3 that are matched by both `window_spans`
320    // and `spans_active_at_start`
321    let rows = stitch_spans(window_spans, spans_active_at_start);
322
323    let summary = write_trace_to_output(rows, start_us, end_us, &output)?;
324
325    eprintln!(
326        "Wrote {} spans from {} actors on {} procs.",
327        summary.span_count, summary.actor_count, summary.proc_count
328    );
329
330    let skipped_count = summary.skipped_count();
331    if skipped_count > 0 {
332        eprintln!("Skipped {skipped_count} rows:");
333        for (reason, count) in summary.skipped_counts {
334            eprintln!("  {count} {}", reason.description());
335        }
336    }
337
338    Ok(output)
339}

352/// Return spans active immediately before `start_us`.
353/// Each returned row has its original start and no end.
354fn query_spans_active_at_start(telemetry_url: &str, start_us: i64) -> Result<Vec<SpanRow>> {
355    if !supports_streaming_queries(telemetry_url)? {
356        eprintln!(
357            "Telemetry server does not support streaming queries; omitting spans active at the start of the profile window."
358        );
359        return Ok(Vec::new());
360    }
361
362    let spans_active_at_start = span_ids_active_at_start(telemetry_url, start_us)?;
363
364    if spans_active_at_start.is_empty() {
365        return Ok(Vec::new());
366    }
367
368    let sql = spans_by_id_sql(spans_active_at_start.keys().cloned());
369
370    query_rows::<SpanMetadataRow>(telemetry_url, &sql)?
371        .into_iter()
372        .map(|span| {
373            let key = SpanKey {
374                process_id: span.process_id.clone(),
375                id: span.id,
376            };
377
378            let start_us = spans_active_at_start
379                .get(&key)
380                .copied()
381                .context("span metadata query returned an unexpected span")?;
382
383            Ok(SpanRow {
384                process_id: span.process_id,
385                id: span.id,
386                name: span.name,
387                target: span.target,
388                fields_json: span.fields_json,
389                start_us: Some(start_us),
390                end_us: None,
391            })
392        })
393        .collect()
394}

396/// Return whether the telemetry server supports streamed Arrow query results.
397fn supports_streaming_queries(telemetry_url: &str) -> Result<bool> {
398    let client = telemetry_client(Some(QUERY_TIMEOUT))?;
399    let url = format!("{}/api/query", telemetry_url.trim_end_matches('/'));
400
401    let response = client
402        .post(&url)
403        .header(ACCEPT, ARROW_STREAM_CONTENT_TYPE)
404        .json(&QueryRequest { sql: "SELECT 1" })
405        .send()
406        .with_context(|| format!("failed to query {url}"))?;
407
408    let status = response.status();
409    if status == StatusCode::NOT_ACCEPTABLE || status == StatusCode::NOT_IMPLEMENTED {
410        return Ok(false);
411    }
412    if !status.is_success() {
413        let body = response.text().with_context(|| {
414            format!("failed to read telemetry query response with status {status}")
415        })?;
416        bail!("telemetry query failed with {status}: {body}");
417    }
418
419    let content_type = response
420        .headers()
421        .get(CONTENT_TYPE)
422        .and_then(|value| value.to_str().ok())
423        .unwrap_or_default();
424
425    let supports_streaming = content_type.starts_with(ARROW_STREAM_CONTENT_TYPE);
426    if supports_streaming {
427        response
428            .bytes()
429            .context("failed to finish telemetry streaming capability query")?;
430    }
431
432    Ok(supports_streaming)
433}

435/// Return retained spans active immediately before `start_us`, keyed by span ID.
436/// Each value is the span's original enter timestamp.
437fn span_ids_active_at_start(telemetry_url: &str, start_us: i64) -> Result<BTreeMap<SpanKey, i64>> {
438    let client = telemetry_client(Some(QUERY_TIMEOUT))?;
439
440    let sql = format!(
441        r#"
442            SELECT process_id, id, timestamp_us, event_type
443            FROM span_events
444            WHERE timestamp_us < {start_us}
445                AND event_type IN ('enter', 'exit')
446        "#
447    );
448
449    let url = format!("{}/api/query", telemetry_url.trim_end_matches('/'));
450
451    let response = client
452        .post(&url)
453        .header(ACCEPT, ARROW_STREAM_CONTENT_TYPE)
454        .json(&QueryRequest { sql: &sql })
455        .send()
456        .with_context(|| format!("failed to query {url}"))?;
457
458    let status = response.status();
459
460    if !status.is_success() {
461        let body = response.text().with_context(|| {
462            format!("failed to read telemetry query response with status {status}")
463        })?;
464        bail!("telemetry query failed with {status}: {body}");
465    }
466
467    let content_type = response
468        .headers()
469        .get(CONTENT_TYPE)
470        .and_then(|value| value.to_str().ok())
471        .unwrap_or_default();
472
473    if !content_type.starts_with(ARROW_STREAM_CONTENT_TYPE) {
474        bail!(
475            "telemetry query returned {content_type:?} instead of an Arrow stream; restart the job with a streaming telemetry server"
476        );
477    }
478
479    let mut unmatched_enters = HashMap::new();
480
481    let mut unmatched_exits = HashSet::new();
482
483    let mut response = EosTrackingReader::new(response);
484
485    {
486        let reader = StreamReader::try_new(&mut response, None)
487            .context("failed to decode telemetry Arrow stream")?;
488
489        for batch in reader {
490            replay_span_event_batch(batch?, &mut unmatched_enters, &mut unmatched_exits)?;
491        }
492    }
493
494    if !response.ended_with_eos() {
495        bail!("telemetry Arrow stream ended without a completion marker");
496    }
497
498    Ok(unmatched_enters.into_iter().collect())
499}

501/// Update unmatched span entries and exits from one Arrow batch.
502/// Rows may arrive in any order and must contain non-null lifecycle columns.
503fn replay_span_event_batch(
504    batch: RecordBatch,
505    unmatched_enters: &mut HashMap<SpanKey, i64>,
506    unmatched_exits: &mut HashSet<SpanKey>,
507) -> Result<()> {
508    let process_ids = typed_column::<StringArray>(&batch, "process_id")?;
509
510    let ids = typed_column::<UInt64Array>(&batch, "id")?;
511
512    let timestamps = typed_column::<Int64Array>(&batch, "timestamp_us")?;
513
514    let event_types = typed_column::<StringArray>(&batch, "event_type")?;
515
516    for index in 0..batch.num_rows() {
517        if process_ids.is_null(index)
518            || ids.is_null(index)
519            || timestamps.is_null(index)
520            || event_types.is_null(index)
521        {
522            bail!("span event row contains null values");
523        }
524
525        let key = SpanKey {
526            process_id: process_ids.value(index).to_owned(),
527            id: ids.value(index),
528        };
529
530        match event_types.value(index) {
531            "enter" => {
532                if !unmatched_exits.remove(&key) {
533                    unmatched_enters
534                        .entry(key)
535                        .and_modify(|timestamp_us| {
536                            *timestamp_us = (*timestamp_us).min(timestamps.value(index));
537                        })
538                        .or_insert_with(|| timestamps.value(index));
539                }
540            }
541            "exit" => {
542                if unmatched_enters.remove(&key).is_none() {
543                    unmatched_exits.insert(key);
544                }
545            }
546            event_type => bail!("unexpected span event type {event_type:?}"),
547        }
548    }
549
550    Ok(())
551}

553/// Return a named Arrow column as `T`, with context for missing or mismatched columns.
554fn typed_column<'a, T: 'static>(batch: &'a RecordBatch, name: &str) -> Result<&'a T> {
555    batch
556        .column_by_name(name)
557        .with_context(|| format!("telemetry Arrow stream is missing column {name:?}"))?
558        .as_any()
559        .downcast_ref::<T>()
560        .with_context(|| format!("telemetry Arrow column {name:?} has an unexpected type"))
561}

563/// Execute SQL and deserialize the JSON response rows as `T`.
564fn query_rows<T: DeserializeOwned>(telemetry_url: &str, sql: &str) -> Result<Vec<T>> {
565    let client = telemetry_client(Some(QUERY_TIMEOUT))?;
566
567    let response = {
568        let url = format!("{}/api/query", telemetry_url.trim_end_matches('/'));
569
570        client
571            .post(&url)
572            .json(&QueryRequest { sql })
573            .send()
574            .with_context(|| format!("failed to query {url}"))
575    }?;
576
577    let status = response.status();
578
579    let body = response
580        .text()
581        .with_context(|| format!("failed to read telemetry query response with status {status}"))?;
582
583    if !status.is_success() {
584        bail!("telemetry query failed with {status}: {body}");
585    }
586
587    let envelope: QueryEnvelope<T> =
588        serde_json::from_str(&body).context("failed to decode telemetry query response")?;
589
590    if let Some(error) = envelope.error {
591        bail!("telemetry query failed: {error}");
592    }
593
594    Ok(envelope.rows.unwrap_or_default())
595}

597fn profile_sql(start_us: i64, end_us: i64) -> String {
598    format!(
599        r#"
600WITH window_events AS (
601    SELECT process_id, id, timestamp_us, event_type
602    FROM span_events
603    WHERE timestamp_us >= {start_us}
604      AND timestamp_us < {end_us}
605      AND event_type IN ('enter', 'exit')
606),
607profile_events AS (
608    SELECT
609        process_id,
610        id,
611        MIN(CASE WHEN event_type = 'enter' THEN timestamp_us END) AS start_us,
612        MAX(CASE WHEN event_type = 'exit' THEN timestamp_us END) AS end_us
613    FROM window_events
614    GROUP BY process_id, id
615)
616SELECT
617    s.process_id,
618    s.id,
619    s.name,
620    s.target,
621    s.fields_json,
622    e.start_us,
623    e.end_us
624FROM profile_events e
625JOIN spans s ON s.process_id = e.process_id AND s.id = e.id"#
626    )
627}

629/// Build a query for the metadata of a nonempty collection of span IDs.
630fn spans_by_id_sql(span_ids: impl IntoIterator<Item = SpanKey>) -> String {
631    let span_ids_sql = span_ids
632        .into_iter()
633        .map(|key| {
634            format!(
635                "({}, CAST({} AS BIGINT UNSIGNED))",
636                sql_string_literal(&key.process_id),
637                key.id,
638            )
639        })
640        .collect::<Vec<_>>()
641        .join(",\n        ");
642
643    format!(
644        r#"
645        WITH span_ids (process_id, id) AS (
646            VALUES
647                {span_ids_sql}
648        )
649        SELECT
650            s.process_id,
651            s.id,
652            s.name,
653            s.target,
654            s.fields_json
655        FROM span_ids i
656        JOIN spans s ON s.process_id = i.process_id AND s.id = i.id
657    "#
658    )
659}

661/// Return one row for every span intersecting the profile window.
662/// For a span active at the window start that exits inside it, the returned row
663/// combines the earlier start with the in-window end.
664fn stitch_spans(
665    mut window_spans: Vec<SpanRow>,
666    spans_active_at_start: Vec<SpanRow>,
667) -> Vec<SpanRow> {
668    let mut spans_active_at_start = spans_active_at_start
669        .into_iter()
670        .map(|span| {
671            (
672                SpanKey {
673                    process_id: span.process_id.clone(),
674                    id: span.id,
675                },
676                span,
677            )
678        })
679        .collect::<HashMap<_, _>>();
680
681    for span in &mut window_spans {
682        let key = SpanKey {
683            process_id: span.process_id.clone(),
684            id: span.id,
685        };
686
687        if let Some(active_span) = spans_active_at_start.remove(&key)
688            && span.start_us.is_none()
689        {
690            span.start_us = active_span.start_us;
691        }
692    }
693
694    window_spans.extend(spans_active_at_start.into_values());
695    window_spans
696}

716fn write_trace(
717    rows: Vec<SpanRow>,
718    window_start_us: i64,
719    window_end_us: i64,
720    output: fs::File,
721) -> Result<TraceSummary> {
722    let row_count = rows.len();
723    let (mut spans, skipped_counts) = rows
724        .into_iter()
725        .map(|row| prepare_span(row, window_start_us, window_end_us))
726        .fold(
727            (Vec::with_capacity(row_count), BTreeMap::new()),
728            |(mut spans, mut skipped_counts), result| {
729                match result {
730                    Ok(span) => spans.push(span),
731                    Err(reason) => *skipped_counts.entry(reason).or_default() += 1,
732                }
733                (spans, skipped_counts)
734            },
735        );
736
737    spans.sort_by(|left, right| {
738        left.track
739            .cmp(&right.track)
740            .then_with(|| left.original_start_us.cmp(&right.original_start_us))
741            .then_with(|| {
742                right
743                    .original_end_us
744                    .unwrap_or(i64::MAX)
745                    .cmp(&left.original_end_us.unwrap_or(i64::MAX))
746            })
747            .then_with(|| left.id.cmp(&right.id))
748    });
749
750    let proc_ids = spans
751        .iter()
752        .map(|span| span.track.proc_id.clone())
753        .collect::<BTreeSet<_>>();
754
755    let actor_tracks = spans
756        .iter()
757        .map(|span| span.track.clone())
758        .collect::<BTreeSet<_>>();
759
760    let process_count = i32::try_from(proc_ids.len()).context("too many process tracks")?;
761
762    let collector = Collector::from_file(output);
763    let mut ctx = Ctx::new(collector);
764
765    let mut process_track_ids = HashMap::new();
766    for (pid, proc_id) in (1..=process_count).zip(&proc_ids) {
767        let track_id = ctx.new_process_with_name(pid, proc_id.clone());
768        process_track_ids.insert(proc_id.clone(), track_id);
769    }
770
771    let mut actor_track_ids = HashMap::new();
772    for actor_track in &actor_tracks {
773        let process_track = process_track_ids
774            .get(&actor_track.proc_id)
775            .copied()
776            .expect("each actor should have a registered process track");
777
778        let track_id = ctx.next_uuid();
779
780        ctx.new_track(track_id)
781            .name(&actor_track.actor_id)
782            .parent(process_track)
783            .consume();
784
785        actor_track_ids.insert(actor_track.clone(), track_id);
786    }
787
788    let flow_plan = build_flow_plan(&spans, || ctx.next_uuid());
789
790    for boundary in span_boundaries(&spans) {
791        let span = &spans[boundary.span_index];
792        let track_id = actor_track_ids
793            .get(&span.track)
794            .copied()
795            .expect("each span should have a registered actor track");
796
797        match boundary.kind {
798            BoundaryKind::Begin => {
799                let flow_edits = flow_plan.get(&boundary.span_index);
800                let mut event = ctx
801                    .start_slice(track_id, boundary.timestamp_ns)
802                    .name(&span.name);
803
804                event = event.add_annotation("process_id", &Value::from(span.process_id.clone()));
805                event = event.add_annotation("span_id", &Value::from(span.id.to_string()));
806                event = event.add_annotation("target", &Value::from(span.target.clone()));
807
808                if span.start_clipped {
809                    event = event.add_annotation("profile.start_clipped", &Value::Bool(true));
810                    event = event.add_annotation(
811                        "profile.original_start_us",
812                        &Value::from(span.original_start_us),
813                    );
814                }
815
816                if span.end_clipped {
817                    event = event.add_annotation("profile.end_clipped", &Value::Bool(true));
818                    if let Some(original_end_us) = span.original_end_us {
819                        event = event.add_annotation(
820                            "profile.original_end_us",
821                            &Value::from(original_end_us),
822                        );
823                    } else {
824                        event =
825                            event.add_annotation("profile.end_event_missing", &Value::Bool(true));
826                    }
827                }
828                for (name, value) in &span.fields {
829                    event = event.add_annotation(name, value);
830                }
831                if let Some(flow_edits) = flow_edits {
832                    event =
833                        event.with_terminating_flow_ids(&flow_edits.request_terminating_flow_ids);
834                }
835                event.consume();
836
837                if let Some(flow_edits) = flow_edits
838                    && !flow_edits.send_flow_ids.is_empty()
839                {
840                    ctx.instant(track_id, boundary.timestamp_ns)
841                        .name("send")
842                        .add_annotation(
843                            "correlation_id",
844                            span.fields
845                                .get("correlation_id")
846                                .expect("a caller with flows should have a correlation id"),
847                        )
848                        .with_flow_ids(&flow_edits.send_flow_ids)
849                        .consume();
850                }
851            }
852            BoundaryKind::End => {
853                let flow_edits = flow_plan.get(&boundary.span_index);
854                if let Some(flow_edits) = flow_edits
855                    && !flow_edits.complete_terminating_flow_ids.is_empty()
856                {
857                    ctx.instant(track_id, boundary.timestamp_ns)
858                        .name("complete")
859                        .add_annotation(
860                            "correlation_id",
861                            span.fields
862                                .get("correlation_id")
863                                .expect("a caller with flows should have a correlation id"),
864                        )
865                        .with_terminating_flow_ids(&flow_edits.complete_terminating_flow_ids)
866                        .consume();
867                }
868
869                let mut event = ctx.end_slice(track_id, boundary.timestamp_ns);
870                if let Some(flow_edits) = flow_edits {
871                    event = event.with_flow_ids(&flow_edits.response_flow_ids);
872                }
873                event.consume();
874            }
875        }
876    }
877
878    let summary = TraceSummary {
879        span_count: spans.len(),
880        actor_count: actor_tracks.len(),
881        proc_count: proc_ids.len(),
882        skipped_counts,
883    };
884
885    let mut collector = ctx.sink();
886
887    collector.flush()?;
888
889    Ok(summary)
890}
891
892fn span_boundaries(spans: &[PreparedSpan]) -> Vec<Boundary> {
893    let mut boundaries = spans
894        .iter()
895        .enumerate()
896        .flat_map(|(span_index, span)| {
897            std::iter::once(Boundary {
898                timestamp_ns: span.start_ns,
899                span_index,
900                kind: BoundaryKind::Begin,
901            })
902            .chain(span.end_ns.map(|timestamp_ns| Boundary {
903                timestamp_ns,
904                span_index,
905                kind: BoundaryKind::End,
906            }))
907        })
908        .collect::<Vec<_>>();
909
910    boundaries.sort_unstable();
911    boundaries
912}

914fn build_flow_plan(
915    spans: &[PreparedSpan],
916    mut next_flow_id: impl FnMut() -> u64,
917) -> HashMap<usize, SpanFlowEdits> {
918    let mut correlated_spans: HashMap<u64, CorrelatedSpans> = HashMap::new();
919
920    for (span_index, span) in spans.iter().enumerate() {
921        let Some(correlation_id) = span.fields.get("correlation_id").and_then(Value::as_u64) else {
922            continue;
923        };
924
925        let group = correlated_spans.entry(correlation_id).or_default();
926        if span.target == ENDPOINT_TELEMETRY_TARGET {
927            group.callers.push(span_index);
928        } else if span.target == USER_TELEMETRY_TARGET {
929            group.receivers.push(span_index);
930        }
931    }
932
933    let mut flow_plan: HashMap<usize, SpanFlowEdits> = HashMap::new();
934    for group in correlated_spans.into_values() {
935        let [caller_index] = group.callers.as_slice() else {
936            continue;
937        };
938        let caller = &spans[*caller_index];
939        for receiver_index in group.receivers {
940            let receiver = &spans[receiver_index];
941
942            if !caller.start_clipped && !receiver.start_clipped {
943                let flow_id = next_flow_id();
944                flow_plan
945                    .entry(*caller_index)
946                    .or_default()
947                    .send_flow_ids
948                    .push(flow_id);
949                flow_plan
950                    .entry(receiver_index)
951                    .or_default()
952                    .request_terminating_flow_ids
953                    .push(flow_id);
954            }
955
956            if caller.returns_response && !caller.end_clipped && !receiver.end_clipped {
957                let flow_id = next_flow_id();
958                flow_plan
959                    .entry(receiver_index)
960                    .or_default()
961                    .response_flow_ids
962                    .push(flow_id);
963                flow_plan
964                    .entry(*caller_index)
965                    .or_default()
966                    .complete_terminating_flow_ids
967                    .push(flow_id);
968            }
969        }
970    }
971
972    flow_plan
973}

975fn prepare_span(
976    row: SpanRow,
977    window_start_us: i64,
978    window_end_us: i64,
979) -> Result<PreparedSpan, SpanSkipReason> {
980    let original_start_us = row.start_us.ok_or(SpanSkipReason::MissingStartTimestamp)?;
981    let original_end_us = row.end_us;
982
983    let start_clipped = original_start_us < window_start_us;
984    let end_clipped = original_end_us.is_none_or(|end_us| end_us > window_end_us);
985
986    let start_us = original_start_us.max(window_start_us);
987    if start_us >= window_end_us {
988        return Err(SpanSkipReason::StartsOutsideProfileWindow);
989    }
990
991    let end_us = original_end_us.map(|end_us| end_us.min(window_end_us));
992    if end_us.is_some_and(|end_us| end_us <= start_us) {
993        return Err(SpanSkipReason::NonPositiveDuration);
994    }
995
996    let fields = match serde_json::from_str::<Value>(&row.fields_json) {
997        Ok(Value::Object(fields)) => fields,
998        Ok(_) => return Err(SpanSkipReason::FieldsJsonNotObject),
999        Err(_) => return Err(SpanSkipReason::InvalidFieldsJson),
1000    };
1001    let actor_id = match fields.get("actor_id") {
1002        Some(Value::String(actor_id)) => actor_id,
1003        Some(_) => return Err(SpanSkipReason::ActorIdNotString),
1004        None => return Err(SpanSkipReason::MissingActorId),
1005    };
1006    let track = parse_actor_track(actor_id).ok_or(SpanSkipReason::ActorIdMissingProcessId)?;
1007
1008    let start_ns = micros_to_nanos(start_us).map_err(|_| SpanSkipReason::InvalidStartTimestamp)?;
1009    let end_ns = end_us
1010        .map(micros_to_nanos)
1011        .transpose()
1012        .map_err(|_| SpanSkipReason::InvalidEndTimestamp)?;
1013    let returns_response = row.target == ENDPOINT_TELEMETRY_TARGET
1014        && matches!(row.name.as_str(), "call" | "call_one" | "choose");
1015    let name = display_name(&row, &fields);
1016
1017    Ok(PreparedSpan {
1018        process_id: row.process_id,
1019        id: row.id,
1020        name,
1021        target: row.target,
1022        fields,
1023        track,
1024        original_start_us,
1025        original_end_us,
1026        start_clipped,
1027        end_clipped,
1028        returns_response,
1029        start_ns,
1030        end_ns,
1031    })
1032}

1034fn display_name(row: &SpanRow, fields: &Map<String, Value>) -> String {
1035    if row.target == ENDPOINT_TELEMETRY_TARGET {
1036        let mesh = fields.get("mesh").and_then(Value::as_str);
1037        let method = fields.get("method").and_then(Value::as_str);
1038        let call_name = fields.get("call_name").and_then(Value::as_str);
1039
1040        return match (mesh, method, call_name) {
1041            (Some(mesh), Some(method), _) => format!("{mesh}.{method}.{}()", row.name),
1042            (_, _, Some(call_name)) if !call_name.is_empty() => {
1043                format!("{call_name}.{}()", row.name)
1044            }
1045            _ => row.name.clone(),
1046        };
1047    }
1048
1049    fields
1050        .get("name")
1051        .and_then(Value::as_str)
1052        .unwrap_or(&row.name)
1053        .to_string()
1054}

1056fn parse_actor_track(actor_addr: &str) -> Option<ActorTrack> {
1057    let actor_addr = actor_addr
1058        .rsplit_once(',')
1059        .map(|(_, suffix)| suffix)
1060        .unwrap_or(actor_addr)
1061        .trim();
1062
1063    let actor_id = actor_addr
1064        .split_once('@')
1065        .map(|(id, _)| id)
1066        .unwrap_or(actor_addr);
1067    // ActorId displays as `<actor_uid>.<proc_id>`, so the first component is
1068    // the actor and the complete remainder is the process ID.
1069    let (_, proc_id) = actor_id.split_once('.')?;
1070
1071    Some(ActorTrack {
1072        proc_id: proc_id.to_string(),
1073        actor_id: actor_id.to_string(),
1074    })
1075}

1077fn micros_to_nanos(timestamp_us: i64) -> Result<u64> {
1078    u64::try_from(timestamp_us)
1079        .context("trace timestamp is before the Unix epoch")?
1080        .checked_mul(1_000)
1081        .context("trace timestamp exceeds the Perfetto range")
1082}

1084fn resolve_output_path(output: Option<PathBuf>, filename_timestamp_us: i64) -> Result<PathBuf> {
1085    let output = match output {
1086        Some(output) => output,
1087        None => {
1088            let user = std::env::var("USER")
1089                .ok()
1090                .filter(|user| !user.is_empty())
1091                .context("USER is not set; pass an explicit output path")?;
1092
1093            PathBuf::from("/tmp")
1094                .join(user)
1095                .join("monarch_profiles")
1096                .join(format!("monarch-profile-{filename_timestamp_us}.pftrace"))
1097        }
1098    };
1099
1100    if output.is_absolute() {
1101        return Ok(output);
1102    }
1103
1104    Ok(std::env::current_dir()
1105        .context("failed to get the current directory")?
1106        .join(output))
1107}

1109fn create_output_file(output: &Path) -> Result<fs::File> {
1110    create_output_parent(output)?;
1111
1112    match fs::OpenOptions::new()
1113        .write(true)
1114        .create_new(true)
1115        .open(output)
1116    {
1117        Ok(file) => Ok(file),
1118        Err(error) if error.kind() == ErrorKind::AlreadyExists => {
1119            bail!("output already exists: {}", output.display())
1120        }
1121        Err(error) => Err(error).with_context(|| format!("failed to create {}", output.display())),
1122    }
1123}
1124
1125fn create_output_parent(output: &Path) -> Result<()> {
1126    if let Some(parent) = output.parent()
1127        && !parent.as_os_str().is_empty()
1128    {
1129        fs::create_dir_all(parent)
1130            .with_context(|| format!("failed to create {}", parent.display()))?;
1131    }
1132
1133    Ok(())
1134}
