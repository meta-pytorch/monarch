9//! Message types for remote supervision sessions.
10//!
11//! A remote supervision relationship is represented by one worker actor
12//! supervising one child actor on behalf of one supervisor. The supervisor
13//! initiates the relationship by sending [`Supervise`] to the worker. The
14//! `supervisor` port in [`Supervise`] identifies the session endpoint that the
15//! worker uses for all replies and supervision events. The `liveness` spec in
16//! [`Supervise`] identifies the implementation actor that the worker spawns as
17//! the session liveness mechanism. Later supervisor commands are sent to the
18//! worker as [`WorkerCommand`] messages and carry the same `session_id`; the
19//! worker accepts only the active session.

42/// Global actor spawn (gspawn) specification.
43#[derive(Clone, Debug, Serialize, Deserialize, Named, PartialEq, Eq)]
44pub struct Gspawn {
45    actor_type: String,
46    uid: Uid,
47    // Encoded as a byte string rather than a sequence of `u8`s. This is
48    // the same bincode encoding, but avoids per-byte serde calls.
49    #[serde(with = "serde_bytes")]
50    params: Data,
51    environment: ActorEnvironment,
52}
53wirevalue::register_type!(Gspawn);

55/// Spawn parameters for actor type `A`, serialized once so that many spawn
56/// requests can reuse them without re-encoding.
57pub struct SerializedParams<A> {
58    data: Data,
59    _actor: PhantomData<fn() -> A>,
60}
61
62impl<A: RemoteSpawn> SerializedParams<A> {
63    /// Serialize `params` for spawning actors of type `A`.
64    pub fn new(params: &A::Params) -> anyhow::Result<Self> {
65        Ok(Self {
66            data: bincode::serde::encode_to_vec(params, bincode::config::legacy())?,
67            _actor: PhantomData,
68        })
69    }
70}
71
72impl<A> Clone for SerializedParams<A> {
73    fn clone(&self) -> Self {
74        Self {
75            data: self.data.clone(),
76            _actor: PhantomData,
77        }
78    }
79}

81impl Gspawn {
82    /// Create a spawn specification for the registered actor type with a fresh
83    /// uid and the caller's persistent environment.
84    pub fn for_actor<A>(cx: &impl context::Actor, params: A::Params) -> anyhow::Result<Self>
85    where
86        A: RemoteSpawn + Named,
87    {
88        Self::for_actor_uid::<A>(cx, Uid::anonymous(), params)
89    }
90
91    /// Create a spawn specification for the registered actor type with an
92    /// explicit uid and the caller's persistent environment.
93    pub fn for_actor_uid<A>(
94        cx: &impl context::Actor,
95        uid: Uid,
96        params: A::Params,
97    ) -> anyhow::Result<Self>
98    where
99        A: RemoteSpawn + Named,
100    {
101        Ok(Self::for_actor_uid_in_environment::<A>(
102            uid,
103            SerializedParams::new(&params)?,
104            cx.instance().actor_environment().clone(),
105        ))
106    }
107
108    /// Create a spawn specification for the registered actor type with an
109    /// explicit uid and a caller-shaped persistent environment (AENV-4).
110    pub(crate) fn for_actor_uid_in_environment<A>(
111        uid: Uid,
112        params: SerializedParams<A>,
113        environment: ActorEnvironment,
114    ) -> Self
115    where
116        A: RemoteSpawn + Named,
117    {
118        Self::with_uid(A::typename(), uid, params.data, environment)
119    }
120
121    pub(crate) fn with_uid(
122        actor_type: impl Into<String>,
123        uid: Uid,
124        params: Data,
125        environment: ActorEnvironment,
126    ) -> Self {
127        Self {
128            actor_type: actor_type.into(),
129            uid,
130            params,
131            environment,
132        }
133    }
134
135    /// Registered name of the actor to spawn.
136    pub fn actor_type(&self) -> &str {
137        &self.actor_type
138    }
139
140    /// The uid that will identify the spawned actor.
141    pub fn uid(&self) -> &Uid {
142        &self.uid
143    }
144
145    /// The serialized parameters passed to the actor.
146    pub fn params(&self) -> &[u8] {
147        &self.params
148    }
149
150    /// The persistent environment to install on the spawned actor.
151    pub fn actor_environment(&self) -> &ActorEnvironment {
152        &self.environment
153    }
154
155    /// Spawn the actor as a supervised child of `parent`.
156    pub async fn spawn_child<C: hyperactor::context::Actor>(
157        self,
158        parent: &C,
159    ) -> anyhow::Result<hyperactor::AnyActorHandle> {
160        let Self {
161            actor_type,
162            uid,
163            params,
164            environment,
165        } = self;
166        parent
167            .instance()
168            .gspawn_uid_in_environment(&actor_type, uid, params, environment)
169            .await
170    }
171}

306/// Messages sent by the worker actor to the supervisor session port.
307#[derive(Clone, Debug, Serialize, Deserialize, Named, PartialEq, Eq)]
308pub enum SupervisorEvent {
309    /// The worker accepted the session and identified its supervised child.
310    Linked {
311        /// Unique identifier for this supervision session.
312        session_id: Uid,
313        /// Worker actor that owns the supervised child.
314        worker: PortRef<WorkerCommand>,
315        /// Actor address for the supervised child.
316        child: ActorAddr,
317        /// Friendly display name for the supervised child, if available.
318        display_name: Option<String>,
319    },
320    /// The worker rejected the supervise request.
321    SuperviseRejected {
322        /// Unique identifier for this supervision session.
323        session_id: Uid,
324        /// Reason the supervise request failed.
325        reason: String,
326    },
327    /// A supervision event observed by, or synthesized for, the worker.
328    SupervisionEvent {
329        /// Unique identifier for this supervision session.
330        session_id: Uid,
331        /// Event to deliver to the supervisor side.
332        event: ActorSupervisionEvent,
333        /// What the worker knows about the remote actor state.
334        disposition: RemoteActorDisposition,
335    },
336    /// The worker unlinked the session without reporting a supervision event.
337    Unlinked {
338        /// Unique identifier for this supervision session.
339        session_id: Uid,
340        /// Reason for unlinking the session.
341        reason: String,
342    },
343}
344wirevalue::register_type!(SupervisorEvent);

346/// What a worker-side event establishes about the supervised child.

356#[cfg(test)]
357mod tests {
358    use wirevalue::Encoding;
359
360    use super::*;
361
362    /// [`Gspawn`] as encoded before its params were marked as bytes.
363    #[derive(Serialize, Deserialize, Debug, PartialEq)]
364    struct SeqGspawn {
365        actor_type: String,
366        uid: Uid,
367        params: Vec<u8>,
368        environment: ActorEnvironment,
369    }
370
371    #[test]
372    fn gspawn_params_wire_compatible() {
373        let params: Vec<u8> = (0..=255).collect();
374        let gspawn = Gspawn::with_uid(
375            "test",
376            Uid::anonymous(),
377            params.clone(),
378            ActorEnvironment::default(),
379        );
380        let seq = SeqGspawn {
381            actor_type: gspawn.actor_type.clone(),
382            uid: gspawn.uid.clone(),
383            params,
384            environment: gspawn.environment.clone(),
385        };
386
387        let bytes = bincode::serde::encode_to_vec(&gspawn, bincode::config::legacy()).unwrap();
388        assert_eq!(
389            bytes,
390            bincode::serde::encode_to_vec(&seq, bincode::config::legacy()).unwrap()
391        );
392        let (decoded, _): (SeqGspawn, _) =
393            bincode::serde::decode_from_slice(&bytes, bincode::config::legacy()).unwrap();
394        assert_eq!(decoded, seq);
395
396        assert_eq!(
397            serde_json::to_vec(&gspawn).unwrap(),
398            serde_json::to_vec(&seq).unwrap()
399        );
400
401        for encoding in [Encoding::Bincode, Encoding::Json, Encoding::Multipart] {
402            let old =
403                wirevalue::Any::serialize_with_encoding_as::<Gspawn, _>(encoding, &seq).unwrap();
404            assert_eq!(
405                old,
406                wirevalue::Any::serialize_with_encoding(encoding, &gspawn).unwrap(),
407                "{encoding}"
408            );
409            assert_eq!(old.deserialized::<Gspawn>().unwrap(), gspawn, "{encoding}");
410        }
411    }
