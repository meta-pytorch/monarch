9//! Message-based link implementation.
10//!
11//! The link has two actors. `KeepaliveWorker` runs on the worker side and owns
12//! the outbound keepalive stream. `KeepaliveSupervisor` runs on the supervisor
13//! side and acknowledges each keepalive. The worker sends [`Keepalive`] with a
14//! reply port for the worker's [`KeepaliveAck`] handler. The supervisor answers
15//! with [`KeepaliveAck`] on that reply port. Either side may fail the link: the
16//! worker fails if a keepalive is not acknowledged within `timeout`, and the
17//! supervisor fails if no newer keepalive is delivered within `timeout`.
18//!
19//! Keepalives carry monotonically increasing generation numbers. A generation
20//! identifies one worker-issued keepalive and the local timers derived from it.
21//! We use generations instead of canceling timers: if a later generation has
22//! already been observed, a delayed timer for an older generation is stale and
23//! is ignored. This relies on Hyperactor's in-order delivery guarantee for
24//! messages sent along a single actor-to-actor path. In particular, the
25//! supervisor observes keepalives from one worker in generation order, and each
26//! actor observes its own scheduled control messages in the order in which they
27//! become deliverable. The generation checks make late timers harmless, while
28//! ordered delivery preserves the meaning of "no newer generation arrived before
29//! this deadline."
30//!
31//! The worker uses two internal self messages. [`SendKeepalive`] advances the
32//! stream after `interval` if its generation is still current. [`AckDeadline`]
33//! checks whether the corresponding generation was acknowledged. The supervisor
34//! uses [`Deadline`] to check whether a newer generation arrived.
35//!
36//! Each side splits its `timeout` into a deadline, three quarters of it, and a
37//! grace period. An expired deadline fails the link only if the grace period,
38//! which starts when the actor handles the deadline, also passes without
39//! progress. A side that was itself stalled past its deadline, for example
40//! because its process was suspended, thus handles the messages that queued
41//! during the stall before it can fail the link: its own stall does not count
42//! against its peer.

86/// Split a failure-detection `timeout` into a deadline and a grace period that
87/// together span it. On a link whose interval is at most half its timeout, a
88/// newer keepalive supersedes each supervisor deadline before it expires.
89fn detection_phases(timeout: Duration) -> (Duration, Duration) {
90    let grace = timeout / 4;
91    (timeout - grace, grace)
92}

163    /// Spawn the supervisor side and return the worker-side link spec.
164    pub fn spawn_supervisor<C: context::Actor>(
165        self,
166        this: &C,
167    ) -> anyhow::Result<(ActorHandle<KeepaliveSupervisor>, LinkSpec)> {
168        self.spawn_supervisor_uid(this, Uid::anonymous())
169    }

171    /// Spawn the supervisor side and return the worker-side link spec,
172    /// used to instantiate the worker side of the link.
173    ///
174    /// The passed uid determines the uid of the worker side implementation
175    /// actor; it is specified by the supervisor to enable efficient group-style
176    /// supervision.
177    pub fn spawn_supervisor_uid<C: context::Actor>(
178        self,
179        this: &C,
180        uid: Uid,
181    ) -> anyhow::Result<(ActorHandle<KeepaliveSupervisor>, LinkSpec)> {
182        let params = self.params;
183        let supervisor =
184            this.instance()
185                .spawn(KeepaliveSupervisor::new(KeepaliveSupervisorParams::new(
186                    self.supervisor_timeout,
187                )));
188        let worker = KeepaliveWorkerParams::new(supervisor.port::<Keepalive>().bind(), params)
189            .link_spec_uid(uid)?;
190        Ok((supervisor, worker))
191    }

267/// Supervisor-side deadline for receiving a newer keepalive.
268///
269/// `generation` is the supervisor's current generation when the deadline is
270/// scheduled. If the supervisor still has the same current generation when this
271/// message is handled, no newer keepalive arrived before the deadline: the
272/// supervisor starts the grace period, or, if `grace` is set, fails the link.
273/// If the current generation is larger, this message is stale.
274#[derive(Clone, Debug, Serialize, Deserialize, Named)]
275struct Deadline {
276    generation: u64,
277    grace: bool,
278}
279wirevalue::register_type!(Deadline);

281/// Worker-side timer for sending the next keepalive.
282///
283/// `generation` is the keepalive generation that scheduled this timer. The
284/// worker sends the next keepalive only if this generation is still current. If
285/// the worker has already advanced, this message is stale.
286#[derive(Clone, Debug, Serialize, Deserialize, Named)]
287struct SendKeepalive {
288    generation: u64,
289}
290wirevalue::register_type!(SendKeepalive);

292/// Worker-side deadline for receiving a keepalive acknowledgment.
293///
294/// `generation` is the keepalive generation that scheduled this deadline. If
295/// the worker has not recorded an acknowledgment for this generation, or any
296/// later generation, when the message is handled, the worker starts the grace
297/// period, or, if `grace` is set, fails the link. If the acknowledged
298/// generation is equal or larger, this deadline is satisfied.
299#[derive(Clone, Debug, Serialize, Deserialize, Named)]
300struct AckDeadline {
301    generation: u64,
302    grace: bool,
303}
304wirevalue::register_type!(AckDeadline);

306/// Worker-side keepalive link actor.
307///
308/// `KeepaliveWorker` owns the generation sequence and handles supervisor
309/// [`KeepaliveAck`] messages directly. On startup, the worker sends generation
310/// 1 and schedules [`SendKeepalive`] and [`AckDeadline`] for that generation.
311/// Each live [`SendKeepalive`] advances to the next generation, sends a new
312/// [`Keepalive`], and schedules the next pair of timers. Each [`KeepaliveAck`]
313/// records progress. An [`AckDeadline`] and its grace period together prove
314/// that the matching generation failed to receive an acknowledgment before
315/// `timeout`, unless a same-or-newer acknowledgment has been recorded.
316#[derive(Debug)]
317#[hyperactor::export]
318pub struct KeepaliveWorker {
319    supervisor: PortRef<Keepalive>,
320    interval: Duration,
321    deadline: Duration,
322    grace: Duration,
323    generation: u64,
324    acked_generation: u64,
325}

377#[async_trait]
378impl Handler<AckDeadline> for KeepaliveWorker {
379    async fn handle(&mut self, cx: &Context<Self>, message: AckDeadline) -> anyhow::Result<()> {
380        if self.acked_generation >= message.generation {
381            return Ok(());
382        }
383        if !message.grace {
384            cx.post_after(
385                cx,
386                AckDeadline {
387                    generation: message.generation,
388                    grace: true,
389                },
390                self.grace,
391            );
392            return Ok(());
393        }
394        cx.kill("keepalive acknowledgment missed")?;
395        Ok(())
396    }
397}

399impl KeepaliveWorker {
400    fn send_keepalive(&mut self, this: &Instance<Self>) -> anyhow::Result<()> {
401        self.generation += 1;
402        let generation = self.generation;
403        let mut reply = this.port::<KeepaliveAck>().bind().into_once();
404        // The worker can stop before an in-flight keepalive is acknowledged.
405        reply.return_undeliverable(false);
406        (&self.supervisor).post(this, Keepalive { generation, reply });
407
408        this.post_after(this, SendKeepalive { generation }, self.interval);
409        this.post_after(
410            this,
411            AckDeadline {
412                generation,
413                grace: false,
414            },
415            self.deadline,
416        );
417        Ok(())
418    }
419}

421/// Supervisor-side keepalive link actor.
422///
423/// `KeepaliveSupervisor` records the maximum keepalive generation it has
424/// accepted and acknowledges every [`Keepalive`] with the same generation. Each
425/// accepted keepalive schedules a [`Deadline`] for the current generation. A
426/// deadline is active only while its generation remains current; if a newer
427/// keepalive has arrived, the deadline is stale.

429pub struct KeepaliveSupervisor {
430    deadline: Duration,
431    grace: Duration,
432    generation: u64,
433    supervision: Option<KeepaliveSupervision>,
434}

474#[async_trait]
475impl Handler<Deadline> for KeepaliveSupervisor {
476    async fn handle(&mut self, cx: &Context<Self>, message: Deadline) -> anyhow::Result<()> {
477        // A deadline is scheduled for the generation that was current at the
478        // time. Equality means no newer keepalive arrived before the deadline.
479        // A smaller deadline generation is stale. A larger deadline generation
480        // cannot be produced by this actor before it observes that generation.
481        if message.generation != self.generation {
482            return Ok(());
483        }
484        if !message.grace {
485            cx.post_after(
486                cx,
487                Deadline {
488                    generation: message.generation,
489                    grace: true,
490                },
491                self.grace,
492            );
493            return Ok(());
494        }
495        let reason = format!("keepalive missed for generation {}", message.generation);
496        if let Some(supervision) = &self.supervision {
497            let target = &supervision.target;
498            let event = match self.terminal_status(cx.instance(), &target.actor_id).await {
499                Some(status) => ActorSupervisionEvent::new(
500                    target.actor_id.clone(),
501                    target.display_name.clone(),
502                    status,
503                    None,
504                ),
505                None => ActorSupervisionEvent::new(
506                    cx.self_addr().clone(),
507                    None,
508                    ActorStatus::generic_failure(format!(
509                        "actor explicitly aborted due to: {reason}"
510                    )),
511                    None,
512                )
513                .with_transparent(true)
514                .with_observation(Some(SupervisionObservation::LinkLost(target.clone()))),
515            };
516            supervision.supervisor.post(
517                cx,
518                SupervisorEvent::SupervisionEvent {
519                    session_id: supervision.session_id.clone(),
520                    event,
521                    disposition: RemoteActorDisposition::Unreachable,
522                },
523            );
524            cx.exit("supervision link outcome reported")?;
525            return Ok(());
526        }
527        cx.kill(&reason)?;
528        Ok(())
529    }
530}

532impl KeepaliveSupervisor {
533    /// Create a supervisor-side keepalive actor.
534    pub fn new(params: KeepaliveSupervisorParams) -> Self {
535        let (deadline, grace) = detection_phases(params.timeout);
536        Self {
537            deadline,
538            grace,
539            generation: 0,
540            supervision: None,
541        }
542    }
543
544    async fn terminal_status(
545        &self,
546        this: &Instance<Self>,
547        target: &ActorAddr,
548    ) -> Option<ActorStatus> {
549        let (reply, receiver) = this.mailbox().open_once_port::<Option<ActorStatus>>();
550        let mut reply_ref = reply.bind();
551        reply_ref.return_undeliverable(false);
552        let mut status_port = target.status_port();
553        status_port.return_undeliverable(false);
554        status_port.post(
555            this,
556            StatusMessage::GetStatus {
557                reply: reply_ref,
558                delivery: None,
559            },
560        );
561        match tokio::time::timeout(LINK_LOST_STATUS_QUERY_TIMEOUT, receiver.recv()).await {
562            Ok(Ok(status)) => status.filter(|status| status.is_terminal()),
563            _ => None,
564        }
565    }
566
567    fn schedule_deadline(&self, this: &Instance<Self>) -> anyhow::Result<()> {
568        this.post_after(
569            this,
570            Deadline {
571                generation: self.generation,
572                grace: false,
573            },
574            self.deadline,
575        );
576        Ok(())
577    }

878    // A suspended process resumes with its expired deadlines and the messages
879    // that queued during the suspension both ready. Blocking the test's
880    // single-threaded runtime suspends every actor in it; a message posted
881    // after the actor handles its expired deadline stands in for one that
882    // queued in a socket while the actor's process was stopped.
883    #[tokio::test]
884    async fn test_keepalive_supervisor_survives_own_stall() {
885        let proc = Proc::isolated();
886        let client = proc.client("client");
887        let (events, mut event_rx) = client.open_port::<ActorSupervisionEvent>();
888        let (handle_tx, handle_rx) = tokio::sync::oneshot::channel();
889        let timeout = Duration::from_secs(1);
890        let parent: ActorHandle<ParentActor> = proc.spawn(ParentActor {
891            spawn: Some(ParentSpawn::ExposedSupervisor(
892                KeepaliveSupervisor::new(KeepaliveSupervisorParams::new(timeout)),
893                handle_tx,
894            )),
895            events: events.bind(),
896        });
897        let supervisor = handle_rx.await.unwrap();
898        let keepalive = |generation| {
899            let (reply, ack_rx) = client.open_once_port::<KeepaliveAck>();
900            supervisor.post(
901                &client,
902                Keepalive {
903                    generation,
904                    reply: reply.bind(),
905                },
906            );
907            ack_rx.recv()
908        };
909        assert_eq!(keepalive(1).await.unwrap().generation, 1);
910
911        std::thread::sleep(timeout + Duration::from_millis(200));
912        tokio::time::sleep(Duration::from_millis(10)).await;
913        assert_no_event(&mut event_rx);
914        assert_eq!(keepalive(2).await.unwrap().generation, 2);
915        tokio::time::sleep(timeout / 2).await;
916
917        assert_no_event(&mut event_rx);
918        parent.stop("test").unwrap();
919        parent.await;
920    }
