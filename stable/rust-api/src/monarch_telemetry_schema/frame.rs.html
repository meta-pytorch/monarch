9//! Telemetry frame codec shared by socket ingest and the persist log.
10//!
11//! A frame is a `u16` big-endian name length, the UTF-8 name, a `u32`
12//! big-endian payload length, and the payload. Data frames name a table and
13//! carry one Arrow IPC stream. A persist log's clean-close marker is an
14//! `__end` frame with an empty payload.
15
16use std::io::ErrorKind;
17use std::io::Read;
18
19use crate::MAX_FRAME_LEN;
20use crate::MAX_TABLE_NAME_LEN;
21use crate::write_frame_header;
22
23/// Empty control frame written only on clean close.
24pub const END_FRAME: &str = "__end";
25
26/// Whether `name` is reserved for control frames.
27pub fn is_reserved(name: &str) -> bool {
28    name.starts_with("__")
29}
30
31/// Validate a frame's table-name length.
32pub fn check_table_name_len(len: usize) -> anyhow::Result<()> {
33    if len == 0 || len > MAX_TABLE_NAME_LEN {
34        anyhow::bail!("invalid table name length {len}");
35    }
36    Ok(())
37}
38
39/// Validate a frame's payload length.
40pub fn check_payload_len(len: usize) -> anyhow::Result<()> {
41    if len == 0 || len > MAX_FRAME_LEN {
42        anyhow::bail!("invalid frame length {len}");
43    }
44    Ok(())
45}
46
47/// Append one complete frame to `buf`.
48pub fn encode_frame(buf: &mut Vec<u8>, name: &str, payload: &[u8]) -> anyhow::Result<()> {
49    write_frame_header(buf, name, payload.len())?;
50    buf.extend_from_slice(payload);
51    Ok(())
52}
53
54/// One undecoded frame.
55#[derive(Debug)]
56pub struct RawFrame {
57    /// Table or control-frame name.
58    pub name: String,
59    /// Frame payload.
60    pub payload: Vec<u8>,
61}
62
63/// Result of reading the next frame.
64#[derive(Debug)]
65pub enum ReadOutcome {
66    /// A complete frame.
67    Frame(RawFrame),
68    /// Clean end of input at a frame boundary.
69    End,
70    /// An incomplete or invalid frame starting at byte `offset`.
71    Torn { offset: u64 },
72}
73
74/// Sequential frame reader over a byte stream.
75pub struct FrameReader<R> {
76    reader: R,
77    offset: u64,
78}
79
80impl<R: Read> FrameReader<R> {
81    /// Read frames from `reader`, starting at offset zero.
82    pub fn new(reader: R) -> Self {
83        Self { reader, offset: 0 }
84    }
85
86    /// Bytes consumed so far, which is where the next frame starts.
87    pub fn offset(&self) -> u64 {
88        self.offset
89    }
90
91    /// Read the next frame, reporting a torn tail instead of failing.
92    pub fn read_frame(&mut self) -> anyhow::Result<ReadOutcome> {
93        let start = self.offset;
94        let torn = ReadOutcome::Torn { offset: start };
95
96        let mut name_len = [0; 2];
97        match self.fill(&mut name_len)? {
98            0 => return Ok(ReadOutcome::End),
99            2 => {}
100            _ => return Ok(torn),
101        }
102        let name_len = u16::from_be_bytes(name_len) as usize;
103        if check_table_name_len(name_len).is_err() {
104            return Ok(torn);
105        }
106
107        let mut name = vec![0; name_len];
108        if self.fill(&mut name)? != name_len {
109            return Ok(torn);
110        }
111        let Ok(name) = String::from_utf8(name) else {
112            return Ok(torn);
113        };
114
115        let mut payload_len = [0; 4];
116        if self.fill(&mut payload_len)? != 4 {
117            return Ok(torn);
118        }
119        let payload_len = u32::from_be_bytes(payload_len) as usize;
120        if (name != END_FRAME || payload_len != 0) && check_payload_len(payload_len).is_err() {
121            return Ok(torn);
122        }
123
124        let mut payload = vec![0; payload_len];
125        if self.fill(&mut payload)? != payload_len {
126            return Ok(torn);
127        }
128        Ok(ReadOutcome::Frame(RawFrame { name, payload }))
129    }
130
131    /// Read until `buf` is full or input ends, returning the bytes read.
132    fn fill(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
133        let mut filled = 0;
134        while filled < buf.len() {
135            match self.reader.read(&mut buf[filled..]) {
136                Ok(0) => break,
137                Ok(n) => filled += n,
138                Err(error) if error.kind() == ErrorKind::Interrupted => {}
139                Err(error) => return Err(error),
140            }
141        }
142        self.offset += filled as u64;
143        Ok(filled)
144    }
145}

147#[cfg(test)]
148mod tests {
149    use std::io::Cursor;
150
151    use super::*;
152
153    fn read_all(bytes: Vec<u8>) -> Vec<ReadOutcome> {
154        let mut reader = FrameReader::new(Cursor::new(bytes));
155        let mut outcomes = Vec::new();
156        loop {
157            let outcome = reader.read_frame().unwrap();
158            let done = !matches!(outcome, ReadOutcome::Frame(_));
159            outcomes.push(outcome);
160            if done {
161                return outcomes;
162            }
163        }
164    }
165
166    #[test]
167    fn encode_then_read_round_trips() {
168        let mut buf = Vec::new();
169        encode_frame(&mut buf, "t", &[1, 2, 3]).unwrap();
170        encode_frame(&mut buf, END_FRAME, &[]).unwrap();
171
172        let outcomes = read_all(buf);
173
174        assert_eq!(outcomes.len(), 3);
175        let ReadOutcome::Frame(first) = &outcomes[0] else {
176            panic!("expected a frame, got {:?}", outcomes[0]);
177        };
178        assert_eq!(first.name, "t");
179        assert_eq!(first.payload, vec![1, 2, 3]);
180        let ReadOutcome::Frame(second) = &outcomes[1] else {
181            panic!("expected a frame, got {:?}", outcomes[1]);
182        };
183        assert_eq!(second.name, END_FRAME);
184        assert!(second.payload.is_empty());
185        assert!(matches!(outcomes[2], ReadOutcome::End));
186    }
187
188    #[test]
189    fn reader_reports_torn_header() {
190        let mut buf = Vec::new();
191        encode_frame(&mut buf, "t", &[1, 2, 3]).unwrap();
192        let first_len = buf.len() as u64;
193        buf.push(0x00);
194
195        let outcomes = read_all(buf);
196
197        assert!(matches!(outcomes[0], ReadOutcome::Frame(_)));
198        assert!(
199            matches!(outcomes[1], ReadOutcome::Torn { offset } if offset == first_len),
200            "half a name length after the first frame is torn at its end: {:?}",
201            outcomes[1]
202        );
203    }
204
205    #[test]
206    fn reader_reports_invalid_length_as_torn() {
207        let outcomes = read_all(0u16.to_be_bytes().to_vec());
208
209        assert!(
210            matches!(outcomes[0], ReadOutcome::Torn { offset: 0 }),
211            "a zero name length is invalid: {:?}",
212            outcomes[0]
213        );
214    }

216    #[test]
217    fn reserved_names() {
218        assert!(is_reserved(END_FRAME));
219        assert!(!is_reserved("spans"));
220        assert!(!is_reserved("_x"));
