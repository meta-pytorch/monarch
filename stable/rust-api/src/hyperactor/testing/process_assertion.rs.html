9use std::future::Future;
10use std::process::Command;
11
12const CHILD_TEST: &str = "HYPERACTOR_TERMINATION_TEST_CHILD";
13
14/// Re-run the current libtest test in a fresh process and verify its exit code.
15///
16/// The child executes `f` with its own Tokio runtime. Forking an existing runtime
17/// would inherit invalid I/O driver state on macOS. Call this once per test;
18/// setup before this call is repeated in the child process.
19/// Returning from `f` instead of terminating the process fails the assertion.
20pub async fn assert_termination<F, Fut>(f: F, expected_code: i32) -> anyhow::Result<()>
21where
22    F: FnOnce() -> Fut,
23    Fut: Future<Output = ()>,
24{
25    let thread = std::thread::current();
26    let test_name = thread
27        .name()
28        .ok_or_else(|| anyhow::anyhow!("termination assertion requires a named libtest thread"))?;
29    if std::env::var(CHILD_TEST).ok().as_deref() == Some(test_name) {
30        f().await;
31        eprintln!("termination callback returned without exiting");
32        std::process::exit(if expected_code == 0 { 1 } else { 0 });
33    }
34
35    let status = Command::new(std::env::current_exe()?)
36        .args(["--exact", test_name, "--nocapture"])
37        .env(CHILD_TEST, test_name)
38        .status()?;
39    anyhow::ensure!(
40        status.code() == Some(expected_code),
41        "expected exit code {expected_code}, got {status}"
42    );
43    Ok(())
44}
45
46#[cfg(test)]
47mod tests {
48    use super::*;
49
50    #[tokio::test]
51    async fn test_return_without_termination_is_rejected() {
52        let result = assert_termination(|| async {}, 0).await;
53        if std::env::var(CHILD_TEST).is_err() {
54            assert!(result.is_err());
55        }
56    }
57
58    #[tokio::test]
59    async fn test_explicit_zero_exit_succeeds() {
60        assert_termination(|| async { std::process::exit(0) }, 0)
61            .await
62            .unwrap();
63    }
64}
