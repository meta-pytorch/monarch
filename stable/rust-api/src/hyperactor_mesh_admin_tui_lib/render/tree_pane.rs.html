23const UNSELECTED_MARKER: &str = "  ";
24const PANE_BORDER_WIDTH: usize = 2;
25
26fn topology_row_text(rows: &VisibleRows<'_>, vis_idx: usize) -> String {
27    let row = &rows.as_slice()[vis_idx];
28    let node = row.node;
29    let indent = "  ".repeat(row.depth);
30    let connector = if row.depth == 0 {
31        ""
32    } else if rows.has_sibling_after(vis_idx, row.depth) {
33        "├─ "
34    } else {
35        "└─ "
36    };
37    let fold = if node.has_children {
38        if node.expanded { "▼ " } else { "▶ " }
39    } else {
40        "  "
41    };
42
43    format!("{}{}{}{}", indent, connector, fold, node.label)
44}

46/// Return the topology pane width, capped by its former percentage allocation.
47///
48/// TUI-28: sizing excludes cursor and scroll state. The marker allowance is
49/// stable even if a future theme's selection caret is not two cells wide.
50pub(crate) fn topology_pane_width(
51    rows: &VisibleRows<'_>,
52    pane_title: &str,
53    selection_caret: &str,
54    ceiling: u16,
55) -> u16 {
56    let marker_width = Line::from(selection_caret)
57        .width()
58        .max(Line::from(UNSELECTED_MARKER).width());
59    let content_width = rows
60        .as_slice()
61        .iter()
62        .enumerate()
63        .map(|(vis_idx, _)| marker_width + Line::from(topology_row_text(rows, vis_idx)).width())
64        .chain(std::iter::once(Line::from(pane_title).width()))
65        .max()
66        .unwrap_or_default()
67        .saturating_add(PANE_BORDER_WIDTH);
68
69    u16::try_from(content_width)
70        .unwrap_or(u16::MAX)
71        .min(ceiling)
72}

74/// Render the topology tree (left pane).
75///
76/// Uses `visible_rows()` to display only expanded nodes. Each row
77/// includes indentation/connectors, an expand/collapse glyph for
78/// nodes with children, and color-coding by `NodeType`, with the
79/// selected row highlighted.
80pub(crate) fn render_topology_tree(
81    frame: &mut ratatui::Frame<'_>,
82    area: Rect,
83    app: &App,
84    rows: &VisibleRows<'_>,
85) {
86    let scheme = &app.theme.scheme;
87
88    // When any overlay is active the tree is non-interactive.
89    // Render it uniformly dim so the user can see it is inactive.
90    // This covers both the py-spy overlay and the config overlay.
91    let pane_inactive = app.overlay.is_some();
92
93    let items: Vec<ListItem> = rows
94        .as_slice()
95        .iter()
96        .enumerate()
97        .map(|(vis_idx, row)| {
98            let node = row.node;
99
100            // Style precedence: inactive > selected > failed > stopped >
101            // system > node-type.  When pane_inactive the entire pane is dimmed
102            // and the selection/failed/system colours must not bleed through.
103            let style = if pane_inactive {
104                scheme.detail_stopped
105            } else if vis_idx == app.cursor.pos() {
106                scheme.stat_selection.add_modifier(Modifier::BOLD)
107            } else if node.failed {
108                scheme.node_failed
109            } else if node.stopped {
110                scheme.detail_stopped
111            } else if node.is_system {
112                scheme.node_system_actor
113            } else {
114                scheme.node_user_actor
115            };
116
117            // Hide the selection caret while the pane is inactive.
118            let marker = if !pane_inactive && vis_idx == app.cursor.pos() {
119                app.theme.labels.selection_caret
120            } else {
121                UNSELECTED_MARKER
122            };
123
124            ListItem::new(Line::from(Span::styled(
125                format!("{}{}", marker, topology_row_text(rows, vis_idx)),
126                style,
127            )))
128        })
129        .collect();
130
131    let block = if pane_inactive {
132        Block::default()
133            .title(Span::styled(
134                app.theme.labels.pane_topology,
135                scheme.detail_stopped,
136            ))
137            .borders(Borders::ALL)
138            .border_style(scheme.detail_stopped)
139    } else {
140        Block::default()
141            .title(app.theme.labels.pane_topology)
142            .borders(Borders::ALL)
143            .border_style(scheme.border)
144    };
145
146    let list = List::new(items)
147        .block(block)
148        .highlight_style(Style::default());
149    let mut list_state = ListState::default()
150        .with_selected(Some(app.cursor.pos()))
151        .with_offset(app.tree_scroll_offset);
152    frame.render_stateful_widget(list, area, &mut list_state);
153}

168    fn node(label: &str, has_children: bool, expanded: bool) -> TreeNode {
169        TreeNode {
170            reference: NodeRef::Root,
171            label: label.to_string(),
172            node_type: NodeType::Root,
173            expanded,
174            fetched: true,
175            has_children,
176            stopped: false,
177            failed: false,
178            is_system: false,
179            children: Vec::new(),
180        }
181    }
182
183    fn rows<'a>(nodes: &'a [TreeNode], depths: &[usize]) -> VisibleRows<'a> {
184        assert_eq!(nodes.len(), depths.len());
185        VisibleRows::new(
186            nodes
187                .iter()
188                .zip(depths)
189                .map(|(node, &depth)| FlatRow {
190                    node,
191                    depth,
192                    owning_proc: None,
193                })
194                .collect(),
195        )
196    }
197
198    // TUI-28: short content shrinks to the title width plus borders.
199    #[test]
200    fn topology_width_hugs_short_content() {
201        let nodes = [node("x", false, false)];
202        let rows = rows(&nodes, &[0]);
203
204        assert_eq!(topology_pane_width(&rows, "Topology", "▸ ", 40), 10);
205    }
206
207    // TUI-28: nested row width includes indentation, connector, fold, and borders.
208    #[test]
209    fn topology_width_includes_tree_structure() {
210        let nodes = [node("p", true, true), node("child", false, false)];
211        let rows = rows(&nodes, &[0, 1]);
212
213        assert_eq!(topology_pane_width(&rows, "Topology", "▸ ", 40), 16);
214    }
215
216    // TUI-28: Unicode labels are measured in terminal cells, including when empty.
217    #[test]
218    fn topology_width_uses_unicode_display_cells() {
219        let empty_rows = VisibleRows::new(Vec::new());
220        assert_eq!(topology_pane_width(&empty_rows, "拓扑", "▸ ", 40), 6);
221
222        let nodes = [node("节点", false, false)];
223        let rows = rows(&nodes, &[0]);
224        assert_eq!(topology_pane_width(&rows, "拓扑", "▸ ", 40), 10);
225    }
226
227    // TUI-28: the cap is Ratatui's exact former allocation, including rounding.
228    #[test]
229    fn topology_width_uses_former_percentage_allocation_as_ceiling() {
230        let area = Rect::new(0, 0, 102, 10);
231        let ceiling = Layout::default()
232            .direction(Direction::Horizontal)
233            .constraints([Constraint::Percentage(40), Constraint::Percentage(60)])
234            .split(area)[0]
235            .width;
236        assert_eq!(ceiling, 41);
237
238        let nodes = [node(&"x".repeat(100), false, false)];
239        let rows = rows(&nodes, &[0]);
240        assert_eq!(topology_pane_width(&rows, "Topology", "▸ ", ceiling), 41);
241    }
