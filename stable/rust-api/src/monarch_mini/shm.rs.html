9//! Shared-memory slab machinery for the machine-local large-message transport.
10//!
11//! Three objects with distinct owners and lifetimes:
12//!
13//! - [`Allocator`] — the bump + per-size-freelist allocator over a single slab
14//!   `memfd`. It owns the slab file, grows it, and hands out offsets; it never
15//!   maps anything.
16//! - [`ShmServer`] — the allocation *authority* (one per gateway actor). It runs
17//!   a unix dgram server that answers allocation requests against an `Allocator`,
18//!   and per grant watches a liveness pipe so a slab block is freed exactly once,
19//!   when the last holder of its token closes.
20//! - [`ShmClient`] — a tiny `Copy` pair of raw fds (the dgram request socket and
21//!   the slab object). Per actor; not the authority, just a handle.
22//! - [`ShmMapper`] — the context-global address-space manager. It reserves a huge
23//!   `PROT_NONE` range per slab once and grows a `MAP_FIXED` mapping into it so
24//!   pointers never move, and unmaps everything when the context is destroyed.
25//!
26//! The bottom of the file holds the libc mechanisms (`memfd_create`, `ftruncate`,
27//! `mmap` reserve/grow, `pipe2`, `socketpair`, and the `SCM_RIGHTS`
28//! `sendmsg`/`recvmsg` pair), each wrapped in a checked helper that turns the
29//! syscall's `-1`/`EAGAIN` into a `Result`/[`io::ErrorKind::WouldBlock`].
30
31// The shared-memory machinery is wired into the transport across later stages, so
32// in a non-test library build these APIs are momentarily unused; the unit tests
33// exercise them directly. (Removed once non-test callers wire them in.)
34#![cfg_attr(
35    not(test),
36    expect(
37        dead_code,
38        reason = "wired into the unix transport in a later stage of the shared-memory plan"
39    )
40)]

58/// Virtual address space reserved (PROT_NONE) per slab object, once. The slab
59/// file's `MAP_FIXED` mapping grows into the front of this reservation so the
60/// base never moves as the file grows. 128 GiB is far more than any slab will
61/// reach but costs nothing until touched.
62const RESERVE: usize = 1 << 37; // 128 GiB
63/// Initial slab file size, and the granularity we grow it by. Both a power of two
64/// so growth is a handful of `ftruncate`s even for large slabs.
65const INITIAL: u64 = 1 << 20; // 1 MiB
66const GROW: u64 = 1 << 20; // 1 MiB
67/// Every allocation is rounded up to this alignment, which also keys the
68/// per-size freelist (so a freed block can only satisfy a same-size request).
69const ALIGN: u64 = 64;
70
71/// Parts at least this large are moved through the slab (one memcpy + a tiny
72/// descriptor) instead of streamed inline over the socket; smaller parts stay
73/// inline, where the per-message fd/liveness overhead would not pay off.
74pub(crate) const SHM_THRESHOLD: u64 = 256 * 1024;
75
76/// A per-actor slot holding its gateway's [`ShmClient`] once learned. Lives on
77/// the actor and is shared (cloned) with that actor's transport coroutines, so
78/// shared memory turns on as soon as the gateway state arrives.
79pub(crate) type ShmClientSlot = Arc<Mutex<Option<ShmClient>>>;

81/// The slab object's name. `memfd_create` names are purely informational (they
82/// appear in `/proc/<pid>/fd` link targets as `memfd:<name>`); nothing collides.
83const SLAB_NAME: &CStr = c"monarch_mini_shm";
84
85/// Round `value` up to the next multiple of `align` (a power of two).
86fn align_up(value: u64, align: u64) -> u64 {
87    (value + align - 1) & !(align - 1)
88}

90/// Checked end of a range that must fit inside one mapper reservation.
91fn checked_range_end(offset: u64, len: usize) -> io::Result<u64> {
92    let len = u64::try_from(len)
93        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "shared-memory length overflow"))?;
94    offset
95        .checked_add(len)
96        .filter(|&end| end <= RESERVE as u64)
97        .ok_or_else(|| {
98            io::Error::new(
99                io::ErrorKind::InvalidData,
100                "shared-memory range exceeds reservation",
101            )
102        })
103}

105/// Create an anonymous, *unnamed* in-memory file. Nothing leaks in `/dev/shm`,
106/// and the kernel reclaims it once the last fd (this one plus any distributed
107/// dups) closes — so no path leaks on any crash.
108fn create_slab() -> io::Result<OwnedFd> {
109    // SAFETY: `SLAB_NAME` is a valid NUL-terminated C string that outlives the
110    // call. `memfd_create` either returns a fresh owned fd or -1; on success we
111    // take sole ownership of that fd via `OwnedFd::from_raw_fd`.
112    let fd = unsafe { libc::memfd_create(SLAB_NAME.as_ptr(), libc::MFD_CLOEXEC) };
113    if fd < 0 {
114        return Err(io::Error::last_os_error());
115    }
116    // SAFETY: `fd` is a fresh, owned, valid file descriptor returned by
117    // `memfd_create` above and not owned by anything else.
118    Ok(unsafe { OwnedFd::from_raw_fd(fd) })
119}
120
121/// Set the slab file's size to `size` bytes (only ever called to grow it).
122fn set_slab_size(fd: RawFd, size: u64) -> io::Result<()> {
123    // SAFETY: `fd` is the live slab memfd owned by the `Allocator`; `ftruncate`
124    // only changes the file's length and has no other effect on the process.
125    let r = unsafe { libc::ftruncate(fd, size as libc::off_t) };
126    if r < 0 {
127        return Err(io::Error::last_os_error());
128    }
129    Ok(())
130}

132/// Bump + per-size-freelist allocator over a single slab `memfd`. Owned by the
133/// allocation-authority side (the future `ShmServer`). It only ever creates and
134/// grows the file and hands out offsets; it never maps anything.
135struct Allocator {
136    /// The slab memfd. `Arc` so the same object can be shared (e.g. handed to a
137    /// mapper) without re-opening; the allocator only needs it to `ftruncate`.
138    slab_fd: Arc<OwnedFd>,
139    /// Current on-disk size of the slab file. `top` never exceeds this.
140    file_size: u64,
141    /// Bump pointer: the next never-yet-allocated offset.
142    top: u64,
143    /// Freed blocks, keyed by their aligned size. A request reuses a same-size
144    /// freed block before bumping, so steady same-size traffic does not grow the
145    /// file unboundedly.
146    free: HashMap<u64, Vec<u64>>,
147}
148
149impl Allocator {
150    /// Create a fresh slab, grown to [`INITIAL`] so the first allocations need no
151    /// `ftruncate`.
152    fn new() -> io::Result<Self> {
153        let slab_fd = create_slab()?;
154        set_slab_size(slab_fd.as_raw_fd(), INITIAL)?;
155        Ok(Self {
156            slab_fd: Arc::new(slab_fd),
157            file_size: INITIAL,
158            top: 0,
159            free: HashMap::new(),
160        })
161    }
162
163    /// The slab object, for whoever maps or forwards it.
164    fn slab_fd(&self) -> &Arc<OwnedFd> {
165        &self.slab_fd
166    }
167
168    /// Allocate `len` bytes, returning the slab offset. Reuses a freed same-size
169    /// block if one exists; otherwise bumps and grows the file (rounded up to
170    /// [`GROW`]) so the returned range is backed before the caller maps it. No
171    /// `mmap` happens here.
172    fn alloc(&mut self, len: u64) -> io::Result<u64> {
173        let size = align_up(len.max(1), ALIGN);
174
175        if let Some(offset) = self.free.get_mut(&size).and_then(Vec::pop) {
176            return Ok(offset);
177        }
178
179        let offset = self.top;
180        let new_top = offset + size;
181        if new_top > self.file_size {
182            let new_size = align_up(new_top, GROW);
183            set_slab_size(self.slab_fd.as_raw_fd(), new_size)?;
184            self.file_size = new_size;
185        }
186        self.top = new_top;
187        Ok(offset)
188    }
189
190    /// Return a previously-allocated block to its size's freelist. `len` must be
191    /// the same length passed to the [`alloc`](Self::alloc) that produced
192    /// `offset` (it is re-aligned identically, so the freelist key matches).
193    fn free(&mut self, offset: u64, len: u64) {
194        let size = align_up(len.max(1), ALIGN);
195        self.free.entry(size).or_default().push(offset);
196    }
197}

200// ShmClient — a tiny, Copy handle to a gateway's slab (per actor)

203/// A non-owning pair of raw fds: where to send allocation requests (the dgram
204/// socket to the [`ShmServer`]) and which slab to map/forward. Deliberately
205/// `Copy` and lifetime-free — the owning `ShmServer` (or, on a child, the process
206/// itself) keeps the fds alive. An actor holds one once it learns its gateway.
207#[derive(Clone, Copy)]
208pub(crate) struct ShmClient {
209    /// Dgram socket to the `ShmServer` for `Alloc` requests. Set non-blocking by
210    /// its owner; shared by every actor under the same gateway (many writers, one
211    /// reader), so a reply cannot be addressed over it — see [`ShmClient::allocate`].
212    dgram_fd: RawFd,
213    /// The slab object, for the mapper and for forwarding to children.
214    slab_fd: RawFd,
215}
216
217impl ShmClient {
218    /// Reconstruct a client from fds learned via gateway-state propagation (the
219    /// dgram request socket and the slab object). The fds are non-owning and must
220    /// stay open for the process lifetime (the caller leaks the received owned fds
221    /// into raw fds before calling this).
222    pub(crate) fn from_raw(dgram_fd: RawFd, slab_fd: RawFd) -> Self {
223        Self { dgram_fd, slab_fd }
224    }
225
226    /// The slab object fd (for mapping or forwarding).
227    pub(crate) fn slab_fd(&self) -> RawFd {
228        self.slab_fd
229    }
230
231    /// The dgram request socket fd (for forwarding via gateway-state propagation).
232    pub(crate) fn dgram_fd(&self) -> RawFd {
233        self.dgram_fd
234    }
235
236    /// Request `len` bytes from the server, returning `(offset, token)`. Because
237    /// the dgram socket is shared (one reader, many writers) the grant cannot ride
238    /// back on it; instead we mint a private pipe, hand the server its `write_end`
239    /// alongside the request, and read the granted offset back from our `read_end`.
240    /// That `read_end` is then the **liveness token**: as long as any copy of it is
241    /// open the server holds the block; when the last copy closes (delivered and
242    /// consumed, or a holder died) the server frees it.
243    pub(crate) async fn allocate(&self, len: u64) -> io::Result<(u64, OwnedFd)> {
244        let (read_end, write_end) = make_pipe()?;
245        set_nonblocking(read_end.as_raw_fd())?;
246
247        // Send `Alloc{len}` + the pipe's write end; then drop our write end so the
248        // server holds the only copy (its hangup watch keys off that).
249        let len_bytes = len.to_le_bytes();
250        send_dgram_with_fd(self.dgram_fd, &len_bytes, write_end.as_raw_fd()).await?;
251        drop(write_end);
252
253        let offset = read_grant(read_end.as_raw_fd()).await?;
254        Ok((offset, read_end))
255    }
256}

259// ShmServer — the allocation authority (one per gateway actor)
260// ---------------------------------------------------------------------------

262/// Owns the slab `memfd` and the `Allocator`, and runs a dgram server that
263/// answers allocation requests and frees each grant when its liveness pipe hangs
264/// up. Dropping it aborts the server task and releases the slab.
265pub(crate) struct ShmServer {
266    /// The server loop; aborted on drop, which drops the `Allocator` and the
267    /// server's end of the dgram socket, releasing the slab once all distributed
268    /// fds are gone.
269    server_task: JoinHandle<()>,
270    /// Clonable handle to this gateway's slab, handed to actors / children.
271    client: ShmClient,
272    /// Number of grants freed (a grant's liveness pipe hung up). Observability for
273    /// tests and diagnostics; never affects behavior.
274    freed: Arc<std::sync::atomic::AtomicU64>,
275    /// Owned client end of the dgram socket. Kept alive here so the raw `dgram_fd`
276    /// inside `client` stays valid for the server's lifetime.
277    _client_end: OwnedFd,
278    /// Owned slab fd. Kept alive here so the raw `slab_fd` inside `client` stays
279    /// valid (the allocator inside the task holds another `Arc` clone).
280    _slab_fd: Arc<OwnedFd>,
281}
282
283impl Drop for ShmServer {
284    fn drop(&mut self) {
285        self.server_task.abort();
286    }
287}
288
289impl ShmServer {
290    /// Build a slab + dgram server and spawn its loop. Must be called on the
291    /// context's tokio runtime (it uses `spawn_local` and the io driver).
292    pub(crate) fn new() -> io::Result<Self> {
293        let allocator = Allocator::new()?;
294        let slab_fd = Arc::clone(allocator.slab_fd());
295
296        let (server_end, client_end) = make_dgram_pair()?;
297        set_nonblocking(server_end.as_raw_fd())?;
298        set_nonblocking(client_end.as_raw_fd())?;
299
300        let client = ShmClient {
301            dgram_fd: client_end.as_raw_fd(),
302            slab_fd: slab_fd.as_raw_fd(),
303        };
304        let freed = Arc::new(std::sync::atomic::AtomicU64::new(0));
305
306        let server_task =
307            tokio::task::spawn_local(server_loop(server_end, allocator, Arc::clone(&freed)));
308
309        Ok(Self {
310            server_task,
311            client,
312            freed,
313            _client_end: client_end,
314            _slab_fd: slab_fd,
315        })
316    }
317
318    /// A `Copy` handle to this gateway's slab.
319    pub(crate) fn client(&self) -> ShmClient {
320        self.client
321    }
322
323    /// Number of grants freed so far (observability).
324    fn freed_count(&self) -> u64 {
325        self.freed.load(std::sync::atomic::Ordering::Relaxed)
326    }
327}

329/// The server loop: read allocation requests off `server_end`, allocate, write the
330/// granted offset back down the request's pipe, and spawn a per-grant watcher that
331/// frees the block when the pipe hangs up. Frees are funneled back here over an
332/// mpsc so the single `Allocator` is only touched from this task.
333async fn server_loop(
334    server_end: OwnedFd,
335    mut allocator: Allocator,
336    freed: Arc<std::sync::atomic::AtomicU64>,
337) {
338    let server_fd = server_end.as_raw_fd();
339    let afd = match AsyncFd::with_interest(server_end, Interest::READABLE) {
340        Ok(afd) => afd,
341        Err(err) => {
342            tracing::error!("shm server: registering dgram socket failed: {err}");
343            return;
344        }
345    };
346    let (free_tx, mut free_rx) = mpsc::unbounded_channel::<(u64, u64)>();
347
348    loop {
349        tokio::select! {
350            request = recv_alloc_request(&afd, server_fd) => {
351                let Ok((len, write_end)) = request else {
352                    return; // socket error: the gateway is going away
353                };
354                let offset = match allocator.alloc(len) {
355                    Ok(offset) => offset,
356                    Err(err) => {
357                        tracing::error!("shm server: alloc({len}) failed: {err}");
358                        // Drop write_end: the client's read_grant sees EOF and errors.
359                        continue;
360                    }
361                };
362                // Grant the offset down the private pipe (8 bytes into a fresh pipe
363                // never blocks), then watch the write end for the all-tokens-closed
364                // hangup that frees the block.
365                if write_all(write_end.as_raw_fd(), &offset.to_le_bytes()).is_err() {
366                    allocator.free(offset, len);
367                    continue;
368                }
369                tokio::task::spawn_local(watch_grant(write_end, offset, len, free_tx.clone()));
370            }
371            freed_block = free_rx.recv() => {
372                // free_tx is held here too, so recv never returns None.
373                let Some((offset, len)) = freed_block else { return };
374                allocator.free(offset, len);
375                freed.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
376            }
377        }
378    }
379}

381/// Watch a grant's pipe `write_end` for hangup — i.e. every copy of the matching
382/// `read_end` token closing — then report the block free. A write end is always
383/// writable, so plain writability never fires; we wait for the edge-triggered
384/// readiness whose `is_write_closed()` flags the closed read ends
385/// (`EPOLLERR`/`EPOLLHUP`), ignoring the initial writable wake.
386async fn watch_grant(
387    write_end: OwnedFd,
388    offset: u64,
389    len: u64,
390    free_tx: mpsc::UnboundedSender<(u64, u64)>,
391) {
392    if set_nonblocking(write_end.as_raw_fd()).is_err() {
393        return;
394    }
395    let afd = match AsyncFd::with_interest(write_end, Interest::WRITABLE) {
396        Ok(afd) => afd,
397        Err(_) => return,
398    };
399    loop {
400        let mut guard = match afd.ready(Interest::WRITABLE).await {
401            Ok(guard) => guard,
402            Err(_) => return,
403        };
404        if guard.ready().is_write_closed() {
405            let _ = free_tx.send((offset, len));
406            return;
407        }
408        // The expected initial "writable" edge; clear it and park for the next
409        // edge, which (mio is edge-triggered) only comes on hangup.
410        guard.clear_ready();
411    }
412}

414/// Await one allocation request off the server dgram socket: `[u64 len]` + one fd
415/// (the grant pipe's write end). Loops past `WouldBlock`; returns `Err` on a real
416/// socket error or a malformed request.
417async fn recv_alloc_request(
418    afd: &AsyncFd<OwnedFd>,
419    server_fd: RawFd,
420) -> io::Result<(u64, OwnedFd)> {
421    loop {
422        let mut guard = afd.readable().await?;
423        let mut buf = [0u8; 8];
424        // An allocation request carries exactly one fd: the grant pipe's write end.
425        match guard.try_io(|_| recv_with_fds(server_fd, &mut buf, 1)) {
426            Ok(Ok((n, mut fds))) => {
427                if n != 8 || fds.len() != 1 {
428                    return Err(io::Error::new(
429                        io::ErrorKind::InvalidData,
430                        "malformed shm alloc request",
431                    ));
432                }
433                let len = u64::from_le_bytes(buf);
434                let write_end = fds.pop().expect("exactly one fd checked above");
435                return Ok((len, write_end));
436            }
437            Ok(Err(err)) => return Err(err),
438            Err(_would_block) => continue,
439        }
440    }
441}

443/// Send `Alloc` request bytes + the grant pipe's write end over the (non-blocking)
444/// dgram socket, awaiting writability on `WouldBlock`. The socket is shared by
445/// many writers, so a transient borrowed `AsyncFd` is used only on backpressure.
446async fn send_dgram_with_fd(dgram_fd: RawFd, data: &[u8], fd: RawFd) -> io::Result<()> {
447    loop {
448        match send_with_fds(dgram_fd, data, &[fd]) {
449            Ok(_) => return Ok(()),
450            Err(err) if err.kind() == io::ErrorKind::WouldBlock => {
451                // SAFETY: `dgram_fd` is a live socket owned elsewhere for the whole
452                // call; the transient `BorrowedFd`/`AsyncFd` only watch readiness
453                // and never close it.
454                let borrowed = unsafe { BorrowedFd::borrow_raw(dgram_fd) };
455                let afd = AsyncFd::with_interest(borrowed, Interest::WRITABLE)?;
456                let _ = afd.writable().await?;
457            }
458            Err(err) => return Err(err),
459        }
460    }
461}

463/// Read the 8-byte granted offset back from the request's private (non-blocking)
464/// pipe `read_fd`, awaiting readability as needed. EOF before 8 bytes means the
465/// server dropped the write end without granting (an allocation error).
466async fn read_grant(read_fd: RawFd) -> io::Result<u64> {
467    // SAFETY: `read_fd` is the live read end owned by the caller for the whole
468    // call; the `AsyncFd` watches readiness over a borrow and never closes it.
469    let borrowed = unsafe { BorrowedFd::borrow_raw(read_fd) };
470    let afd = AsyncFd::with_interest(borrowed, Interest::READABLE)?;
471    let mut buf = [0u8; 8];
472    let mut filled = 0;
473    while filled < buf.len() {
474        let mut guard = afd.readable().await?;
475        match guard.try_io(|_| read_fd_bytes(read_fd, &mut buf[filled..])) {
476            Ok(Ok(0)) => {
477                return Err(io::Error::new(
478                    io::ErrorKind::UnexpectedEof,
479                    "shm grant pipe closed before reply",
480                ));
481            }
482            Ok(Ok(n)) => filled += n,
483            Ok(Err(err)) => return Err(err),
484            Err(_would_block) => continue,
485        }
486    }
487    Ok(u64::from_le_bytes(buf))
488}

494/// Shared handle to the per-context [`ShmMapper`].
495pub(crate) type MapperHandle = Arc<Mutex<ShmMapper>>;

497/// One slab's persistent mapping: a large `PROT_NONE` reservation whose front is
498/// progressively backed by the slab file via `MAP_FIXED`, so `base` never moves
499/// as the file grows.
500struct Reservation {
501    /// Owned dup of the slab fd, kept so the mapping can keep growing even after
502    /// the client's fd closes. (Existing mappings survive the fd closing; only
503    /// growth needs a live fd.)
504    _fd: OwnedFd,
505    /// Start of the reserved address range.
506    base: *mut u8,
507    /// How many bytes from `base` are currently backed by the file.
508    mapped: u64,
509}
510
511impl Drop for Reservation {
512    fn drop(&mut self) {
513        // SAFETY: `base` came from a `mmap` of `RESERVE` bytes in `ShmMapper::map`
514        // and is owned solely by this `Reservation`; unmapping it exactly once on
515        // drop is sound. Any pointers handed out from it must not outlive the
516        // context (and thus the mapper), which is the mapper's contract.
517        unsafe {
518            libc::munmap(self.base.cast(), RESERVE);
519        }
520    }
521}
522
523/// Maps slab ranges into this context's address space and keeps them mapped until
524/// the context is destroyed. One reservation per distinct slab (keyed by inode),
525/// so clones of the same `ShmClient` share a single mapping.
526pub(crate) struct ShmMapper {
527    reservations: HashMap<u64, Reservation>,
528}
529
530impl ShmMapper {
531    pub(crate) fn new() -> Self {
532        Self {
533            reservations: HashMap::new(),
534        }
535    }
536
537    /// Ensure `[offset, offset+len)` of the slab `slab_fd` is mapped and return a
538    /// pointer to `base + offset`. Reserves the slab's address range on first use
539    /// and grows the backed extent in place as needed; the base never moves.
540    ///
541    /// # Safety
542    /// `slab_fd` must be a valid fd referring to the slab whose allocator produced
543    /// `offset`, and `[offset, offset+len)` must lie within a region the allocator
544    /// has grown the file to cover. The returned pointer is valid only while this
545    /// mapper (i.e. the context) lives, and only `len` bytes are accessible.
546    pub(crate) unsafe fn map(
547        &mut self,
548        slab_fd: RawFd,
549        offset: u64,
550        len: usize,
551    ) -> io::Result<*mut u8> {
552        let need = checked_range_end(offset, len)?;
553        let (inode, _size) = fstat_ino_size(slab_fd)?;
554
555        let reservation = match self.reservations.entry(inode) {
556            std::collections::hash_map::Entry::Occupied(e) => e.into_mut(),
557            std::collections::hash_map::Entry::Vacant(e) => {
558                let base = mmap_reserve(RESERVE)?;
559                let fd = dup_fd(slab_fd)?;
560                e.insert(Reservation {
561                    _fd: fd,
562                    base,
563                    mapped: 0,
564                })
565            }
566        };
567
568        if need > reservation.mapped {
569            let target = align_up(need, page_size());
570            // SAFETY: `base` is the start of a `RESERVE`-byte reservation; `target`
571            // cannot exceed `RESERVE` because `checked_range_end` bounded `need` and
572            // `RESERVE` is page-aligned. `target` and `mapped` are page-aligned, so the
573            // `MAP_FIXED` overlay lands inside the reservation at a page boundary backed
574            // by the slab file (which the allocator has grown to cover `need`). `_fd` is
575            // a live slab fd.
576            unsafe {
577                mmap_fixed(
578                    reservation.base,
579                    reservation.mapped,
580                    target - reservation.mapped,
581                    reservation._fd.as_raw_fd(),
582                )?;
583            }
584            reservation.mapped = target;
585        }
586
587        // SAFETY: `base` is a valid mapping of at least `reservation.mapped >=
588        // offset + len` bytes (ensured just above), so `base + offset` points
589        // within it.
590        Ok(unsafe { reservation.base.add(offset as usize) })
591    }
592}

605/// `O_NONBLOCK` the given fd (preserving its other flags).
606fn set_nonblocking(fd: RawFd) -> io::Result<()> {
607    // SAFETY: `fd` is a live fd owned by the caller; fcntl F_GETFL/F_SETFL only
608    // read and write its status flags.
609    let flags = unsafe { libc::fcntl(fd, libc::F_GETFL) };
610    if flags < 0 {
611        return Err(io::Error::last_os_error());
612    }
613    let r = unsafe { libc::fcntl(fd, libc::F_SETFL, flags | libc::O_NONBLOCK) };
614    if r < 0 {
615        return Err(io::Error::last_os_error());
616    }
617    Ok(())
618}

620/// `dup` an fd into an owned handle.
621fn dup_fd(fd: RawFd) -> io::Result<OwnedFd> {
622    // SAFETY: `fd` is live; `dup` returns a fresh owned fd or -1.
623    let dup = unsafe { libc::dup(fd) };
624    if dup < 0 {
625        return Err(io::Error::last_os_error());
626    }
627    // SAFETY: `dup` is a fresh, owned, valid fd not owned by anything else.
628    Ok(unsafe { OwnedFd::from_raw_fd(dup) })
629}

631/// `fstat` for the inode (mapping dedup key) and current size.
632fn fstat_ino_size(fd: RawFd) -> io::Result<(u64, u64)> {
633    // SAFETY: `st` is fully written by a successful fstat; `fd` is a live fd.
634    let mut st: libc::stat = unsafe { std::mem::zeroed() };
635    let r = unsafe { libc::fstat(fd, &mut st) };
636    if r < 0 {
637        return Err(io::Error::last_os_error());
638    }
639    Ok((st.st_ino as u64, st.st_size as u64))
640}

649/// Create a `(read_end, write_end)` pipe with `O_CLOEXEC`.
650fn make_pipe() -> io::Result<(OwnedFd, OwnedFd)> {
651    let mut fds = [0 as RawFd; 2];
652    // SAFETY: `fds` is a valid 2-element array; pipe2 fills it on success.
653    let r = unsafe { libc::pipe2(fds.as_mut_ptr(), libc::O_CLOEXEC) };
654    if r < 0 {
655        return Err(io::Error::last_os_error());
656    }
657    // SAFETY: both entries are fresh, owned, valid fds from a successful pipe2.
658    Ok(unsafe { (OwnedFd::from_raw_fd(fds[0]), OwnedFd::from_raw_fd(fds[1])) })
659}

661/// Create a `(server_end, client_end)` `AF_UNIX`/`SOCK_DGRAM` socket pair.
662fn make_dgram_pair() -> io::Result<(OwnedFd, OwnedFd)> {
663    let mut fds = [0 as RawFd; 2];
664    // SAFETY: `fds` is a valid 2-element array; socketpair fills it on success.
665    let r = unsafe {
666        libc::socketpair(
667            libc::AF_UNIX,
668            libc::SOCK_DGRAM | libc::SOCK_CLOEXEC,
669            0,
670            fds.as_mut_ptr(),
671        )
672    };
673    if r < 0 {
674        return Err(io::Error::last_os_error());
675    }
676    // SAFETY: both entries are fresh, owned, valid fds from a successful socketpair.
677    Ok(unsafe { (OwnedFd::from_raw_fd(fds[0]), OwnedFd::from_raw_fd(fds[1])) })
678}

680/// Write all of `data` to `fd` (used only for the 8-byte grant into a fresh pipe,
681/// which never partials/blocks, but loop on short writes for correctness).
682fn write_all(fd: RawFd, data: &[u8]) -> io::Result<()> {
683    let mut written = 0;
684    while written < data.len() {
685        // SAFETY: `fd` is a live writable fd; we pass a valid pointer/length into
686        // the remaining slice and only read `len` bytes from it.
687        let n = unsafe { libc::write(fd, data[written..].as_ptr().cast(), data.len() - written) };
688        if n < 0 {
689            return Err(io::Error::last_os_error());
690        }
691        written += n as usize;
692    }
693    Ok(())
694}

696/// One non-blocking `read` into `buf`, mapping `EAGAIN` to `WouldBlock`.
697fn read_fd_bytes(fd: RawFd, buf: &mut [u8]) -> io::Result<usize> {
698    // SAFETY: `fd` is a live readable fd; `buf` is a valid mutable slice that read
699    // fills up to its length.
700    let n = unsafe { libc::read(fd, buf.as_mut_ptr().cast(), buf.len()) };
701    if n < 0 {
702        return Err(io::Error::last_os_error());
703    }
704    Ok(n as usize)
705}

707/// Reserve `len` bytes of address space (`PROT_NONE`, anonymous, no commit).
708fn mmap_reserve(len: usize) -> io::Result<*mut u8> {
709    // SAFETY: a fresh anonymous PROT_NONE reservation has no preconditions; we
710    // check for MAP_FAILED before returning the pointer.
711    let p = unsafe {
712        libc::mmap(
713            std::ptr::null_mut(),
714            len,
715            libc::PROT_NONE,
716            libc::MAP_PRIVATE | libc::MAP_ANONYMOUS | libc::MAP_NORESERVE,
717            -1,
718            0,
719        )
720    };
721    if p == libc::MAP_FAILED {
722        return Err(io::Error::last_os_error());
723    }
724    Ok(p.cast())
725}

727/// Overlay `[at, at+len)` of the reservation at `base` with the slab file's bytes
728/// at file offset `at` (read/write, shared). `base+at` and `at` must be page
729/// aligned and the range must lie within the reservation.
730///
731/// # Safety
732/// `base` must be the start of a live `RESERVE`-byte reservation, `at + len <=
733/// RESERVE`, both `at` and `base` page-aligned, and `fd` a live slab fd whose file
734/// is at least `at + len` bytes.
735unsafe fn mmap_fixed(base: *mut u8, at: u64, len: u64, fd: RawFd) -> io::Result<()> {
736    // SAFETY: per this fn's contract `base.add(at)` is page-aligned and inside the
737    // reservation, so MAP_FIXED overlays a valid sub-range; the file is backed to
738    // `at + len`. We check MAP_FAILED before returning.
739    let p = unsafe {
740        libc::mmap(
741            base.add(at as usize).cast(),
742            len as usize,
743            libc::PROT_READ | libc::PROT_WRITE,
744            libc::MAP_SHARED | libc::MAP_FIXED,
745            fd,
746            at as libc::off_t,
747        )
748    };
749    if p == libc::MAP_FAILED {
750        return Err(io::Error::last_os_error());
751    }
752    Ok(())
753}

755/// `sendmsg` `data` plus `fds` (via one `SCM_RIGHTS` control message) on `sock`.
756/// `MSG_NOSIGNAL` so a dead peer yields `EPIPE` rather than a signal.
757pub(crate) fn send_with_fds(sock: RawFd, data: &[u8], fds: &[RawFd]) -> io::Result<usize> {
758    let fd_bytes = std::mem::size_of_val(fds);
759    // SAFETY: the whole block builds a single msghdr with one iovec over `data` and
760    // one SCM_RIGHTS cmsg holding `fds`, exactly as the cmsg macros require, then
761    // calls sendmsg. `data`/`fds` outlive the call; the control buffer is sized via
762    // CMSG_SPACE and zeroed; we copy the fds into CMSG_DATA bytewise (no alignment
763    // assumption). The result is checked before use.
764    unsafe {
765        let mut iov = libc::iovec {
766            iov_base: data.as_ptr() as *mut libc::c_void,
767            iov_len: data.len(),
768        };
769        let control_len = libc::CMSG_SPACE(fd_bytes as libc::c_uint) as usize;
770        let mut control = vec![0u8; control_len];
771
772        let mut msg: libc::msghdr = std::mem::zeroed();
773        msg.msg_iov = &mut iov;
774        msg.msg_iovlen = 1;
775        msg.msg_control = control.as_mut_ptr().cast();
776        msg.msg_controllen = control_len as _;
777
778        let cmsg = libc::CMSG_FIRSTHDR(&msg);
779        (*cmsg).cmsg_level = libc::SOL_SOCKET;
780        (*cmsg).cmsg_type = libc::SCM_RIGHTS;
781        (*cmsg).cmsg_len = libc::CMSG_LEN(fd_bytes as libc::c_uint) as _;
782        std::ptr::copy_nonoverlapping(fds.as_ptr().cast::<u8>(), libc::CMSG_DATA(cmsg), fd_bytes);
783
784        let n = libc::sendmsg(sock, &msg, libc::MSG_NOSIGNAL);
785        if n < 0 {
786            return Err(io::Error::last_os_error());
787        }
788        Ok(n as usize)
789    }
790}

792/// `recvmsg` into `data`, collecting any `SCM_RIGHTS` fds as fresh owned fds
793/// (`MSG_CMSG_CLOEXEC`). Non-blocking (`MSG_DONTWAIT`); `EAGAIN` -> `WouldBlock`.
794/// Returns `(bytes_read, fds)`; `0` bytes means EOF. `max_fds` sizes the control
795/// buffer: the caller knows from the frame header exactly how many fds the sender
796/// attached, so there is no fixed cap — too small a buffer would truncate the
797/// ancillary data (`MSG_CTRUNC`) and the caller's count check would catch it.
798pub(crate) fn recv_with_fds(
799    sock: RawFd,
800    data: &mut [u8],
801    max_fds: usize,
802) -> io::Result<(usize, Vec<OwnedFd>)> {
803    let fd_size = std::mem::size_of::<RawFd>();
804    // SAFETY: the block builds a msghdr with one iovec over `data` and a control
805    // buffer sized for `max_fds` cmsg fds, calls recvmsg, then walks the returned
806    // cmsgs copying out each SCM_RIGHTS fd (installed by the kernel as a fresh fd we
807    // take ownership of). All pointers come from the just-filled msghdr.
808    unsafe {
809        let mut iov = libc::iovec {
810            iov_base: data.as_mut_ptr().cast(),
811            iov_len: data.len(),
812        };
813        let control_len = libc::CMSG_SPACE((max_fds * fd_size) as libc::c_uint) as usize;
814        let mut control = vec![0u8; control_len];
815
816        let mut msg: libc::msghdr = std::mem::zeroed();
817        msg.msg_iov = &mut iov;
818        msg.msg_iovlen = 1;
819        msg.msg_control = control.as_mut_ptr().cast();
820        msg.msg_controllen = control_len as _;
821
822        let n = libc::recvmsg(sock, &mut msg, libc::MSG_CMSG_CLOEXEC | libc::MSG_DONTWAIT);
823        if n < 0 {
824            return Err(io::Error::last_os_error());
825        }
826
827        let mut fds = Vec::new();
828        let mut cmsg = libc::CMSG_FIRSTHDR(&msg);
829        while !cmsg.is_null() {
830            if (*cmsg).cmsg_level == libc::SOL_SOCKET && (*cmsg).cmsg_type == libc::SCM_RIGHTS {
831                let payload = (*cmsg).cmsg_len as usize - libc::CMSG_LEN(0) as usize;
832                let count = payload / fd_size;
833                let src = libc::CMSG_DATA(cmsg);
834                for i in 0..count {
835                    let mut raw: RawFd = 0;
836                    std::ptr::copy_nonoverlapping(
837                        src.add(i * fd_size),
838                        (&mut raw as *mut RawFd).cast::<u8>(),
839                        fd_size,
840                    );
841                    fds.push(OwnedFd::from_raw_fd(raw));
842                }
843            }
844            cmsg = libc::CMSG_NXTHDR(&msg, cmsg);
845        }
846        Ok((n as usize, fds))
847    }
848}
