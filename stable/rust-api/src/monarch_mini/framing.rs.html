9//! Wire framing for the QUIC transport (and, later, tcp). The UNIX transport has
10//! its own framing (see [`crate::unix_framing`]) so machine-local shared-memory
11//! and fd-passing concerns never leak into the cross-machine wire format.
12//!
13//! A frame is a length-prefixed bincode header ([`WireFrame`]) holding only small
14//! routing/control metadata — never part bytes. An actor message's routing header is
15//! followed by a per-part plan ([`WirePart`], each part inline or striped) and then the
16//! inline parts' bytes: a small part streams inline right after the plan (from the owning
17//! `MsgPart`), a large part is **striped** across the connection's data streams and named
18//! only by its per-shard lengths (see [`crate::dataio`]).
19//!
20//! This module owns the header/plan wire format and streams *inline* part bytes; the
21//! parts themselves go through the [`PartReader`]/[`PartWriter`] the read/write functions
22//! are handed (per part), so the shared-memory and striping concerns — a large part read
23//! straight into a slab block, or striped across data streams — never involve framing.
24//!
25//! The reader/writer are generic over tokio's [`AsyncRead`]/[`AsyncWrite`], so the
26//! same code drives a QUIC stream's `RecvStream`/`SendStream`. Liveness policy
27//! (heartbeats, timeouts, EOF) lives in each transport; this module only serializes
28//! frames — data-stream [`WireFrame`]s, and the bare [`Heartbeat`] probe that rides
29//! its own stream (see [`write_heartbeat`]), which is never a [`ConnectionCommand`].

52/// One frame header on the data stream. There is a variant per [`ConnectionCommand`]
53/// that crosses the pipe (including `Establish`, which is how the two ends exchange
54/// identity). A header is routing/control only — it carries no message-part bytes; an
55/// [`ActorMessage`](WireFrame::ActorMessage)'s parts follow it via
56/// [`PartWriter::write_message_parts`]/[`PartReader::read_message_parts`]. Heartbeats are *not* here: they ride
57/// a dedicated stream, serialized as a bare [`Heartbeat`] (see
58/// [`write_heartbeat`]/[`read_heartbeat`]).
59#[derive(Serialize, Deserialize)]
60enum WireFrame {
61    Establish {
62        role: Role,
63        ident: Option<Vec<u8>>,
64        name_for_other: Option<Vec<u8>>,
65        alive: bool,
66    },
67    /// An actor message: this header carries only the destination; the parts follow.
68    ActorMessage {
69        destination_ident: Vec<u8>,
70    },
71    /// A monitor fire — the only other [`SendMessage`](ConnectionCommand::SendMessage);
72    /// it has no parts, so it is a plain header (kept separate from `ActorMessage` so
73    /// only the message path deals with parts).
74    FireMonitorMessage {
75        destination_ident: Vec<u8>,
76        to_monitor: Vec<u8>,
77        is_timeout: bool,
78    },
79    PublishRoutes {
80        live: Vec<Vec<u8>>,
81        dead: Vec<Vec<u8>>,
82    },
83    UpdateMonitorSubscription {
84        listener: Vec<u8>,
85        target: Vec<u8>,
86        op: MonitorOp,
87    },
88    Severed {
89        reason: Vec<u8>,
90    },
91    PublishGatewayRoutes {
92        live: Vec<Vec<u8>>,
93    },
94    GatewayDied {
95        dead: Vec<Vec<u8>>,
96    },
97}

99/// How one message part is carried on the wire. A small part is streamed **inline**
100/// on the control stream right after the header (zero-copy, as always). A large part
101/// is **striped**: its bytes travel out-of-band on the connection's data streams (see
102/// [`crate::dataio`]), and the header carries only the per-shard byte lengths — one
103/// per data stream — which fully describe the split so the data streams themselves
104/// stay unframed. The sender alone decides inline-vs-striped (from its size heuristic
105/// and stream count), so the receiver never has to agree on a threshold: it just
106/// reads whichever form the descriptor names.
107#[derive(Clone, Serialize, Deserialize)]
108pub(crate) enum WirePart {
109    Inline { len: u64 },
110    Striped { shard_lens: Vec<u64> },
111}

113/// One frame header on a gateway side-channel's *message* stream — the wire form of a
114/// [`SideChannelMessage`], mirroring [`WireFrame`]. A `Send` of an actor message
115/// (`SendActorMessage`) is a routing-only header whose parts follow via
116/// [`PartWriter::write_message_parts`]/[`PartReader::read_message_parts`], exactly like a connection message;
117/// every other case is header-only. Delegated heartbeats are not here — they ride the
118/// companion heartbeat stream as a bare [`SideChannelHeartbeat`].
119#[derive(Serialize, Deserialize)]
120enum SideChannelFrame {
121    /// A `Send` of an actor message: this header carries only the gateway; parts follow.
122    SendActorMessage { gateway_for_actor: Vec<u8> },
123    /// A `Send` of a monitor fire — no parts, so a plain header.
124    SendFireMonitor {
125        gateway_for_actor: Vec<u8>,
126        to_monitor: Vec<u8>,
127        is_timeout: bool,
128    },
129    UpdateRemoteMonitorState {
130        gateway_for_actor: Vec<u8>,
131        listener: Vec<u8>,
132        op: MonitorOp,
133    },
134    AckRemoteMonitor {
135        gateway_for_actor: Vec<u8>,
136        monitoring: Vec<u8>,
137    },
138}

144/// Encode `value` as bincode and write it length-prefixed as `[u64 header_len]
145/// [header]`. Every framed write goes through this for its header — the frame header,
146/// and (for a message) the per-part plan written by [`PartWriter::write_message_parts`]. Callers
147/// layer their own concerns on top: trailing inline part bytes and the flush (every
148/// caller decides when to flush; this helper never does). The length prefix is
149/// little-endian, matching [`read_header`].
150pub(crate) async fn write_header<W: AsyncWrite + Unpin, T: Serialize>(
151    writer: &mut W,
152    value: &T,
153) -> std::io::Result<()> {
154    let header = bincode::serde::encode_to_vec(value, bincode_config())
155        .map_err(|err| std::io::Error::new(std::io::ErrorKind::InvalidData, err))?;
156    writer
157        .write_all(&(header.len() as u64).to_le_bytes())
158        .await?;
159    writer.write_all(&header).await
160}

162/// Read a `[u64 header_len][header]`-framed bincode value written by
163/// [`write_header`] and decode it as `T`. The single place every framed read goes
164/// through for its header; callers that carry trailing message-part bytes read them
165/// after this (see [`read_frame`]).
166pub(crate) async fn read_header<R: AsyncRead + Unpin, T: DeserializeOwned>(
167    reader: &mut R,
168) -> std::io::Result<T> {
169    let mut len_buf = [0u8; 8];
170    reader.read_exact(&mut len_buf).await?;
171    let header_len = u64::from_le_bytes(len_buf) as usize;
172    let mut header = vec![0u8; header_len];
173    reader.read_exact(&mut header).await?;
174    let (value, _) = bincode::serde::decode_from_slice::<T, _>(&header, bincode_config())
175        .map_err(|err| std::io::Error::new(std::io::ErrorKind::InvalidData, err))?;
176    Ok(value)
177}

179/// The first frame a QUIC *joiner* (the connecting side) writes on the **first**
180/// bi-stream it opens, declaring what kind of connection this is. Every connection
181/// carries two bi-streams — a data/message stream and a companion heartbeat stream
182/// — so heartbeats never queue behind a large data transfer on a single ordered
183/// stream. Only the first stream carries a preamble: the second is *always* the
184/// heartbeat stream (opened right after the first), so the acceptor classifies it
185/// by order and reads no preamble on it.
186///
187/// - `Join` — an ordinary parent/child connection; the command loop takes over
188///   (identity exchange, hello, routing) over the data stream.
189/// - `SideChannel` — a gateway-to-gateway channel: the accepting gateway reads
190///   [`SideChannelFrame`] frames and routes them locally. Never paired with a
191///   serve, never establishes a parent/child link.
192///
193/// Only the joiner writes the preamble (one direction); the server never does, so
194/// the joiner's own readers keep reading frames as before.
195#[derive(Serialize, Deserialize)]
196pub(crate) enum Preamble {
197    Join,
198    SideChannel,
199}
200
201/// Write the connection preamble (see [`Preamble`]). Length-prefixed bincode,
202/// the same scheme as [`write_frame`], so a reader can decode it with
203/// [`read_preamble`] before switching to the per-frame loop.
204pub(crate) async fn write_preamble<W: AsyncWrite + Unpin>(
205    writer: &mut W,
206    preamble: Preamble,
207) -> std::io::Result<()> {
208    write_header(writer, &preamble).await?;
209    writer.flush().await
210}

212/// Read the connection preamble written by [`write_preamble`].
213pub(crate) async fn read_preamble<R: AsyncRead + Unpin>(
214    reader: &mut R,
215) -> std::io::Result<Preamble> {
216    read_header(reader).await
217}

219/// A delegated-heartbeat beat/ack/release carried on a gateway side-channel's
220/// dedicated heartbeat stream. It is the bare wire form on that stream (which
221/// carries only these), and is *never* surfaced to ctx — the receiving quic reader
222/// hands it straight to the heartbeat subsystem — so it stays out of ctx's
223/// [`SideChannelMessage`]/`SideChannelAction` routing types.
224#[derive(Serialize, Deserialize)]
225pub(crate) struct SideChannelHeartbeat {
226    /// The actor whose gateway received the beat — i.e. the one the addressed
227    /// heartbeat coroutine belongs to as a child.
228    pub(crate) recipient: Vec<u8>,
229    pub(crate) from: Vec<u8>,
230    pub(crate) conn_id: ConnectionId,
231    pub(crate) kind: BeatKind,
232}

234/// Serialize and write one command. An actor message writes a routing-only header then
235/// its parts via [`PartWriter::write_message_parts`] (planned and, if large, striped by
236/// `part_writer`); every other command is a header-only frame. Flushes so the peer (and,
237/// for quic, its heartbeat deadline) sees it promptly.
238pub(crate) async fn write_command<W, N>(
239    writer: &mut W,
240    command: ConnectionCommand,
241    part_writer: &mut PartWriter<N>,
242) -> std::io::Result<()>
243where
244    W: AsyncWrite + Unpin,
245    N: Net,
246{
247    let frame = match command {
248        ConnectionCommand::SendMessage {
249            destination_ident,
250            payload: SendPayload::ActorMessage(parts),
251        } => {
252            // Routing-only header, then the parts (plan + bytes) via the part writer.
253            write_header(writer, &WireFrame::ActorMessage { destination_ident }).await?;
254            return part_writer.write_message_parts(writer, parts).await;
255        }
256        ConnectionCommand::SendMessage {
257            destination_ident,
258            payload:
259                SendPayload::FireMonitor {
260                    to_monitor,
261                    is_timeout,
262                },
263        } => WireFrame::FireMonitorMessage {
264            destination_ident,
265            to_monitor,
266            is_timeout,
267        },
268        ConnectionCommand::Establish {
269            role,
270            ident,
271            name_for_other,
272            alive,
273        } => WireFrame::Establish {
274            role,
275            ident,
276            name_for_other,
277            alive,
278        },
279        ConnectionCommand::PublishRoutes { live, dead } => WireFrame::PublishRoutes { live, dead },
280        ConnectionCommand::UpdateMonitorSubscription {
281            listener,
282            target,
283            op,
284        } => WireFrame::UpdateMonitorSubscription {
285            listener,
286            target,
287            op,
288        },
289        ConnectionCommand::Severed { reason } => WireFrame::Severed { reason },
290        ConnectionCommand::PublishGatewayRoutes { live } => {
291            WireFrame::PublishGatewayRoutes { live }
292        }
293        ConnectionCommand::GatewayDied { dead } => WireFrame::GatewayDied { dead },
294        // Shared memory is machine-local, so quic drops gateway state rather than
295        // forwarding it: skip without writing anything to the wire.
296        ConnectionCommand::GatewayState { .. } => return Ok(()),
297    };
298    write_frame(writer, &frame).await
299}

320/// Write a header-only frame: the length-prefixed bincode header, then flush. An actor
321/// message instead writes its routing header and then its parts via
322/// [`PartWriter::write_message_parts`]; every command/action that reaches here is header-only.
323async fn write_frame<W: AsyncWrite + Unpin>(
324    writer: &mut W,
325    frame: &impl Serialize,
326) -> std::io::Result<()> {
327    write_header(writer, frame).await?;
328    writer.flush().await
329}

331/// Read and decode one control-stream frame into its [`ConnectionCommand`]. An actor
332/// message's routing header comes back here; its parts are read through `part_reader`
333/// (which owns the shared memory and striping), so those concerns stay out of framing. A
334/// striped part's [`MsgPart`] is returned before its bytes have all landed; the caller
335/// gates delivery on the part reader's batch (see [`crate::dataio`]). Heartbeats never
336/// arrive here — they ride their own stream (see [`read_heartbeat`]).
337pub(crate) async fn read_frame<R, N>(
338    reader: &mut R,
339    part_reader: &mut PartReader<N>,
340) -> std::io::Result<ConnectionCommand>
341where
342    R: AsyncRead + Unpin,
343    N: Net,
344{
345    Ok(match read_header::<_, WireFrame>(reader).await? {
346        WireFrame::ActorMessage { destination_ident } => ConnectionCommand::SendMessage {
347            destination_ident,
348            payload: SendPayload::ActorMessage(part_reader.read_message_parts(reader).await?),
349        },
350        WireFrame::FireMonitorMessage {
351            destination_ident,
352            to_monitor,
353            is_timeout,
354        } => ConnectionCommand::SendMessage {
355            destination_ident,
356            payload: SendPayload::FireMonitor {
357                to_monitor,
358                is_timeout,
359            },
360        },
361        WireFrame::Establish {
362            role,
363            ident,
364            name_for_other,
365            alive,
366        } => ConnectionCommand::Establish {
367            role,
368            ident,
369            name_for_other,
370            alive,
371        },
372        WireFrame::PublishRoutes { live, dead } => ConnectionCommand::PublishRoutes { live, dead },
373        WireFrame::UpdateMonitorSubscription {
374            listener,
375            target,
376            op,
377        } => ConnectionCommand::UpdateMonitorSubscription {
378            listener,
379            target,
380            op,
381        },
382        WireFrame::Severed { reason } => ConnectionCommand::Severed { reason },
383        WireFrame::PublishGatewayRoutes { live } => {
384            ConnectionCommand::PublishGatewayRoutes { live }
385        }
386        WireFrame::GatewayDied { dead } => ConnectionCommand::GatewayDied { dead },
387    })
388}

390/// Serialize and write one [`SideChannelMessage`]. A `Send` of an actor message writes a
391/// routing-only header then its parts via [`PartWriter::write_message_parts`] (planned and, if
392/// large, striped by `part_writer`); every other action is a header-only frame.
393pub(crate) async fn write_side_channel<W, N>(
394    writer: &mut W,
395    message: SideChannelMessage,
396    part_writer: &mut PartWriter<N>,
397) -> std::io::Result<()>
398where
399    W: AsyncWrite + Unpin,
400    N: Net,
401{
402    let SideChannelMessage {
403        gateway_for_actor,
404        action,
405    } = message;
406    let frame = match action {
407        SideChannelAction::Send(SendPayload::ActorMessage(parts)) => {
408            write_header(
409                writer,
410                &SideChannelFrame::SendActorMessage { gateway_for_actor },
411            )
412            .await?;
413            return part_writer.write_message_parts(writer, parts).await;
414        }
415        SideChannelAction::Send(SendPayload::FireMonitor {
416            to_monitor,
417            is_timeout,
418        }) => SideChannelFrame::SendFireMonitor {
419            gateway_for_actor,
420            to_monitor,
421            is_timeout,
422        },
423        SideChannelAction::UpdateRemoteMonitorState { listener, op } => {
424            SideChannelFrame::UpdateRemoteMonitorState {
425                gateway_for_actor,
426                listener,
427                op,
428            }
429        }
430        SideChannelAction::AckRemoteMonitor { monitoring } => SideChannelFrame::AckRemoteMonitor {
431            gateway_for_actor,
432            monitoring,
433        },
434    };
435    write_frame(writer, &frame).await
436}

457/// Read and decode one side-channel frame into its [`SideChannelMessage`]. A `Send` of an
458/// actor message reads its parts through `parts` (exactly like [`read_frame`]); every
459/// other action is header-only. Heartbeats travel on the companion heartbeat stream (see
460/// [`read_side_channel_heartbeat`]) and never appear here.
461pub(crate) async fn read_side_channel<R, N>(
462    reader: &mut R,
463    part_reader: &mut PartReader<N>,
464) -> std::io::Result<SideChannelMessage>
465where
466    R: AsyncRead + Unpin,
467    N: Net,
468{
469    Ok(match read_header::<_, SideChannelFrame>(reader).await? {
470        SideChannelFrame::SendActorMessage { gateway_for_actor } => SideChannelMessage {
471            gateway_for_actor,
472            action: SideChannelAction::Send(SendPayload::ActorMessage(
473                part_reader.read_message_parts(reader).await?,
474            )),
475        },
476        SideChannelFrame::SendFireMonitor {
477            gateway_for_actor,
478            to_monitor,
479            is_timeout,
480        } => SideChannelMessage {
481            gateway_for_actor,
482            action: SideChannelAction::Send(SendPayload::FireMonitor {
483                to_monitor,
484                is_timeout,
485            }),
486        },
487        SideChannelFrame::UpdateRemoteMonitorState {
488            gateway_for_actor,
489            listener,
490            op,
491        } => SideChannelMessage {
492            gateway_for_actor,
493            action: SideChannelAction::UpdateRemoteMonitorState { listener, op },
494        },
495        SideChannelFrame::AckRemoteMonitor {
496            gateway_for_actor,
497            monitoring,
498        } => SideChannelMessage {
499            gateway_for_actor,
500            action: SideChannelAction::AckRemoteMonitor { monitoring },
501        },
502    })
503}
