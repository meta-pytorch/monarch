25pub(crate) struct ActorEntry {
26    pub(crate) name: ActorName,
27    pub(crate) delivery: Delivery,
28    pub(crate) parent: Option<Connection>,
29    pub(crate) children: SlotMap<ChildConnectionKey, Connection>,
30    /// Routing table: per destination ident, where it lives (or that it is dead)
31    /// *and* the monitor subscriptions this actor is responsible for on it. Monitors
32    /// live on the route because they share its key and travel the same paths: a
33    /// monitor registered against an `Unknown` route is forwarded up exactly like a
34    /// buffered message when this actor gains a parent, and one on a `Connection`
35    /// route fires when that route transitions to `Dead`.
36    pub(crate) routes: HashMap<Vec<u8>, Route>,
37    /// Monitors this actor *created* (it is the monitoring/`dest` side), grouped by
38    /// the *monitored target* ident. All of an actor's monitors on one target share
39    /// a single upstream subscription: the first creates it, the rest just join,
40    /// and they all fire together when that target dies (a fire carries only the
41    /// dead ident — no per-monitor id is needed on the wire). The matching
42    /// `Unsubscribe` is debounced via [`TargetMonitors::unsub_timer`], so rapid
43    /// monitor/cancel churn on one target sends no per-cycle traffic.
44    pub(crate) monitored: HashMap<Vec<u8>, TargetMonitors>,
45    /// Whether this actor is a gateway, and if so the state only a gateway holds.
46    /// A gateway is the entry point for its process group, with no parent or a
47    /// network (quic/tcp) parent; it owns the shared-memory slab for the actors that
48    /// route through it. Set once at creation and never flipped, so gateway-only
49    /// state is unrepresentable on a non-gateway. Joining a gateway to a non-network
50    /// parent is rejected.
51    pub(crate) gateway: GatewayState,
52    /// Per child connection, the set of *gateway* tags reachable through it. Unlike
53    /// [`Self::routes`] (which stops at gateway boundaries), gateway routes are
54    /// carried all the way up to the root and exist solely to publish gateway
55    /// deaths: when a child connection fails, every gateway tag it carried is
56    /// implicitly dead, and a death broadcast fans back out down exactly these
57    /// connections. Held even at non-gateway actors, so a nested gateway under a
58    /// non-gateway parent is still announced when that parent (and thus the
59    /// connection to the gateway) dies.
60    pub(crate) gateway_routes: HashMap<ChildConnectionKey, HashSet<Vec<u8>>>,
61    /// Root-only handle to this actor's gateway-death coalescing coroutine, spawned
62    /// lazily on the first death that turns around here. A death wave climbs to the
63    /// root and, rather than fanning out one broadcast per detected death
64    /// (O(deaths x gateways) — a "dead-gateway storm" when many machines fail at
65    /// once), each dead-tag packet is written to this channel; the coroutine owns
66    /// all the timing and emits one coalesced downward broadcast per window. `None`
67    /// until the first turn-around (and always `None` on non-root actors); dropping
68    /// it on actor teardown closes the channel and ends the coroutine.
69    pub(crate) gateway_death_tx: Option<UnboundedSender<Vec<Vec<u8>>>>,
70    pub(crate) alive: bool,
71    /// This actor's gateway shared-memory client, once it learns its gateway (the
72    /// context's shared slab for a gateway, or a parent's slab received via
73    /// gateway-state handoff). The value is `Option<ShmClient>`; it lives behind an
74    /// `Arc<Mutex<_>>` only so the actor's transport coroutines can read it and the
75    /// reader can set it the moment a handoff arrives.
76    pub(crate) shm_client: ShmClientSlot,
77}

79/// State of this actor's monitoring of one target ident. The variant *is* the
80/// state — no generation bookkeeping — so a debounced unsubscribe is correct by
81/// construction: re-monitoring aborts the pending task and flips back to
82/// `Active`, and the timer's command only acts if the entry is still
83/// `PendingUnsubscribe` when it is processed.
84pub(crate) enum TargetMonitors {
85    /// Live local monitors: monitor id → its failure-message prefix. The prefix
86    /// never travels; the full message is reconstructed here when the target
87    /// dies. (For an unnamed actor these are recorded but the upstream `Subscribe`
88    /// is held until naming.)
89    Active {
90        entries: HashMap<u64, Vec<MsgPart>>,
91        /// Effective non-existence timeout (ms; 0 = none) for this target, set by
92        /// the *first* monitor on it. Kept here so a `Subscribe` deferred while
93        /// the actor is unnamed can still carry the timeout once it is named.
94        timeout_ms: u64,
95    },
96    /// Every monitor on this target was cancelled; the upstream subscription is
97    /// still live, but a debounced unsubscribe task is scheduled. Holds that
98    /// task's abort handle so re-monitoring (or a fire) cancels it.
99    PendingUnsubscribe(tokio::task::AbortHandle),
100}
101
102/// Whether an actor is a gateway, carrying the gateway-only state inline in the
103/// `Gateway` variant so it cannot exist on a non-gateway. Set once at creation and
104/// never flipped; callers `match` on it directly to reach the fields.
105pub(crate) enum GatewayState {
106    NotAGateway,
107    Gateway {
108        /// Per *owning* gateway tag, the cross-gateway monitor state this gateway
109        /// holds for it. A tag with no entry is assumed alive and carries no
110        /// monitors; a [`GatewayMonitors::Dead`] entry both records the death (so a
111        /// repeated death wave is deduplicated and a new subscribe fires at once)
112        /// and replaces the base's separate dead-gateway set. The map is consulted
113        /// when a death is broadcast (to fire the monitors held for the now-dead
114        /// gateway) and when filtering stale routes to a known-dead gateway.
115        gateway_state: HashMap<Vec<u8>, GatewayMonitors>,
116    },
117}
118
119/// The cross-gateway monitor state a gateway holds for one *owning* gateway tag.
120pub(crate) enum GatewayMonitors {
121    /// The owning gateway is known dead. Encodes "already dead" directly: a new
122    /// subscribe to a target on it fires immediately, and a repeat death broadcast
123    /// for the tag is deduplicated.
124    Dead,
125    /// Live monitors this (subscribing) gateway holds on actors owned by the tag's
126    /// gateway. Each fires its `listener` when the owning gateway is known dead.
127    Subscribed(Vec<MonitorToFire>),
128}
129
130/// One cross-gateway monitor held at the *subscribing* gateway. Fires `listener`
131/// when `monitoring` (on the owning gateway) is known to have died.
132pub(crate) struct MonitorToFire {
133    /// Whether the owning gateway has acknowledged this registration. Until then
134    /// the "must exist" timer may declare the gateway dead (the gateway never
135    /// answered, so it is treated as unreachable).
136    pub(crate) acked: bool,
137    /// The actor to fire to (the monitoring actor, `== dest`).
138    pub(crate) listener: Vec<u8>,
139    /// The monitored target ident on the owning gateway.
140    pub(crate) monitoring: Vec<u8>,
141}

143impl ActorEntry {
144    pub(crate) fn new(ident: Option<Vec<u8>>, gateway: bool) -> Self {
145        Self {
146            name: ActorName::new(ident),
147            delivery: Delivery::NoPoller {
148                buffered: VecDeque::new(),
149            },
150            parent: None,
151            children: SlotMap::with_key(),
152            routes: HashMap::new(),
153            monitored: HashMap::new(),
154            gateway: if gateway {
155                GatewayState::Gateway {
156                    gateway_state: HashMap::new(),
157                }
158            } else {
159                GatewayState::NotAGateway
160            },
161            gateway_routes: HashMap::new(),
162            gateway_death_tx: None,
163            alive: true,
164            shm_client: Arc::new(Mutex::new(None)),
165        }
166    }
167
168    /// Whether this actor is a gateway.
169    pub(crate) fn is_gateway(&self) -> bool {
170        matches!(self.gateway, GatewayState::Gateway { .. })
171    }
172
173    /// Record that `ident` is live through child connection `child`, so that if that
174    /// connection later fails we can mark exactly those idents dead in
175    /// O(idents-handled) instead of scanning the whole routing table. Only live
176    /// idents are tracked; ones already dead need no re-killing on failure.
177    pub(crate) fn record_routed_via_child(&mut self, child: ChildConnectionKey, ident: &[u8]) {
178        if let Some(Connection::Established { routed_idents, .. }) = self.children.get_mut(child) {
179            routed_idents.insert(ident.to_vec());
180        }
181    }
182
183    /// This actor's ident, or `None` if it has not been named yet.
184    pub(crate) fn name(&self) -> Option<&[u8]> {
185        match &self.name {
186            ActorName::Named(name) => Some(name),
187            ActorName::Unknown { .. } => None,
188        }
189    }
190
191    pub(crate) fn subscribe(
192        &mut self,
193        poller_key: PollerKey,
194        index: usize,
195    ) -> anyhow::Result<VecDeque<Vec<MsgPart>>> {
196        let old = std::mem::replace(&mut self.delivery, Delivery::Poller { poller_key, index });
197
198        let Delivery::NoPoller { buffered } = old else {
199            self.delivery = old;
200            anyhow::bail!("actor is already subscribed to a poller");
201        };
202
203        Ok(buffered)
204    }
205
206    pub(crate) fn unsubscribe(&mut self) {
207        self.delivery = Delivery::NoPoller {
208            buffered: VecDeque::new(),
209        };
210    }
211
212    pub(crate) fn poller_subscription(&self) -> Option<(PollerKey, usize)> {
213        match &self.delivery {
214            Delivery::NoPoller { .. } => None,
215            Delivery::Poller {
216                poller_key, index, ..
217            } => Some((*poller_key, *index)),
218        }
219    }
220
221    /// Buffer a payload addressed to a destination whose route is not yet known
222    /// here, so it can be flushed once the route becomes known or this actor gains
223    /// a parent. This is the only place payloads enter a [`Route::Unknown`] buffer.
224    pub(crate) fn buffer_unrouted(&mut self, destination: Vec<u8>, payload: SendPayload) {
225        match self
226            .routes
227            .entry(destination)
228            .or_insert_with(Route::new_unknown)
229        {
230            Route::Unknown { messages, .. } => messages.push(payload),
231            _ => unreachable!("a known route would have been used to send instead of buffering"),
232        }
233    }
234
235    /// Register a monitor subscription on `to_monitor`'s route. Held on the route
236    /// (creating an `Unknown` buffer if there is none yet) until the route turns
237    /// `Dead` (fired) or this actor gains a parent (forwarded up, for `Unknown`).
238    /// The caller guarantees the route is not `Dead` (a dead route fires at once).
239    pub(crate) fn buffer_monitor(&mut self, to_monitor: Vec<u8>, sub: MonitorSub) {
240        match self
241            .routes
242            .entry(to_monitor)
243            .or_insert_with(Route::new_unknown)
244        {
245            Route::Unknown { monitors, .. } | Route::Connection { monitors, .. } => {
246                monitors.push(sub)
247            }
248            Route::Dead => unreachable!("a dead route fires immediately and never registers"),
249        }
250    }
251}

253/// A destination ident's entry in an actor's routing table. Carries the monitor
254/// subscriptions this actor is responsible for on that destination alongside its
255/// reachability, so the two stay in lock-step (same key, same propagation path).
256pub(crate) enum Route {
257    /// The destination is not (yet) known here — it may not have been created.
258    /// Messages and monitor subscriptions for it are buffered until a route is
259    /// published (then carried into [`Route::Connection`]) or this actor gains a
260    /// parent (then forwarded up).
261    Unknown {
262        messages: Vec<SendPayload>,
263        monitors: Vec<MonitorSub>,
264    },
265    /// The destination is reachable through this child connection. Monitors here
266    /// fire when the route transitions to [`Route::Dead`].
267    Connection {
268        child: ChildConnectionKey,
269        monitors: Vec<MonitorSub>,
270    },
271    /// The destination was reachable but has since died (directly, or because a
272    /// connection on the path to it failed). Messages for it are dropped; its
273    /// monitors were fired and removed at the moment of transition.
274    Dead,
275}
276
277impl Route {
278    pub(crate) fn new_unknown() -> Self {
279        Route::Unknown {
280            messages: Vec::new(),
281            monitors: Vec::new(),
282        }
283    }
284
285    /// The monitor subscriptions held on this route, if it can hold any (i.e. it is
286    /// not `Dead`).
287    pub(crate) fn monitors_mut(&mut self) -> Option<&mut Vec<MonitorSub>> {
288        match self {
289            Route::Unknown { monitors, .. } | Route::Connection { monitors, .. } => Some(monitors),
290            Route::Dead => None,
291        }
292    }
293}

295/// A subscription held at the responsible common ancestor `R`: when the watched
296/// ident dies, route a fire down to `dest` (the monitoring actor). One per
297/// monitoring actor per target — the dead ident itself identifies the fire, so
298/// no per-monitor id is carried.
299pub(crate) struct MonitorSub {
300    pub(crate) dest: Vec<u8>,
301    /// Non-existence timeout in ms; 0 = none. Travels with the subscription as it
302    /// is forwarded up the tree, so each ancestor that (re)installs it can arm a
303    /// fresh timer wherever the subscription currently lives.
304    pub(crate) timeout_ms: u64,
305}

307/// An actor's ident, which may not be known at creation (a child can be named by
308/// its parent when it joins).
309pub(crate) enum ActorName {
310    /// Not named yet. Monitors created in the meantime cannot address their
311    /// fire-backs (which route to this actor's own ident), so their upstream
312    /// `Subscribe` is held (in `monitored`) until a name is assigned, then sent.
313    Unknown {},
314    Named(Vec<u8>),
315}
316
317impl ActorName {
318    pub(crate) fn new(ident: Option<Vec<u8>>) -> Self {
319        match ident {
320            Some(name) => ActorName::Named(name),
321            None => ActorName::Unknown {},
322        }
323    }
324}
