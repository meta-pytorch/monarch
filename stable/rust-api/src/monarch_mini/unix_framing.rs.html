9//! Wire framing for the UNIX-socket transport, including shared-memory parts.
10//!
11//! Deliberately **separate** from the generic [`crate::framing`] used by quic:
12//! the UNIX transport is where machine-local shared-memory handling lives, so its
13//! wire format carries fd-passing and slab part descriptors that have no business
14//! in the cross-machine framing.
15//!
16//! A frame is `[u64 header_len][header][maybe fd-exchange][inline part bytes]`.
17//! The header is a bincode [`UnixFrame`]; the header and inline bytes are
18//! read/written with plain readiness-driven `try_read`/`try_write`. Only if the
19//! header says fds are present (one per [`PartDesc::Shm`] part) do we do one extra
20//! `sendmsg`/`recvmsg` step — a 1-byte payload carrying all the fds via
21//! `SCM_RIGHTS`. No-fd frames never touch the fd-passing syscalls.
22//!
23//! Per part on send: a part already in the slab is relayed by descriptor (no map,
24//! no copy); a large owned part on a connection that has a [`ShmClient`] is
25//! allocated, `memcpy`d into the slab, and sent as a descriptor + liveness token;
26//! everything else streams inline. On receive, an inline part is read into owned
27//! bytes and a slab part is reconstructed as *unmapped* metadata (mapped only when
28//! the user takes the bytes).
29

51/// One frame on the UNIX wire. A variant per [`ConnectionCommand`] that crosses
52/// the pipe (including `Establish`). Message-part *bytes* are never carried here —
53/// only inline-part lengths and slab-part descriptors.
54#[derive(Serialize, Deserialize)]
55enum UnixFrame {
56    Establish {
57        role: Role,
58        ident: Option<Vec<u8>>,
59        name_for_other: Option<Vec<u8>>,
60        alive: bool,
61    },
62    Message {
63        destination_ident: Vec<u8>,
64        payload: UnixPayload,
65    },
66    PublishRoutes {
67        live: Vec<Vec<u8>>,
68        dead: Vec<Vec<u8>>,
69    },
70    UpdateMonitorSubscription {
71        listener: Vec<u8>,
72        target: Vec<u8>,
73        op: MonitorOp,
74    },
75    Severed {
76        reason: Vec<u8>,
77    },
78    PublishGatewayRoutes {
79        live: Vec<Vec<u8>>,
80    },
81    GatewayDied {
82        dead: Vec<Vec<u8>>,
83    },
84    /// Gateway-state handoff: the header carries nothing; two fds (the slab object
85    /// then the dgram request socket) ride the fd-exchange step.
86    GatewayState,
87}

89/// Wire form of a message payload. An actor message carries a descriptor per part
90/// (inline length or slab offset/length); a monitor fire carries the small dead
91/// target ident inline.
92#[derive(Serialize, Deserialize)]
93enum UnixPayload {
94    ActorMessage {
95        parts: Vec<PartDesc>,
96    },
97    FireMonitor {
98        to_monitor: Vec<u8>,
99        is_timeout: bool,
100    },
101}
102
103/// How one message part is represented on the wire. `Inline` parts have their
104/// bytes streamed raw after the header; `Shm` parts carry only an offset+length
105/// into the slab and are accompanied (in part order) by one liveness-token fd in
106/// the fd-exchange step.
107#[derive(Serialize, Deserialize)]
108enum PartDesc {
109    Inline { len: u64 },
110    Shm { offset: u64, len: u64 },
111}
112
113fn bincode_config() -> bincode::config::Configuration {
114    bincode::config::standard()
115}

117/// Serialize and write one command over `stream`. For a message, large owned
118/// parts are moved into the slab (allocate via `client`, `memcpy` via `mapper`)
119/// and sent as descriptors + token fds; slab parts are relayed by descriptor;
120/// small/owned parts stream inline. `client`/`mapper` are unused by non-message
121/// commands and by messages with no large parts.
122pub(crate) async fn write_command(
123    stream: &UnixStream,
124    command: ConnectionCommand,
125    mapper: &MapperHandle,
126    client: Option<ShmClient>,
127) -> std::io::Result<()> {
128    match command {
129        ConnectionCommand::SendMessage {
130            destination_ident,
131            payload: SendPayload::ActorMessage(parts),
132        } => write_actor_message(stream, destination_ident, parts, mapper, client).await,
133        ConnectionCommand::SendMessage {
134            destination_ident,
135            payload:
136                SendPayload::FireMonitor {
137                    to_monitor,
138                    is_timeout,
139                },
140        } => {
141            let frame = UnixFrame::Message {
142                destination_ident,
143                payload: UnixPayload::FireMonitor {
144                    to_monitor,
145                    is_timeout,
146                },
147            };
148            write_header(stream, &frame).await
149        }
150        ConnectionCommand::Establish {
151            role,
152            ident,
153            name_for_other,
154            alive,
155        } => {
156            write_header(
157                stream,
158                &UnixFrame::Establish {
159                    role,
160                    ident,
161                    name_for_other,
162                    alive,
163                },
164            )
165            .await
166        }
167        ConnectionCommand::PublishRoutes { live, dead } => {
168            write_header(stream, &UnixFrame::PublishRoutes { live, dead }).await
169        }
170        ConnectionCommand::UpdateMonitorSubscription {
171            listener,
172            target,
173            op,
174        } => {
175            write_header(
176                stream,
177                &UnixFrame::UpdateMonitorSubscription {
178                    listener,
179                    target,
180                    op,
181                },
182            )
183            .await
184        }
185        ConnectionCommand::Severed { reason } => {
186            write_header(stream, &UnixFrame::Severed { reason }).await
187        }
188        ConnectionCommand::PublishGatewayRoutes { live } => {
189            write_header(stream, &UnixFrame::PublishGatewayRoutes { live }).await
190        }
191        ConnectionCommand::GatewayDied { dead } => {
192            write_header(stream, &UnixFrame::GatewayDied { dead }).await
193        }
194        ConnectionCommand::GatewayState { client } => {
195            write_header(stream, &UnixFrame::GatewayState).await?;
196            // Order matters: slab fd first, then the dgram request socket — the
197            // reader reconstructs in the same order.
198            send_fd_exchange(stream, &[client.slab_fd(), client.dgram_fd()]).await
199        }
200    }
201}

203/// Build and write a `Message` frame: classify each part (relay slab / shm-ify
204/// large owned / inline), write the header, hand over the token fds in one
205/// fd-exchange step, then stream the inline bytes.
206async fn write_actor_message(
207    stream: &UnixStream,
208    destination_ident: Vec<u8>,
209    parts: Vec<MsgPart>,
210    mapper: &MapperHandle,
211    client: Option<ShmClient>,
212) -> std::io::Result<()> {
213    let mut descs = Vec::with_capacity(parts.len());
214    let mut tokens: Vec<OwnedFd> = Vec::new();
215    let mut inline: Vec<MsgPart> = Vec::new();
216
217    for part in parts {
218        if part.is_shm() {
219            // Already in the slab: relay the descriptor and its token, no copy.
220            let (offset, len, token) = part.into_shm();
221            descs.push(PartDesc::Shm { offset, len });
222            tokens.push(token);
223            continue;
224        }
225        let len = part.len() as u64;
226        match client {
227            Some(client) if len >= SHM_THRESHOLD => {
228                let (offset, token) = client.allocate(len).await?;
229                let dst = {
230                    let mut mapper = mapper.lock().expect("shm mapper mutex poisoned");
231                    // SAFETY: `offset` was just granted for `len` bytes against
232                    // `client`'s slab, so the file covers it and the mapper can map it.
233                    unsafe { mapper.map(client.slab_fd(), offset, len as usize)? }
234                };
235                let src = part.as_bytes();
236                // SAFETY: `dst` is a writable mapping of `len` bytes (just mapped)
237                // and `src` is exactly `len` bytes; the regions do not overlap (one
238                // is the user buffer, the other the slab).
239                unsafe { std::ptr::copy_nonoverlapping(src.as_ptr(), dst, len as usize) };
240                descs.push(PartDesc::Shm { offset, len });
241                tokens.push(token);
242                // `part` (the owned bytes) drops here, freeing the user buffer.
243            }
244            _ => {
245                descs.push(PartDesc::Inline { len });
246                inline.push(part);
247            }
248        }
249    }
250
251    let frame = UnixFrame::Message {
252        destination_ident,
253        payload: UnixPayload::ActorMessage { parts: descs },
254    };
255    write_header(stream, &frame).await?;
256
257    if !tokens.is_empty() {
258        let raw: Vec<RawFd> = tokens.iter().map(AsRawFd::as_raw_fd).collect();
259        send_fd_exchange(stream, &raw).await?;
260        // The kernel has duplicated the tokens to the peer; drop our copies so the
261        // grant's liveness now rides with the peer (and any relay hop).
262        drop(tokens);
263    }
264
265    for part in &inline {
266        write_all(stream, part.as_bytes()).await?;
267    }
268    Ok(())
269}

271/// Serialize `frame` and write `[u64 len][header]`.
272async fn write_header(stream: &UnixStream, frame: &UnixFrame) -> std::io::Result<()> {
273    let header = bincode::serde::encode_to_vec(frame, bincode_config())
274        .map_err(|err| std::io::Error::new(std::io::ErrorKind::InvalidData, err))?;
275    write_all(stream, &(header.len() as u64).to_le_bytes()).await?;
276    write_all(stream, &header).await
277}

279/// Read one frame and map it back to a [`ConnectionCommand`]. Inline parts are
280/// read into owned buffers; slab parts are reconstructed as unmapped metadata
281/// using this connection's `mapper` and `client` (for the slab fd) plus the token
282/// fds received in the fd-exchange step.
283pub(crate) async fn read_command(
284    stream: &UnixStream,
285    mapper: &MapperHandle,
286    client: Option<ShmClient>,
287) -> std::io::Result<ConnectionCommand> {
288    let mut len_buf = [0u8; 8];
289    read_exact(stream, &mut len_buf).await?;
290    let header_len = u64::from_le_bytes(len_buf) as usize;
291
292    let mut header = vec![0u8; header_len];
293    read_exact(stream, &mut header).await?;
294    let (frame, _) = bincode::serde::decode_from_slice::<UnixFrame, _>(&header, bincode_config())
295        .map_err(|err| std::io::Error::new(std::io::ErrorKind::InvalidData, err))?;
296
297    Ok(match frame {
298        UnixFrame::Message {
299            destination_ident,
300            payload,
301        } => {
302            let payload = read_message_payload(stream, payload, mapper, client).await?;
303            ConnectionCommand::SendMessage {
304                destination_ident,
305                payload,
306            }
307        }
308        UnixFrame::Establish {
309            role,
310            ident,
311            name_for_other,
312            alive,
313        } => ConnectionCommand::Establish {
314            role,
315            ident,
316            name_for_other,
317            alive,
318        },
319        UnixFrame::PublishRoutes { live, dead } => ConnectionCommand::PublishRoutes { live, dead },
320        UnixFrame::UpdateMonitorSubscription {
321            listener,
322            target,
323            op,
324        } => ConnectionCommand::UpdateMonitorSubscription {
325            listener,
326            target,
327            op,
328        },
329        UnixFrame::Severed { reason } => ConnectionCommand::Severed { reason },
330        UnixFrame::PublishGatewayRoutes { live } => {
331            ConnectionCommand::PublishGatewayRoutes { live }
332        }
333        UnixFrame::GatewayDied { dead } => ConnectionCommand::GatewayDied { dead },
334        UnixFrame::GatewayState => {
335            // Two fds in the order they were sent: slab object, then dgram socket.
336            // They must stay open for the process lifetime, so leak the received
337            // owned fds into raw fds the (non-owning, `Copy`) ShmClient holds.
338            let mut fds = recv_fd_exchange(stream, 2).await?.into_iter();
339            let slab = fds.next().expect("two fds checked by recv_fd_exchange");
340            let dgram = fds.next().expect("two fds checked by recv_fd_exchange");
341            ConnectionCommand::GatewayState {
342                client: ShmClient::from_raw(dgram.into_raw_fd(), slab.into_raw_fd()),
343            }
344        }
345    })
346}

348/// Reconstruct a message payload's parts: collect token fds (one per slab part)
349/// in the single fd-exchange step, then walk the descriptors in order, reading
350/// inline bytes and pairing each slab descriptor with the next token fd.
351async fn read_message_payload(
352    stream: &UnixStream,
353    payload: UnixPayload,
354    mapper: &MapperHandle,
355    client: Option<ShmClient>,
356) -> std::io::Result<SendPayload> {
357    let descs = match payload {
358        UnixPayload::ActorMessage { parts } => parts,
359        UnixPayload::FireMonitor {
360            to_monitor,
361            is_timeout,
362        } => {
363            return Ok(SendPayload::FireMonitor {
364                to_monitor,
365                is_timeout,
366            });
367        }
368    };
369
370    let n_fds = descs
371        .iter()
372        .filter(|d| matches!(d, PartDesc::Shm { .. }))
373        .count();
374    let mut tokens = if n_fds > 0 {
375        recv_fd_exchange(stream, n_fds).await?.into_iter()
376    } else {
377        Vec::new().into_iter()
378    };
379
380    let mut parts = Vec::with_capacity(descs.len());
381    for desc in descs {
382        match desc {
383            PartDesc::Inline { len } => {
384                let mut buf = vec![0u8; len as usize];
385                read_exact(stream, &mut buf).await?;
386                parts.push(MsgPart::from_bytes(buf));
387            }
388            PartDesc::Shm { offset, len } => {
389                let token = tokens.next().ok_or_else(|| {
390                    std::io::Error::new(std::io::ErrorKind::InvalidData, "missing shm token fd")
391                })?;
392                let client = client.ok_or_else(|| {
393                    std::io::Error::new(
394                        std::io::ErrorKind::InvalidData,
395                        "received a slab part but have no shm client",
396                    )
397                })?;
398                parts.push(MsgPart::new_shm(
399                    mapper.clone(),
400                    client.slab_fd(),
401                    token,
402                    offset,
403                    len,
404                ));
405            }
406        }
407    }
408    Ok(SendPayload::ActorMessage(parts))
409}

415/// Fill `buf` exactly, awaiting readability between partial reads. EOF before the
416/// buffer is full is an error (a truncated frame).
417async fn read_exact(stream: &UnixStream, buf: &mut [u8]) -> std::io::Result<()> {
418    let mut filled = 0;
419    while filled < buf.len() {
420        stream.readable().await?;
421        match stream.try_read(&mut buf[filled..]) {
422            Ok(0) => {
423                return Err(std::io::Error::new(
424                    std::io::ErrorKind::UnexpectedEof,
425                    "unix frame truncated",
426                ));
427            }
428            Ok(n) => filled += n,
429            Err(err) if err.kind() == std::io::ErrorKind::WouldBlock => continue,
430            Err(err) => return Err(err),
431        }
432    }
433    Ok(())
434}
435
436/// Write all of `buf`, awaiting writability between partial writes.
437async fn write_all(stream: &UnixStream, buf: &[u8]) -> std::io::Result<()> {
438    let mut sent = 0;
439    while sent < buf.len() {
440        stream.writable().await?;
441        match stream.try_write(&buf[sent..]) {
442            Ok(n) => sent += n,
443            Err(err) if err.kind() == std::io::ErrorKind::WouldBlock => continue,
444            Err(err) => return Err(err),
445        }
446    }
447    Ok(())
448}

450/// The fd-exchange step on send: one `sendmsg` of a single marker byte carrying
451/// all `fds` via `SCM_RIGHTS`, driven under write readiness.
452async fn send_fd_exchange(stream: &UnixStream, fds: &[RawFd]) -> std::io::Result<()> {
453    loop {
454        stream.writable().await?;
455        match stream.try_io(Interest::WRITABLE, || {
456            send_with_fds(stream.as_raw_fd(), &[0u8], fds)
457        }) {
458            Ok(_) => return Ok(()),
459            Err(err) if err.kind() == std::io::ErrorKind::WouldBlock => continue,
460            Err(err) => return Err(err),
461        }
462    }
463}
464
465/// The fd-exchange step on receive: one `recvmsg` of the single marker byte,
466/// collecting the `expected` token fds. The 1-byte iovec bounds the read to the
467/// marker so following inline bytes stay in the stream.
468async fn recv_fd_exchange(stream: &UnixStream, expected: usize) -> std::io::Result<Vec<OwnedFd>> {
469    loop {
470        stream.readable().await?;
471        let mut marker = [0u8; 1];
472        match stream.try_io(Interest::READABLE, || {
473            recv_with_fds(stream.as_raw_fd(), &mut marker, expected)
474        }) {
475            Ok((0, _)) => {
476                return Err(std::io::Error::new(
477                    std::io::ErrorKind::UnexpectedEof,
478                    "unix frame truncated before fd exchange",
479                ));
480            }
481            Ok((_, fds)) => {
482                if fds.len() != expected {
483                    return Err(std::io::Error::new(
484                        std::io::ErrorKind::InvalidData,
485                        "unexpected fd count in fd exchange",
486                    ));
487                }
488                return Ok(fds);
489            }
490            Err(err) if err.kind() == std::io::ErrorKind::WouldBlock => continue,
491            Err(err) => return Err(err),
492        }
493    }
494}

510    fn run_local<F: std::future::Future>(future: F) -> F::Output {
511        let rt = tokio::runtime::Builder::new_current_thread()
512            .enable_all()
513            .build()
514            .expect("build runtime");
515        tokio::task::LocalSet::new().block_on(&rt, future)
516    }
517
518    fn mapper() -> MapperHandle {
519        Arc::new(Mutex::new(ShmMapper::new()))
520    }

522    /// A `[small header, big payload]` message: the header stays inline and the
523    /// payload travels through the slab (a descriptor + token on the socket, not
524    /// the bytes), and both arrive intact and in order. The two mappers stand in
525    /// for the sender's and receiver's separate contexts mapping the one slab.
526    #[test]
527    fn large_part_goes_through_slab_small_part_inline() {
528        run_local(async {
529            let server = ShmServer::new().expect("shm server");
530            let client = server.client();
531            let send_mapper = mapper();
532            let recv_mapper = mapper();
533            let (a, b) = UnixStream::pair().expect("socketpair");
534
535            let header = b"header".to_vec();
536            let payload = vec![0x42u8; SHM_THRESHOLD as usize + 100];
537            let command = ConnectionCommand::SendMessage {
538                destination_ident: b"dest".to_vec(),
539                payload: SendPayload::ActorMessage(vec![
540                    MsgPart::from_bytes(header.clone()),
541                    MsgPart::from_bytes(payload.clone()),
542                ]),
543            };
544
545            let (write_res, read_res) = tokio::join!(
546                write_command(&a, command, &send_mapper, Some(client)),
547                read_command(&b, &recv_mapper, Some(client)),
548            );
549            write_res.expect("write");
550            let received = read_res.expect("read");
551
552            let ConnectionCommand::SendMessage {
553                destination_ident,
554                payload: SendPayload::ActorMessage(parts),
555            } = received
556            else {
557                panic!("expected an actor message");
558            };
559            assert_eq!(destination_ident, b"dest");
560            assert_eq!(parts.len(), 2, "both parts reconstructed");
561            assert_eq!(
562                parts[0].as_bytes(),
563                header.as_slice(),
564                "inline header intact"
565            );
566            assert_eq!(
567                parts[1].as_bytes(),
568                payload.as_slice(),
569                "slab payload intact and in order"
570            );
571        });
572    }

574    /// With no shm client, even a large part falls back to streaming inline.
575    #[test]
576    fn large_part_without_client_streams_inline() {
577        run_local(async {
578            let map = mapper();
579            let (a, b) = UnixStream::pair().expect("socketpair");
580            let payload = vec![0x7u8; SHM_THRESHOLD as usize + 100];
581            let command = ConnectionCommand::SendMessage {
582                destination_ident: b"d".to_vec(),
583                payload: SendPayload::ActorMessage(vec![MsgPart::from_bytes(payload.clone())]),
584            };
585
586            let (write_res, read_res) = tokio::join!(
587                write_command(&a, command, &map, None),
588                read_command(&b, &map, None),
589            );
590            write_res.expect("write");
591            let received = read_res.expect("read");
592
593            let ConnectionCommand::SendMessage {
594                payload: SendPayload::ActorMessage(parts),
595                ..
596            } = received
597            else {
598                panic!("expected an actor message");
599            };
600            assert_eq!(parts.len(), 1);
601            assert_eq!(
602                parts[0].as_bytes(),
603                payload.as_slice(),
604                "inline fallback intact"
605            );
606        });
607    }
