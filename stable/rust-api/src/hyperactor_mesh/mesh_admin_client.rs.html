9//! Shared TLS-aware `reqwest` client construction for mesh-admin
10//! HTTP clients.
11//!
12//! The mesh admin server requires mutual TLS. Building a correct
13//! `reqwest::Client` against it is subtle because fbcode Buck builds
14//! compile both `native-tls` and `rustls` features into reqwest, and
15//! the two backends use incompatible `Identity` constructors.
16//!
17//! This module centralizes that logic so that every mesh-admin client
18//! (admin TUI, integration tests, future tooling) shares the same
19//! correct code path.
20//!
21//! # Transport selection (highest priority first)
22//!
23//! 0. `--plaintext`: plain HTTP, ignoring any TLS material.
24//! 1. Explicit `http://` scheme: plain HTTP. Combining it with
25//!    `--tls-ca` / `--tls-cert` / `--tls-key` is a hard error.
26//! 2. Explicit `--tls-ca` (with optional `--tls-cert` + `--tls-key`
27//!    for mutual TLS): an unreadable or invalid CA, a cert/key that
28//!    cannot be read, or only one of cert/key, is a hard error.
29//!    `--tls-cert`/`--tls-key` without `--tls-ca` is also an error,
30//!    since they are unusable without a CA.
31//! 3. Auto-detection via [`hyperactor::channel::try_tls_pem_bundle`]
32//!    (OSS config first, then Meta well-known paths).
33//! 4. If no CA is found and TLS is expected (an `https://` scheme or,
34//!    at Meta, the default), that is a hard error; otherwise (OSS with
35//!    no certs) fall back to plain HTTP.
36//!
37//! Transport is **fail-closed**: when TLS is expected but cannot be
38//! configured, [`build_client`] returns an error rather than silently
39//! downgrading to plain HTTP.
40

43/// Connection options for a mesh admin server, mirroring the
44/// `--tls-ca`, `--tls-cert`, `--tls-key`, and `--plaintext` CLI flags.
45#[derive(Debug, Clone, Default)]
46pub struct AdminClientOptions {
47    /// Admin server address: `host:port`, `https://host:port`, or
48    /// `http://host:port`.
49    pub addr: String,
50    /// Path to a PEM CA certificate for TLS server verification.
51    pub tls_ca: Option<String>,
52    /// Path to a PEM client certificate for mutual TLS.
53    pub tls_cert: Option<String>,
54    /// Path to a PEM client private key for mutual TLS.
55    pub tls_key: Option<String>,
56    /// Disable TLS and use plain HTTP, overriding TLS auto-detection.
57    pub plaintext: bool,
58}

60/// Configure TLS on a `reqwest::ClientBuilder` by adding a root CA,
61/// and optionally a client identity (cert + key) for mutual TLS.
62///
63/// - `ca_bytes` must be a PEM-encoded CA certificate; if it cannot be
64///   parsed, this returns `(builder, false)` and leaves the builder
65///   unchanged.
66/// - If both `cert_bytes` and `key_bytes` are provided, they are
67///   concatenated and parsed as a PEM identity. Identity parse
68///   failures are non-fatal: the root CA remains installed and the
69///   function still returns `true`.
70///
71/// Returns `(updated_builder, ca_installed)`.
72pub fn add_tls(
73    builder: reqwest::ClientBuilder,
74    ca_bytes: &[u8],
75    cert_bytes: Option<Vec<u8>>,
76    key_bytes: Option<Vec<u8>>,
77) -> (reqwest::ClientBuilder, bool) {
78    let root_cert = match reqwest::Certificate::from_pem(ca_bytes) {
79        Ok(c) => c,
80        Err(e) => {
81            eprintln!("TLS: invalid CA PEM: {}", e);
82            return (builder, false);
83        }
84    };
85    let mut builder = builder.add_root_certificate(root_cert);
86
87    if let (Some(cert), Some(key)) = (cert_bytes, key_bytes) {
88        // reqwest's Identity type is backend-specific: from_pkcs8_pem
89        // creates a native-tls identity, from_pem creates a rustls
90        // identity. When both features are compiled (fbcode Buck builds),
91        // using the wrong variant silently fails at connect time with
92        // "incompatible TLS identity type".
93        //
94        // Meta's server.pem bundles certs + key in one file.
95        // from_pkcs8_pem requires the key as a separate buffer, so we
96        // split it out by finding the private key marker.
97        let combined = if cert == key {
98            cert
99        } else {
100            let mut c = cert;
101            c.extend_from_slice(&key);
102            c
103        };
104        let identity_result = {
105            // Split PEM into cert-only and key-only buffers for native-tls.
106            // reqwest 0.11 with both native-tls and rustls features compiled
107            // (fbcode Buck builds) defaults to the native-tls connector.
108            // Identity::from_pem creates a rustls-flavored identity that is
109            // silently rejected by native-tls at connect time. We must use
110            // from_pkcs8_pem (native-tls) in fbcode, and from_pem (rustls)
111            // in OSS where native-tls is excluded (D93626607).
112            let combined_str = String::from_utf8_lossy(&combined);
113            let key_markers = [
114                // @lint-ignore PRIVATEKEY
115                "-----BEGIN PRIVATE KEY-----",
116                // @lint-ignore PRIVATEKEY
117                "-----BEGIN RSA PRIVATE KEY-----",
118                // @lint-ignore PRIVATEKEY
119                "-----BEGIN EC PRIVATE KEY-----",
120            ];
121            let key_pos = key_markers
122                .iter()
123                .filter_map(|m| combined_str.find(m))
124                .min();
125            #[cfg(fbcode_build)]
126            {
127                if let Some(key_start) = key_pos {
128                    let cert_pem = combined_str[..key_start].trim().as_bytes();
129                    let key_pem = combined_str[key_start..].trim().as_bytes();
130                    reqwest::Identity::from_pkcs8_pem(cert_pem, key_pem)
131                } else {
132                    reqwest::Identity::from_pem(&combined)
133                }
134            }
135            #[cfg(not(fbcode_build))]
136            {
137                let _ = key_pos; // suppress unused warning
138                reqwest::Identity::from_pem(&combined)
139            }
140        };
141        match identity_result {
142            Ok(identity) => {
143                builder = builder.identity(identity);
144            }
145            Err(e) => eprintln!(
146                "WARNING: TLS: failed to parse client identity PEM: {}. \
147                 The mesh admin server requires mTLS — connection will fail \
148                 without a valid client certificate.",
149                e
150            ),
151        }
152    }
153
154    (builder, true)
155}

157/// Read all bytes from a [`Pem`](hyperactor::config::Pem), returning
158/// `None` if it can't be opened/read or if the result is empty.
159fn read_pem(pem: &hyperactor::config::Pem) -> Option<Vec<u8>> {
160    use std::io::Read;
161    let mut buf = Vec::new();
162    pem.reader().ok()?.read_to_end(&mut buf).ok()?;
163    if buf.is_empty() { None } else { Some(buf) }
164}

166/// Split an address into an optional `http`/`https` scheme and the
167/// remaining host part.
168///
169/// If `addr` starts with `https://` or `http://`, returns
170/// `(Some(scheme), rest)`. Otherwise returns `(None, addr)`. The
171/// returned `&str` values are slices of `addr`.
172fn parse_addr(addr: &str) -> (Option<&str>, &str) {
173    if let Some(host) = addr.strip_prefix("https://") {
174        (Some("https"), host)
175    } else if let Some(host) = addr.strip_prefix("http://") {
176        (Some("http"), host)
177    } else {
178        (None, addr)
179    }
180}

182/// Configure TLS on a `reqwest::ClientBuilder` using PEM files
183/// supplied via CLI (`--tls-ca`, and optionally `--tls-cert` +
184/// `--tls-key`).
185///
186/// Fail-closed: an unreadable or invalid CA is an error, as is a
187/// `--tls-cert`/`--tls-key` that cannot be read or that is supplied
188/// without its counterpart. (A cert/key that reads but fails to parse
189/// is handled by [`add_tls`], which keeps CA-only TLS; that still fails
190/// the handshake against an mTLS server rather than downgrading to
191/// plain HTTP.)
192fn add_tls_from_paths(
193    builder: reqwest::ClientBuilder,
194    ca_path: &str,
195    cert_path: Option<&str>,
196    key_path: Option<&str>,
197) -> io::Result<reqwest::ClientBuilder> {
198    let ca_bytes = std::fs::read(ca_path)
199        .map_err(|e| io::Error::other(format!("cannot read TLS CA file {ca_path}: {e}")))?;
200
201    let (cert_bytes, key_bytes) = match (cert_path, key_path) {
202        (Some(cert), Some(key)) => {
203            let cert_bytes = std::fs::read(cert)
204                .map_err(|e| io::Error::other(format!("cannot read TLS cert file {cert}: {e}")))?;
205            let key_bytes = std::fs::read(key)
206                .map_err(|e| io::Error::other(format!("cannot read TLS key file {key}: {e}")))?;
207            (Some(cert_bytes), Some(key_bytes))
208        }
209        (None, None) => (None, None),
210        _ => {
211            return Err(io::Error::other(
212                "--tls-cert and --tls-key must be provided together",
213            ));
214        }
215    };
216
217    let (builder, ca_installed) = add_tls(builder, &ca_bytes, cert_bytes, key_bytes);
218    if !ca_installed {
219        return Err(io::Error::other(format!(
220            "TLS CA file {ca_path} is not a valid PEM certificate"
221        )));
222    }
223    Ok(builder)
224}

226/// Configure TLS on a `reqwest::ClientBuilder` using an auto-detected
227/// hyperactor [`PemBundle`](hyperactor::config::PemBundle).
228///
229/// Fail-closed: a bundle whose CA is missing or is not a valid PEM is
230/// an error. A client cert/key are applied when both are readable.
231fn add_tls_from_bundle(
232    builder: reqwest::ClientBuilder,
233    bundle: &hyperactor::config::PemBundle,
234) -> io::Result<reqwest::ClientBuilder> {
235    let ca_bytes = read_pem(&bundle.ca)
236        .ok_or_else(|| io::Error::other("auto-detected TLS CA is missing or unreadable"))?;
237
238    let (builder, ca_installed) = add_tls(
239        builder,
240        &ca_bytes,
241        read_pem(&bundle.cert),
242        read_pem(&bundle.key),
243    );
244    if !ca_installed {
245        return Err(io::Error::other(
246            "auto-detected TLS CA is not a valid PEM certificate",
247        ));
248    }
249    Ok(builder)
250}

252/// Build a `reqwest` client and a base URL for a mesh admin server.
253///
254/// Returns `(base_url, client)` where `base_url` always includes the
255/// selected scheme (`http://...` or `https://...`). See the module
256/// docs for the transport-selection order.
257///
258/// Transport is **fail-closed**: when TLS is expected (an `https://`
259/// scheme, an explicit `--tls-ca`, or the Meta default) but no usable
260/// CA can be configured, this returns an error instead of silently
261/// using plain HTTP.
262pub fn build_client(config: &AdminClientOptions) -> io::Result<(String, reqwest::Client)> {
263    let (explicit_scheme, host) = parse_addr(&config.addr);
264    let has_tls_paths =
265        config.tls_ca.is_some() || config.tls_cert.is_some() || config.tls_key.is_some();
266
267    // TP-7: no client-level timeout. All timeout enforcement is
268    // per-operation via tokio::time::timeout at the call boundary.
269
270    // --plaintext is the explicit override: plain HTTP, ignore any TLS material.
271    if config.plaintext {
272        let client = reqwest::Client::builder()
273            .build()
274            .map_err(io::Error::other)?;
275        return Ok((format!("http://{host}"), client));
276    }
277
278    // An explicit http:// scheme also selects plain HTTP; combining it with TLS
279    // paths is contradictory.
280    if explicit_scheme == Some("http") {
281        if has_tls_paths {
282            return Err(io::Error::other(
283                "http:// scheme requested but TLS certificate paths were also given; use https:// or drop --tls-*",
284            ));
285        }
286        let client = reqwest::Client::builder()
287            .build()
288            .map_err(io::Error::other)?;
289        return Ok((format!("http://{host}"), client));
290    }
291
292    // A client cert/key is only consumed alongside a --tls-ca; without one the
293    // paths would be silently ignored, so reject the combination outright.
294    if config.tls_ca.is_none() && (config.tls_cert.is_some() || config.tls_key.is_some()) {
295        return Err(io::Error::other("--tls-cert/--tls-key require --tls-ca"));
296    }
297
298    // TLS is expected for an https:// scheme, an explicit CA, or (at Meta) by
299    // default. In OSS with no CA, plain HTTP is a legitimate fallback.
300    let tls_expected =
301        explicit_scheme == Some("https") || config.tls_ca.is_some() || cfg!(fbcode_build);
302
303    let mut builder = reqwest::Client::builder();
304    let use_tls = if let Some(ca_path) = &config.tls_ca {
305        builder = add_tls_from_paths(
306            builder,
307            ca_path,
308            config.tls_cert.as_deref(),
309            config.tls_key.as_deref(),
310        )?;
311        true
312    } else if let Some(bundle) = hyperactor::channel::try_tls_pem_bundle() {
313        builder = add_tls_from_bundle(builder, &bundle)?;
314        true
315    } else {
316        false
317    };
318
319    if !use_tls && tls_expected {
320        return Err(io::Error::other(
321            "TLS is required (an https:// scheme or the Meta default) but no CA certificate was found; pass --tls-ca or use --plaintext",
322        ));
323    }
324
325    let scheme = if use_tls { "https" } else { "http" };
326    let client = builder.build().map_err(io::Error::other)?;
327    Ok((format!("{scheme}://{host}"), client))
328}

362    /// A temp PEM file removed on drop, so a panicking assertion never
363    /// leaks it (RAII cleanup, no `tempfile` dep).
364    struct TempPem(PathBuf);
365
366    impl TempPem {
367        fn new(contents: &str) -> Self {
368            static N: AtomicU64 = AtomicU64::new(0);
369            let path = std::env::temp_dir().join(format!(
370                "mesh_admin_client_test_{}_{}.pem",
371                std::process::id(),
372                N.fetch_add(1, Ordering::Relaxed)
373            ));
374            std::fs::write(&path, contents).expect("write temp file");
375            Self(path)
376        }
377
378        fn path(&self) -> String {
379            self.0.to_string_lossy().into_owned()
380        }
381    }
382
383    impl Drop for TempPem {
384        fn drop(&mut self) {
385            std::fs::remove_file(&self.0).ok();
386        }
387    }
