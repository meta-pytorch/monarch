9//! Host lifecycle management for procs running on one machine.
10//!
11//! A [`Host`] owns one [`Gateway`] and one [`ProcManager`]. The manager makes
12//! procs real, either by spawning them in-process for tests or by launching
13//! separate OS processes. The gateway owns connectivity: it serves the backend
14//! and frontend endpoints, multiplexes inbound traffic to in-process procs,
15//! and routes spawned child proc gateways through peer routes.
16//!
17//! Use [`Host::new`] or [`Host::new_with_gateway`] to construct a host, start its
18//! backend and frontend accept loops, then [`Host::spawn`] to create child procs.
19//! Spawned children are returned as [`ProcAddr`] values whose location is
20//! advertised through this host's gateway.
21//!
22//! ## Gateway topology
23//!
24//! A host gateway exposes a frontend endpoint `*` and serves a backend
25//! endpoint `#` for child proc gateways. Each spawned child has its own
26//! gateway endpoint (`#1`, `#2`, ...), uses `#` as its forwarder, and is
27//! advertised as `Via(child_uid, host_location)`. The `host_location` is the
28//! host gateway's advertised location: the newest active frontend serve or
29//! `serve_via` session.
30//!
31//! The host gateway keeps one peer route per child uid. When it receives a
32//! message for `Via(child_uid, host_location)`, it peels the child hop and
33//! forwards the envelope to the child's gateway. In-process service and local
34//! procs are delivered through the gateway's local proc table.
35//!
36//! ```text
37//! inbound to host_location
38//! (*, or Via(...) when attached)
39//!          |
40//!          v
41//!   +---------------+     peer child_uid_1 -> #1     +----------------------+
42//!   | Host Gateway  |------------------------------->| child gateway/proc 1 |
43//!   | local procs:  |                                | addr #1              |
44//!   | service/local |<-------------------------------| forwarder -> #       |
45//!   +---------------+          host backend #         +----------------------+
46//!          |
47//!          | peer child_uid_2 -> #2
48//!          v
49//!   +----------------------+
50//!   | child gateway/proc 2 |
51//!   | addr #2              |
52//!   | forwarder -> #       |
53//!   +----------------------+
54//! ```
55//!
56//! The built-in service and local procs are created during construction and
57//! run in-process on the host gateway. The local proc starts with no actors; a
58//! `ProcAgent` and root client actor are added lazily when
59//! `HostMeshAgent::handle(GetLocalProc)` first asks for it.

199/// Client transport handles transferred to the bootstrap shutdown task.
200///
201/// Dropping this value stops any handles that have not already been joined,
202/// so cancellation cannot detach a live frontend or via transport.
203#[derive(Debug)]
204pub(crate) struct HostShutdownHandles {
205    frontend: Option<GatewayServeHandle>,
206    via: Option<GatewayServeHandle>,
207}
208
209impl HostShutdownHandles {
210    pub(crate) fn new(frontend: GatewayServeHandle, via: Option<GatewayServeHandle>) -> Self {
211        Self {
212            frontend: Some(frontend),
213            via,
214        }
215    }
216
217    pub(crate) async fn stop_and_join(mut self, reason: &str) {
218        if let Some(mut frontend) = self.frontend.take() {
219            frontend.stop(reason);
220            if let Err(error) = frontend.join().await {
221                tracing::warn!(%error, "failed to join host shutdown frontend");
222            }
223        }
224        if let Some(mut via) = self.via.take() {
225            via.stop(reason);
226            if let Err(error) = via.join().await {
227                tracing::warn!(%error, "failed to join host shutdown via transport");
228            }
229        }
230    }
231}

244struct ProcHandleKillGuard<H: ProcHandle> {
245    handle: Option<H>,
246}
247
248impl<H: ProcHandle> ProcHandleKillGuard<H> {
249    fn handle(&self) -> &H {
250        self.handle
251            .as_ref()
252            .expect("active spawn cleanup owns the proc handle")
253    }
254
255    fn disarm(mut self) {
256        self.handle = None;
257    }
258
259    fn maybe_kill(&mut self) -> impl Future<Output = ()> + Send + 'static {
260        let handle = self.handle.take();
261
262        async move {
263            let Some(handle) = handle else {
264                return;
265            };
266            match handle.kill().await {
267                Ok(_) | Err(TerminateError::AlreadyTerminated(_)) => {}
268                Err(error) => {
269                    tracing::warn!(proc = %handle.proc_addr(), %error, "failed to clean up proc spawn");
270                }
271            }
272        }
273    }
274}

276impl<H: ProcHandle> Drop for ProcHandleKillGuard<H> {
277    fn drop(&mut self) {
278        if self.handle.is_some() {
279            tokio::spawn(self.maybe_kill());
280        }
281    }

284/// Lifecycle manager for the procs on one machine.
285///
286/// The host delegates all connectivity to its [`Gateway`]. It creates
287/// built-in service/local procs, asks its [`ProcManager`] to spawn children,
288/// and keeps the gateway peer registrations for those children alive.
289pub struct Host<M> {
290    /// Peer guards for spawned child procs, keyed by name. The stored
291    /// [`PeerAttachGuard`] keeps the gateway peer route for the child
292    /// alive; dropping it removes the entry (used by
293    /// [`Host::terminate_children`] to free slots).
294    procs: Arc<StdMutex<HashMap<String, PeerAttachGuard>>>,
295    frontend_addr: ChannelAddr,
296    backend_addr: ChannelAddr,
297    /// Connectivity for every proc owned by this host.
298    ///
299    /// The built-in procs share the gateway in-process. Spawned children have
300    /// their own gateways and are registered here with
301    /// [`Gateway::attach_peer`].
302    gateway: Gateway,
303    frontend_handle: Option<GatewayServeHandle>,
304    backend_handle: Option<GatewayServeHandle>,
305    /// Duplex `serve_via` session to a remote gateway, present when this
306    /// host was bootstrapped out-of-cluster. Kept alive for the host's
307    /// lifetime so the cluster route and outbound forwarder persist.
308    via_handle: Option<GatewayServeHandle>,
309    manager: Arc<M>,
310    service_proc: Proc,
311    local_proc: Proc,
312}

314impl<M: ProcManager> Host<M> {
315    /// Construct a host and start its gateway frontend server on `addr`.
316    pub async fn new(manager: M, addr: ChannelAddr) -> Result<Self, HostError> {
317        Self::new_with_default(manager, addr, None).await
318    }

320    /// Like [`new`], but optionally uses an already-bound listener.
321    ///
322    /// When `listener` is `Some`, it is used as the frontend listening socket
323    /// instead of binding a new one.
324    pub async fn new_with_default(
325        manager: M,
326        addr: ChannelAddr,
327        listener: Option<std::net::TcpListener>,
328    ) -> Result<Self, HostError> {
329        // Default to the process-wide global gateway so procs on this
330        // host share one routing table with the rest of the process.
331        // Callers that need a different gateway (e.g. via attach) build
332        // it externally and pass it to [`new_with_gateway`].
333        Self::new_with_gateway(
334            manager,
335            ProcAddr::new(service_proc_id(), addr.into()),
336            listener,
337            Gateway::global().clone(),
338            None,
339        )
340        .await
341    }

343    /// Like [`new_with_default`], but uses a caller-provided [`Gateway`] and
344    /// service [`ProcAddr`].
345    ///
346    /// `addr` is a startup specification: its proc id becomes the service proc
347    /// identity, and its direct location is used to serve the frontend. The
348    /// final service proc location comes from the gateway after binding, so it
349    /// may differ because of an alias, an ephemeral address, or `via`. A
350    /// source-routed [`Location::Via`] is therefore rejected here.
351    ///
352    /// When `via` is `Some`, the gateway attaches to that remote duplex
353    /// address with [`Gateway::serve_via`] *after* the local
354    /// frontend/backend serves but *before* minting the built-in procs,
355    /// so the via session is the newest active serve. That ordering
356    /// makes every ref minted on this host advertise the routable
357    /// `Via` location rather than the bare local frontend, which is
358    /// what lets an out-of-cluster client receive return traffic over
359    /// the duplex.
360    #[hyperactor::instrument(fields(addr=addr.to_string()))]
361    pub async fn new_with_gateway(
362        manager: M,
363        addr: ProcAddr,
364        listener: Option<std::net::TcpListener>,
365        gateway: Gateway,
366        via: Option<ChannelAddr>,
367    ) -> Result<Self, HostError> {
368        assert!(
369            addr.uid().is_instance(),
370            "host service proc must have an instance id"
371        );
372        let service_proc_id = addr.id().clone();
373        let addr = match addr.location() {
374            Location::Addr(addr) => addr.clone(),
375            Location::Via(..) => {
376                return Err(HostError::InvalidFrontendLocation(addr.location().clone()));
377            }
378        };
379        let mut backend_handle = Gateway::serve(&gateway, ChannelAddr::any(manager.transport()))?;
380        let backend_addr = gateway.default_location().addr().clone();
381
382        // Serve the frontend. Kernel-socket (net) transports use a muxed
383        // listener so simplex clients and duplex attach clients share one
384        // address; the gateway owns both the simplex and (`AttachWire`)
385        // duplex accept paths, so there is a single attach protocol whether
386        // a frontend is muxed or a plain duplex endpoint. `serve_mux`
387        // requires a net transport, so we branch on `is_net()` rather than
388        // `supports_duplex()` (the in-process `Local` transport supports
389        // duplex but is not a kernel socket). Both paths register the bound
390        // address as the gateway's active serve location. The muxed frontend
391        // owns a listener-scoped runtime control registration.
392        let frontend_result: Result<GatewayServeHandle, ChannelError> = if addr.transport().is_net()
393        {
394            gateway.serve_mux_with_listener(addr, listener)
395        } else {
396            gateway.serve_with_listener(addr, listener)
397        };
398        let mut frontend_handle = match frontend_result {
399            Ok(handle) => handle,
400            Err(error) => {
401                backend_handle.stop("host setup failed");
402                if let Err(join_error) = backend_handle.join().await {
403                    tracing::warn!(
404                        error = %join_error,
405                        "failed to join backend server after host setup error"
406                    );
407                }
408                return Err(error.into());
409            }
410        };
411        let frontend_addr = gateway.default_location().addr().clone();
412        assert!(
413            !matches!(frontend_addr, ChannelAddr::Alias { .. }),
414            "gateway frontend serve must canonicalize aliases"
415        );
416
417        // Attach to the remote gateway after the local serves are live
418        // but before any proc or actor ref is minted below. `serve_via`
419        // must be the newest active serve so it supplies the gateway's
420        // `default_location`; otherwise the local frontend serve above
421        // would win, and refs would advertise a bare, cluster-
422        // unreachable address — the out-of-cluster return-path bug.
423        let via_handle = match via {
424            Some(via_addr) => match gateway.serve_via(via_addr).await {
425                Ok(handle) => Some(handle),
426                Err(error) => {
427                    frontend_handle.stop("host setup failed");
428                    if let Err(join_error) = frontend_handle.join().await {
429                        tracing::warn!(
430                            error = %join_error,
431                            "failed to join frontend server after via attach error"
432                        );
433                    }
434                    backend_handle.stop("host setup failed");
435                    if let Err(join_error) = backend_handle.join().await {
436                        tracing::warn!(
437                            error = %join_error,
438                            "failed to join backend server after via attach error"
439                        );
440                    }
441                    return Err(HostError::ViaAttachFailure(error));
442                }
443            },
444            None => None,
445        };
446
447        // Set up the system proc and the local client proc after the
448        // gateway servers are live. The HostAgent is published only
449        // after it binds its handler, so the brief unroutable window is
450        // before normal clients can discover this host.
451        let expected_service_proc_addr =
452            ProcAddr::new(service_proc_id.clone(), gateway.default_location());
453        let service_proc = Proc::builder()
454            .proc_id(service_proc_id)
455            .shared_gateway(gateway.clone())
456            .build()
457            .map_err(HostError::ProcConstructionFailure)?;
458        frontend_handle
459            .register_proc_owner(&service_proc)
460            .map_err(HostError::ProcConstructionFailure)?;
461        if let Some(handle) = &via_handle {
462            handle
463                .register_proc_owner(&service_proc)
464                .map_err(HostError::ProcConstructionFailure)?;
465        }
466        assert_eq!(
467            service_proc.proc_addr(),
468            expected_service_proc_addr,
469            "service proc must use the gateway's active advertised location"
470        );
471        let local_proc = Proc::builder()
472            .proc_id(ProcId::instance(Label::strip(LOCAL_PROC_NAME)))
473            .shared_gateway(gateway.clone())
474            .build()
475            .map_err(HostError::ProcConstructionFailure)?;
476        let service_proc_id = service_proc.proc_addr().clone();
477        let local_proc_id = local_proc.proc_addr().clone();
478
479        tracing::info!(
480            frontend_addr = frontend_addr.to_string(),
481            backend_addr = backend_addr.to_string(),
482            service_proc_id = service_proc_id.to_string(),
483            local_proc_id = local_proc_id.to_string(),
484            "serving host"
485        );
486
487        Ok(Host {
488            procs: Arc::new(StdMutex::new(HashMap::new())),
489            frontend_addr,
490            backend_addr,
491            gateway,
492            frontend_handle: Some(frontend_handle),
493            backend_handle: Some(backend_handle),
494            via_handle,
495            manager: Arc::new(manager),
496            service_proc,
497            local_proc,
498        })
499    }

525    /// Spawn a child proc with the given `name`.
526    ///
527    /// On success, the proc is ready and reachable through the returned
528    /// [`ProcAddr`]. The proc id is derived from `name`; its location is
529    /// advertised through this host's frontend gateway using a `Via(child_uid,
530    /// host_location)` source route.
531    /// The caller must ensure that no other spawn with the same `name` is in
532    /// flight. This method only rejects names that are already registered with the host.
533    pub(crate) fn spawn(
534        &self,
535        name: String,
536        config: M::Config,
537    ) -> impl Future<Output = Result<(ProcAddr, ActorRef<ManagerAgent<M>>), HostError>> + Send + 'static
538    where
539        M: Send + Sync + 'static,
540        M::Config: Send + 'static,
541    {
542        let procs = Arc::clone(&self.procs);
543        let backend_addr = self.backend_addr.clone();
544        let gateway = self.gateway.clone();
545        let manager = Arc::clone(&self.manager);
546
547        async move {
548            if procs
549                .lock()
550                .map_err(|_| HostError::ProcRegistryPoisoned)?
551                .contains_key(&name)
552            {
553                return Err(HostError::ProcExists(name));
554            }
555
556            // Preserve the route to this host, then append the child hop that
557            // is peeled by this host's gateway.
558            let resource_id = ResourceId::from_name(&name);
559            let proc_uid = resource_id.uid().clone();
560            let proc_id =
561                resource_id.proc_addr(gateway.default_location().append_via(proc_uid.clone()));
562            let handle = manager.spawn(proc_id.clone(), backend_addr, config).await?;
563            let mut proc_handle_guard = ProcHandleKillGuard {
564                handle: Some(handle),
565            };
566
567            let created = async {
568                let timeout: Duration =
569                    hyperactor_config::global::get(hyperactor::config::HOST_SPAWN_READY_TIMEOUT);
570
571                let ready = if timeout == Duration::from_secs(0) {
572                    ReadyProc::ensure(proc_handle_guard.handle()).await
573                } else {
574                    match tokio::time::timeout(
575                        timeout,
576                        ReadyProc::ensure(proc_handle_guard.handle()),
577                    )
578                    .await
579                    {
580                        Ok(result) => result,
581                        Err(_elapsed) => Err(ReadyProcError::Timeout),
582                    }
583                }
584                .map_err(|e| {
585                    HostError::ProcessConfigurationFailure(
586                        proc_id.clone(),
587                        anyhow::anyhow!("{e:?}"),
588                    )
589                })?;
590
591                let child_sender = MailboxClient::dial(ready.addr().clone()).map_err(|e| {
592                    HostError::ProcessConfigurationFailure(
593                        proc_id.clone(),
594                        anyhow::anyhow!("failed to dial spawned proc at {}: {}", ready.addr(), e),
595                    )
596                })?;
597                let agent_ref = ready.agent_ref().clone();
598                drop(ready);
599
600                let guard = gateway.attach_peer(proc_uid.clone(), child_sender.into_boxed());
601                match procs
602                    .lock()
603                    .map_err(|_| HostError::ProcRegistryPoisoned)?
604                    .entry(name.clone())
605                {
606                    Entry::Vacant(entry) => {
607                        entry.insert(guard);
608                    }
609                    Entry::Occupied(_) => return Err(HostError::ProcExists(name.clone())),
610                }
611
612                // Once the proc exits, answer for its actors at the gateway
613                // (see `PeerAttachGuard::mark_dead`), until the route is
614                // released.
615                if let Some(exited) = proc_handle_guard.handle().terminal_actor_status() {
616                    let procs = Arc::downgrade(&procs);
617                    let name = name.clone();
618                    let proc_id = proc_id.clone();
619                    tokio::spawn(async move {
620                        let status = exited.await;
621                        if let Some(procs) = procs.upgrade()
622                            && let Some(guard) =
623                                procs.lock().expect("procs mutex poisoned").get(&name)
624                        {
625                            tracing::info!(%proc_id, %status, "proc exited; answering for its actors");
626                            guard.mark_dead(status);
627                        }
628                    });
629                }
630
631                Ok((proc_id.clone(), agent_ref))
632            }
633            .await;
634
635            match created {
636                Ok(created) => {
637                    proc_handle_guard.disarm();
638                    Ok(created)
639                }
640                Err(error) => {
641                    proc_handle_guard.maybe_kill().await;
642                    Err(error)
643                }
644            }
645        }
646    }

907    ///   1) to prevent resource spikes (I/O, CPU, file descriptors,
908    ///   etc.).
909    async fn terminate_all(

1018/// Minimal uniform surface for a spawned-**proc** handle returned by
1019/// a `ProcManager`. Each manager can return its own concrete handle,
1020/// as long as it exposes these. A **proc** is the Hyperactor runtime
1021/// + its actors (lifecycle controlled via `Proc` APIs such as
1022/// `destroy_and_wait`). A proc **may** be hosted *inside* an OS
1023/// **process**, but it is conceptually distinct:
1024///
1025/// - `LocalProcManager`: runs the proc **in this OS process**; there
1026///   is no child process to signal. Lifecycle is entirely proc-level.
1027/// - `ProcessProcManager` (test-only here): launches an **external OS
1028///   process** which hosts the proc, but this toy manager does
1029///   **not** wire a control plane for shutdown, nor an exit monitor.
1030///
1031/// This trait is therefore written in terms of the **proc**
1032/// lifecycle:
1033///
1034/// - `ready()` resolves when the proc is Ready (mailbox bound; agent
1035///   available).
1036/// - `wait()` resolves with the proc's terminal status
1037///   (Stopped/Killed/Failed).
1038/// - `terminate()` requests a graceful shutdown of the *proc* and
1039///   waits up to the deadline; managers that also own a child OS
1040///   process may escalate to `SIGKILL` if the proc does not exit in
1041///   time.
1042/// - `kill()` requests an immediate, forced termination. For
1043///    in-process procs, this may be implemented as an immediate
1044///    drain/abort of actor tasks. For external procs, this is
1045///    typically a `SIGKILL`.
1046///
1047/// The shape of the terminal value is `Self::TerminalStatus`.
1048/// Managers that track rich info (exit code, signal, address, agent)
1049/// can expose it; trivial managers may use `()`.
1050///
1051/// Managers that do not support signaling must return `Unsupported`.
1052#[async_trait]
1053pub trait ProcHandle: Clone + Send + Sync + 'static {
1054    /// The agent actor type installed in the proc by the manager.
1055    /// Must implement both:
1056    /// - [`Actor`], because the agent actually runs inside the proc,
1057    ///   and
1058    /// - [`Referable`], so callers can hold `ActorRef<Self::Agent>`.
1059    type Agent: Actor + Referable;
1060
1061    /// The type of terminal status produced when the proc exits.
1062    ///
1063    /// For example, an external proc manager may use a rich status
1064    /// enum (e.g. `ProcStatus`), while an in-process manager may use
1065    /// a trivial unit type. This is the value returned by
1066    /// [`ProcHandle::wait`] and carried by [`ReadyError::Terminal`].
1067    type TerminalStatus: std::fmt::Debug + Clone + Send + Sync + 'static;
1068
1069    /// The proc's logical address on this host.
1070    fn proc_addr(&self) -> &ProcAddr;
1071
1072    /// The proc's address (the one callers bind into the host
1073    /// router). May return `None` before `ready()` completes.
1074    /// Guaranteed to return `Some` after `ready()` succeeds.
1075    ///
1076    /// **Prefer [`ready_proc()`]** for type-safe access that
1077    /// guarantees availability at compile time.
1078    fn addr(&self) -> Option<ChannelAddr>;
1079
1080    /// The agent actor reference hosted in the proc. May return
1081    /// `None` before `ready()` completes. Guaranteed to return `Some`
1082    /// after `ready()` succeeds.
1083    ///
1084    /// **Prefer [`ready_proc()`]** for type-safe access that
1085    /// guarantees availability at compile time.
1086    fn agent_ref(&self) -> Option<ActorRef<Self::Agent>>;
1087
1088    /// Resolves when the proc becomes Ready. Multi-waiter,
1089    /// non-consuming.
1090    async fn ready(&self) -> Result<(), ReadyError<Self::TerminalStatus>>;
1091
1092    /// Resolves with the terminal status (Stopped/Killed/Failed/etc).
1093    /// Multi-waiter, non-consuming.
1094    async fn wait(&self) -> Result<Self::TerminalStatus, WaitError>;
1095
1096    /// Politely stop the proc before the deadline; managers that own
1097    /// a child OS process may escalate to a forced kill at the
1098    /// deadline. Idempotent and race-safe: concurrent callers
1099    /// coalesce; the first terminal outcome wins and all callers
1100    /// observe it via `wait()`.
1101    ///
1102    /// Returns the single terminal status the proc reached (the same
1103    /// value `wait()` will return). Never fabricates terminal states:
1104    /// this is only returned after the exit monitor observes
1105    /// termination.
1106    ///
1107    /// # Parameters
1108    /// - `cx`: The actor context for sending messages.
1109    /// - `timeout`: Grace period before escalation.
1110    /// - `reason`: Human-readable reason for termination.
1111    async fn terminate(
1112        &self,
1113        cx: &impl context::Actor,
1114        timeout: Duration,
1115        reason: &str,
1116    ) -> Result<Self::TerminalStatus, TerminateError<Self::TerminalStatus>>;
1117
1118    /// Force the proc down immediately. For in-process managers this
1119    /// may abort actor tasks; for external managers this typically
1120    /// sends `SIGKILL`. Also idempotent/race-safe; the terminal
1121    /// outcome is the one observed by `wait()`.
1122    async fn kill(&self) -> Result<Self::TerminalStatus, TerminateError<Self::TerminalStatus>>;
1123
1124    /// A future that resolves once the proc has exited, with the terminal
1125    /// status that each of its actors reports from then on: a clean stop if
1126    /// the host had begun stopping or killing the proc when its exit was
1127    /// observed, the exit as a failure otherwise, even if the proc's owner
1128    /// or mesh had asked for a stop. [`Host`] answers status queries for the proc's
1129    /// actors with it.
1130    ///
1131    /// Returns `None` if the handle does not observe its proc's exit; the
1132    /// host then keeps routing to the proc.
1133    fn terminal_actor_status(&self) -> Option<BoxFuture<'static, ActorStatus>> {
1134        None

1195/// A ProcManager that spawns **in-process** procs (test-only).
1196///
1197/// The proc runs inside this same OS process; there is **no** child
1198/// process to signal. Lifecycle is purely proc-level:
1199/// - `terminate(timeout)`: delegates to
1200///   `Proc::destroy_and_wait(timeout)`, which drains and, at the
1201///   deadline, aborts remaining actors.
1202/// - `kill()`: uses a zero deadline to emulate a forced stop via
1203///   `destroy_and_wait(Duration::ZERO)`.
1204/// - `wait()`: trivial (no external lifecycle to observe).
1205///
1206///   No OS signals are sent or required.
1207pub struct LocalProcManager<S> {
1208    procs: Arc<Mutex<HashMap<ProcAddr, Proc>>>,
1209    stopping: Arc<Mutex<HashMap<ProcAddr, tokio::sync::watch::Sender<LocalProcStatus>>>>,
1210    spawn: S,
1211}
1212
1213impl<S> LocalProcManager<S> {
1214    /// Create a new in-process proc manager with the given agent
1215    /// params.
1216    pub fn new(spawn: S) -> Self {
1217        Self {
1218            procs: Arc::new(Mutex::new(HashMap::new())),
1219            stopping: Arc::new(Mutex::new(HashMap::new())),
1220            spawn,
1221        }
1222    }
1223
1224    async fn discard_proc(&self, proc_id: &ProcAddr, reason: &str) {
1225        let proc = self.procs.lock().await.remove(proc_id);
1226        if let Some(mut proc) = proc
1227            && let Err(error) = proc.destroy_and_wait(Duration::ZERO, reason).await
1228        {
1229            tracing::warn!(%proc_id, %error, "failed to discard local proc");
1230        }
1231    }
1232
1233    /// Non-blocking stop: remove the proc and spawn a background task
1234    /// that tears it down.
1235    ///
1236    /// Status transitions through `Stopping` -> `Stopped` and is
1237    /// observable via [`local_proc_status`] and [`watch`]. Idempotent:
1238    /// no-ops if the proc is already stopping or stopped.
1239    pub async fn request_stop(&self, proc: &ProcAddr, timeout: Duration, reason: &str) {
1240        {
1241            let guard = self.stopping.lock().await;
1242            if guard.contains_key(proc) {
1243                return;
1244            }
1245        }
1246
1247        let mut proc_handle = {
1248            let mut guard = self.procs.lock().await;
1249            match guard.remove(proc) {
1250                Some(p) => p,
1251                None => return,
1252            }
1253        };
1254
1255        let proc_ref: ProcAddr = proc_handle.proc_addr().clone();
1256        let (tx, _) = tokio::sync::watch::channel(LocalProcStatus::Stopping);
1257        self.stopping.lock().await.insert(proc_ref.clone(), tx);
1258
1259        let stopping = Arc::clone(&self.stopping);
1260        let reason = reason.to_string();
1261        tokio::spawn(async move {
1262            if let Err(e) = proc_handle.destroy_and_wait(timeout, &reason).await {
1263                tracing::warn!(error = %e, "request_stop(local): destroy_and_wait failed");
1264            }
1265            if let Some(tx) = stopping.lock().await.get(&proc_ref) {
1266                let _ = tx.send(LocalProcStatus::Stopped);
1267            }
1268        });
1269    }
1270
1271    /// Query the lifecycle status of a proc that was stopped via
1272    /// [`request_stop`].
1273    ///
1274    /// Returns `None` if the proc was never stopped through this path.
1275    pub async fn local_proc_status(&self, proc: &ProcAddr) -> Option<LocalProcStatus> {
1276        self.stopping.lock().await.get(proc).map(|tx| *tx.borrow())

1295#[async_trait]
1296impl<S> BulkTerminate for LocalProcManager<S>
1297where
1298    S: Send + Sync,
1299{
1300    async fn terminate_all(
1301        &self,
1302        _cx: &impl context::Actor,
1303        timeout: std::time::Duration,
1304        max_in_flight: usize,
1305        reason: &str,
1306    ) -> TerminateSummary {
1307        // Drain procs so we don't hold the lock across awaits and subsequent
1308        // calls to terminate_all don't try to re-terminate.
1309        let procs: Vec<Proc> = {
1310            let mut guard = self.procs.lock().await;
1311            guard.drain().map(|(_, v)| v).collect()
1312        };
1313
1314        let attempted = procs.len();
1315
1316        let results = stream::iter(procs.into_iter().map(|mut p| async move {
1317            // For local manager, graceful proc-level stop.
1318            match p.destroy_and_wait(timeout, reason).await {
1319                Ok(_) => true,
1320                Err(e) => {
1321                    tracing::warn!(error=%e, "terminate_all(local): destroy_and_wait failed");
1322                    false
1323                }
1324            }
1325        }))
1326        .buffer_unordered(max_in_flight.max(1))
1327        .collect::<Vec<bool>>()
1328        .await;
1329
1330        let ok = results.into_iter().filter(|b| *b).count();
1331
1332        TerminateSummary {
1333            attempted,
1334            ok,
1335            failed: attempted.saturating_sub(ok),
1336        }
1337    }
1338}

1365/// A lightweight [`ProcHandle`] for procs managed **in-process** via
1366/// [`LocalProcManager`].
1367///
1368/// This handle wraps the minimal identifying state of a spawned proc:
1369/// - its [`ProcId`] (logical identity on the host),
1370/// - the proc's [`ChannelAddr`] (the address callers bind into the

1373///
1374/// Unlike external handles, `LocalHandle` does **not** manage an OS

1376/// `addr()`, `agent_ref()`) and implements `terminate()`/`kill()` by

1378/// **proc-level** shutdown.

1381/// bound (`Actor + Referable`).

1386    procs: Arc<Mutex<HashMap<ProcAddr, Proc>>>,
1387}

1401#[async_trait]
1402impl<A: Actor + Referable> ProcHandle for LocalHandle<A> {
1403    /// `Agent = A` (inherits `Actor + Referable` from the trait
1404    /// bound).
1405    type Agent = A;
1406    type TerminalStatus = ();
1407
1408    fn proc_addr(&self) -> &ProcAddr {
1409        &self.proc_id
1410    }
1411
1412    fn addr(&self) -> Option<ChannelAddr> {
1413        Some(self.addr.clone())
1414    }
1415
1416    fn agent_ref(&self) -> Option<ActorRef<Self::Agent>> {
1417        Some(self.agent_ref.clone())
1418    }
1419
1420    /// Always resolves immediately: a local proc is created
1421    /// in-process and is usable as soon as the handle exists.
1422    async fn ready(&self) -> Result<(), ReadyError<Self::TerminalStatus>> {
1423        Ok(())
1424    }
1425    /// Always resolves immediately with `()`: a local proc has no
1426    /// external lifecycle to await. There is no OS child process
1427    /// behind this handle.
1428    async fn wait(&self) -> Result<Self::TerminalStatus, WaitError> {
1429        Ok(())
1430    }
1431
1432    async fn terminate(
1433        &self,
1434        _cx: &impl context::Actor,
1435        timeout: Duration,
1436        reason: &str,
1437    ) -> Result<(), TerminateError<Self::TerminalStatus>> {
1438        let mut proc = {
1439            let guard = self.procs.lock().await;
1440            match guard.get(self.proc_addr()) {
1441                Some(p) => p.clone(),
1442                None => {
1443                    // The proc was already removed; treat as already
1444                    // terminal.
1445                    return Err(TerminateError::AlreadyTerminated(()));
1446                }
1447            }
1448        };
1449
1450        // Graceful stop of the *proc* (actors) with a deadline. This
1451        // will drain and then abort remaining actors at expiry.
1452        let _ = proc
1453            .destroy_and_wait(timeout, reason)
1454            .await
1455            .map_err(TerminateError::Io)?;
1456
1457        Ok(())
1458    }
1459
1460    async fn kill(&self) -> Result<(), TerminateError<Self::TerminalStatus>> {
1461        // Forced stop == zero deadline; `destroy_and_wait` will
1462        // immediately abort remaining actors and return.
1463        let mut proc = {
1464            let guard = self.procs.lock().await;
1465            match guard.get(self.proc_addr()) {
1466                Some(p) => p.clone(),
1467                None => return Err(TerminateError::AlreadyTerminated(())),
1468            }
1469        };
1470
1471        let _ = proc
1472            .destroy_and_wait(Duration::from_millis(0), "kill")
1473            .await
1474            .map_err(TerminateError::Io)?;
1475
1476        Ok(())
1477    }

1480/// Local, in-process ProcManager.
1481///
1482/// **Type bounds:**
1483/// - `A: Actor + Referable + Binds<A>`
1484///   - `Actor`: the agent actually runs inside the proc.
1485///   - `Referable`: callers hold `ActorRef<A>` to the agent; this
1486///     bound is required for typed remote refs.
1487///   - `Binds<A>`: lets the runtime wire the agent's handler ports.
1488/// - `F: Future<Output = anyhow::Result<ActorHandle<A>>> + Send`:
1489///   the spawn closure returns a Send future (we `tokio::spawn` it).
1490/// - `S: Fn(Proc) -> F + Sync`: the factory can be called from
1491///   concurrent contexts.
1492///
1493/// Result handle is `LocalHandle<A>` (whose `Agent = A` via `ProcHandle`).
1494#[async_trait]
1495impl<A, S, F> ProcManager for LocalProcManager<S>
1496where
1497    A: Actor + Referable + Binds<A>,
1498    F: Future<Output = anyhow::Result<ActorHandle<A>>> + Send,
1499    S: Fn(Proc) -> F + Sync,
1500{
1501    type Handle = LocalHandle<A>;
1502
1503    fn transport(&self) -> ChannelTransport {
1504        ChannelTransport::Local
1505    }
1506
1507    #[hyperactor::instrument(fields(proc_id=proc_id.to_string(), addr=forwarder_addr.to_string()))]
1508    async fn spawn(
1509        &self,
1510        proc_id: ProcAddr,
1511        forwarder_addr: ChannelAddr,
1512        _config: (),
1513    ) -> Result<Self::Handle, HostError> {
1514        let transport = forwarder_addr.transport();
1515        let proc = Proc::configured(
1516            proc_id.clone(),
1517            MailboxClient::dial(forwarder_addr)?.into_boxed(),
1518        );
1519        let (proc_addr, rx) = channel::serve(ChannelAddr::any(transport))?;
1520        self.procs
1521            .lock()
1522            .await
1523            .insert(proc_id.clone(), proc.clone());
1524        let _handle = proc.clone().serve(rx);
1525        let agent_handle = match AssertUnwindSafe((self.spawn)(proc)).catch_unwind().await {
1526            Ok(Ok(agent_handle)) => agent_handle,
1527            Ok(Err(error)) => {
1528                self.discard_proc(&proc_id, "proc agent failed to spawn")
1529                    .await;
1530                return Err(HostError::AgentSpawnFailure(proc_id, error));
1531            }
1532            Err(panic) => {
1533                self.discard_proc(&proc_id, "proc agent spawn panicked")
1534                    .await;
1535                std::panic::resume_unwind(panic);
1536            }
1537        };
1538
1539        Ok(LocalHandle {
1540            proc_id,
1541            addr: proc_addr,
1542            agent_ref: agent_handle.bind(),
1543            procs: Arc::clone(&self.procs),
1544        })
1545    }

1548/// A ProcManager that manages each proc as a **separate OS process**
1549/// (test-only toy).
1550///
1551/// This implementation launches a child via `Command` and relies on
1552/// `kill_on_drop(true)` so that children are SIGKILLed if the manager
1553/// (or host) drops. There is **no** graceful proc control plane or
1554/// exit monitor wired here. `terminate()` and `kill()` therefore
1555/// remove and force-kill the child directly, while `wait()` remains
1556/// trivial and does not report real lifecycle state.
1557///
1558/// It follows a simple protocol:
1559///
1560/// Each process is launched with the following environment variables:
1561/// - `HYPERACTOR_HOST_BACKEND_ADDR`: the backend address to which all messages are forwarded,
1562/// - `HYPERACTOR_HOST_PROC_ID`: the proc id to assign the launched proc, and
1563/// - `HYPERACTOR_HOST_CALLBACK_ADDR`: the channel address with which to return the proc's address
1564///
1565/// The launched proc should also spawn an actor to manage it - the details of this are
1566/// implementation dependent, and outside the scope of the process manager.
1567///
1568/// The function [`boot_proc`] provides a convenient implementation of the
1569/// protocol.
1570pub struct ProcessProcManager<A> {
1571    program: std::path::PathBuf,
1572    children: Arc<Mutex<HashMap<ProcAddr, Child>>>,
1573    _phantom: PhantomData<A>,
1574}
1575
1576struct ProcessChildCleanup {
1577    children: Arc<Mutex<HashMap<ProcAddr, Child>>>,
1578    proc_id: Option<ProcAddr>,
1579}
1580
1581impl ProcessChildCleanup {
1582    fn disarm(mut self) {
1583        self.proc_id = None;
1584    }
1585}
1586
1587impl Drop for ProcessChildCleanup {
1588    fn drop(&mut self) {
1589        let Some(proc_id) = self.proc_id.take() else {
1590            return;
1591        };
1592        let children = Arc::clone(&self.children);
1593
1594        tokio::spawn(async move {
1595            if let Some(mut child) = children.lock().await.remove(&proc_id) {
1596                if let Err(error) = child.kill().await {
1597                    tracing::warn!(%proc_id, %error, "failed to kill interrupted process spawn");
1598                }
1599
1600                if let Err(error) = child.wait().await {
1601                    tracing::warn!(%proc_id, %error, "failed to reap interrupted process spawn");
1602                }
1603            }
1604        });
1605    }
1606}

1725#[async_trait]
1726impl<A> ProcManager for ProcessProcManager<A>
1727where
1728    // Agent actor runs in the proc (`Actor`) and must be
1729    // referenceable (`Referable`).
1730    A: Actor + Referable + Sync,
1731{
1732    type Handle = ProcessHandle<A>;
1733
1734    fn transport(&self) -> ChannelTransport {
1735        ChannelTransport::Unix
1736    }
1737
1738    #[hyperactor::instrument(fields(proc_id=proc_id.to_string(), addr=forwarder_addr.to_string()))]
1739    async fn spawn(
1740        &self,
1741        proc_id: ProcAddr,
1742        forwarder_addr: ChannelAddr,
1743        _config: (),
1744    ) -> Result<Self::Handle, HostError> {
1745        let (callback_addr, mut callback_rx) =
1746            channel::serve(ChannelAddr::any(ChannelTransport::Unix))?;
1747
1748        let mut cmd = Command::new(&self.program);
1749        cmd.env("HYPERACTOR_HOST_PROC_ID", proc_id.to_string());
1750        cmd.env("HYPERACTOR_HOST_BACKEND_ADDR", forwarder_addr.to_string());
1751        cmd.env("HYPERACTOR_HOST_CALLBACK_ADDR", callback_addr.to_string());
1752
1753        // Lifetime strategy: mark the child with
1754        // `kill_on_drop(true)` so the OS will send SIGKILL if the
1755        // handle is dropped and retain the `Child` in
1756        // `self.children`, tying its lifetime to the manager/host.
1757        //
1758        // This is the simplest viable policy to avoid orphaned
1759        // subprocesses in CI; more sophisticated lifecycle control
1760        // (graceful shutdown, restart) will be layered on later.
1761
1762        // Kill the child when its handle is dropped.
1763        cmd.kill_on_drop(true);
1764
1765        let child = cmd.spawn().map_err(|e| {
1766            HostError::ProcessSpawnFailure(proc_id.clone(), self.program.display().to_string(), e)
1767        })?;
1768
1769        // Retain the handle so it lives for the life of the
1770        // manager/host.
1771        {
1772            let mut children = self.children.lock().await;
1773            children.insert(proc_id.clone(), child);
1774        }
1775        let cleanup = ProcessChildCleanup {
1776            children: Arc::clone(&self.children),
1777            proc_id: Some(proc_id.clone()),
1778        };
1779
1780        // Wait for the child's callback with (addr, agent_ref)
1781        let (proc_addr, agent_ref) = callback_rx.recv().await?;
1782
1783        // TODO(production): For a non-test implementation, plumb a
1784        // shutdown path:
1785        // - expose a proc-level graceful stop RPC on the agent and
1786        //   implement `terminate(timeout)` by invoking it and, on
1787        //   deadline, call `Child::kill()`; implement `kill()` as
1788        //   immediate `Child::kill()`.
1789        // - wire an exit monitor so `wait()` resolves with a real
1790        //   terminal status.
1791        let handle = ProcessHandle {
1792            proc_id,
1793            addr: proc_addr,
1794            agent_ref,
1795            children: Arc::clone(&self.children),
1796        };
1797        cleanup.disarm();
1798        Ok(handle)
1799    }

1835/// Spawn a proc at `proc_id` with an `A`-typed agent actor,
1836/// forwarding messages to the provided `backend_addr`,
1837/// and returning the proc's address and agent actor on
1838/// the provided `callback_addr`.
1839#[hyperactor::instrument(fields(proc_id=proc_id.to_string(), addr=backend_addr.to_string(), callback_addr=callback_addr.to_string()))]
1840pub async fn spawn_proc<A, S, F>(
1841    proc_id: ProcAddr,
1842    backend_addr: ChannelAddr,
1843    callback_addr: ChannelAddr,
1844    spawn: S,
1845) -> Result<Proc, HostError>
1846where
1847    // `Actor`: runs in the proc; `Referable`: allows ActorRef<A>;
1848    // `Binds<A>`: wires ports
1849    A: Actor + Referable + Binds<A>,
1850    S: FnOnce(Proc) -> F,
1851    F: Future<Output = Result<ActorHandle<A>, anyhow::Error>>,
1852{
1853    let backend_transport = backend_addr.transport();
1854    let proc = Proc::configured(
1855        proc_id.clone(),
1856        MailboxClient::dial(backend_addr)?.into_boxed(),
1857    );
1858
1859    let agent_handle = spawn(proc.clone())
1860        .await
1861        .map_err(|e| HostError::AgentSpawnFailure(proc_id.clone(), e))?;
1862
1863    // Finally serve the proc on the same transport as the backend address,
1864    // and call back.
1865    let (proc_addr, proc_rx) = channel::serve(ChannelAddr::any(backend_transport))?;
1866    proc.clone().serve(proc_rx);
1867    let agent_ref: ActorRef<A> = agent_handle.bind::<A>();
1868    channel::dial::<(ChannelAddr, ActorRef<A>)>(callback_addr)?
1869        .send((proc_addr, agent_ref))
1870        .await
1871        .map_err(ChannelError::from)?;
1872
1873    Ok(proc)
1874}
