9//! Private test-support bindings.
10//!
11//! Private fixtures live here:
12//!
13//! 1. `TestStruct`, a minimal PyO3 struct for testing `@rust_struct` mixin
14//!    patching. The `#[pyclass]` module is set to the Python file that defines
15//!    the `@rust_struct`-decorated class so the name-validation check passes.
16//! 2. `_HandleProbe`, a closed probe that mints a real Rust-produced `Handle`
17//!    for the `Future`/`Handle` contract suite.
18//! 3. `_make_delayed_handle`, a `Handle` that publishes after registering its
19//!    first waiter, for the `returns_future` overhead benchmark.
20//! 4. `_handle_observer_count`, a probe for proving that a pending `get()`
21//!    has registered its local wait before a test requests an interrupt.
22//! 5. `_call_in_tokio_context`, a synchronous probe call with the runtime-entered
23//!    state that makes nested blocking unsafe, without a coroutine driver.
24//!
25//! Everything here is private support: nothing is re-exported from `monarch`
26//! and no production Python imports it.

60/// Send one of the fixed delivery-failure witnesses through the actor runtime.
61/// This tests Python callback dispatch, not how a destination fails delivery.
62#[pyfunction]
63fn _post_delivery_failure(instance: &PyInstance, kind: &str, report: bool) -> PyResult<()> {
64    let dest = instance
65        .self_addr()
66        .port_addr(hyperactor::Port::handler::<crate::actor::PythonMessage>());
67    let failure = match kind {
68        "stopped" | "failed" | "unbound" => DeliveryFailure::new(InvalidReference::new(
69            dest.clone(),
70            match kind {
71                "stopped" => InvalidReferenceReason::ActorStopped,
72                "failed" => InvalidReferenceReason::ActorFailed,
73                _ => InvalidReferenceReason::HandlerNotBound,
74            },
75        )),
76        "expired" => DeliveryFailure::new(ExpiredDelivery::new(dest.clone())),
77        "port_gone" => DeliveryFailure::new(UndeliverableReason::PortGone(PortGone::new(
78            dest.clone(),
79            None,
80        ))),
81        "channel_closed" => {
82            DeliveryFailure::new(UndeliverableReason::Transport(TransportFailure::new(
83                dest.clone(),
84                TransportFailureReason::ChannelClosed {
85                    addr: hyperactor::channel::ChannelAddr::any(
86                        hyperactor::channel::ChannelTransport::Local,
87                    ),
88                },
89            )))
90        }
91        _ => return Err(PyValueError::new_err("unknown delivery-failure witness")),
92    };
93    let message = if report {
94        Undeliverable::report(DeliveryFailureReport::new(
95            instance.self_addr().clone(),
96            EndpointLocation::Port(dest),
97            None,
98            failure,
99        ))
100    } else {
101        let mut envelope =
102            MessageEnvelope::serialize(instance.self_addr().clone(), dest, &(), Flattrs::new())
103                .map_err(|error| PyValueError::new_err(error.to_string()))?;
104        envelope.push_delivery_failure(failure);
105        Undeliverable::message(envelope)
106    };
107    instance.port().post(&**instance, message);
108    Ok(())
109}

115/// Count a paired Handle's observers without adding one. A fresh pair has
116/// one. A pending `get()` adds a second before releasing the GIL; a Python
117/// helper thread can then observe that the native call is active. Completion
118/// consumes the publisher, after which this returns zero.
119#[pyfunction]
120fn _handle_observer_count(completer: &PyHandleCompleter) -> usize {
121    completer.observer_count_for_test()

124/// The closed set of terminal outcomes the probe can produce.
125///
126/// Deliberately not a caller-supplied coroutine, awaitable, callable, future or
127/// producer function: the probe exists to put a real `PyHandle` into one of
128/// three reviewed terminal states, not to drive arbitrary work.
129#[derive(Clone, Copy)]
130enum ProbeOutcome {
131    Success,
132    Exception,
133    BaseException,
134}
135
136impl ProbeOutcome {
137    fn parse(name: &str) -> PyResult<Self> {
138        match name {
139            "success" => Ok(Self::Success),
140            "exception" => Ok(Self::Exception),
141            "base_exception" => Ok(Self::BaseException),
142            other => Err(PyValueError::new_err(format!(
143                "unknown probe outcome {other:?}; expected one of \
144                 'success', 'exception', 'base_exception'"
145            ))),
146        }
147    }
148
149    fn into_result(self) -> PyResult<i64> {
150        match self {
151            Self::Success => Ok(PROBE_SUCCESS_VALUE),
152            Self::Exception => Err(PyValueError::new_err("probe failure")),
153            Self::BaseException => Err(PyKeyboardInterrupt::new_err("probe base failure")),
154        }
155    }
156}

158/// A closed control object over one Rust-produced `Handle`.
159///
160/// The handle comes from the real [`PyHandle::spawn`] path (HDL-13), so the
161/// contract suite observes a direct Rust producer. The producer parks on a
162/// release gate, so a test decides exactly when the terminal state becomes
163/// observable without sleeping.
164///
165/// Two limits are load-bearing, not stylistic:
166///
167/// * It must never accept arbitrary Python work -- no coroutine, awaitable,
168///   callable, future or producer function. Widening it to take one turns a
169///   contract fixture into a general async escape hatch, and every caller then
170///   depends on this module staying compiled into production.
171/// * It must never be cited as a real producer's oracle. Its timing is a
172///   release gate and its outcomes are three fixed cases, so it says nothing
173///   about when a real producer starts work or whether that work is abandoned
174///   when nobody observes it. Those belong to the producer's own tests.
175#[pyclass(
176    name = "_HandleProbe",
177    module = "monarch._rust_bindings.monarch_hyperactor.testing"
178)]
179pub struct PyHandleProbe {
180    handle: Py<PyHandle>,
181    release: Mutex<Option<oneshot::Sender<()>>>,
182    started: Arc<AtomicBool>,
183    completed: Arc<AtomicBool>,
184}
185
186#[pymethods]
187impl PyHandleProbe {
188    /// The observe-only `Handle` under test. Cloning the reference does not
189    /// consume it, so repeated lookups and observations are safe.
190    #[getter]
191    fn _handle(&self, py: Python<'_>) -> Py<PyHandle> {
192        self.handle.clone_ref(py)
193    }
194
195    /// Whether the producer body has begun.
196    ///
197    /// This can be true immediately after construction: the producer is eager,
198    /// so it enters its body and sets this *before* waiting on `_release`. Only
199    /// completion is gated, so `_completed` is the witness a test can assert a
200    /// negative on.
201    #[getter]
202    fn _started(&self) -> bool {
203        self.started.load(Ordering::SeqCst)
204    }
205
206    /// Whether the producer body has finished.
207    ///
208    /// Set just before the terminal value is published, so it is a witness for
209    /// assertions, not a synchronization primitive; use `_wait_completed` to
210    /// wait for the value to be observable.
211    #[getter]
212    fn _completed(&self) -> bool {
213        self.completed.load(Ordering::SeqCst)
214    }
215
216    /// Open the release gate so the producer can run to its terminal state.
217    ///
218    /// Idempotent: a second call is a no-op, so a test may release
219    /// unconditionally in a fixture teardown.
220    fn _release(&self) {
221        if let Some(tx) = self.release.lock().expect("probe mutex poisoned").take() {
222            let _ = tx.send(());
223        }
224    }
225
226    /// Block until the terminal state is published, discarding it.
227    ///
228    /// This is how a test reaches a genuinely *ready* `Handle` deterministically
229    /// -- the alternative is sleeping, which the contract suite forbids. The
230    /// terminal value is dropped here; the test observes it through the handle.
231    /// Must be called off a Tokio thread, like any other blocking observation.
232    fn _wait_completed(&self, py: Python<'_>) -> PyResult<()> {
233        let waiter = self.handle.bind(py).borrow().wait_completion();
234        // The probe waits for publication; whether the outcome was success or
235        // error is the test's business, observed through the handle.
236        signal_safe_block_on(py, async move {
237            let _ = waiter.await;
238        })
239    }
240}

242/// Build a probe whose handle will reach `outcome` once released.
243#[pyfunction]
244fn _make_handle_probe(py: Python<'_>, outcome: &str) -> PyResult<PyHandleProbe> {
245    let outcome = ProbeOutcome::parse(outcome)?;
246    let (tx, rx) = oneshot::channel::<()>();
247    let started = Arc::new(AtomicBool::new(false));
248    let completed = Arc::new(AtomicBool::new(false));
249    let started_producer = Arc::clone(&started);
250    let completed_producer = Arc::clone(&completed);
251
252    let handle = PyHandle::spawn(async move {
253        started_producer.store(true, Ordering::SeqCst);
254        // A dropped sender releases the gate too, so the producer cannot wedge
255        // if the probe is garbage collected mid-test.
256        let _ = rx.await;
257        let result = outcome.into_result();
258        completed_producer.store(true, Ordering::SeqCst);
259        result
260    });
261
262    Ok(PyHandleProbe {
263        handle: Py::new(py, handle)?,
264        release: Mutex::new(Some(tx)),
265        started,
266        completed,
267    })
268}

270/// A `Handle` that publishes `value` after registering its first waiter.
271///
272/// The producer publishes through its `Publisher` rather than
273/// [`PyHandle::spawn`], which takes the GIL to convert its value on
274/// completion, so completion takes no GIL. The Handle is one observer of the
275/// core, and `get()` or `await` adds another; `poll()` does not. Once the
276/// producer detects the second, it waits at least `delay` before publishing. A
277/// later observer waits only for the remaining time. The producer yields
278/// between checks, so it remains runnable on Tokio until the first waiter is
279/// registered, and ends if the Handle is dropped unobserved. The time from its
280/// first poll until it detects that waiter is added to the gate totals read by
281/// `_delayed_handle_gate_stats`.
282fn delayed_handle(delay: Duration, value: Py<PyAny>) -> PyHandle {
283    let (publisher, handle) = PyHandle::pending();
284    get_tokio_runtime().spawn(async move {
285        let first_poll = Instant::now();
286        loop {
287            match publisher.observer_count() {
288                0 => return,
289                1 => tokio::task::yield_now().await,
290                _ => break,
291            }
292        }
293        let gate = u64::try_from(first_poll.elapsed().as_nanos()).unwrap_or(u64::MAX);
294        {
295            let mut stats = GATE_STATS
296                .lock()
297                .expect("should not be poisoned: no holder can panic");
298            stats.0 += 1;
299            stats.1 = stats.1.saturating_add(gate);
300        }
301        if !delay.is_zero() {
302            tokio::time::sleep(delay).await;
303        }
304        publisher.publish(Ok(value));
305    });
306    handle
307}

309/// Build a `Handle` that publishes `None` no sooner than `delay` seconds after
310/// detecting its first registered waiter.
311#[pyfunction]
312fn _make_delayed_handle(py: Python<'_>, delay: f64) -> PyResult<PyHandle> {
313    let delay = Duration::try_from_secs_f64(delay)
314        .map_err(|e| PyValueError::new_err(format!("invalid delay {delay}: {e}")))?;
315    Ok(delayed_handle(delay, py.None()))
316}

318/// Delayed Handles whose producer has detected its first waiter, and their
319/// total gate time in nanoseconds, since the last `_delayed_handle_gate_stats`.
320/// One lock guards both, so a read is a snapshot.
321static GATE_STATS: Mutex<(u64, u64)> = Mutex::new((0, 0));
322
323/// Return the number of delayed Handles whose gate opened, and their total
324/// gate time in nanoseconds, since the last call; then reset both.
325///
326/// Gate time is elapsed time, including any time other tasks ran between the
327/// producer's checks, so it is an upper bound on the gate's CPU.
328#[pyfunction]
329fn _delayed_handle_gate_stats() -> (u64, u64) {
330    std::mem::take(
331        &mut *GATE_STATS
332            .lock()
333            .expect("should not be poisoned: no holder can panic"),
334    )
335}

363/// Call a synchronous test probe with Tokio's runtime-entered state set.
364///
365/// `block_on` polls this future on the calling thread: it starts no task and
366/// drives no Python coroutine. The probe must not wait for work that needs the
367/// GIL, since it keeps the GIL throughout the call. Tokio restores the previous
368/// runtime context when the probe returns or raises a Python exception.
369#[pyfunction]
370fn _call_in_tokio_context(f: &Bound<'_, PyAny>) -> PyResult<Py<PyAny>> {
371    get_tokio_runtime().block_on(async { f.call0().map(Bound::unbind) })
372}

374pub fn register_python_bindings(module: &Bound<'_, PyModule>) -> PyResult<()> {
375    module.add_function(wrap_pyfunction!(_post_delivery_failure, module)?)?;
376    module.add_class::<PyTestStruct>()?;
377    module.add_function(wrap_pyfunction!(_make_test_struct, module)?)?;
378    module.add_class::<PyHandleProbe>()?;
379    module.add_function(wrap_pyfunction!(_make_handle_probe, module)?)?;
380    module.add_function(wrap_pyfunction!(_handle_observer_count, module)?)?;
381    module.add("_PROBE_SUCCESS_VALUE", PROBE_SUCCESS_VALUE)?;
382    module.add_function(wrap_pyfunction!(_make_delayed_handle, module)?)?;
383    module.add_function(wrap_pyfunction!(_delayed_handle_gate_stats, module)?)?;
384    module.add_function(wrap_pyfunction!(_call_in_tokio_context, module)?)?;
385    crate::actor::register_testing_python_bindings(module)?;
386    Ok(())
387}

390mod tests {
391    use pyo3::types::PyCFunction;

393    use super::*;
394    use crate::runtime::GilSite;
395    use crate::runtime::ensure_python;
396    use crate::runtime::monarch_with_gil_blocking;

398    #[test]
399    fn tokio_context_probe_enters_the_nested_blocking_guard() {
400        ensure_python();
401        monarch_with_gil_blocking(GilSite::Test, |py| {
402            assert!(!crate::runtime::is_in_tokio_runtime());
403            let probe = PyCFunction::new_closure(py, None, None, |_args, _kwargs| {
404                assert!(crate::runtime::is_in_tokio_runtime());
405                // Merely entering a Tokio Handle would allow this nested wait.
406                // The helper must establish the runtime-entered guard too.
407                assert!(
408                    std::panic::catch_unwind(|| get_tokio_runtime().block_on(async {})).is_err(),
409                    "nested block_on should be rejected inside the probe"
410                );
411                Ok::<(), PyErr>(())
412            })
413            .expect("the synchronous probe should construct");
414            _call_in_tokio_context(probe.as_any()).expect("the probe should return");
415            assert!(!crate::runtime::is_in_tokio_runtime());
416        });
417    }

419    const DELAYS: [Duration; 2] = [Duration::ZERO, Duration::from_millis(10)];
420
421    fn none() -> Py<PyAny> {
422        monarch_with_gil_blocking(GilSite::Test, |py| py.None())
423    }
424
425    // With no observer, the producer never starts its delay.
426    #[test]
427    fn delayed_handle_waits_for_an_observer() {
428        ensure_python();
429        for delay in DELAYS {
430            let handle = delayed_handle(delay, none());
431            std::thread::sleep(Duration::from_millis(30));
432            assert!(
433                monarch_with_gil_blocking(GilSite::Test, |py| handle.poll_ready(py))
434                    .unwrap()
435                    .is_none(),
436                "an unobserved Handle with delay {delay:?} should still be pending"
437            );
438        }
439    }
440
441    // Once observed, it publishes `None`, no sooner than the delay.
442    #[test]
443    fn delayed_handle_publishes_after_its_delay() {
444        ensure_python();
445        for delay in DELAYS {
446            let handle = delayed_handle(delay, none());
447            let observed = Instant::now();
448            let wait = handle.wait_future();
449            // Bounded, so a broken gate fails here rather than at the target's
450            // timeout.
451            let value = get_tokio_runtime()
452                .block_on(async { tokio::time::timeout(Duration::from_secs(5), wait).await })
453                .expect("the Handle should publish within 5 s")
454                .unwrap();
455            assert!(
456                observed.elapsed() >= delay,
457                "the Handle published before its delay of {delay:?}"
458            );
459            monarch_with_gil_blocking(GilSite::Test, |py| {
460                assert!(value.is_none(py), "the Handle should publish None");
461            });
462        }
463    }
