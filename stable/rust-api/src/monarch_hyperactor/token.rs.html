9//! Connect to an existing Python actor mesh using a token string.
10//!
11//! An actor publishes a token for a Python actor mesh. Another process
12//! uses it to obtain a reference and call the mesh's endpoints.
13//! Connecting does not add actors to the mesh.
14//!
15//! The actor whose instance publishes the token owns its rendezvous actor
16//! and receives each connecting actor's address. It need not be an actor
17//! in the published mesh. Once the publishing actor stops, the token no
18//! longer accepts connections.
19//!
20//! `connect()` returns a pending mesh with the region advertised in the token.
21//! Its calls wait for the rendezvous to supply the actual mesh reference.
22//!
23//! ## Token invariants (PYTK-*)
24//!
25//! - **PYTK-1 (parse before sending).** An invalid timeout or mesh token,
26//!   including an invalid region, raises `ValueError` before opening a reply
27//!   port or sending a join. Parsing preserves a valid region's offset and strides.
28//! - **PYTK-2 (reference only through join).** The token supplies the region,
29//!   not the mesh reference. `connect()` returns without waiting; queued calls
30//!   wait for a joined reference whose region matches the token's region.
31//! - **PYTK-3 (connection failures stay local).** A failed join is shared by
32//!   initialization and reply-bearing endpoint calls. A reply timeout closes
33//!   the reply port, not the caller's actor, and does not retract the request.
34//!   Fire-and-forget calls return no failure outcome.
35//! - **PYTK-4 (publisher owns rendezvous).** Publication neither changes mesh
36//!   ownership nor permits the rendezvous to outlive its publishing actor.

74/// A token for connecting to an existing Python actor mesh.
75///
76/// The region lets `connect()` construct the mesh before the rendezvous replies.
77/// Calls wait for that reply to supply the published actors' mesh reference.
78#[derive(Clone, Debug)]
79struct ActorMeshToken {
80    /// The token used to contact the publisher's rendezvous actor.
81    /// A join returns the published mesh reference and reports the connecting
82    /// actor's address to the publisher.
83    rendezvous_token: Token<ActorMeshRef<PythonActor>, ActorAddr>,
84    /// The published mesh's dimensions and selected ranks.
85    /// Used to construct the pending mesh, then checked against the reference
86    /// returned by the join.
87    region: Region,
88}
89
90/// The JSON representation stored inside an actor-mesh token string.
91///
92/// This enum has one case so serde writes a `kind` field and rejects a
93/// missing or different kind when reading. The tag identifies a Python
94/// actor-mesh token, not a bare rendezvous token.
95///
96/// Decoding this payload does not validate its region. `ActorMeshToken`'s
97/// parser checks the region before constructing the usable token.
98#[derive(Serialize, Deserialize)]
99#[serde(tag = "kind")]
100enum ActorMeshTokenPayload {
101    #[serde(rename = "monarch_hyperactor::token::ActorMeshToken")]
102    Mesh {
103        /// The generic rendezvous token, encoded as its canonical string.
104        token: Token<ActorMeshRef<PythonActor>, ActorAddr>,
105        /// The published mesh's region, still unchecked after decoding.
106        region: Region,
107    },
108}
109
110impl FromStr for ActorMeshToken {
111    type Err = anyhow::Error;
112
113    /// Read either the compact token or its display form with a `#JSON` suffix.
114    ///
115    /// Only the base64 text before `#` is decoded; the suffix is for people to read.
116    /// Reject invalid encoding, the wrong token kind, or an invalid rendezvous
117    /// token or region before constructing `ActorMeshToken`.
118    fn from_str(text: &str) -> Result<Self, Self::Err> {
119        let encoded = text.split_once('#').map_or(text, |(encoded, _)| encoded);
120        let json = BASE64_STANDARD.decode(encoded)?;
121        let ActorMeshTokenPayload::Mesh {
122            token: rendezvous_token,
123            region,
124        } = serde_json::from_slice(&json)?;
125        // PYTK-1: serde's Region/Slice derives bypass their constructors.
126        let (labels, slice) = region.into_inner();
127        let (offset, sizes, strides) = slice.into_inner();
128        let region = Region::try_from_parts(labels, offset, sizes, strides)?;
129        Ok(Self {
130            rendezvous_token,
131            region,
132        })
133    }
134}

136impl fmt::Display for ActorMeshToken {
137    /// Write the token as base64 JSON followed by `#` and that same JSON.
138    ///
139    /// The base64 part is the compact token; the suffix lets people inspect its
140    /// contents. The parser accepts either the compact part or the full string.
141    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
142        let payload = ActorMeshTokenPayload::Mesh {
143            token: self.rendezvous_token.clone(),
144            region: self.region.clone(),
145        };
146        let json = serde_json::to_string(&payload).map_err(|_| fmt::Error)?;
147        write!(f, "{}#{}", BASE64_STANDARD.encode(&json), json)
148    }
149}

151/// The result of publishing: a token string and its connection notifications.
152///
153/// Share `token` with other processes. `next_peer()` returns a `Handle`
154/// for the actor address supplied by the next connecting actor.
155/// The actor represented by `create_token()`'s `instance` owns the
156/// rendezvous actor; this Python object does not own it or keep it running.
157#[pyclass(
158    frozen,
159    name = "PublishedToken",
160    module = "monarch._rust_bindings.monarch_hyperactor.token"
161)]
162pub struct PyPublishedToken {
163    /// Token string other processes use to request the published mesh reference.
164    token: String,
165    /// Receiver for a port opened on the actor that publishes the token.
166    /// The rendezvous sends each connecting actor's address to this port.
167    /// This object and the Tokio tasks started by `next_peer()` share the
168    /// receiver through the `Arc`. The mutex serializes receives, so each
169    /// notification is consumed once, not delivered to every task.
170    /// A task retains the receiver until its receive finishes, even if
171    /// Python drops this `PublishedToken` object.
172    connection_notifications: Arc<tokio::sync::Mutex<PortReceiver<token::Joined<ActorAddr>>>>,
173}
174
175#[pymethods]
176impl PyPublishedToken {
177    /// The token string to hand to peers.
178    #[getter]
179    fn token(&self) -> String {
180        self.token.clone()
181    }
182
183    /// Start receiving one connection notification and return its `PyHandle`.
184    /// The result is the connecting actor's address as a string.
185    fn next_peer(&self) -> PyHandle {
186        let connection_notifications = Arc::clone(&self.connection_notifications);
187        PyHandle::spawn(async move {
188            let connected = connection_notifications
189                .lock()
190                .await
191                .recv()
192                .await
193                .map_err(|e| PyRuntimeError::new_err(e.to_string()))?;
194            Ok(connected.peer.to_string())
195        })
196    }
197}

204/// Create the rendezvous actor behind a token as a child of `owner`.
205/// `owner` is the actor instance publishing the token, not necessarily
206/// an actor in the mesh that the Python binding publishes.
207/// Each connecting actor receives `value_from_owner`. The value it
208/// supplies in return is sent to `connection_notifications`.
209///
210/// Generic over the owner's actor type and the two exchanged values.
211/// The Python binding exchanges a mesh reference and the connecting actor's
212/// address; Rust lifecycle tests exchange actor addresses instead, testing
213/// ownership without building a Python mesh.
214///
215/// Reject publication if the owner is stopping or has ended. Otherwise,
216/// hold its status read lock until the new child is linked to it. This
217/// prevents the owner from stopping between the check and child creation,
218/// which could leave the token running after its owner has stopped.
219///
220/// `before_create` lets tests request a stop inside that protected interval.
221/// Production callers pass a no-op.
222fn create_owned_rendezvous<A, C, J>(
223    owner: &Instance<A>,
224    value_from_owner: C,
225    connection_notifications: PortRef<token::Joined<J>>,
226    before_create: impl FnOnce(),
227) -> anyhow::Result<Token<C, J>>
228where
229    A: Actor,
230    C: token::TokenPeer + Clone,
231    J: token::TokenPeer,
232{
233    let status_receiver = owner.status();
234    // Read the current status and hold its read lock until this function returns.
235    let status_guard = status_receiver.borrow();
236    // PYTK-4: hold this status guard until the rendezvous is linked as a child.
237    anyhow::ensure!(
238        accepts_children(&status_guard),
239        "the token owner {} is {}",
240        owner.self_addr(),
241        *status_guard
242    );
243    // Tests can request a competing stop here, while we still hold the lock.
244    before_create();
245
246    // `token::create()` spawns a rendezvous actor as `owner`'s child and
247    // returns a token pointing to it. When another actor connects, the
248    // rendezvous sends that actor's supplied value to
249    // `connection_notifications`, then replies with `value_from_owner`.
250    // Keep the status lock held until the child is registered with `owner`,
251    // so the owner's shutdown cannot miss it.
252    token::create(
253        owner,
254        value_from_owner,
255        connection_notifications,
256        TokenOptions::default(),
257    )
258}

260/// Publish a reference to `mesh` and return a `PublishedToken`.
261///
262/// Called by Python's `publish(mesh)` on the publishing side. `instance`
263/// represents the calling actor, or the process's root client when called
264/// outside an actor. The rendezvous runs as that actor's child, so stopping
265/// it prevents further connections. Publishing does not change mesh ownership.
266///
267/// Other processes pass the returned object's `token` string to Python's
268/// `connect()`. Its `next_peer()` method reports the address supplied by
269/// each connecting actor or root client.
270///
271/// The mesh must already exist; this function does not wait for creation.
272/// A pending mesh raises `ValueError`: wait for `initialized` first.
273/// A failed mesh propagates its creation error. If the publishing actor
274/// is stopping or has ended, raise `RuntimeError`.
275#[pyfunction]
276fn create_token(instance: &PyInstance, mesh: &PythonActorMesh) -> PyResult<PyPublishedToken> {
277    // Obtain the existing mesh's serializable reference without waiting
278    // or starting its creation.
279    let mesh_ref = mesh.get_inner().ready_mesh_ref()?;
280
281    // Make Monarch's Tokio runtime current on this thread. Creating the
282    // rendezvous actor below schedules work through that runtime.
283    let _runtime = get_tokio_runtime().enter();
284
285    // Give the rendezvous a port to notify before accepting connections.
286    // Each notification carries the connecting actor's address.
287    let (notification_port, connection_notifications) =
288        instance.open_port::<token::Joined<ActorAddr>>();
289
290    let region = mesh_ref.region().clone();
291
292    // Create the rendezvous as a child of the publishing actor (`instance`).
293    // It gives each connecting actor `mesh_ref` and sends that actor's address
294    // to `notification_port`. Hold the owner's status lock through child creation
295    // so shutdown cannot miss the rendezvous. The empty closure is a test hook
296    // left unused here. Report creation failures to Python as `RuntimeError`.
297    let rendezvous_token =
298        create_owned_rendezvous(&**instance, mesh_ref, notification_port.bind(), || {})
299            .map_err(|e| PyRuntimeError::new_err(e.to_string()))?;
300    let token = ActorMeshToken {
301        rendezvous_token,
302        region,
303    };
304
305    // Return the shareable token string and retain the notification receiver.
306    // `next_peer()` tasks share this receiver and serialize access to it.
307    Ok(PyPublishedToken {
308        token: token.to_string(),
309        connection_notifications: Arc::new(tokio::sync::Mutex::new(connection_notifications)),
310    })
311}

313/// Request the mesh reference published through `token`.
314///
315/// Called by Python's `connect(token, Class, ...)` on the connecting side.
316/// `instance` represents that caller's actor or root client, not the actor
317/// that published the token. Send its address to the token's rendezvous
318/// actor and return a pending mesh and the region advertised in the token.
319/// Python's `connect()` wraps them using `Class` without waiting for the reply.
320///
321/// Raise `ValueError` for an invalid timeout or token before sending. A rejected
322/// join, timeout, receive error or region mismatch fails mesh initialization and
323/// its reply-bearing calls. The timeout bounds the join, not endpoint calls.
324#[pyfunction]
325fn connect_token(
326    instance: &PyInstance,
327    token: &str,
328    timeout_secs: f64,
329) -> PyResult<(PythonActorMesh, PyRegion)> {
330    // PYTK-1: both checks precede port creation and sending.
331    let timeout = Duration::try_from_secs_f64(timeout_secs)
332        .map_err(|e| PyValueError::new_err(format!("invalid timeout {timeout_secs}: {e}")))?;
333
334    // Decode the rendezvous reference and check the token's exchanged Rust
335    // types. This does not check the Python actor class passed to `connect()`.
336    let token: ActorMeshToken = token
337        .parse()
338        .map_err(|e: anyhow::Error| PyValueError::new_err(format!("invalid token: {e}")))?;
339
340    // Use Monarch's runtime for the port operations and the task spawned below.
341    let _runtime = get_tokio_runtime().enter();
342
343    // Open a reply port on the connecting actor. `result.bind()` gives the
344    // rendezvous an address to reply to; `results` receives its answer here.
345    let (result, results) = instance.open_port::<token::JoinResult<ActorMeshRef<PythonActor>>>();
346
347    // Send the connecting actor's address and the reply-port reference now,
348    // without waiting for the answer. If the token's owner has stopped, its
349    // rendezvous actor is gone: drop the request rather than return it to
350    // `instance`, where it would fail the connecting actor (the root client
351    // outside an actor). The caller then times out, as when the owner's
352    // process exits.
353    //
354    // TODO: `Token::join` returns an undeliverable request to the caller, so
355    // this posts `Join` on the rendezvous's port directly. Give `join` a
356    // policy for undeliverable requests (or make dropping them its only
357    // behavior) and call it here instead.
358    let mut join_port = token
359        .rendezvous_token
360        .rendezvous()
361        .port::<token::Join<ActorMeshRef<PythonActor>, ActorAddr>>();
362    // PYTK-3: a stopped rendezvous must not fault the connecting actor.
363    join_port.return_undeliverable(false);
364    join_port.post(
365        &**instance,
366        token::Join {
367            joiner: instance.self_addr().clone(),
368            result: result.bind(),
369        },
370    );
371
372    let region = PyRegion::from(token.region.clone());
373    // PYTK-2: the existing queue drives this shared future and holds calls
374    // until the join supplies the reference. No observation starts another join.
375    let joined_mesh = await_joined_mesh(results, token.region, timeout);
376    let mesh = PythonActorMesh::new(joined_mesh, true);
377    Ok((mesh, region))
378}

380/// Finish a pending connection from the rendezvous's reply.
381///
382/// The returned reference must have exactly the region advertised in the token.
383/// A timeout, rejection, receive error or region mismatch fails the connection
384/// instead of producing a usable mesh.
385async fn await_joined_mesh(
386    mut results: PortReceiver<token::JoinResult<ActorMeshRef<PythonActor>>>,
387    expected_region: Region,
388    timeout: Duration,
389) -> PyResult<Box<dyn SupervisableActorMesh>> {
390    // This deadline bounds the connection's reply wait, independently
391    // of any later `Handle.get()` timeout. Returning drops `results`
392    // and closes the reply port; it does not retract the sent request.
393    let reply = tokio::time::timeout(timeout, results.recv())
394        .await
395        .map_err(|_| PyTimeoutError::new_err("no answer from the token's owner; is it running?"))?
396        .map_err(|e| PyRuntimeError::new_err(e.to_string()))?;
397    match reply {
398        token::JoinResult::Joined { peer } => {
399            // PYTK-2: never dispatch using an unconfirmed advertised region.
400            if peer.region() != &expected_region {
401                return Err(PyRuntimeError::new_err(
402                    "the published mesh's region does not match the token",
403                ));
404            }
405            Ok(Box::new(PythonActorMeshImpl::new_ref(peer)))
406        }
407        // PYTK-3: the creation future shares failures with mesh observers.
408        token::JoinResult::Rejected { reason } => Err(PyValueError::new_err(format!(
409            "the token's owner rejected the connection: {reason}"
410        ))),
411    }
412}

414pub fn register_python_bindings(module: &Bound<'_, PyModule>) -> PyResult<()> {
415    module.add_class::<PyPublishedToken>()?;
416    for function in [
417        wrap_pyfunction!(create_token, module)?,
418        wrap_pyfunction!(connect_token, module)?,
419    ] {
420        function.setattr(
421            "__module__",
422            "monarch._rust_bindings.monarch_hyperactor.token",
423        )?;
424        module.add_function(function)?;
425    }
426    Ok(())
427}

589        let client = proc.client("client");
590        let (instance_tx, mut instance_rx) = tokio::sync::mpsc::unbounded_channel();
591        let (cleaned_up_tx, cleaned_up) = mpsc::sync_channel(1);
592        let handle = proc.spawn(Owner {
593            instance: instance_tx,
594            cleaned_up: cleaned_up_tx,
595        });
596        let owner = tokio::time::timeout(WAIT, instance_rx.recv())
597            .await
598            .expect("the owner starts")
599            .expect("the owner sends its instance");
600
601        let (peer_port, _peers) = client.open_port::<token::Joined<ActorAddr>>();
602        let creator = client.self_addr().clone();
603        let stopper = handle.clone();
604        let (token, cleaned_up_during_create) = tokio::task::spawn_blocking(move || {
605            let mut cleaned_up_during_create = false;
606            let token: Token<ActorAddr, ActorAddr> =
607                create_owned_rendezvous(&owner, creator, peer_port.bind(), || {
608                    stopper.drain_and_stop("racing publication").unwrap();
609                    // Do not read the owner's status here: that read would
610                    // queue behind the stop's pending status write.
611                    cleaned_up_during_create =
612                        cleaned_up.recv_timeout(Duration::from_millis(500)).is_ok();
613                })
614                .expect("the owner was running when publication checked");
615            (token, cleaned_up_during_create)
616        })
617        .await
618        .unwrap();
619        assert!(
620            !cleaned_up_during_create,
621            "the owner must not finish stopping while publication holds its status"
622        );
623
624        tokio::time::timeout(WAIT, handle)
625            .await
626            .expect("the owner stops after publication");
627        // A stopped actor leaves the proc's registry; one still registered
628        // must be on its way to a terminal status. Without the status guard,
629        // the rendezvous stays registered and running, so this wait fails.
630        if let Some(rendezvous) = proc.get_instance(token.rendezvous().actor_addr()) {
631            let mut status = rendezvous.status().clone();
632            tokio::time::timeout(WAIT, status.wait_for(ActorStatus::is_terminal))
633                .await
634                .expect("the owner's teardown stops the rendezvous")
635                .unwrap();
636        }
637    }
638}
