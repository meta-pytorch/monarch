93    /// Format a Python object for inclusion in error messages.
94    ///
95    /// Prefers `str(obj)` (more user-friendly for exceptions), and
96    /// falls back to `repr(obj)` if `str()` fails. If both fail,
97    /// returns a fixed placeholder.
98    pub(super) fn pyany_to_error_string(obj: &Bound<'_, PyAny>) -> String {
99        obj.str()
100            .map(|s| s.to_string())
101            .or_else(|_| obj.repr().map(|r| r.to_string()))
102            .unwrap_or_else(|_| "<error formatting exception>".into())
103    }

117/// Decoding of exit results returned by the Python proc spawner.
118///
119/// The spawner replies on an explicit exit port with a pickled
120/// payload. That payload must be a `ProcExitResult` dataclass
121/// (from `monarch._src.actor.proc_launcher`) with the standard
122/// exit-reporting attributes.

124/// Decoding is *strict*: all required attributes must be present
125/// with correct types.

137    /// Field names for the `ProcExitResult` dataclass attributes.
138    const K_EXIT_CODE: &str = "exit_code";
139    const K_SIGNAL: &str = "signal";
140    const K_CORE_DUMPED: &str = "core_dumped";
141    const K_FAILED_REASON: &str = "failed_reason";
142    const K_STDERR_TAIL: &str = "stderr_tail";
143
144    /// Required attributes for a valid `ProcExitResult` dataclass.
145    const REQUIRED_ATTRS: [&str; 5] = [
146        K_EXIT_CODE,
147        K_SIGNAL,
148        K_CORE_DUMPED,
149        K_FAILED_REASON,
150        K_STDERR_TAIL,
151    ];

153    /// Intermediate representation of exit information decoded from
154    /// the Python spawner.
155    ///
156    /// This struct separates:
157    /// (1) parsing/validation of a Python dataclass payload, from
158    /// (2) policy decisions when mapping into [`ProcExitResult`] /
159    ///     [`ProcExitKind`].
160    #[derive(Debug)]
161    pub(super) struct DecodedExit {
162        /// Process exit code, if known.
163        pub exit_code: Option<i32>,
164        /// Terminating signal number, if the process was killed by a signal.
165        pub signal: Option<i32>,
166        /// Whether a core dump was produced for a signaled termination.
167        pub core_dumped: bool,
168        /// Failure reason reported by the spawner.
169        pub failed_reason: Option<String>,
170        /// Trailing stderr lines captured by the spawner.
171        pub stderr_tail: Vec<String>,
172    }
173
174    /// Validate that the object has all required attributes.
175    fn validate_shape(obj: &Bound<'_, PyAny>) -> Result<(), ProcLauncherError> {
176        for k in REQUIRED_ATTRS {
177            if !obj
178                .hasattr(k)
179                .map_err(|e| ProcLauncherError::Other(format!("hasattr {k}: {e}")))?
180            {
181                return Err(ProcLauncherError::Other(format!(
182                    "ProcExitResult must be a ProcExitResult dataclass; missing attribute {k}"
183                )));
184            }
185        }
186        Ok(())
187    }
188
189    /// Extract fields from a validated `ProcExitResult` dataclass.
190    fn extract_fields(obj: &Bound<'_, PyAny>) -> Result<DecodedExit, ProcLauncherError> {
191        let exit_code = obj
192            .getattr(K_EXIT_CODE)
193            .map_err(|e| ProcLauncherError::Other(format!("getattr {K_EXIT_CODE}: {e}")))?
194            .extract::<Option<i32>>()
195            .map_err(|e| ProcLauncherError::Other(format!("extract {K_EXIT_CODE}: {e}")))?;
196
197        let signal = obj
198            .getattr(K_SIGNAL)
199            .map_err(|e| ProcLauncherError::Other(format!("getattr {K_SIGNAL}: {e}")))?
200            .extract::<Option<i32>>()
201            .map_err(|e| ProcLauncherError::Other(format!("extract {K_SIGNAL}: {e}")))?;
202
203        let core_dumped = obj
204            .getattr(K_CORE_DUMPED)
205            .map_err(|e| ProcLauncherError::Other(format!("getattr {K_CORE_DUMPED}: {e}")))?
206            .extract::<bool>()
207            .map_err(|e| ProcLauncherError::Other(format!("extract {K_CORE_DUMPED}: {e}")))?;
208
209        let failed_reason = obj
210            .getattr(K_FAILED_REASON)
211            .map_err(|e| ProcLauncherError::Other(format!("getattr {K_FAILED_REASON}: {e}")))?
212            .extract::<Option<String>>()
213            .map_err(|e| ProcLauncherError::Other(format!("extract {K_FAILED_REASON}: {e}")))?;
214
215        let stderr_tail = obj
216            .getattr(K_STDERR_TAIL)
217            .map_err(|e| ProcLauncherError::Other(format!("getattr {K_STDERR_TAIL}: {e}")))?
218            .extract::<Vec<String>>()
219            .map_err(|e| ProcLauncherError::Other(format!("extract {K_STDERR_TAIL}: {e}")))?;
220
221        Ok(DecodedExit {
222            exit_code,
223            signal,
224            core_dumped,
225            failed_reason,
226            stderr_tail,
227        })
228    }

230    /// Decode exit data from a `ProcExitResult` dataclass.
231    ///
232    /// Decoding is *strict*:
233    /// - All required attributes must exist (missing attributes are protocol errors).
234    /// - Attribute values must have the expected types (or be `None`
235    ///   for optional fields).
236    pub(super) fn decode_exit_obj(
237        obj: &Bound<'_, PyAny>,
238    ) -> Result<DecodedExit, ProcLauncherError> {
239        validate_shape(obj)?;
240        extract_fields(obj)
241    }

243    /// Convert a decoded Python exit payload into a
244    /// [`ProcExitResult`].
245    ///
246    /// Mapping rules:
247    /// - If `failed_reason` is set, the proc is treated as
248    ///   [`ProcExitKind::Failed`].
249    /// - Else if `signal` is set, the proc is treated as
250    ///   [`ProcExitKind::Signaled`] (propagating `core_dumped`).
251    /// - Else the proc is treated as [`ProcExitKind::Exited`]; if
252    ///   `exit_code` is missing, `-1` is used as a sentinel for
253    ///   "unknown".
254    ///
255    /// `stderr_tail` is always populated from the decoded payload
256    /// (possibly empty).
257    fn decoded_to_exit_result(d: DecodedExit, observed_at: Instant) -> ProcExitResult {
258        let kind = if let Some(reason) = d.failed_reason {
259            ProcExitKind::Failed { reason }
260        } else if let Some(sig) = d.signal {
261            ProcExitKind::Signaled {
262                signal: sig,
263                core_dumped: d.core_dumped,
264            }
265        } else {
266            // -1 is a sentinel for "exit_code missing/unknown"
267            ProcExitKind::Exited {
268                code: d.exit_code.unwrap_or(-1),
269            }
270        };
271
272        ProcExitResult {
273            observed_at,
274            kind,
275            stderr_tail: Some(d.stderr_tail),
276        }
277    }

279    /// Map a Python exception value into a failed [`ProcExitResult`].
280    ///
281    /// This is used when the spawner reports an exception (rather than a
282    /// normal exit payload). The exception is formatted using
283    /// [`super::py::pyany_to_error_string`] and embedded in
284    /// [`ProcExitKind::Failed`]. No stderr tail is attached because the
285    /// failure originated in the spawner logic rather than the launched
286    /// process.
287    fn exception_to_exit_result(
288        err_obj: &Bound<'_, PyAny>,
289        observed_at: Instant,
290    ) -> ProcExitResult {
291        let reason = format!(
292            "spawner raised: {}",
293            super::py::pyany_to_error_string(err_obj)
294        );
295        ProcExitResult {
296            observed_at,
297            kind: ProcExitKind::Failed { reason },
298            stderr_tail: None,
299        }
300    }

302    /// Convert a spawner response [`PythonMessage`] into a
303    /// [`ProcExitResult`].
304    ///
305    /// The spawner replies on the exit port with a pickled payload:
306    /// - [`PythonMessageKind::Result`]: a pickled `ProcExitResult`-shaped
307    ///   object (dataclass), which is decoded via [`decode_exit_obj`]
308    ///   and mapped with [`decoded_to_exit_result`].
309    /// - [`PythonMessageKind::Exception`]: a pickled exception object,
310    ///   which is mapped to [`ProcExitKind::Failed`] via
311    ///   [`exception_to_exit_result`].
312    ///
313    /// Messages carrying a pending pickle state are rejected (exit
314    /// results must be fully materialized), and any unexpected message
315    /// kind is treated as a protocol error.
316    pub(super) fn convert_py_exit_result(
317        msg: crate::actor::PythonMessage,
318        observed_at: Instant,
319    ) -> Result<ProcExitResult, ProcLauncherError> {
320        use crate::actor::PythonMessageKind;
321
322        monarch_with_gil_blocking(GilSite::Convert, |py| {
323            let cloudpickle = super::py::import_cloudpickle(py)?;
324
325            match msg.kind {
326                PythonMessageKind::Result { .. } => {
327                    let obj = cloudpickle
328                        .call_method1("loads", (msg.message.to_bytes().as_ref(),))
329                        .map_err(|e| ProcLauncherError::Other(format!("cloudpickle.loads: {e}")))?;
330                    let decoded = decode_exit_obj(&obj)?;
331                    Ok(decoded_to_exit_result(decoded, observed_at))
332                }
333                PythonMessageKind::Exception { .. } => {
334                    let err_obj = cloudpickle
335                        .call_method1("loads", (msg.message.to_bytes().as_ref(),))
336                        .map_err(|e| {
337                            ProcLauncherError::Other(format!("cloudpickle.loads exception: {e}"))
338                        })?;
339                    Ok(exception_to_exit_result(&err_obj, observed_at))
340                }
341                _ => Err(ProcLauncherError::Other(
342                    "unexpected message kind in exit result".into(),
343                )),
344            }
345        })
346    }

568/// A [`ProcLauncher`] implemented by delegating proc lifecycle
569/// operations to a Python actor.
570///
571/// The `spawner` actor must implement the `ProcLauncher` ABC from
572/// `monarch._src.actor.proc_launcher`, and is responsible for
573/// actually spawning and controlling OS processes (Docker, VMs,
574/// etc.). Rust retains the *lifecycle wiring* expected by
575/// [`BootstrapProcManager`]: it initiates launch/terminate/kill
576/// requests and exposes an [`oneshot::Receiver`] (`exit_rx`) that
577/// resolves when the spawner reports exit.
578///
579/// ## Semantics
580///
581/// - **PID is optional**: the Python spawner may not expose a real
582///   PID, so [`LaunchResult::pid`] is `None`.
583/// - **Exit reporting is required**: the spawner must send exactly
584///   one exit result on the provided exit port. If the port is closed
585///   or the payload cannot be decoded, the receiver resolves to a
586///   [`ProcExitKind::Failed`] result.
587/// - **Termination is best-effort**: `terminate` and `kill` are
588///   forwarded to the spawner; success only means the request was
589///   delivered.
590///
591/// ## Context requirement
592///
593/// [`ProcLauncher`] methods don't take a context parameter, but
594/// sending actor messages does. This launcher stores an
595/// [`Instance<()>`] ("client-only" actor) to use as the send context.
596/// The instance is created via [`Instance::child()`] and must remain
597/// valid for the lifetime of the launcher.
598#[derive(Debug)]
599pub struct ActorProcLauncher {
600    /// Handle to the Python spawner actor that implements the
601    /// ProcLauncher ABC.
602    spawner: ActorHandle<PythonActor>,
603
604    /// Mailbox used to allocate the one-shot exit port per launched
605    /// proc.
606    mailbox: Mailbox,
607
608    /// Client-only actor instance used as the send context for all
609    /// messages to `spawner`.
610    ///
611    /// Created via `Instance::child()`. The `()` type indicates this
612    /// is not a real actor—just a sending context. Must outlive the
613    /// launcher.
614    instance: Instance<()>,
615
616    /// Debug-only tracking of procs launched via this instance.
617    ///
618    /// Not used for correctness; used for diagnostics and sanity
619    /// checks.
620    active_procs: Arc<Mutex<HashSet<hyperactor::ProcAddr>>>,
621
622    /// How the host and the Python launcher's procs reach each other, as
623    /// declared by its class.
624    reachability: ProcReachability,
625}
626
627impl ActorProcLauncher {
628    /// Create a new actor-based proc launcher.
629    ///
630    /// # Arguments
631    ///
632    /// * `spawner` - Handle to the Python actor implementing the
633    ///   `ProcLauncher` ABC.
634    /// * `mailbox` - Mailbox used to create one-shot exit ports.
635    /// * `instance` - Send context for `ActorHandle::send` (typically
636    ///   from `Instance::child()`). Any valid instance granting send
637    ///   capability is sufficient; it need not be
638    ///   `Instance<PythonActor>`. Must remain valid for the
639    ///   launcher's lifetime.
640    /// * `reachability` - How the host and the launcher's procs reach each
641    ///   other.
642    pub fn new(
643        spawner: ActorHandle<PythonActor>,
644        mailbox: Mailbox,
645        instance: Instance<()>,
646        reachability: ProcReachability,
647    ) -> Self {
648        Self {
649            spawner,
650            mailbox,
651            instance,
652            active_procs: Arc::new(Mutex::new(HashSet::new())),
653            reachability,
654        }
655    }

658#[async_trait]
659impl ProcLauncher for ActorProcLauncher {
660    fn reachability(&self) -> ProcReachability {
661        self.reachability
662    }

664    /// Spawn a proc by delegating to the Python spawner actor.
665    ///
666    /// This method:
667    /// 1) opens a one-shot mailbox port used for the spawner's exit
668    ///    notification,
669    /// 2) serializes `(proc_id, LaunchOptions)` with `cloudpickle`,
670    /// 3) sends a `CallMethod { launch, ExplicitPort(..) }` message
671    ///    to the spawner,
672    /// 4) returns immediately with a [`LaunchResult`] whose `exit_rx`
673    ///    completes once the spawner reports process termination (or the
674    ///    port closes).
675    ///
676    /// ## Notes
677    /// - `pid` is always `None`: the Rust side does not assume an OS
678    ///   PID exists.
679    /// - Exit is observed asynchronously via `exit_rx`;
680    ///   termination/kill are best-effort requests to the spawner actor
681    ///   rather than direct OS signals.
682    /// - If decoding the exit payload fails, the returned `exit_rx`
683    ///   resolves to `ProcExitKind::Failed` with a decode error reason.
684    async fn launch(
685        &self,
686        proc_id: &hyperactor::ProcAddr,
687        opts: LaunchOptions,
688    ) -> Result<LaunchResult, ProcLauncherError> {
689        let spawner_status = self.spawner.status().borrow().clone();
690        if spawner_status.is_terminal() {
691            return Err(ProcLauncherError::Other(format!(
692                "launcher actor stopped: {spawner_status}"
693            )));
694        }
695
696        let (exit_port, exit_port_rx) = self.mailbox.open_once_port::<PythonMessage>();
697
698        let pickled_args = monarch_with_gil_blocking(
699            GilSite::Bootstrap,
700            |py| -> Result<Vec<u8>, ProcLauncherError> {
701                let cloudpickle = import_cloudpickle(py)?;
702
703                let mod_ = py
704                    .import("monarch._src.actor.proc_launcher")
705                    .map_err(|e| ProcLauncherError::Other(format!("import proc_launcher: {e}")))?;
706                let launch_opts_cls = mod_
707                    .getattr("LaunchOptions")
708                    .map_err(|e| ProcLauncherError::Other(format!("getattr LaunchOptions: {e}")))?;
709
710                let program = opts.command.program.to_str().ok_or_else(|| {
711                    ProcLauncherError::Other("program path is not valid UTF-8".into())
712                })?;
713                let vsock_uds_path = opts
714                    .vsock_uds_path
715                    .as_deref()
716                    .map(|path| {
717                        path.to_str().ok_or_else(|| {
718                            ProcLauncherError::Other("vsock uds path is not valid UTF-8".into())
719                        })
720                    })
721                    .transpose()?;
722
723                let env = pyo3::types::PyDict::new(py);
724                for (k, v) in &opts.command.env {
725                    env.set_item(k, v)
726                        .map_err(|e| ProcLauncherError::Other(format!("set env item: {e}")))?;
727                }
728
729                let py_proc_bind = opts.proc_bind.as_ref().map(|bind| {
730                    let d = pyo3::types::PyDict::new(py);
731                    if let Some(v) = &bind.cpunodebind {
732                        d.set_item("cpunodebind", v).unwrap();
733                    }
734                    if let Some(v) = &bind.membind {
735                        d.set_item("membind", v).unwrap();
736                    }
737                    if let Some(v) = &bind.physcpubind {
738                        d.set_item("physcpubind", v).unwrap();
739                    }
740                    if let Some(v) = &bind.cpus {
741                        d.set_item("cpus", v).unwrap();
742                    }
743                    d
744                });
745
746                let launch_kwargs = pyo3::types::PyDict::new(py);
747                launch_kwargs
748                    .set_item("vsock_uds_path", vsock_uds_path)
749                    .map_err(|e| ProcLauncherError::Other(format!("set vsock_uds_path: {e}")))?;
750                let py_opts = launch_opts_cls
751                    .call(
752                        (
753                            &opts.bootstrap_payload,
754                            &opts.process_name,
755                            program,
756                            opts.command.arg0.as_deref(),
757                            &opts.command.args,
758                            env,
759                            opts.want_stdio,
760                            opts.tail_lines,
761                            opts.log_channel.as_ref().map(|a| a.to_string()),
762                            py_proc_bind,
763                            opts.command.inherit_env,
764                            opts.command.env_remove.iter().collect::<Vec<_>>(),
765                        ),
766                        Some(&launch_kwargs),
767                    )
768                    .map_err(|e| {
769                        ProcLauncherError::Other(format!("construct LaunchOptions: {e}"))
770                    })?;
771
772                let args = (proc_id.to_string(), py_opts);
773                let kwargs = pyo3::types::PyDict::new(py);
774                let pickled = cloudpickle
775                    .call_method1("dumps", ((args, kwargs),))
776                    .map_err(|e| ProcLauncherError::Other(format!("cloudpickle: {e}")))?;
777
778                pickled
779                    .extract::<Vec<u8>>()
780                    .map_err(|e| ProcLauncherError::Other(format!("extract bytes: {e}")))
781            },
782        )?;
783
784        let bound_port = exit_port.bind();
785        let message = PythonMessage {
786            kind: PythonMessageKind::CallMethod {
787                name: MethodSpecifier::ExplicitPort {
788                    name: "launch".into(),
789                },
790                response_port: Some(EitherPortRef::Once(PythonOncePortRef::from(bound_port))),
791                correlation_id: None,
792            },
793            message: pickled_args.into(),
794            refs: Vec::new(),
795        };
796
797        self.spawner.post(&self.instance, message);
798
799        let mut active_procs: tokio::sync::MutexGuard<'_, HashSet<hyperactor::ProcAddr>> =
800            self.active_procs.lock().await;
801        active_procs.insert(proc_id.clone());
802        drop(active_procs);
803
804        let (exit_tx, exit_rx) = oneshot::channel();
805        let active_procs: Arc<Mutex<HashSet<hyperactor::ProcAddr>>> =
806            Arc::clone(&self.active_procs);
807        let proc_id_clone = proc_id.clone();
808        let mut spawner_status = self.spawner.status();
809
810        tokio::spawn(async move {
811            // An exit report that has already arrived wins over the
812            // spawner having stopped since.
813            let result = tokio::select! {
814                biased;
815                reply = exit_port_rx.recv() => {
816                    // Python decoding and active-proc cleanup can delay
817                    // delivery, so preserve when the report was received.
818                    let observed_at = Instant::now();
819                    match reply {
820                        Ok(py_message) => {
821                            convert_py_exit_result(py_message, observed_at).unwrap_or_else(|e| ProcExitResult {
822                                observed_at,
823                                kind: ProcExitKind::Failed {
824                                    reason: format!("failed to decode exit result: {e}"),
825                                },
826                                stderr_tail: None,
827                            })
828                        }
829                        Err(_) => ProcExitResult {
830                            observed_at,
831                            kind: ProcExitKind::Failed {
832                                reason: "exit port closed (spawner crashed or forgot to send)".into(),
833                            },
834                            stderr_tail: None,
835                        },
836                    }
837                },
838                stopped = wait_terminal(&mut spawner_status) => ProcExitResult {
839                    observed_at: Instant::now(),
840                    kind: ProcExitKind::Failed {
841                        reason: format!("launcher actor stopped: {stopped}"),
842                    },
843                    stderr_tail: None,
844                },
845            };
846            active_procs.lock().await.remove(&proc_id_clone);
847            let _ = exit_tx.send(result);
848        });
849
850        Ok(LaunchResult {
851            pid: None,
852            started_at: std::time::SystemTime::now(),
853            stdio: StdioHandling::ManagedByLauncher,
854            exit_rx,
855        })
856    }
857
858    /// Request graceful termination of a proc, with a best-effort
859    /// timeout.
860    ///
861    /// This delegates to the Python spawner actor's
862    /// `terminate(proc_id, timeout_secs)` method. The request is sent
863    /// fire-and-forget: we do not wait for an acknowledgment, and
864    /// there is no guarantee the proc will actually exit within
865    /// `timeout`.
866    ///
867    /// ## Errors
868    ///
869    /// Returns `ProcLauncherError::Terminate` if we fail to:
870    /// - import/serialize the request via `cloudpickle`, or
871    /// - send the message to the spawner actor.
872    async fn terminate(
873        &self,
874        proc_id: &hyperactor::ProcAddr,
875        timeout: Duration,
876    ) -> Result<(), ProcLauncherError> {
877        let pickled =
878            monarch_with_gil_blocking(GilSite::Stop, |py| -> Result<Vec<u8>, ProcLauncherError> {
879                let cloudpickle = import_cloudpickle(py)
880                    .map_err(|e| ProcLauncherError::Terminate(format!("{e}")))?;
881                let args = (proc_id.to_string(), timeout.as_secs_f64());
882                let kwargs = pyo3::types::PyDict::new(py);
883                cloudpickle
884                    .call_method1("dumps", ((args, kwargs),))
885                    .map_err(|e| ProcLauncherError::Terminate(format!("cloudpickle: {e}")))?
886                    .extract()
887                    .map_err(|e| ProcLauncherError::Terminate(format!("extract: {e}")))
888            })?;
889
890        let message = PythonMessage {
891            kind: PythonMessageKind::CallMethod {
892                name: MethodSpecifier::ReturnsResponse {
893                    name: "terminate".into(),
894                },
895                response_port: None,
896                correlation_id: None,
897            },
898            message: pickled.into(),
899            refs: Vec::new(),
900        };
901
902        self.spawner.post(&self.instance, message);
903        Ok(())
904    }
905
906    /// Forcefully kill a proc.
907    ///
908    /// This delegates to the Python spawner actor's `kill(proc_id)`
909    /// method. Like `terminate`, this is best-effort and
910    /// fire-and-forget: success here means the request was serialized
911    /// and delivered to the spawner actor, not that the process is
912    /// already dead.
913    ///
914    /// ## Errors
915    ///
916    /// Returns `ProcLauncherError::Kill` if we fail to:
917    /// - import/serialize the request via `cloudpickle`, or
918    /// - send the message to the spawner actor.
919    async fn kill(&self, proc_id: &hyperactor::ProcAddr) -> Result<(), ProcLauncherError> {
920        let pickled =
921            monarch_with_gil_blocking(GilSite::Stop, |py| -> Result<Vec<u8>, ProcLauncherError> {
922                let cloudpickle =
923                    import_cloudpickle(py).map_err(|e| ProcLauncherError::Kill(format!("{e}")))?;
924                let args = (proc_id.to_string(),);
925                let kwargs = pyo3::types::PyDict::new(py);
926                cloudpickle
927                    .call_method1("dumps", ((args, kwargs),))
928                    .map_err(|e| ProcLauncherError::Kill(format!("cloudpickle: {e}")))?
929                    .extract()
930                    .map_err(|e| ProcLauncherError::Kill(format!("extract: {e}")))
931            })?;
932
933        let message = PythonMessage {
934            kind: PythonMessageKind::CallMethod {
935                name: MethodSpecifier::ReturnsResponse {
936                    name: "kill".into(),
937                },
938                response_port: None,
939                correlation_id: None,
940            },
941            message: pickled.into(),
942            refs: Vec::new(),
943        };
944
945        self.spawner.post(&self.instance, message);
946        Ok(())
947    }
948}

950/// Wait until `status` reports a terminal state and return it. A closed
951/// channel means the actor is gone, so its last reported status is returned.
952async fn wait_terminal(status: &mut watch::Receiver<ActorStatus>) -> ActorStatus {
953    loop {
954        let current = status.borrow_and_update().clone();
955        if current.is_terminal() {
956            return current;
957        }
958        if status.changed().await.is_err() {
959            return status.borrow().clone();
960        }
961    }
962}

965/// each host's system proc.

968/// Install a Python proc launcher on every host of `hosts`.
969///
970/// Hosts are installed concurrently. Succeeds only if every host installed
971/// the launcher; otherwise the error names each failed host. Hosts that did
972/// install keep the launcher.
973pub(crate) async fn install_proc_launcher(
974    cx: &impl context::Actor,
975    hosts: &HostMeshRef,
976    launcher: PythonActorParams,
977    reachability: ProcReachability,
978) -> anyhow::Result<()> {
979    let installer_type = Remote::global()
980        .name_of::<ProcLauncherInstaller>()
981        .context("ProcLauncherInstaller is not registered")?;
982    let params = bincode::serde::encode_to_vec((), bincode::config::legacy())?;
983    let num_hosts = hosts.region().num_ranks();
984
985    let failures: Vec<String> = stream::iter(hosts.values().enumerate())
986        .map(|(rank, agent)| {
987            let launcher = launcher.clone();
988            let params = params.clone();
989            async move {
990                let host = agent.actor_addr().proc_addr().clone();
991                install_on_host(cx, agent, installer_type, params, launcher, reachability)
992                    .await
993                    .map_err(|e| format!("host rank {rank} ({host}): {e:#}"))
994            }
995        })
996        .buffer_unordered(num_hosts.max(1))
997        .filter_map(|result| async move { result.err() })
998        .collect()
999        .await;
1000
1001    if failures.is_empty() {
1002        return Ok(());
1003    }
1004    anyhow::bail!(
1005        "set_proc_launcher failed on {} of {} hosts:\n{}",
1006        failures.len(),
1007        num_hosts,
1008        failures.join("\n")
1009    )
1010}

1018    params: Vec<u8>,
1019    launcher: PythonActorParams,
1020    reachability: ProcReachability,

1037/// Installs a Python proc launcher on the host whose system proc it runs on.
1038///
1039/// It spawns the launcher actor as its child, waits for the launcher's
1040/// constructor to succeed, wraps the actor in an [`ActorProcLauncher`] and
1041/// hands that to the host agent. After a successful install it stays alive
1042/// as the launcher actor's parent and absorbs that actor's failures, so a
1043/// faulty launcher cannot fail the system proc. After a failed install it
1044/// stops, which stops the launcher actor with it.
1045#[derive(Debug, Default)]
1046#[hyperactor::export(handlers = [InstallProcLauncher])]
1047#[hyperactor::spawnable]
1048pub struct ProcLauncherInstaller;
1049
1050#[async_trait]
1051impl Actor for ProcLauncherInstaller {
1052    async fn handle_supervision_event(
1053        &mut self,
1054        _this: &Instance<Self>,
1055        event: &ActorSupervisionEvent,
1056    ) -> Result<bool, anyhow::Error> {
1057        if event.is_error() {
1058            tracing::error!(%event, "proc launcher actor failed; procs it launched are no longer supervised");
1059        }
1060        Ok(true)
1061    }
1062
1063    async fn handle_undeliverable_message(
1064        &mut self,
1065        _cx: &Instance<Self>,
1066        reason: UndeliverableReason,
1067        _undeliverable: Undeliverable<MessageEnvelope>,
1068    ) -> Result<(), anyhow::Error> {
1069        tracing::warn!(
1070            ?reason,
1071            "message for the proc launcher actor was undeliverable"
1072        );
1073        Ok(())
1074    }
1075}

1077/// Ask a [`ProcLauncherInstaller`] to install a Python launcher on its host.
1078#[derive(Debug, Serialize, Deserialize, Named, Handler, HandleClient, RefClient)]
1079pub struct InstallProcLauncher {
1080    /// Spawn parameters of the Python launcher actor.
1081    pub launcher: PythonActorParams,
1082    /// How the host and the launcher's procs reach each other.
1083    pub reachability: ProcReachability,
1084    /// The host agent on the installer's system proc.
1085    pub host_agent: ActorRef<HostAgent>,
1086    /// Receives `Ok(())` once installed, or why installation failed.
1087    #[reply]
1088    pub result: OncePortRef<Result<(), String>>,
1089}
1090wirevalue::register_type!(InstallProcLauncher);
1091
1092#[async_trait]
1093impl Handler<InstallProcLauncher> for ProcLauncherInstaller {
1094    async fn handle(
1095        &mut self,
1096        cx: &Context<Self>,
1097        InstallProcLauncher {
1098            launcher,
1099            reachability,
1100            host_agent,
1101            result,
1102        }: InstallProcLauncher,
1103    ) -> anyhow::Result<()> {
1104        let outcome = install(cx, launcher, reachability, &host_agent).await;
1105        let failed = outcome.is_err();
1106        result.post(cx, outcome);
1107        if failed && let Err(e) = cx.stop("proc launcher installation failed") {
1108            tracing::warn!(%e, "failed to stop proc launcher installer");
1109        }
1110        Ok(())
1111    }
1112}

1114/// Spawn the launcher actor, wait for its constructor, and install it on
1115/// `host_agent`'s host.
1116async fn install(
1117    cx: &Context<'_, ProcLauncherInstaller>,
1118    launcher: PythonActorParams,
1119    reachability: ProcReachability,
1120    host_agent: &ActorRef<HostAgent>,
1121) -> Result<(), String> {
1122    let (actor_type, init_message, kind) = launcher.into_parts();
1123    let mut init_message = init_message.ok_or("launcher spawn parameters carry no Init message")?;
1124    let (init_reply, init_reply_rx) = cx.mailbox().open_once_port::<PythonMessage>();
1125    init_message.kind = PythonMessageKind::CallMethod {
1126        name: MethodSpecifier::Init {},
1127        response_port: Some(EitherPortRef::Once(PythonOncePortRef::from(
1128            init_reply.bind(),
1129        ))),
1130        correlation_id: None,
1131    };
1132    let point = extent!()
1133        .point_of_rank(0)
1134        .expect("the empty extent should have rank 0");
1135    let actor = PythonActor::new(actor_type, Some(init_message), Some(point), kind)
1136        .map_err(|e| format!("failed to construct the launcher actor: {e}"))?;
1137    let launcher_actor = cx.spawn(actor);
1138
1139    let mut status = launcher_actor.status();
1140    let reply = tokio::select! {
1141        biased;
1142        reply = init_reply_rx.recv() => reply
1143            .map_err(|e| format!("launcher actor never acknowledged its construction: {e}"))?,
1144        stopped = wait_terminal(&mut status) => {
1145            return Err(format!("launcher actor stopped during construction: {stopped}"));
1146        }
1147    };
1148    check_init_reply(&reply)?;
1149
1150    let (send_context, _) = cx.child();
1151    let mailbox = send_context.mailbox().clone();
1152    let proc_launcher: Arc<dyn ProcLauncher> = Arc::new(ActorProcLauncher::new(
1153        launcher_actor,
1154        mailbox,
1155        send_context,
1156        reachability,
1157    ));
1158    let agent = host_agent
1159        .downcast_handle(cx)
1160        .ok_or("installer is not co-located with the host agent")?;
1161    agent
1162        .set_proc_launcher(cx, proc_launcher)
1163        .await
1164        .map_err(|e| format!("host agent did not answer: {e}"))?
1165}
1166
1167/// Map the launcher's reply to its `Init` message: success, or the
1168/// constructor's exception.
1169fn check_init_reply(reply: &PythonMessage) -> Result<(), String> {
1170    match &reply.kind {
1171        PythonMessageKind::Result { .. } => Ok(()),
1172        PythonMessageKind::Exception { .. } => {
1173            Err(monarch_with_gil_blocking(GilSite::Convert, |py| {
1174                let decoded =
1175                    import_cloudpickle(py)
1176                        .map_err(|e| e.to_string())
1177                        .and_then(|cloudpickle| {
1178                            cloudpickle
1179                                .call_method1("loads", (reply.message.to_bytes().as_ref(),))
1180                                .map_err(|e| e.to_string())
1181                        });
1182                match decoded {
1183                    Ok(error) => format!(
1184                        "launcher constructor failed: {}",
1185                        py::pyany_to_error_string(&error)
1186                    ),
1187                    Err(e) => {
1188                        format!("launcher constructor failed; its error could not be decoded: {e}")
1189                    }
1190                }
1191            }))
1192        }
1193        _ => Err("unexpected reply to the launcher's construction".to_string()),
1194    }
1195}
