9//! Multicast liveness monitoring.
10//!
11//! [`MeshMonitor`] mirrors [`hyperactor::ActorMonitor`] but observes a set of
12//! ranked actors rather than a single one. It composes: one `ActorMonitor` per
13//! rank, fanning each per-rank [`MonitorFailure`] out into a mesh-level
14//! [`MeshFailure`] that names the failed rank. Multicast monitoring is, for now,
15//! many unicast monitors; a later diff can optimize the fan-out with casting.

45/// A liveness monitor over a set of ranked actors.
46///
47/// A `MeshMonitor` logically monitors one actor per rank in a region. Monitoring
48/// runs until the `MeshMonitor` is dropped. A monitor from [`MeshMonitor::spawn_persistent`]
49/// is persistent and polls throughout; one from [`MeshMonitor::spawn_dormant`]
50/// starts dormant and polls only while active. Awaiting a `&MeshMonitor`
51/// resolves with the first rank failure, reported as a [`MeshFailure`] that
52/// names the rank; use [`MeshMonitor::guard`] to run a future until it completes
53/// or a rank fails.
54pub struct MeshMonitor {
55    mesh_id: Option<ActorMeshId>,
56    /// The display name of each rank's actor, keyed by the mesh's region.
57    display_names: Option<ValueMesh<String>>,
58    /// One monitor per rank, keyed by the mesh's region.
59    monitors: ValueMesh<ActorMonitor>,
60}
61
62impl MeshMonitor {
63    /// Spawn one persistent [`ActorMonitor`] per rank as a child of `cx`.
64    ///
65    /// `actors` holds the actor addresses in the mesh's region, so the resulting
66    /// monitors share that region: each rank reports failures under the same rank
67    /// it holds in the mesh.
68    pub fn spawn_persistent(cx: &impl context::Actor, actors: ValueMesh<ActorAddr>) -> Self {
69        Self::spawn_with(actors, |actor| ActorMonitor::spawn_persistent(cx, actor))
70    }
71
72    /// Spawn one [`ActorMonitor`] per rank as a child of `cx`, starting dormant.
73    ///
74    /// Each rank observes terminal-status notifications in either state, but
75    /// polls status only while active: while a [`MeshMonitorActivation`] from
76    /// [`Self::activate`] is live.
77    pub fn spawn_dormant(cx: &impl context::Actor, actors: ValueMesh<ActorAddr>) -> Self {
78        Self::spawn_with(actors, |actor| ActorMonitor::spawn_dormant(cx, actor))
79    }
80
81    fn spawn_with(actors: ValueMesh<ActorAddr>, spawn: impl Fn(ActorAddr) -> ActorMonitor) -> Self {
82        Self {
83            mesh_id: None,
84            display_names: None,
85            monitors: actors.map_into(|actor| spawn(actor.clone())),
86        }
87    }
88
89    /// Identify the mesh in every failure reported by this monitor.
90    pub fn with_mesh_id(mut self, mesh_id: ActorMeshId) -> Self {
91        self.mesh_id = Some(mesh_id);
92        self
93    }
94
95    /// Name each rank's actor in the failures reported by this monitor.
96    ///
97    /// # Errors
98    ///
99    /// Returns [`crate::Error::DisplayNamesRegionMismatch`] if
100    /// `display_names` does not share the monitored region.
101    pub fn with_display_names(
102        mut self,
103        display_names: Option<ValueMesh<String>>,
104    ) -> crate::Result<Self> {
105        if let Some(display_names) = &display_names
106            && display_names.region() != self.region()
107        {
108            return Err(crate::Error::DisplayNamesRegionMismatch {
109                expected: self.region().clone(),
110                actual: display_names.region().clone(),
111            });
112        }
113        self.display_names = display_names;
114        Ok(self)
115    }
116
117    /// Keep every rank active until the returned activation is dropped.
118    ///
119    /// See [`ActorMonitor::activate`].
120    pub fn activate(&self, cx: &impl context::Actor) -> MeshMonitorActivation {
121        MeshMonitorActivation {
122            activations: self.monitors.map_into(|monitor| monitor.activate(cx)),
123        }
124    }
125
126    /// Return the monitored region.
127    pub fn region(&self) -> &Region {
128        self.monitors.region()
129    }
130
131    /// Return the latest observed status for `rank`.
132    pub fn status(&self, rank: usize) -> Option<ActorStatus> {
133        self.monitors
134            .get(rank)
135            .map(|monitor| match monitor.status() {
136                MonitorStatus::Checking => ActorStatus::Unknown,
137                MonitorStatus::Alive(status) => status,
138                MonitorStatus::Failed(failure) => monitor_failure_to_actor_status(&failure),
139            })
140    }
141
142    /// Wait for the first monitored rank to fail, reported as a [`MeshFailure`].
143    ///
144    /// Internal helper behind the [`IntoFuture`] impl for `&MeshMonitor` and
145    /// [`Self::guard`], the public ways to observe failures. An empty monitor set
146    /// never resolves: a mesh with no ranks has no rank that can fail.
147    async fn wait_for_failure(&self) -> MeshFailure {
148        if self.region().num_ranks() == 0 {
149            return std::future::pending().await;
150        }
151        let waits = (0..self.region().num_ranks()).map(|rank| {
152            let monitor = self
153                .monitors
154                .get(rank)
155                .expect("dense actor monitor mesh has one value per rank");
156            Box::pin(async move {
157                // Guarding a never-completing future resolves only when the
158                // monitor reports a failure.
159                let failure = monitor
160                    .guard(std::future::pending::<()>())
161                    .await
162                    .expect_err("pending future never completes");
163                (rank, failure)
164            })
165        });
166        let ((rank, failure), _index, _rest) = select_all(waits).await;
167        let mut failure = monitor_failure_to_mesh_failure(rank, &failure);
168        failure.actor_mesh_name = self.mesh_id.as_ref().map(ToString::to_string);
169        failure.event.display_name = self
170            .display_names
171            .as_ref()
172            .and_then(|names| names.get(rank))
173            .cloned();
174        failure
175    }
176
177    /// Wait until every monitored rank is terminal, returning each rank's
178    /// outcome in rank order. Hold a [`MeshMonitorActivation`] while waiting so
179    /// that an unreachable rank resolves with a status timeout instead of
180    /// waiting for a terminal notification that never arrives.
181    pub async fn wait_all_terminal(&self) -> Vec<MonitorFailure> {
182        join_all((0..self.region().num_ranks()).map(|rank| {
183            let monitor = self
184                .monitors
185                .get(rank)
186                .expect("dense actor monitor mesh has one value per rank");
187            async move {
188                monitor
189                    .guard(std::future::pending::<()>())
190                    .await
191                    .expect_err("pending future never completes")
192            }
193        }))
194        .await
195    }
196
197    /// Run `fut` until it completes or any monitored rank fails.
198    pub async fn guard<F>(&self, fut: F) -> Result<F::Output, MeshFailure>
199    where
200        F: Future,
201    {
202        tokio::pin!(fut);
203        tokio::select! {
204            result = fut => Ok(result),
205            failure = self.wait_for_failure() => Err(failure),
206        }
207    }
208
209    /// Cast `message` and collect one response per rank unless a rank fails.
210    ///
211    /// `receiver` must receive exactly one response from each actor in
212    /// `actor_mesh`. Responses are returned in arrival order. The outer result
213    /// reports cast or mailbox errors; the inner result distinguishes all-rank
214    /// completion from a monitored rank failure. This monitor must have been
215    /// created from `actor_mesh`.
216    pub async fn cast_and_collect<A, M, R>(
217        &self,
218        actor_mesh: &ActorMeshRef<A>,
219        cx: &impl context::Actor,
220        message: M,
221        receiver: &mut PortReceiver<R>,
222    ) -> crate::Result<Result<Vec<R>, MeshFailure>>
223    where
224        A: Referable + RemoteHandles<M>,
225        M: RemoteMessage + Clone,
226        R: Message,
227    {
228        actor_mesh.cast(cx, message)?;
229        let num_ranks = actor_mesh.region().num_ranks();
230        let replies = async {
231            let mut replies = Vec::with_capacity(num_ranks);
232            for _ in 0..num_ranks {
233                replies.push(receiver.recv().await?);
234            }
235            Ok::<_, hyperactor::mailbox::MailboxError>(replies)
236        };
237
238        match self.guard(replies).await {
239            Ok(Ok(replies)) => Ok(Ok(replies)),
240            Ok(Err(error)) => Err(Box::new(error).into()),
241            Err(failure) => Ok(Err(failure)),
242        }
243    }
244}

246/// Keeps a [`MeshMonitor`] active; see [`MeshMonitor::activate`].
247#[must_use = "the monitor goes dormant once its last activation is dropped"]
248pub struct MeshMonitorActivation {
249    #[expect(
250        dead_code,
251        reason = "held only so dropping it releases each rank's activation"
252    )]
253    activations: ValueMesh<MonitorActivation>,
254}
255
256/// Awaiting a `&MeshMonitor` resolves with the first rank failure. The borrow
257/// keeps the monitors alive for the duration of the await.
258impl<'a> IntoFuture for &'a MeshMonitor {
259    type Output = MeshFailure;
260    type IntoFuture = Pin<Box<dyn Future<Output = MeshFailure> + Send + 'a>>;
261
262    fn into_future(self) -> Self::IntoFuture {
263        Box::pin(self.wait_for_failure())
264    }
265}
266
267fn monitor_failure_to_mesh_failure(rank: usize, failure: &MonitorFailure) -> MeshFailure {
268    let actor_id = failure.actor_id().clone();
269    let actor_status = monitor_failure_to_actor_status(failure);
270    // The failed actor is identified by `event.actor_id` and the rank; the
271    // monitor adds the mesh and display names it was given.
272    MeshFailure {
273        actor_mesh_name: None,
274        event: ActorSupervisionEvent::new(actor_id, None, actor_status, None),
275        crashed_ranks: vec![rank],
276        // Synthesized locally from direct per-rank monitoring, not a
277        // controller report.
278        reporting_controller: None,
279    }
280}
281
282fn monitor_failure_to_actor_status(failure: &MonitorFailure) -> ActorStatus {
283    match failure {
284        MonitorFailure::ActorStopped { status, .. } => status.clone(),
285        MonitorFailure::ActorFailed { status, .. } if status.is_failed() => status.clone(),
286        failure => ActorStatus::generic_failure(failure.to_string()),
287    }
288}

307    #[tokio::test]
308    async fn test_mesh_monitor_names_failed_rank() {
309        let proc = Proc::isolated();
310        let client = proc.client("client");
311        let alive = client.spawn_with_label("rank0", testactor::TestActor);
312        let stopped = client.spawn_with_label("rank1", testactor::TestActor);
313        let region: Region = extent!(replicas = 2).into();
314        let actors = ValueMesh::new(
315            region.clone(),
316            vec![alive.actor_addr().clone(), stopped.actor_addr().clone()],
317        )
318        .unwrap();
319        let display_names = ValueMesh::new(
320            region,
321            vec![
322                "workers{'replicas': 0/2}".to_string(),
323                "workers{'replicas': 1/2}".to_string(),
324            ],
325        )
326        .unwrap();
327        let monitor = MeshMonitor::spawn_persistent(&client, actors)
328            .with_display_names(Some(display_names))
329            .expect("display names must share the monitored region");
330
331        stopped
332            .drain_and_stop("rank complete")
333            .expect("target should accept stop");
334        let failure = tokio::time::timeout(Duration::from_secs(10), &monitor)
335            .await
336            .expect("timed out waiting for mesh monitor failure");
337        assert_eq!(failure.crashed_ranks, vec![1]);
338        assert_eq!(
339            failure.event.display_name.as_deref(),
340            Some("workers{'replicas': 1/2}"),
341            "the failure should carry the stopped rank's display name"
342        );
343        assert!(
344            failure
345                .event
346                .to_string()
347                .contains("workers{'replicas': 1/2}"),
348            "the rendered failure should name the actor by display name: {}",
349            failure.event
350        );
351    }

353    #[tokio::test]
354    async fn test_mesh_monitor_reports_rank_failure() {
355        let proc = Proc::isolated();
356        let client = proc.client("client");
357        let target = client.spawn_with_label("rank0", testactor::TestActor);
358        let region: Region = extent!(replicas = 1).into();
359        let actors = ValueMesh::new(region, vec![target.actor_addr().clone()]).unwrap();
360        let monitor = MeshMonitor::spawn_persistent(&client, actors);
361        assert_eq!(monitor.status(0), Some(ActorStatus::Unknown));
362
363        // No failure should be observed while the actor is alive.
364        let mut wait = (&monitor).into_future();
365        tokio::select! {
366            biased;
367            failure = &mut wait => panic!("unexpected failure before stop: {failure:?}"),
368            _ = tokio::task::yield_now() => {}
369        }
370
371        target
372            .drain_and_stop("rank complete")
373            .expect("target should accept stop");
374
375        let failure = tokio::time::timeout(Duration::from_secs(10), wait)
376            .await
377            .expect("timed out waiting for mesh monitor failure");
378        assert_eq!(failure.crashed_ranks, vec![0]);
379        assert!(matches!(
380            failure.event.actor_status,
381            ActorStatus::Stopped(ref reason) if reason == "rank complete"
382        ));
383        assert!(matches!(
384            monitor.status(0).expect("rank 0 should remain monitored"),
385            ActorStatus::Stopped(ref reason) if reason == "rank complete"
386        ));
387
388        tokio::time::timeout(Duration::from_secs(5), target)
389            .await
390            .expect("timed out waiting for target to stop");
391    }

394    async fn test_cast_and_collect_returns_all_replies_or_rank_failure() {
395        let proc = Proc::isolated();
396        proc.spawn_bound_with_uid(
397            Uid::singleton(Label::strip(CAST_ACTOR_NAME)),
398            CastActor::default(),
399        )
400        .expect("isolated test proc should spawn its CastActor");
401        let client = proc.client("client");
402        let rank0 = client.spawn_with_label("rank0", testactor::TestActor);
403        let rank1 = client.spawn_with_label("rank1", testactor::TestActor);
404        let actor_refs: Vec<ActorRef<testactor::TestActor>> = vec![rank0.bind(), rank1.bind()];
405        let expected_actor_addrs: HashSet<_> = actor_refs
406            .iter()
407            .map(|actor_ref| actor_ref.actor_addr().clone())
408            .collect();
409        let region: Region = extent!(replicas = 2).into();
410        let actor_mesh = ActorMeshRef::try_new_data(
411            ActorMeshId::instance(Label::new("replicas").unwrap()),
412            region,
413            actor_refs,
414        )
415        .expect("data actor mesh should be valid");
416        let monitor = actor_mesh.monitor(&client);
417
418        let (port, mut receiver) = client.open_port();
419        let replies = tokio::time::timeout(
420            Duration::from_secs(10),
421            monitor.cast_and_collect(
422                &actor_mesh,
423                &client,
424                testactor::GetActorId(port.bind()),
425                &mut receiver,
426            ),
427        )
428        .await
429        .expect("timed out waiting for all rank replies")
430        .expect("cast and receive should succeed")
431        .expect("all ranks should remain alive");
432        let actor_addrs: HashSet<_> = replies
433            .into_iter()
434            .map(|(actor_addr, _seq_info)| actor_addr)
435            .collect();
436        assert_eq!(actor_addrs, expected_actor_addrs);
437
438        let (_port, mut receiver): (_, PortReceiver<()>) = client.open_port();
439        let mut pending =
440            std::pin::pin!(monitor.cast_and_collect(&actor_mesh, &client, (), &mut receiver,));
441        tokio::select! {
442            biased;
443            result = &mut pending => panic!("cast unexpectedly completed: {result:?}"),
444            _ = tokio::task::yield_now() => {}
445        }
446        rank1
447            .drain_and_stop("rank complete")
448            .expect("rank 1 should accept stop");
449        let failure = tokio::time::timeout(Duration::from_secs(10), pending)
450            .await
451            .expect("timed out waiting for a rank failure")
452            .expect("cast should initiate")
453            .expect_err("rank failure should win before all replies arrive");
454        assert_eq!(failure.actor_mesh_name, Some(actor_mesh.id().to_string()));
455        assert_eq!(failure.crashed_ranks, vec![1]);
456        assert!(matches!(
457            failure.event.actor_status,
458            ActorStatus::Stopped(ref reason) if reason == "rank complete"
459        ));
460
461        rank0
462            .drain_and_stop("test complete")
463            .expect("rank 0 should accept stop");
464        tokio::time::timeout(Duration::from_secs(5), rank0)
465            .await
466            .expect("timed out waiting for rank 0 to stop");
467        tokio::time::timeout(Duration::from_secs(5), rank1)
468            .await
469            .expect("timed out waiting for rank 1 to stop");
470    }
