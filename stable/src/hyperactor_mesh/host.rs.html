9//! Host lifecycle management for procs running on one machine.
10//!
11//! A [`Host`] owns one [`Gateway`] and one [`ProcManager`]. The manager makes
12//! procs real, either by spawning them in-process for tests or by launching
13//! separate OS processes. The gateway owns connectivity: it serves the backend
14//! and frontend endpoints, multiplexes inbound traffic to in-process procs,
15//! and routes spawned child proc gateways through peer routes.
16//!
17//! Use [`Host::new`] or [`Host::new_with_gateway`] to construct a host, start its
18//! backend and frontend accept loops, then [`Host::spawn`] to create child procs.
19//! Spawned children are returned as [`ProcAddr`] values whose location is
20//! advertised through this host's gateway.
21//!
22//! ## Gateway topology
23//!
24//! A host gateway exposes a frontend endpoint `*` and serves a backend
25//! endpoint `#` for child proc gateways. Each spawned child has its own
26//! gateway endpoint (`#1`, `#2`, ...), uses `#` as its forwarder, and is
27//! advertised as `Via(child_uid, host_location)`. The `host_location` is the
28//! host gateway's advertised location: the newest active frontend serve or
29//! `serve_via` session.
30//!
31//! The host gateway keeps one peer route per child uid. When it receives a
32//! message for `Via(child_uid, host_location)`, it peels the child hop and
33//! forwards the envelope to the child's gateway. In-process service and local
34//! procs are delivered through the gateway's local proc table.
35//!
36//! ```text
37//! inbound to host_location
38//! (*, or Via(...) when attached)
39//!          |
40//!          v
41//!   +---------------+     peer child_uid_1 -> #1     +----------------------+
42//!   | Host Gateway  |------------------------------->| child gateway/proc 1 |
43//!   | local procs:  |                                | addr #1              |
44//!   | service/local |<-------------------------------| forwarder -> #       |
45//!   +---------------+          host backend #         +----------------------+
46//!          |
47//!          | peer child_uid_2 -> #2
48//!          v
49//!   +----------------------+
50//!   | child gateway/proc 2 |
51//!   | addr #2              |
52//!   | forwarder -> #       |
53//!   +----------------------+
54//! ```
55//!
56//! The built-in service and local procs are created during construction and
57//! run in-process on the host gateway. The local proc starts with no actors; a
58//! `ProcAgent` and root client actor are added lazily when
59//! `HostMeshAgent::handle(GetLocalProc)` first asks for it.

187/// Client transport handles transferred to the bootstrap shutdown task.
188///
189/// Dropping this value stops any handles that have not already been joined,
190/// so cancellation cannot detach a live frontend or via transport.
191#[derive(Debug)]
192pub(crate) struct HostShutdownHandles {
193    frontend: Option<GatewayServeHandle>,
194    via: Option<GatewayServeHandle>,
195}
196
197impl HostShutdownHandles {
198    pub(crate) fn new(frontend: GatewayServeHandle, via: Option<GatewayServeHandle>) -> Self {
199        Self {
200            frontend: Some(frontend),
201            via,
202        }
203    }
204
205    pub(crate) async fn stop_and_join(mut self, reason: &str) {
206        if let Some(mut frontend) = self.frontend.take() {
207            frontend.stop(reason);
208            if let Err(error) = frontend.join().await {
209                tracing::warn!(%error, "failed to join host shutdown frontend");
210            }
211        }
212        if let Some(mut via) = self.via.take() {
213            via.stop(reason);
214            if let Err(error) = via.join().await {
215                tracing::warn!(%error, "failed to join host shutdown via transport");
216            }
217        }
218    }
219}

221impl Drop for HostShutdownHandles {
222    fn drop(&mut self) {
223        if let Some(frontend) = &mut self.frontend {
224            frontend.stop("host shutdown handles dropped");
225        }
226        if let Some(via) = &mut self.via {
227            via.stop("host shutdown handles dropped");
228        }
229    }

232struct ProcHandleKillGuard<H: ProcHandle> {
233    handle: Option<H>,
234}
235
236impl<H: ProcHandle> ProcHandleKillGuard<H> {
237    fn handle(&self) -> &H {
238        self.handle
239            .as_ref()
240            .expect("active spawn cleanup owns the proc handle")
241    }
242
243    fn disarm(mut self) {
244        self.handle = None;
245    }
246
247    fn maybe_kill(&mut self) -> impl Future<Output = ()> + Send + 'static {
248        let handle = self.handle.take();
249
250        async move {
251            let Some(handle) = handle else {
252                return;
253            };
254            match handle.kill().await {
255                Ok(_) | Err(TerminateError::AlreadyTerminated(_)) => {}
256                Err(error) => {
257                    tracing::warn!(proc = %handle.proc_addr(), %error, "failed to clean up proc spawn");
258                }
259            }
260        }
261    }
262}
263
264impl<H: ProcHandle> Drop for ProcHandleKillGuard<H> {
265    fn drop(&mut self) {
266        if self.handle.is_some() {
267            tokio::spawn(self.maybe_kill());
268        }
269    }

272/// Lifecycle manager for the procs on one machine.
273///
274/// The host delegates all connectivity to its [`Gateway`]. It creates
275/// built-in service/local procs, asks its [`ProcManager`] to spawn children,
276/// and keeps the gateway peer registrations for those children alive.
277pub struct Host<M> {
278    /// Peer guards for spawned child procs, keyed by name. The stored
279    /// [`PeerAttachGuard`] keeps the gateway peer route for the child
280    /// alive; dropping it removes the entry (used by
281    /// [`Host::terminate_children`] to free slots).
282    procs: Arc<StdMutex<HashMap<String, PeerAttachGuard>>>,
283    frontend_addr: ChannelAddr,
284    backend_addr: ChannelAddr,
285    /// Connectivity for every proc owned by this host.
286    ///
287    /// The built-in procs share the gateway in-process. Spawned children have
288    /// their own gateways and are registered here with
289    /// [`Gateway::attach_peer`].
290    gateway: Gateway,
291    frontend_handle: Option<GatewayServeHandle>,
292    backend_handle: Option<GatewayServeHandle>,
293    /// Duplex `serve_via` session to a remote gateway, present when this
294    /// host was bootstrapped out-of-cluster. Kept alive for the host's
295    /// lifetime so the cluster route and outbound forwarder persist.
296    via_handle: Option<GatewayServeHandle>,
297    manager: Arc<M>,
298    service_proc: Proc,
299    local_proc: Proc,
300}

304    pub async fn new(manager: M, addr: ChannelAddr) -> Result<Self, HostError> {

308    /// Like [`new`], but optionally uses an already-bound listener.
309    ///
310    /// When `listener` is `Some`, it is used as the frontend listening socket
311    /// instead of binding a new one.
312    pub async fn new_with_default(
313        manager: M,
314        addr: ChannelAddr,
315        listener: Option<std::net::TcpListener>,
316    ) -> Result<Self, HostError> {
317        // Default to the process-wide global gateway so procs on this
318        // host share one routing table with the rest of the process.
319        // Callers that need a different gateway (e.g. via attach) build
320        // it externally and pass it to [`new_with_gateway`].
321        Self::new_with_gateway(
322            manager,
323            ProcAddr::new(legacy_service_proc_id(), addr.into()),
324            listener,
325            Gateway::global().clone(),
326            None,
327        )
328        .await
329    }

331    /// Like [`new_with_default`], but uses a caller-provided [`Gateway`] and
332    /// service [`ProcAddr`].
333    ///
334    /// `addr` is a startup specification: its proc id becomes the service proc
335    /// identity, and its direct location is used to serve the frontend. The
336    /// final service proc location comes from the gateway after binding, so it
337    /// may differ because of an alias, an ephemeral address, or `via`. A
338    /// source-routed [`Location::Via`] is therefore rejected here.
339    ///
340    /// When `via` is `Some`, the gateway attaches to that remote duplex
341    /// address with [`Gateway::serve_via`] *after* the local
342    /// frontend/backend serves but *before* minting the built-in procs,
343    /// so the via session is the newest active serve. That ordering
344    /// makes every ref minted on this host advertise the routable
345    /// `Via` location rather than the bare local frontend, which is
346    /// what lets an out-of-cluster client receive return traffic over
347    /// the duplex.
348    #[hyperactor::instrument(fields(addr=addr.to_string()))]
349    pub async fn new_with_gateway(
350        manager: M,
351        addr: ProcAddr,
352        listener: Option<std::net::TcpListener>,
353        gateway: Gateway,
354        via: Option<ChannelAddr>,
355    ) -> Result<Self, HostError> {
356        let service_proc_id = addr.id().clone();
357        let addr = match addr.location() {
358            Location::Addr(addr) => addr.clone(),
359            Location::Via(..) => {
360                return Err(HostError::InvalidFrontendLocation(addr.location().clone()));
361            }
362        };
363        let mut backend_handle = Gateway::serve(&gateway, ChannelAddr::any(manager.transport()))?;
364        let backend_addr = gateway.default_location().addr().clone();
365
366        // Serve the frontend. Kernel-socket (net) transports use a muxed
367        // listener so simplex clients and duplex attach clients share one
368        // address; the gateway owns both the simplex and (`AttachWire`)
369        // duplex accept paths, so there is a single attach protocol whether
370        // a frontend is muxed or a plain duplex endpoint. `serve_mux`
371        // requires a net transport, so we branch on `is_net()` rather than
372        // `supports_duplex()` (the in-process `Local` transport supports
373        // duplex but is not a kernel socket). Both paths register the bound
374        // address as the gateway's active serve location.
375        let frontend_result: Result<GatewayServeHandle, ChannelError> = if addr.transport().is_net()
376        {
377            gateway.serve_mux_with_listener(addr, listener)
378        } else {
379            gateway.serve_with_listener(addr, listener)
380        };
381        let mut frontend_handle = match frontend_result {
382            Ok(handle) => handle,
383            Err(error) => {
384                backend_handle.stop("host setup failed");
385                if let Err(join_error) = backend_handle.join().await {
386                    tracing::warn!(
387                        error = %join_error,
388                        "failed to join backend server after host setup error"
389                    );
390                }
391                return Err(error.into());
392            }
393        };
394        let frontend_addr = gateway.default_location().addr().clone();
395        assert!(
396            !matches!(frontend_addr, ChannelAddr::Alias { .. }),
397            "gateway frontend serve must canonicalize aliases"
398        );
399
400        // Attach to the remote gateway after the local serves are live
401        // but before any proc or actor ref is minted below. `serve_via`
402        // must be the newest active serve so it supplies the gateway's
403        // `default_location`; otherwise the local frontend serve above
404        // would win, and refs would advertise a bare, cluster-
405        // unreachable address — the out-of-cluster return-path bug.
406        let via_handle = match via {
407            Some(via_addr) => match gateway.serve_via(via_addr).await {
408                Ok(handle) => Some(handle),
409                Err(error) => {
410                    frontend_handle.stop("host setup failed");
411                    if let Err(join_error) = frontend_handle.join().await {
412                        tracing::warn!(
413                            error = %join_error,
414                            "failed to join frontend server after via attach error"
415                        );
416                    }
417                    backend_handle.stop("host setup failed");
418                    if let Err(join_error) = backend_handle.join().await {
419                        tracing::warn!(
420                            error = %join_error,
421                            "failed to join backend server after via attach error"
422                        );
423                    }
424                    return Err(HostError::ViaAttachFailure(error));
425                }
426            },
427            None => None,
428        };
429
430        // Set up the system proc and the local client proc after the
431        // gateway servers are live. The HostAgent is published only
432        // after it binds its handler, so the brief unroutable window is
433        // before normal clients can discover this host.
434        let expected_service_proc_addr =
435            ProcAddr::new(service_proc_id.clone(), gateway.default_location());
436        let service_proc = Proc::builder()
437            .proc_id(service_proc_id)
438            .shared_gateway(gateway.clone())
439            .build()
440            .map_err(HostError::ProcConstructionFailure)?;
441        assert_eq!(
442            service_proc.proc_addr(),
443            expected_service_proc_addr,
444            "service proc must use the gateway's active advertised location"
445        );
446        let local_proc = Proc::builder()
447            .proc_id(ProcId::instance(Label::strip(LOCAL_PROC_NAME)))
448            .shared_gateway(gateway.clone())
449            .build()
450            .map_err(HostError::ProcConstructionFailure)?;
451        let service_proc_id = service_proc.proc_addr().clone();
452        let local_proc_id = local_proc.proc_addr().clone();
453
454        tracing::info!(
455            frontend_addr = frontend_addr.to_string(),
456            backend_addr = backend_addr.to_string(),
457            service_proc_id = service_proc_id.to_string(),
458            local_proc_id = local_proc_id.to_string(),
459            "serving host"
460        );
461
462        Ok(Host {
463            procs: Arc::new(StdMutex::new(HashMap::new())),
464            frontend_addr,
465            backend_addr,
466            gateway,
467            frontend_handle: Some(frontend_handle),
468            backend_handle: Some(backend_handle),
469            via_handle,
470            manager: Arc::new(manager),
471            service_proc,
472            local_proc,
473        })
474    }

500    /// Spawn a child proc with the given `name`.
501    ///
502    /// On success, the proc is ready and reachable through the returned
503    /// [`ProcAddr`]. The proc id is derived from `name`; its location is
504    /// advertised through this host's frontend gateway using a `Via(child_uid,
505    /// host_location)` source route.
506    /// The caller must ensure that no other spawn with the same `name` is in
507    /// flight. This method only rejects names that are already registered with the host.
508    pub(crate) fn spawn(
509        &self,
510        name: String,
511        config: M::Config,
512    ) -> impl Future<Output = Result<(ProcAddr, ActorRef<ManagerAgent<M>>), HostError>> + Send + 'static
513    where
514        M: Send + Sync + 'static,
515        M::Config: Send + 'static,
516    {
517        let procs = Arc::clone(&self.procs);
518        let backend_addr = self.backend_addr.clone();
519        let gateway = self.gateway.clone();
520        let manager = Arc::clone(&self.manager);
521
522        async move {
523            if procs
524                .lock()
525                .map_err(|_| HostError::ProcRegistryPoisoned)?
526                .contains_key(&name)
527            {
528                return Err(HostError::ProcExists(name));
529            }
530
531            // Preserve the route to this host, then append the child hop that
532            // is peeled by this host's gateway.
533            let resource_id = ResourceId::from_name(&name);
534            let proc_uid = resource_id.uid().clone();
535            let proc_id =
536                resource_id.proc_addr(gateway.default_location().append_via(proc_uid.clone()));
537            let handle = manager.spawn(proc_id.clone(), backend_addr, config).await?;
538            let mut proc_handle_guard = ProcHandleKillGuard {
539                handle: Some(handle),
540            };
541
542            let created = async {
543                let timeout: Duration =
544                    hyperactor_config::global::get(hyperactor::config::HOST_SPAWN_READY_TIMEOUT);
545
546                let ready = if timeout == Duration::from_secs(0) {
547                    ReadyProc::ensure(proc_handle_guard.handle()).await
548                } else {
549                    match tokio::time::timeout(
550                        timeout,
551                        ReadyProc::ensure(proc_handle_guard.handle()),
552                    )
553                    .await
554                    {
555                        Ok(result) => result,
556                        Err(_elapsed) => Err(ReadyProcError::Timeout),
557                    }
558                }
559                .map_err(|e| {
560                    HostError::ProcessConfigurationFailure(
561                        proc_id.clone(),
562                        anyhow::anyhow!("{e:?}"),
563                    )
564                })?;
565
566                let child_sender = MailboxClient::dial(ready.addr().clone()).map_err(|e| {
567                    HostError::ProcessConfigurationFailure(
568                        proc_id.clone(),
569                        anyhow::anyhow!("failed to dial spawned proc at {}: {}", ready.addr(), e),
570                    )
571                })?;
572                let agent_ref = ready.agent_ref().clone();
573                drop(ready);
574
575                let guard = gateway
576                    .attach_peer(proc_uid.clone(), child_sender.into_boxed())
577                    .map_err(|error| HostError::PeerAttachFailure(proc_id.clone(), error))?;
578                match procs
579                    .lock()
580                    .map_err(|_| HostError::ProcRegistryPoisoned)?
581                    .entry(name.clone())
582                {
583                    Entry::Vacant(entry) => {
584                        entry.insert(guard);
585                    }
586                    Entry::Occupied(_) => return Err(HostError::ProcExists(name.clone())),
587                }
588
589                Ok((proc_id.clone(), agent_ref))
590            }
591            .await;
592
593            match created {
594                Ok(created) => {
595                    proc_handle_guard.disarm();
596                    Ok(created)
597                }
598                Err(error) => {
599                    proc_handle_guard.maybe_kill().await;
600                    Err(error)
601                }
602            }
603        }
604    }

642    /// Stop and join every gateway server owned by this host.
643    pub(crate) async fn shutdown_servers(&mut self) {
644        if let Some(mut handle) = self.frontend_handle.take() {
645            handle.stop("host shutdown");
646            if let Err(error) = handle.join().await {
647                tracing::warn!(%error, "failed to join host frontend server");
648            }
649        }
650        if let Some(mut handle) = self.backend_handle.take() {
651            handle.stop("host shutdown");
652            if let Err(error) = handle.join().await {
653                tracing::warn!(%error, "failed to join host backend server");
654            }

975/// Minimal uniform surface for a spawned-**proc** handle returned by
976/// a `ProcManager`. Each manager can return its own concrete handle,
977/// as long as it exposes these. A **proc** is the Hyperactor runtime
978/// + its actors (lifecycle controlled via `Proc` APIs such as
979/// `destroy_and_wait`). A proc **may** be hosted *inside* an OS
980/// **process**, but it is conceptually distinct:
981///
982/// - `LocalProcManager`: runs the proc **in this OS process**; there
983///   is no child process to signal. Lifecycle is entirely proc-level.
984/// - `ProcessProcManager` (test-only here): launches an **external OS
985///   process** which hosts the proc, but this toy manager does
986///   **not** wire a control plane for shutdown, nor an exit monitor.
987///
988/// This trait is therefore written in terms of the **proc**
989/// lifecycle:
990///
991/// - `ready()` resolves when the proc is Ready (mailbox bound; agent
992///   available).
993/// - `wait()` resolves with the proc's terminal status
994///   (Stopped/Killed/Failed).
995/// - `terminate()` requests a graceful shutdown of the *proc* and
996///   waits up to the deadline; managers that also own a child OS
997///   process may escalate to `SIGKILL` if the proc does not exit in
998///   time.
999/// - `kill()` requests an immediate, forced termination. For
1000///    in-process procs, this may be implemented as an immediate
1001///    drain/abort of actor tasks. For external procs, this is
1002///    typically a `SIGKILL`.
1003///
1004/// The shape of the terminal value is `Self::TerminalStatus`.
1005/// Managers that track rich info (exit code, signal, address, agent)
1006/// can expose it; trivial managers may use `()`.
1007///
1008/// Managers that do not support signaling must return `Unsupported`.
1009#[async_trait]
1010pub trait ProcHandle: Clone + Send + Sync + 'static {
1011    /// The agent actor type installed in the proc by the manager.
1012    /// Must implement both:
1013    /// - [`Actor`], because the agent actually runs inside the proc,
1014    ///   and
1015    /// - [`Referable`], so callers can hold `ActorRef<Self::Agent>`.
1016    type Agent: Actor + Referable;
1017
1018    /// The type of terminal status produced when the proc exits.
1019    ///
1020    /// For example, an external proc manager may use a rich status
1021    /// enum (e.g. `ProcStatus`), while an in-process manager may use
1022    /// a trivial unit type. This is the value returned by
1023    /// [`ProcHandle::wait`] and carried by [`ReadyError::Terminal`].
1024    type TerminalStatus: std::fmt::Debug + Clone + Send + Sync + 'static;
1025
1026    /// The proc's logical address on this host.
1027    fn proc_addr(&self) -> &ProcAddr;
1028
1029    /// The proc's address (the one callers bind into the host
1030    /// router). May return `None` before `ready()` completes.
1031    /// Guaranteed to return `Some` after `ready()` succeeds.
1032    ///
1033    /// **Prefer [`ready_proc()`]** for type-safe access that
1034    /// guarantees availability at compile time.
1035    fn addr(&self) -> Option<ChannelAddr>;
1036
1037    /// The agent actor reference hosted in the proc. May return
1038    /// `None` before `ready()` completes. Guaranteed to return `Some`
1039    /// after `ready()` succeeds.
1040    ///
1041    /// **Prefer [`ready_proc()`]** for type-safe access that
1042    /// guarantees availability at compile time.
1043    fn agent_ref(&self) -> Option<ActorRef<Self::Agent>>;
1044
1045    /// Resolves when the proc becomes Ready. Multi-waiter,
1046    /// non-consuming.
1047    async fn ready(&self) -> Result<(), ReadyError<Self::TerminalStatus>>;
1048
1049    /// Resolves with the terminal status (Stopped/Killed/Failed/etc).
1050    /// Multi-waiter, non-consuming.
1051    async fn wait(&self) -> Result<Self::TerminalStatus, WaitError>;
1052
1053    /// Politely stop the proc before the deadline; managers that own
1054    /// a child OS process may escalate to a forced kill at the
1055    /// deadline. Idempotent and race-safe: concurrent callers
1056    /// coalesce; the first terminal outcome wins and all callers
1057    /// observe it via `wait()`.
1058    ///
1059    /// Returns the single terminal status the proc reached (the same
1060    /// value `wait()` will return). Never fabricates terminal states:
1061    /// this is only returned after the exit monitor observes
1062    /// termination.
1063    ///
1064    /// # Parameters
1065    /// - `cx`: The actor context for sending messages.
1066    /// - `timeout`: Grace period before escalation.
1067    /// - `reason`: Human-readable reason for termination.
1068    async fn terminate(
1069        &self,
1070        cx: &impl context::Actor,
1071        timeout: Duration,
1072        reason: &str,
1073    ) -> Result<Self::TerminalStatus, TerminateError<Self::TerminalStatus>>;
1074
1075    /// Force the proc down immediately. For in-process managers this
1076    /// may abort actor tasks; for external managers this typically
1077    /// sends `SIGKILL`. Also idempotent/race-safe; the terminal
1078    /// outcome is the one observed by `wait()`.
1079    async fn kill(&self) -> Result<Self::TerminalStatus, TerminateError<Self::TerminalStatus>>;
1080}

1139/// A ProcManager that spawns **in-process** procs (test-only).
1140///
1141/// The proc runs inside this same OS process; there is **no** child
1142/// process to signal. Lifecycle is purely proc-level:
1143/// - `terminate(timeout)`: delegates to
1144///   `Proc::destroy_and_wait(timeout)`, which drains and, at the
1145///   deadline, aborts remaining actors.
1146/// - `kill()`: uses a zero deadline to emulate a forced stop via
1147///   `destroy_and_wait(Duration::ZERO)`.
1148/// - `wait()`: trivial (no external lifecycle to observe).
1149///
1150///   No OS signals are sent or required.
1151pub struct LocalProcManager<S> {
1152    procs: Arc<Mutex<HashMap<ProcAddr, Proc>>>,
1153    stopping: Arc<Mutex<HashMap<ProcAddr, tokio::sync::watch::Sender<LocalProcStatus>>>>,
1154    spawn: S,
1155}
1156
1157impl<S> LocalProcManager<S> {
1158    /// Create a new in-process proc manager with the given agent
1159    /// params.
1160    pub fn new(spawn: S) -> Self {
1161        Self {
1162            procs: Arc::new(Mutex::new(HashMap::new())),
1163            stopping: Arc::new(Mutex::new(HashMap::new())),
1164            spawn,
1165        }
1166    }
1167
1168    async fn discard_proc(&self, proc_id: &ProcAddr, reason: &str) {
1169        let proc = self.procs.lock().await.remove(proc_id);
1170        if let Some(mut proc) = proc
1171            && let Err(error) = proc.destroy_and_wait(Duration::ZERO, reason).await
1172        {
1173            tracing::warn!(%proc_id, %error, "failed to discard local proc");
1174        }
1175    }
1176
1177    /// Non-blocking stop: remove the proc and spawn a background task
1178    /// that tears it down.
1179    ///
1180    /// Status transitions through `Stopping` -> `Stopped` and is
1181    /// observable via [`local_proc_status`] and [`watch`]. Idempotent:
1182    /// no-ops if the proc is already stopping or stopped.
1183    pub async fn request_stop(&self, proc: &ProcAddr, timeout: Duration, reason: &str) {
1184        {
1185            let guard = self.stopping.lock().await;
1186            if guard.contains_key(proc) {
1187                return;
1188            }
1189        }
1190
1191        let mut proc_handle = {
1192            let mut guard = self.procs.lock().await;
1193            match guard.remove(proc) {
1194                Some(p) => p,
1195                None => return,
1196            }
1197        };
1198
1199        let proc_ref: ProcAddr = proc_handle.proc_addr().clone();
1200        let (tx, _) = tokio::sync::watch::channel(LocalProcStatus::Stopping);
1201        self.stopping.lock().await.insert(proc_ref.clone(), tx);
1202
1203        let stopping = Arc::clone(&self.stopping);
1204        let reason = reason.to_string();
1205        tokio::spawn(async move {
1206            if let Err(e) = proc_handle.destroy_and_wait(timeout, &reason).await {
1207                tracing::warn!(error = %e, "request_stop(local): destroy_and_wait failed");
1208            }
1209            if let Some(tx) = stopping.lock().await.get(&proc_ref) {
1210                let _ = tx.send(LocalProcStatus::Stopped);
1211            }
1212        });
1213    }
1214
1215    /// Query the lifecycle status of a proc that was stopped via
1216    /// [`request_stop`].
1217    ///
1218    /// Returns `None` if the proc was never stopped through this path.
1219    pub async fn local_proc_status(&self, proc: &ProcAddr) -> Option<LocalProcStatus> {
1220        self.stopping.lock().await.get(proc).map(|tx| *tx.borrow())
1221    }
1222
1223    /// Subscribe to lifecycle status changes for a proc that was
1224    /// stopped via [`request_stop`].
1225    ///
1226    /// Returns `None` if the proc was never stopped through this path.
1227    pub async fn watch(
1228        &self,
1229        proc: &ProcAddr,
1230    ) -> Option<tokio::sync::watch::Receiver<LocalProcStatus>> {
1231        self.stopping
1232            .lock()
1233            .await
1234            .get(proc)
1235            .map(|tx| tx.subscribe())
1236    }
1237}

1239#[async_trait]
1240impl<S> BulkTerminate for LocalProcManager<S>
1241where
1242    S: Send + Sync,
1243{
1244    async fn terminate_all(
1245        &self,
1246        _cx: &impl context::Actor,
1247        timeout: std::time::Duration,
1248        max_in_flight: usize,
1249        reason: &str,
1250    ) -> TerminateSummary {
1251        // Drain procs so we don't hold the lock across awaits and subsequent
1252        // calls to terminate_all don't try to re-terminate.
1253        let procs: Vec<Proc> = {
1254            let mut guard = self.procs.lock().await;
1255            guard.drain().map(|(_, v)| v).collect()
1256        };
1257
1258        let attempted = procs.len();
1259
1260        let results = stream::iter(procs.into_iter().map(|mut p| async move {
1261            // For local manager, graceful proc-level stop.
1262            match p.destroy_and_wait(timeout, reason).await {
1263                Ok(_) => true,
1264                Err(e) => {
1265                    tracing::warn!(error=%e, "terminate_all(local): destroy_and_wait failed");
1266                    false
1267                }
1268            }
1269        }))
1270        .buffer_unordered(max_in_flight.max(1))
1271        .collect::<Vec<bool>>()
1272        .await;
1273
1274        let ok = results.into_iter().filter(|b| *b).count();
1275
1276        TerminateSummary {
1277            attempted,
1278            ok,
1279            failed: attempted.saturating_sub(ok),
1280        }
1281    }
1282}

1290        &self,

1309/// A lightweight [`ProcHandle`] for procs managed **in-process** via
1310/// [`LocalProcManager`].
1311///
1312/// This handle wraps the minimal identifying state of a spawned proc:
1313/// - its [`ProcId`] (logical identity on the host),
1314/// - the proc's [`ChannelAddr`] (the address callers bind into the
1315///   host router), and
1316/// - the [`ActorAddr`] to the agent actor hosted in the proc.
1317///
1318/// Unlike external handles, `LocalHandle` does **not** manage an OS
1319/// child process. It provides a uniform surface (`proc_id()`,
1320/// `addr()`, `agent_ref()`) and implements `terminate()`/`kill()` by
1321/// calling into the underlying `Proc::destroy_and_wait`, i.e.,
1322/// **proc-level** shutdown.
1323///
1324/// **Type parameter:** `A` is constrained by the `ProcHandle::Agent`
1325/// bound (`Actor + Referable`).
1326pub struct LocalHandle<A: Actor + Referable> {
1327    proc_id: ProcAddr,
1328    addr: ChannelAddr,
1329    agent_ref: ActorRef<A>,
1330    procs: Arc<Mutex<HashMap<ProcAddr, Proc>>>,
1331}

1334impl<A: Actor + Referable> Clone for LocalHandle<A> {

1337            proc_id: self.proc_id.clone(),
1338            addr: self.addr.clone(),
1339            agent_ref: self.agent_ref.clone(),
1340            procs: Arc::clone(&self.procs),
1341        }
1342    }
1343}

1345#[async_trait]
1346impl<A: Actor + Referable> ProcHandle for LocalHandle<A> {
1347    /// `Agent = A` (inherits `Actor + Referable` from the trait
1348    /// bound).
1349    type Agent = A;
1350    type TerminalStatus = ();
1351
1352    fn proc_addr(&self) -> &ProcAddr {
1353        &self.proc_id
1354    }
1355
1356    fn addr(&self) -> Option<ChannelAddr> {
1357        Some(self.addr.clone())
1358    }
1359
1360    fn agent_ref(&self) -> Option<ActorRef<Self::Agent>> {
1361        Some(self.agent_ref.clone())
1362    }
1363
1364    /// Always resolves immediately: a local proc is created
1365    /// in-process and is usable as soon as the handle exists.
1366    async fn ready(&self) -> Result<(), ReadyError<Self::TerminalStatus>> {
1367        Ok(())
1368    }
1369    /// Always resolves immediately with `()`: a local proc has no
1370    /// external lifecycle to await. There is no OS child process
1371    /// behind this handle.
1372    async fn wait(&self) -> Result<Self::TerminalStatus, WaitError> {
1373        Ok(())
1374    }
1375
1376    async fn terminate(
1377        &self,
1378        _cx: &impl context::Actor,
1379        timeout: Duration,
1380        reason: &str,
1381    ) -> Result<(), TerminateError<Self::TerminalStatus>> {
1382        let mut proc = {
1383            let guard = self.procs.lock().await;
1384            match guard.get(self.proc_addr()) {
1385                Some(p) => p.clone(),
1386                None => {
1387                    // The proc was already removed; treat as already
1388                    // terminal.
1389                    return Err(TerminateError::AlreadyTerminated(()));
1390                }
1391            }
1392        };
1393
1394        // Graceful stop of the *proc* (actors) with a deadline. This
1395        // will drain and then abort remaining actors at expiry.
1396        let _ = proc
1397            .destroy_and_wait(timeout, reason)
1398            .await
1399            .map_err(TerminateError::Io)?;
1400
1401        Ok(())
1402    }
1403
1404    async fn kill(&self) -> Result<(), TerminateError<Self::TerminalStatus>> {
1405        // Forced stop == zero deadline; `destroy_and_wait` will
1406        // immediately abort remaining actors and return.
1407        let mut proc = {
1408            let guard = self.procs.lock().await;
1409            match guard.get(self.proc_addr()) {
1410                Some(p) => p.clone(),
1411                None => return Err(TerminateError::AlreadyTerminated(())),
1412            }
1413        };
1414
1415        let _ = proc
1416            .destroy_and_wait(Duration::from_millis(0), "kill")
1417            .await
1418            .map_err(TerminateError::Io)?;
1419
1420        Ok(())
1421    }
1422}

1424/// Local, in-process ProcManager.
1425///
1426/// **Type bounds:**
1427/// - `A: Actor + Referable + Binds<A>`
1428///   - `Actor`: the agent actually runs inside the proc.
1429///   - `Referable`: callers hold `ActorRef<A>` to the agent; this
1430///     bound is required for typed remote refs.
1431///   - `Binds<A>`: lets the runtime wire the agent's handler ports.
1432/// - `F: Future<Output = anyhow::Result<ActorHandle<A>>> + Send`:
1433///   the spawn closure returns a Send future (we `tokio::spawn` it).
1434/// - `S: Fn(Proc) -> F + Sync`: the factory can be called from
1435///   concurrent contexts.
1436///
1437/// Result handle is `LocalHandle<A>` (whose `Agent = A` via `ProcHandle`).
1438#[async_trait]
1439impl<A, S, F> ProcManager for LocalProcManager<S>
1440where
1441    A: Actor + Referable + Binds<A>,
1442    F: Future<Output = anyhow::Result<ActorHandle<A>>> + Send,
1443    S: Fn(Proc) -> F + Sync,
1444{
1445    type Handle = LocalHandle<A>;
1446
1447    fn transport(&self) -> ChannelTransport {
1448        ChannelTransport::Local
1449    }
1450
1451    #[hyperactor::instrument(fields(proc_id=proc_id.to_string(), addr=forwarder_addr.to_string()))]
1452    async fn spawn(
1453        &self,
1454        proc_id: ProcAddr,
1455        forwarder_addr: ChannelAddr,
1456        _config: (),
1457    ) -> Result<Self::Handle, HostError> {
1458        let transport = forwarder_addr.transport();
1459        let proc = Proc::configured(
1460            proc_id.clone(),
1461            MailboxClient::dial(forwarder_addr)?.into_boxed(),
1462        );
1463        let (proc_addr, rx) = channel::serve(ChannelAddr::any(transport))?;
1464        self.procs
1465            .lock()
1466            .await
1467            .insert(proc_id.clone(), proc.clone());
1468        let _handle = proc.clone().serve(rx);
1469        let agent_handle = match AssertUnwindSafe((self.spawn)(proc)).catch_unwind().await {
1470            Ok(Ok(agent_handle)) => agent_handle,
1471            Ok(Err(error)) => {
1472                self.discard_proc(&proc_id, "proc agent failed to spawn")
1473                    .await;
1474                return Err(HostError::AgentSpawnFailure(proc_id, error));
1475            }
1476            Err(panic) => {
1477                self.discard_proc(&proc_id, "proc agent spawn panicked")
1478                    .await;
1479                std::panic::resume_unwind(panic);
1480            }
1481        };
1482
1483        Ok(LocalHandle {
1484            proc_id,
1485            addr: proc_addr,
1486            agent_ref: agent_handle.bind(),
1487            procs: Arc::clone(&self.procs),
1488        })
1489    }

1492/// A ProcManager that manages each proc as a **separate OS process**
1493/// (test-only toy).
1494///
1495/// This implementation launches a child via `Command` and relies on
1496/// `kill_on_drop(true)` so that children are SIGKILLed if the manager
1497/// (or host) drops. There is **no** graceful proc control plane or
1498/// exit monitor wired here. `terminate()` and `kill()` therefore
1499/// remove and force-kill the child directly, while `wait()` remains
1500/// trivial and does not report real lifecycle state.
1501///
1502/// It follows a simple protocol:
1503///
1504/// Each process is launched with the following environment variables:
1505/// - `HYPERACTOR_HOST_BACKEND_ADDR`: the backend address to which all messages are forwarded,
1506/// - `HYPERACTOR_HOST_PROC_ID`: the proc id to assign the launched proc, and
1507/// - `HYPERACTOR_HOST_CALLBACK_ADDR`: the channel address with which to return the proc's address
1508///
1509/// The launched proc should also spawn an actor to manage it - the details of this are
1510/// implementation dependent, and outside the scope of the process manager.
1511///
1512/// The function [`boot_proc`] provides a convenient implementation of the
1513/// protocol.
1514pub struct ProcessProcManager<A> {
1515    program: std::path::PathBuf,
1516    children: Arc<Mutex<HashMap<ProcAddr, Child>>>,
1517    _phantom: PhantomData<A>,
1518}
1519
1520struct ProcessChildCleanup {
1521    children: Arc<Mutex<HashMap<ProcAddr, Child>>>,
1522    proc_id: Option<ProcAddr>,
1523}
1524
1525impl ProcessChildCleanup {
1526    fn disarm(mut self) {
1527        self.proc_id = None;
1528    }
1529}
1530
1531impl Drop for ProcessChildCleanup {
1532    fn drop(&mut self) {
1533        let Some(proc_id) = self.proc_id.take() else {
1534            return;
1535        };
1536        let children = Arc::clone(&self.children);
1537
1538        tokio::spawn(async move {
1539            if let Some(mut child) = children.lock().await.remove(&proc_id) {
1540                if let Err(error) = child.kill().await {
1541                    tracing::warn!(%proc_id, %error, "failed to kill interrupted process spawn");
1542                }
1543
1544                if let Err(error) = child.wait().await {
1545                    tracing::warn!(%proc_id, %error, "failed to reap interrupted process spawn");
1546                }
1547            }
1548        });
1549    }
1550}

1670impl<A> ProcManager for ProcessProcManager<A>
1671where
1672    // Agent actor runs in the proc (`Actor`) and must be
1673    // referenceable (`Referable`).
1674    A: Actor + Referable + Sync,
1675{
1676    type Handle = ProcessHandle<A>;
1677

1679        ChannelTransport::Unix
1680    }
1681
1682    #[hyperactor::instrument(fields(proc_id=proc_id.to_string(), addr=forwarder_addr.to_string()))]
1683    async fn spawn(
1684        &self,
1685        proc_id: ProcAddr,
1686        forwarder_addr: ChannelAddr,
1687        _config: (),
1688    ) -> Result<Self::Handle, HostError> {
1689        let (callback_addr, mut callback_rx) =
1690            channel::serve(ChannelAddr::any(ChannelTransport::Unix))?;
1691
1692        let mut cmd = Command::new(&self.program);
1693        cmd.env("HYPERACTOR_HOST_PROC_ID", proc_id.to_string());
1694        cmd.env("HYPERACTOR_HOST_BACKEND_ADDR", forwarder_addr.to_string());
1695        cmd.env("HYPERACTOR_HOST_CALLBACK_ADDR", callback_addr.to_string());
1696
1697        // Lifetime strategy: mark the child with
1698        // `kill_on_drop(true)` so the OS will send SIGKILL if the
1699        // handle is dropped and retain the `Child` in
1700        // `self.children`, tying its lifetime to the manager/host.
1701        //
1702        // This is the simplest viable policy to avoid orphaned
1703        // subprocesses in CI; more sophisticated lifecycle control
1704        // (graceful shutdown, restart) will be layered on later.
1705
1706        // Kill the child when its handle is dropped.
1707        cmd.kill_on_drop(true);
1708
1709        let child = cmd.spawn().map_err(|e| {
1710            HostError::ProcessSpawnFailure(proc_id.clone(), self.program.display().to_string(), e)
1711        })?;
1712
1713        // Retain the handle so it lives for the life of the
1714        // manager/host.
1715        {
1716            let mut children = self.children.lock().await;
1717            children.insert(proc_id.clone(), child);
1718        }
1719        let cleanup = ProcessChildCleanup {
1720            children: Arc::clone(&self.children),
1721            proc_id: Some(proc_id.clone()),
1722        };
1723
1724        // Wait for the child's callback with (addr, agent_ref)
1725        let (proc_addr, agent_ref) = callback_rx.recv().await?;
1726
1727        // TODO(production): For a non-test implementation, plumb a
1728        // shutdown path:
1729        // - expose a proc-level graceful stop RPC on the agent and
1730        //   implement `terminate(timeout)` by invoking it and, on
1731        //   deadline, call `Child::kill()`; implement `kill()` as
1732        //   immediate `Child::kill()`.
1733        // - wire an exit monitor so `wait()` resolves with a real
1734        //   terminal status.
1735        let handle = ProcessHandle {
1736            proc_id,
1737            addr: proc_addr,
1738            agent_ref,
1739            children: Arc::clone(&self.children),
1740        };
1741        cleanup.disarm();
1742        Ok(handle)
1743    }

1779/// Spawn a proc at `proc_id` with an `A`-typed agent actor,
1780/// forwarding messages to the provided `backend_addr`,
1781/// and returning the proc's address and agent actor on
1782/// the provided `callback_addr`.
1783#[hyperactor::instrument(fields(proc_id=proc_id.to_string(), addr=backend_addr.to_string(), callback_addr=callback_addr.to_string()))]
1784pub async fn spawn_proc<A, S, F>(
1785    proc_id: ProcAddr,
1786    backend_addr: ChannelAddr,
1787    callback_addr: ChannelAddr,
1788    spawn: S,
1789) -> Result<Proc, HostError>
1790where
1791    // `Actor`: runs in the proc; `Referable`: allows ActorRef<A>;
1792    // `Binds<A>`: wires ports
1793    A: Actor + Referable + Binds<A>,
1794    S: FnOnce(Proc) -> F,
1795    F: Future<Output = Result<ActorHandle<A>, anyhow::Error>>,
1796{
1797    let backend_transport = backend_addr.transport();
1798    let proc = Proc::configured(
1799        proc_id.clone(),
1800        MailboxClient::dial(backend_addr)?.into_boxed(),
1801    );
1802
1803    let agent_handle = spawn(proc.clone())
1804        .await
1805        .map_err(|e| HostError::AgentSpawnFailure(proc_id.clone(), e))?;
1806
1807    // Finally serve the proc on the same transport as the backend address,
1808    // and call back.
1809    let (proc_addr, proc_rx) = channel::serve(ChannelAddr::any(backend_transport))?;
1810    proc.clone().serve(proc_rx);
1811    let agent_ref: ActorRef<A> = agent_handle.bind::<A>();
1812    channel::dial::<(ChannelAddr, ActorRef<A>)>(callback_addr)?
1813        .send((proc_addr, agent_ref))
1814        .await
1815        .map_err(ChannelError::from)?;
1816
1817    Ok(proc)
1818}

2542    /// Repro for the OSS broken-link issue: when the host's duplex
2543    /// frontend shuts down with messages still on the wire, the
2544    /// simplex peer must see a clean close (and pending acks must
2545    /// flush) rather than retry-looping for `MESSAGE_DELIVERY_TIMEOUT`.
2546    ///
2547    /// Before the fix: the peer's `NetTx` got no acks for in-flight
2548    /// messages and no `Closed` response, so it spent the full 30 s
2549    /// `MESSAGE_DELIVERY_TIMEOUT` reconnecting against a dead host.

2552    /// asserts the simplex `NetTx` transitions to `Closed` quickly —

2557            Ok(proc.spawn_with_label::<EchoActor>("host_agent", EchoActor))
2558        });
2559        let mut host =
2560            Host::new_with_default(proc_manager, ChannelAddr::any(ChannelTransport::Unix), None)
2561                .await
2562                .unwrap();
2563        let mut serve_handle = host.take_frontend_handle().unwrap();
2564
2565        // Spawn an EchoActor and send a request from a simplex client.
2566        let echo_handle = host.system_proc().spawn(EchoActor);
2567        let echo_ref = echo_handle.bind::<EchoActor>();
2568
2569        let dial_router = DialMailboxRouter::new();
2570        dial_router.bind(
2571            Addr::from(host.system_proc().proc_addr().clone()),
2572            host.addr().clone(),
2573        );
2574        let client_addr = ChannelAddr::any(ChannelTransport::Unix);
2575        let (client_listen_addr, client_rx) = channel::serve(client_addr).unwrap();
2576        let client_proc_id = ResourceId::proc_addr_from_name(client_listen_addr, "client");
2577        let client_proc = Proc::configured(client_proc_id, dial_router.into_boxed());
2578        let _client_handle = client_proc.clone().serve(client_rx);
2579
2580        let client_inst = client_proc.client("requester");
2581        let (reply_port, reply_handle) = client_inst.mailbox().open_once_port::<ActorAddr>();
2582        let reply_port = reply_port.bind();
2583        echo_ref
2584            .port::<OncePortRef<ActorAddr>>()
2585            .post(&client_inst, reply_port);
2586        let _ = tokio::time::timeout(Duration::from_secs(5), reply_handle.recv())
2587            .await
2588            .expect("baseline round-trip timed out")
2589            .expect("baseline recv failed");
2590
2591        // Snapshot the client's outbound NetTx status before shutdown.
2592        let host_tx = channel::dial::<MessageEnvelope>(host.addr().clone()).unwrap();
2593        // Push one message so the lazy-connect kicks in.
2594        let dummy_dest = host
2595            .system_proc()
2596            .proc_addr()
2597            .actor_addr("noop")
2598            .port_addr(Port::from(0u64));
2599        let mut envelope = MessageEnvelope::serialize(
2600            client_inst.self_addr().clone(),
2601            dummy_dest,
2602            &"warmup".to_string(),
2603            Default::default(),
2604        )
2605        .unwrap();
2606        envelope.set_return_undeliverable(false);
2607        host_tx.post(envelope);
2608        // Wait briefly for connection to establish.
2609        tokio::time::sleep(Duration::from_millis(200)).await;
2610        assert!(matches!(*host_tx.status().borrow(), TxStatus::Active));
2611
2612        // Shut down the host's frontend. The fix ensures pending
2613        // recv-side acks are flushed AND a `Closed` response is sent,
2614        // so the simplex peer transitions to `Closed` promptly.
2615        serve_handle.stop("test shutdown");
2616        let _ = tokio::time::timeout(Duration::from_secs(5), serve_handle.join())
2617            .await
2618            .expect("serve handle did not resolve");
2619
2620        // The simplex peer should see Closed within a few seconds —
2621        // not the full MESSAGE_DELIVERY_TIMEOUT (30 s). Wait for the
2622        // status watch to flip.
2623        let mut status = host_tx.status().clone();
2624        tokio::time::timeout(Duration::from_secs(10), async {
2625            loop {
2626                if let TxStatus::Closed(_) = *status.borrow() {
2627                    return;
2628                }
2629                if status.changed().await.is_err() {
2630                    return;
2631                }
2632            }
2633        })
2634        .await
2635        .expect("simplex peer did not see Closed within 10s of host shutdown");
2636
2637        match &*host_tx.status().borrow() {
2638            TxStatus::Closed(_) => {}
2639            other => panic!("expected TxStatus::Closed, got {:?}", other),
2640        }
2641    }
