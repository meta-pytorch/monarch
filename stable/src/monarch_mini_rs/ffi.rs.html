9//! Raw, `unsafe` FFI declarations for the minimonarch C ABI.
10//!
11//! These mirror `monarch_mini/minimonarch.h` verbatim: same layout, same
12//! function signatures, same error codes. This is the *only* place that talks
13//! to the C symbols; the safe wrappers in [`crate`] are built exclusively on
14//! top of these declarations so the bindings go through the stable C ABI rather
15//! than the internal Rust implementation.
16//!
17//! Nothing here is safe to call directly — the layout contracts (valid
18//! pointers, ownership transfer of `mm_msg_part_t`, single-threaded poller use)
19//! are documented on the safe wrappers.
20
21#![expect(non_camel_case_types, reason = "mirror the C ABI type names exactly")]
22
23use std::ffi::c_char;
24use std::ffi::c_int;
25use std::ffi::c_void;
26
27/// A single part of a multipart message (mirrors `mm_msg_part_t`).
28///
29/// Ownership of the buffer is transferred *by value* into the runtime on
30/// `mm_actor_send`/`mm_actor_serve`/etc.: after the call the runtime invokes
31/// `deleter(deleter_ctx)` exactly once when the bytes are no longer needed.
32#[repr(C)]
33pub struct mm_msg_part_t {
34    pub data: *const c_void,
35    pub len: usize,
36    /// Called with `deleter_ctx` when this part's memory may be freed. May be
37    /// `None` (NULL in C) for borrowed data with no owned storage.
38    pub deleter: Option<unsafe extern "C" fn(ctx: *mut c_void)>,
39    pub deleter_ctx: *mut c_void,
40}
41
42/// A complete multipart message: an array of `n_parts` parts (mirrors `mm_msg_t`).
43#[repr(C)]
44pub struct mm_msg_t {
45    pub parts: *mut mm_msg_part_t,
46    pub n_parts: usize,
47}
48
49// Opaque handle types. The C header exposes these only as pointers.
50pub enum mm_ctx {}
51pub enum mm_actor {}
52pub enum mm_poller {}
53
54/// A monitor handle is just an opaque integer id, scoped to its actor (mirrors
55/// `mm_monitor_handle_t`). It owns nothing.
56pub type mm_monitor_handle_t = u64;
57
58/// This actor's role in a serve/join pair (mirrors `mm_role_t`).
59#[repr(i32)]
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61pub enum mm_role_t {
62    Child = 0,
63    Parent = 1,
64}
65
66/// Arguments shared by `mm_actor_serve` and `mm_actor_join` (mirrors
67/// `mm_connect_args_t`).
68#[repr(C)]
69pub struct mm_connect_args_t {
70    pub role: mm_role_t,
71    /// Optional: assign a name to the remote actor (NULL to skip). Transferred
72    /// by value like any other part.
73    pub name_for_other: *mut mm_msg_part_t,
74    /// Prefix for the connection-established message (NULL for none).
75    pub hello_prefix: *const mm_msg_t,
76    /// Prefix for connection-failure messages (NULL for none).
77    pub failure_prefix: *const mm_msg_t,
78}
79
80/// Result code returned by the fallible `mm_*` functions (mirrors `mm_err_t`).
81#[repr(i32)]
82#[derive(Debug, Clone, Copy, PartialEq, Eq)]
83#[expect(
84    dead_code,
85    reason = "variants are produced by the C ABI across the FFI boundary and matched on, not constructed in Rust"
86)]
87pub enum mm_err_t {
88    /// Success.
89    Ok = 0,
90    /// No message available (`mm_poller_next`).
91    NoMsg = 1,
92    /// Parts buffer too small; `n_parts_out` holds the required size.
93    BufSz = -1,
94    /// Internal error; see `mm_last_error`.
95    Internal = -2,
96}
