9//! Safe Rust bindings for the minimonarch C ABI.
10//!
11//! This crate is the Rust analogue of the Python bindings (`minimonarch.pyi`):
12//! a thin, idiomatic wrapper over the stable C ABI declared in
13//! [`minimonarch.h`]. Every call goes through the C ABI (see [`ffi`]) — the
14//! bindings never reach into the internal Rust implementation — so the ABI
15//! contract, and long-term compatibility, is the same one C and Python callers
16//! depend on.
17//!
18//! # Shape
19//!
20//! Unlike Python, Rust has no ambient event loop, so the context and poller the
21//! Python API hides are explicit here:
22//!
23//! - [`Context`] owns the runtime (one background event loop). Handles are
24//!   plain owned values with no borrows tying them to it; dropping the context
25//!   tears the runtime down, and any later use of an [`Actor`] simply returns
26//!   an error rather than misbehaving.
27//! - [`Actor`] is an addressable endpoint: [`send`](Actor::send),
28//!   [`serve`](Actor::serve), [`join`](Actor::join), [`die`](Actor::die),
29//!   [`monitor`](Actor::monitor), and [`recv`](Actor::recv) to await this
30//!   actor's own next message (the analogue of Python's `actor.next()`).
31//! - [`Poller`] drives message delivery for *several* actors at once. Subscribe
32//!   actors to it, then drain with [`try_recv`](Poller::try_recv) (non-blocking)
33//!   or [`recv`](Poller::recv) (blocks on the wakeup fd), integrating with any
34//!   fd-based event loop via [`fd`](Poller::fd). [`Actor::recv`] is the
35//!   single-actor convenience built on top: it lazily creates and stores a
36//!   private poller the first time it is called.
37//! - [`Part`] is a multipart-message segment. It owns its bytes (like
38//!   `minimonarch.bytearray`) and can be *moved* into a message zero-copy —
39//!   including forwarding a received part without copying.
40//!
41//! # Example
42//!
43//! ```no_run
44//! use monarch_mini_rs::Context;
45//! use monarch_mini_rs::Part;
46//! use monarch_mini_rs::Role;
47//!
48//! # async fn example() -> Result<(), monarch_mini_rs::Error> {
49//! let ctx = Context::new()?;
50//! let actor = ctx.actor(Some(b"hello-actor"), /*gateway=*/ true)?;
51//!
52//! actor.send(b"hello-actor", vec![Part::copy_from(b"hello, self")])?;
53//! let parts = actor.recv().await?; // lazily creates this actor's own poller
54//! assert_eq!(parts[0].as_bytes(), b"hello, self");
55//! # Ok(())
56//! # }
57//! ```
58

77/// An error from the minimonarch runtime, carrying the C ABI's last-error
78/// string for the failing call.
79#[derive(Debug, thiserror::Error)]
80#[error("minimonarch: {0}")]
81pub struct Error(String);
82
83/// Result alias for the fallible bindings API.
84pub type Result<T> = std::result::Result<T, Error>;
85
86impl Error {
87    /// Capture the current thread's last-error string from the C ABI.
88    fn last() -> Self {
89        // SAFETY: `mm_last_error` returns a valid NUL-terminated pointer owned
90        // by the runtime, valid until the next `mm_*` call on this thread; we
91        // copy it out immediately.
92        let ptr = unsafe { ffi::mm_last_error() };
93        let msg = if ptr.is_null() {
94            String::new()
95        } else {
96            // SAFETY: `ptr` is a valid C string per the ABI contract above.
97            unsafe { std::ffi::CStr::from_ptr(ptr) }
98                .to_string_lossy()
99                .into_owned()
100        };
101        Error(msg)
102    }
103}
104
105/// Translate a `mm_err_t` from a fallible non-poller call into a `Result`.
106fn check(err: ffi::mm_err_t) -> Result<()> {
107    match err {
108        ffi::mm_err_t::Ok => Ok(()),
109        _ => Err(Error::last()),
110    }
111}

113// A few C calls (`mm_ctx_destroy`, `mm_actor_create`, `mm_poller_create`,
114// `mm_poller_subscribe`) block briefly for a round-trip to minimonarch's own
115// runtime thread. minimonarch is linked as an opaque native library (a static
116// archive with its own std/tokio; see build.rs), so that wait is invisible to
117// the caller's async runtime — there is no shared thread-local runtime context to
118// trip over. These calls therefore just block the calling thread, exactly like
119// any synchronous C function, on any runtime flavor and with no special
120// handling. (The same round-trip is what the Python bindings block the asyncio
121// loop thread for.)
122
123/// This actor's role in a serve/join pair. Exactly one side of a connection
124/// must be [`Role::Parent`].
125#[derive(Debug, Clone, Copy, PartialEq, Eq)]
126pub enum Role {
127    /// This actor becomes the parent of the remote actor.
128    Parent,
129    /// This actor becomes the child of the remote actor.
130    Child,
131}
132
133impl From<Role> for ffi::mm_role_t {
134    fn from(role: Role) -> Self {
135        match role {
136            Role::Parent => ffi::mm_role_t::Parent,
137            Role::Child => ffi::mm_role_t::Child,
138        }
139    }
140}

146/// A single part of a multipart message: an owned byte buffer.
147///
148/// A `Part` owns its storage through a C-style deleter, mirroring
149/// `minimonarch.bytearray`. It is either bytes allocated on the Rust side or a
150/// buffer adopted from a received message (possibly backed by a shared-memory
151/// slab). Either way, sending a `Part` *moves* it into the message zero-copy —
152/// so a received part can be forwarded onward without copying its bytes.
153///
154/// `Part` is intentionally neither `Send` nor `Sync`: parts are built and
155/// consumed inline with a [`send`](Actor::send) or drained from a [`Poller`],
156/// so they never need to cross a thread boundary on their own.
157pub struct Part {
158    data: *const c_void,
159    len: usize,
160    deleter: Option<unsafe extern "C" fn(*mut c_void)>,
161    deleter_ctx: *mut c_void,
162}
163
164/// Deleter for a `Part` backed by a Rust `Box<Vec<u8>>`.
165unsafe extern "C" fn drop_boxed_vec(ctx: *mut c_void) {
166    // SAFETY: `ctx` was produced by `Part::from_vec` via
167    // `Box::into_raw(Box<Vec<u8>>)`, installed as the sole deleter, so it runs
168    // exactly once here.
169    unsafe {
170        drop(Box::from_raw(ctx.cast::<Vec<u8>>()));
171    }
172}
173
174impl Part {
175    /// An empty part.
176    pub fn empty() -> Self {
177        Part {
178            data: std::ptr::null(),
179            len: 0,
180            deleter: None,
181            deleter_ctx: std::ptr::null_mut(),
182        }
183    }
184
185    /// Take ownership of `bytes`, moving it into a part with no copy of the
186    /// underlying buffer.
187    pub fn from_vec(bytes: Vec<u8>) -> Self {
188        if bytes.is_empty() {
189            return Part::empty();
190        }
191        let boxed = Box::new(bytes);
192        let data = boxed.as_ptr().cast();
193        let len = boxed.len();
194        Part {
195            data,
196            len,
197            deleter: Some(drop_boxed_vec),
198            deleter_ctx: Box::into_raw(boxed).cast(),
199        }
200    }
201
202    /// Copy `bytes` into a new owned part.
203    pub fn copy_from(bytes: &[u8]) -> Self {
204        Part::from_vec(bytes.to_vec())
205    }
206
207    /// The part's bytes.
208    pub fn as_bytes(&self) -> &[u8] {
209        if self.data.is_null() || self.len == 0 {
210            return &[];
211        }
212        // SAFETY: `data`/`len` describe a buffer this part owns for its whole
213        // lifetime (Rust-allocated or adopted from a received part); we only
214        // read it immutably.
215        unsafe { std::slice::from_raw_parts(self.data.cast::<u8>(), self.len) }
216    }
217
218    /// The part's length in bytes.
219    pub fn len(&self) -> usize {
220        self.len
221    }
222
223    /// Whether the part is empty.
224    pub fn is_empty(&self) -> bool {
225        self.len == 0
226    }
227
228    /// Copy the part's bytes into an owned `Vec`.
229    pub fn to_vec(&self) -> Vec<u8> {
230        self.as_bytes().to_vec()
231    }
232
233    /// Adopt a raw part handed back from the C ABI (a received message part),
234    /// taking over its buffer and deleter.
235    ///
236    /// # Safety
237    /// `raw` must be a part just produced by the runtime (e.g. via
238    /// `mm_poller_next`) whose ownership is being transferred exactly once.
239    unsafe fn from_raw(raw: mm_msg_part_t) -> Self {
240        Part {
241            data: raw.data,
242            len: raw.len,
243            deleter: raw.deleter,
244            deleter_ctx: raw.deleter_ctx,
245        }
246    }
247
248    /// Transfer ownership of the buffer to the C ABI, suppressing this part's
249    /// drop so the deleter runs once, on the runtime side.
250    fn into_raw(self) -> mm_msg_part_t {
251        let me = std::mem::ManuallyDrop::new(self);
252        mm_msg_part_t {
253            data: me.data,
254            len: me.len,
255            deleter: me.deleter,
256            deleter_ctx: me.deleter_ctx,
257        }
258    }
259}
260
261impl Drop for Part {
262    fn drop(&mut self) {
263        if let Some(deleter) = self.deleter {
264            // SAFETY: `deleter_ctx` was paired with `deleter` at construction
265            // and this is the sole owner, so the deleter runs exactly once.
266            unsafe { deleter(self.deleter_ctx) };
267        }
268    }
269}
270
271impl From<Vec<u8>> for Part {
272    fn from(bytes: Vec<u8>) -> Self {
273        Part::from_vec(bytes)
274    }
275}
276
277impl From<&[u8]> for Part {
278    fn from(bytes: &[u8]) -> Self {
279        Part::copy_from(bytes)
280    }
281}

283/// A borrowed multipart message: a temporary `Vec<mm_msg_part_t>` that keeps
284/// the parts' ownership until the C call consumes them. Dropping it *without*
285/// leaking runs every remaining part's deleter, so it is always leaked
286/// (`ManuallyDrop`) once handed to a C call that takes ownership.
287struct RawParts {
288    parts: Vec<mm_msg_part_t>,
289}
290
291impl RawParts {
292    /// Move `parts` into their raw C representation.
293    fn new(parts: Vec<Part>) -> Self {
294        RawParts {
295            parts: parts.into_iter().map(Part::into_raw).collect(),
296        }
297    }
298
299    /// Build owned copies of a slice of byte-slice prefixes.
300    fn copied(prefix: &[&[u8]]) -> Self {
301        RawParts::new(prefix.iter().map(|b| Part::copy_from(b)).collect())
302    }
303
304    fn as_msg(&mut self) -> mm_msg_t {
305        mm_msg_t {
306            parts: self.parts.as_mut_ptr(),
307            n_parts: self.parts.len(),
308        }
309    }
310}
311
312impl Drop for RawParts {
313    fn drop(&mut self) {
314        // Run each not-yet-consumed part's deleter by rehydrating it as a
315        // `Part`. Only reached if a C call was *not* made (e.g. an error before
316        // the send); a successful call transfers ownership and this drains an
317        // already-empty vec.
318        for raw in self.parts.drain(..) {
319            // SAFETY: each `raw` came from `Part::into_raw`, so reconstructing
320            // the owning `Part` restores the single-owner invariant.
321            drop(unsafe { Part::from_raw(raw) });
322        }
323    }
324}

330/// The minimonarch runtime for a process: a single background event loop plus
331/// the actors and pollers created from it.
332///
333/// A `Context` is a handle to that event loop; every operation is dispatched to
334/// it over an internal channel, so the handle is safe to move and share across
335/// threads.
336pub struct Context {
337    ptr: *mut ffi::mm_ctx,
338}
339
340// SAFETY: a `Context` handle only ever forwards commands to the runtime over an
341// internal `Send + Sync` channel and blocks on a reply; it holds no
342// thread-affine state, so it is safe to move between threads (`Send`) and to
343// share (`Sync`) — concurrent creates each use their own reply channel.
344unsafe impl Send for Context {}
345unsafe impl Sync for Context {}
346
347impl Context {
348    /// Create a new context and its background event loop.
349    pub fn new() -> Result<Self> {
350        let mut ptr: *mut ffi::mm_ctx = std::ptr::null_mut();
351        // SAFETY: `out` points to a valid slot we own; on `Ok` it is populated
352        // with a context pointer we take ownership of.
353        check(unsafe { ffi::mm_ctx_create(&mut ptr) })?;
354        Ok(Context { ptr })
355    }
356
357    /// Create an actor bound to this context.
358    ///
359    /// `ident` must be unique across the whole run; if `None`, a name must be
360    /// assigned by the peer in a later `serve`/`join`. `gateway` declares the
361    /// actor as the entry point for its process group (no parent or a network
362    /// parent); it is fixed at creation.
363    pub fn actor(&self, ident: Option<&[u8]>, gateway: bool) -> Result<Actor> {
364        // Keep the ident part alive across the call; the runtime takes ownership
365        // of it (like every other part) when a name is supplied.
366        let mut ident_raw = ident.map(|b| Part::copy_from(b).into_raw());
367        let ident_ptr = ident_raw
368            .as_mut()
369            .map_or(std::ptr::null_mut(), |p| p as *mut mm_msg_part_t);
370
371        let mut ptr: *mut ffi::mm_actor = std::ptr::null_mut();
372        // SAFETY: `self.ptr` is a live context; `ident_ptr` is either null or a
373        // valid part whose ownership transfers to the runtime; `out` is ours.
374        let err = unsafe { ffi::mm_actor_create(self.ptr, ident_ptr, gateway, &mut ptr) };
375        check(err)?;
376        Ok(Actor {
377            ptr,
378            ctx: self.ptr,
379            poller: tokio::sync::Mutex::new(None),
380        })
381    }
382
383    /// Create a [`Poller`] bound to this context.
384    pub fn poller(&self) -> Result<Poller> {
385        let mut ptr: *mut ffi::mm_poller = std::ptr::null_mut();
386        let mut fd: RawFd = -1;
387        // SAFETY: `self.ptr` is a live context; `fd_out` and `out` point to slots
388        // we own and are populated on success.
389        let err = unsafe { ffi::mm_poller_create(self.ptr, &mut fd, &mut ptr) };
390        check(err)?;
391        Ok(Poller {
392            ptr,
393            fd,
394            async_fd: None,
395        })
396    }
397}
398
399impl Drop for Context {
400    fn drop(&mut self) {
401        // SAFETY: `self.ptr` is a live context we own; this consumes it,
402        // flushing pending messages and joining minimonarch's runtime thread.
403        // Actors/pollers are self-contained C handles that stay safe to drop
404        // afterwards (they each hold their own runtime sender), so no
405        // drop-ordering relationship with them is required.
406        unsafe { ffi::mm_ctx_destroy(self.ptr) };
407    }
408}

411// Actor

414/// An addressable messaging endpoint, created from a [`Context`].
415///
416/// An `Actor` is a plain owned handle: it may outlive its [`Context`], and any
417/// method called after the context is dropped returns an error rather than
418/// misbehaving.
419pub struct Actor {
420    ptr: *mut ffi::mm_actor,
421    // The context this actor was created from, kept so [`Actor::recv`] can lazily
422    // build its own [`Poller`]. A raw pointer (not a `Context` handle) so an
423    // actor never keeps the runtime alive; if the context is gone, poller
424    // creation simply errors.
425    ctx: *mut ffi::mm_ctx,
426    // A per-actor poller, created on the first call to [`Actor::recv`] and reused
427    // thereafter, so `actor.recv().await` works without the caller ever managing
428    // a [`Poller`] — the Rust analogue of the Python bindings' `actor.next()`.
429    // Behind a `tokio::sync::Mutex` (whose guard is held across the `await`) so
430    // the actor stays `Send + Sync` even though a `Poller` is `!Sync`.
431    poller: tokio::sync::Mutex<Option<Poller>>,
432}
433
434// SAFETY: every `Actor` method forwards a command to the runtime over its
435// internal `Send + Sync` channel (monitor-id allocation uses an atomic), so an
436// `Actor` is safe to move between threads (`Send`) and to call concurrently from
437// several threads (`Sync`) — the classic "send from many threads" pattern. The
438// raw pointers are just runtime handles; the lazily-created per-actor poller is
439// guarded by a `tokio::sync::Mutex` (itself `Send + Sync` since `Poller: Send`),
440// which serializes the otherwise-`!Sync` poller's use.
441unsafe impl Send for Actor {}
442unsafe impl Sync for Actor {}
443
444impl Actor {
445    /// Send a multipart message to `receiver`. Each [`Part`] is moved into the
446    /// message; `receiver` is copied.
447    pub fn send(&self, receiver: &[u8], parts: Vec<Part>) -> Result<()> {
448        let mut raw = RawParts::new(parts);
449        let msg = raw.as_msg();
450        let receiver_raw = Part::copy_from(receiver).into_raw();
451        // SAFETY: `self.ptr` is a live actor. `receiver_raw` and every part in
452        // `msg` are valid owned parts whose ownership transfers to the runtime
453        // (which always consumes them, even on error), so we leak `raw` to
454        // avoid double-freeing.
455        let err = unsafe { ffi::mm_actor_send(self.ptr, receiver_raw, &msg) };
456        std::mem::forget(raw);
457        check(err)
458    }
459
460    /// Serve (listen) on `url`, taking `role` in the resulting pair.
461    ///
462    /// On success this actor is later delivered `[hello..., self, other]`; on
463    /// failure `[failure..., other, reason]`. `name` optionally assigns a name
464    /// to the peer. See [`Actor::join`] for the joining side.
465    pub fn serve(
466        &self,
467        url: &str,
468        role: Role,
469        name: Option<&[u8]>,
470        hello: &[&[u8]],
471        failure: &[&[u8]],
472    ) -> Result<()> {
473        self.connect(url, role, name, hello, failure, ffi::mm_actor_serve)
474    }
475
476    /// Join (connect to) `url`, taking `role` in the resulting pair. See
477    /// [`Actor::serve`] for the semantics of the arguments.
478    pub fn join(
479        &self,
480        url: &str,
481        role: Role,
482        name: Option<&[u8]>,
483        hello: &[&[u8]],
484        failure: &[&[u8]],
485    ) -> Result<()> {
486        self.connect(url, role, name, hello, failure, ffi::mm_actor_join)
487    }
488
489    /// Shared implementation of `serve`/`join`, which take identical arguments.
490    fn connect(
491        &self,
492        url: &str,
493        role: Role,
494        name: Option<&[u8]>,
495        hello: &[&[u8]],
496        failure: &[&[u8]],
497        // `serve` and `join` have the same signature; pick one.
498        call: unsafe extern "C" fn(
499            *mut ffi::mm_actor,
500            *const std::ffi::c_char,
501            *const ffi::mm_connect_args_t,
502        ) -> ffi::mm_err_t,
503    ) -> Result<()> {
504        let curl = CString::new(url).map_err(|e| Error(e.to_string()))?;
505
506        let mut name_raw = name.map(|b| Part::copy_from(b).into_raw());
507        let name_ptr = name_raw
508            .as_mut()
509            .map_or(std::ptr::null_mut(), |p| p as *mut mm_msg_part_t);
510
511        let mut hello_raw = RawParts::copied(hello);
512        let mut failure_raw = RawParts::copied(failure);
513        let hello_msg = hello_raw.as_msg();
514        let failure_msg = failure_raw.as_msg();
515
516        let args = ffi::mm_connect_args_t {
517            role: role.into(),
518            name_for_other: name_ptr,
519            hello_prefix: &hello_msg,
520            failure_prefix: &failure_msg,
521        };
522        // SAFETY: `self.ptr` is a live actor; `curl` is a valid NUL-terminated
523        // string; `args` and the parts it references are valid for the call,
524        // and the runtime takes ownership of the (owned) parts.
525        let err = unsafe { call(self.ptr, curl.as_ptr(), &args) };
526        std::mem::forget(hello_raw);
527        std::mem::forget(failure_raw);
528        check(err)
529    }
530
531    /// Signal that this actor is dead to its parent, children, and monitors.
532    /// `reason` is a UTF-8 string explaining why.
533    pub fn die(&self, reason: &[u8]) {
534        let reason_raw = Part::copy_from(reason).into_raw();
535        // SAFETY: `self.ptr` is a live actor; `reason_raw` is a valid owned part
536        // whose ownership transfers to the runtime.
537        unsafe { ffi::mm_actor_die(self.ptr, reason_raw) };
538    }
539
540    /// Monitor `ident`. If it dies (or is already dead), this actor is sent
541    /// `[failure..., ident, b"actor died"]`.
542    ///
543    /// If `timeout_for_nonexistence` is non-zero and `ident` is still not known
544    /// anywhere after that many milliseconds, the monitor fires once with
545    /// reason `b"actor does not exist"` and is consumed. `0` disables the
546    /// timeout. Only the first monitor on a given target arms a timeout.
547    pub fn monitor(
548        &self,
549        ident: &[u8],
550        failure: &[&[u8]],
551        timeout_for_nonexistence: u64,
552    ) -> Result<MonitorHandle> {
553        let ident_raw = Part::copy_from(ident).into_raw();
554        let mut failure_raw = RawParts::copied(failure);
555        let failure_msg = failure_raw.as_msg();
556
557        let mut handle: ffi::mm_monitor_handle_t = 0;
558        // SAFETY: `self.ptr` is a live actor; `ident_raw` and the failure parts
559        // are valid owned parts transferred to the runtime; `out` is ours.
560        let err = unsafe {
561            ffi::mm_actor_monitor(
562                self.ptr,
563                ident_raw,
564                &failure_msg,
565                timeout_for_nonexistence,
566                &mut handle,
567            )
568        };
569        std::mem::forget(failure_raw);
570        check(err)?;
571        Ok(MonitorHandle { handle })
572    }
573
574    /// Cancel `monitor`, which must have been created by this actor. Its failure
575    /// message will no longer be delivered, and any already-queued failure is dropped
576    /// before delivery. Consumes the handle; dropping it instead leaves the monitor
577    /// active.
578    pub fn cancel_monitor(&self, monitor: MonitorHandle) {
579        // SAFETY: `self.ptr` is a live actor, and `monitor.handle` is an opaque monitor
580        // id consumed by this call. The cancel is addressed by (actor, id) and frees
581        // nothing.
582        unsafe { ffi::mm_monitor_handle_cancel(self.ptr, monitor.handle) };
583    }
584
585    /// Await this actor's next delivered message — the Rust analogue of the
586    /// Python bindings' awaitable `actor.next()`.
587    ///
588    /// On the first call, the actor lazily creates a private [`Poller`],
589    /// subscribes itself to it, and stores it on the handle; subsequent calls
590    /// reuse it. This is the convenience path for the common "just read this
591    /// actor's messages" case, so the caller never has to manage a [`Poller`]
592    /// or actor indices. Must be called from within a Tokio runtime.
593    ///
594    /// Do not also subscribe this actor to a separate [`Poller`]: the C ABI
595    /// allows an actor on at most one poller at a time, so mixing the two paths
596    /// makes this call (or the manual `subscribe`) fail. For fan-in over many
597    /// actors, use an explicit [`Poller`] instead of per-actor `recv`.
598    pub async fn recv(&self) -> Result<Vec<Part>> {
599        let mut guard = self.poller.lock().await;
600        if guard.is_none() {
601            *guard = Some(self.make_poller()?);
602        }
603        // The guard is held across the await so the `!Sync` poller is used by
604        // one task at a time; concurrent `recv`s on the same actor serialize.
605        let poller = guard.as_mut().expect("poller just created");
606        let (_index, parts) = poller.recv().await?;
607        Ok(parts)
608    }
609
610    /// Build the actor's private poller and subscribe itself at index 0.
611    fn make_poller(&self) -> Result<Poller> {
612        let mut ptr: *mut ffi::mm_poller = std::ptr::null_mut();
613        let mut fd: RawFd = -1;
614        // SAFETY: `self.ctx` is the context this actor was created from; `fd_out`
615        // and `out` point to slots we own and are populated on success. (If the
616        // context has been dropped, this returns an error instead.)
617        check(unsafe { ffi::mm_poller_create(self.ctx, &mut fd, &mut ptr) })?;
618        let poller = Poller {
619            ptr,
620            fd,
621            async_fd: None,
622        };
623        poller.subscribe(0, self)?;
624        Ok(poller)
625    }
626
627    /// The raw actor pointer, for [`Poller::subscribe`].
628    fn as_ptr(&self) -> *mut ffi::mm_actor {
629        self.ptr
630    }
631}
632
633impl Drop for Actor {
634    fn drop(&mut self) {
635        // Drop the lazily-created per-actor poller (if any) first: it holds this
636        // actor subscribed and owns a wakeup fd, so it must be torn down before
637        // the actor. `get_mut` needs no lock — we have exclusive access here.
638        self.poller.get_mut().take();
639        // SAFETY: `self.ptr` is a live actor we own; this consumes it.
640        unsafe { ffi::mm_actor_destroy(self.ptr) };
641    }
642}

646// ---------------------------------------------------------------------------
647
648/// Handle to a registered monitor, analogous to a [`tokio::task::JoinHandle`]:
649/// **dropping it does not cancel the monitor** — it detaches, and the monitor
650/// keeps running (it will still fire on the target's death/absence). Keep the
651/// handle only if you may want to pass it to [`Actor::cancel_monitor`] later.
652///
653/// (Consistent with the Python bindings, whose handle also does not cancel on
654/// GC.) The handle owns nothing — it is just the monitor's id — so dropping it
655/// is free and leaks nothing.
656pub struct MonitorHandle {
657    handle: ffi::mm_monitor_handle_t,
658}

660// No `Drop` impl: dropping a `MonitorHandle` detaches (JoinHandle semantics) and
661// must not cancel the monitor.

663// ---------------------------------------------------------------------------
664// Poller
665// ---------------------------------------------------------------------------
666
667/// Drives message delivery for a set of subscribed actors, integrating with a
668/// Tokio runtime.
669///
670/// [`recv`](Poller::recv) is `async`: it awaits the poller's wakeup fd on the
671/// caller's Tokio reactor (via [`AsyncFd`]), draining delivered messages — the
672/// same fd-reader model the Python bindings install on the asyncio loop.
673///
674/// Unlike [`Context`] and [`Actor`], a `Poller` is `Send` but **not** `Sync`:
675/// the C ABI requires that calls on a single poller be externally serialized.
676/// It can be owned by, or moved to, one task/thread that drives it, but it
677/// cannot be shared for concurrent use. `!Sync` (from the raw pointer) enforces
678/// this; draining still goes through `&mut self`.
679pub struct Poller {
680    ptr: *mut ffi::mm_poller,
681    fd: RawFd,
682    // The wakeup fd registered with the Tokio reactor, created lazily on the
683    // first `recv` (registration must happen inside a runtime).
684    async_fd: Option<AsyncFd<WakeupFd>>,
685}
686
687// SAFETY: a `Poller` owns only a runtime handle, a `Send` receiver, and a
688// non-thread-affine wakeup fd, so it is sound to move to another thread
689// (`Send`). It is deliberately left `!Sync` (the `*mut` field withholds the
690// auto-impl): the C ABI requires calls on one poller to be externally
691// serialized, so it must not be shared for concurrent access.
692unsafe impl Send for Poller {}
693
694/// Borrows the poller's wakeup fd for Tokio registration *without owning it* —
695/// the fd belongs to the poller (the C ABI says not to close it), and dropping
696/// a `RawFd` does nothing, so the `AsyncFd` only registers/deregisters it.
697struct WakeupFd(RawFd);
698
699impl AsRawFd for WakeupFd {
700    fn as_raw_fd(&self) -> RawFd {
701        self.0
702    }
703}
704
705impl Poller {
706    /// The wakeup file descriptor. It becomes readable when more delivered
707    /// messages may exist; suitable for `select`/`poll`/`epoll`. Owned by the
708    /// poller — do not close it.
709    pub fn fd(&self) -> RawFd {
710        self.fd
711    }
712
713    /// Watch `actor` for incoming messages, associating it with `index`. An
714    /// actor may be subscribed to at most one poller at a time.
715    pub fn subscribe(&self, index: usize, actor: &Actor) -> Result<()> {
716        // SAFETY: `self.ptr` and `actor` are live handles into the same runtime.
717        check(unsafe { ffi::mm_poller_subscribe(self.ptr, index, actor.as_ptr()) })
718    }
719
720    /// Stop watching the actor previously assigned `index`.
721    pub fn unsubscribe(&self, index: usize) {
722        // SAFETY: `self.ptr` is a live poller we own.
723        unsafe { ffi::mm_poller_unsubscribe(self.ptr, index) };
724    }
725
726    /// Read the next delivered message without blocking, returning
727    /// `Ok(Some((index, parts)))` for a message, or `Ok(None)` if none is
728    /// available. Arms the wakeup [`fd`](Poller::fd) when it returns `None`.
729    pub fn try_recv(&mut self) -> Result<Option<(usize, Vec<Part>)>> {
730        // Grow the parts buffer on demand: `mm_poller_next` reports the required
731        // size via `n_parts_out` on `BufSz` without consuming the message.
732        let mut cap = 8usize;
733        loop {
734            let mut index_out: usize = 0;
735            let mut n_parts_out: usize = 0;
736            let mut buf: Vec<mm_msg_part_t> = Vec::with_capacity(cap);
737            // SAFETY: `self.ptr` is a live poller; `buf` has `cap` writable
738            // slots; the out-params point to locals we own. On `Ok` the runtime
739            // writes `n_parts_out` parts whose ownership transfers to us.
740            let err = unsafe {
741                ffi::mm_poller_next(
742                    self.ptr,
743                    &mut index_out,
744                    buf.as_mut_ptr(),
745                    cap,
746                    &mut n_parts_out,
747                )
748            };
749            match err {
750                ffi::mm_err_t::Ok => {
751                    // SAFETY: the runtime initialized `n_parts_out` parts.
752                    unsafe { buf.set_len(n_parts_out) };
753                    let parts = buf
754                        .into_iter()
755                        // SAFETY: each part was just produced by the runtime and
756                        // its ownership transferred to us exactly once.
757                        .map(|raw| unsafe { Part::from_raw(raw) })
758                        .collect();
759                    return Ok(Some((index_out, parts)));
760                }
761                ffi::mm_err_t::NoMsg => return Ok(None),
762                ffi::mm_err_t::BufSz => {
763                    // Nothing was written or consumed; retry with a big-enough
764                    // buffer.
765                    cap = n_parts_out;
766                    continue;
767                }
768                ffi::mm_err_t::Internal => return Err(Error::last()),
769            }
770        }
771    }
772
773    /// Await the next delivered message — the Rust analogue of the Python API's
774    /// awaitable `next()`.
775    ///
776    /// Drains [`try_recv`](Poller::try_recv); when empty, awaits the poller's
777    /// wakeup fd on the current Tokio reactor and retries. Must be called from
778    /// within a Tokio runtime (the fd is registered lazily on first use).
779    pub async fn recv(&mut self) -> Result<(usize, Vec<Part>)> {
780        loop {
781            if let Some(message) = self.try_recv()? {
782                return Ok(message);
783            }
784            // `try_recv` drained the wakeup fd and armed the poller; await the
785            // runtime signalling it (an enqueued message writes the eventfd).
786            let async_fd = self.ensure_async_fd()?;
787            let mut guard = async_fd
788                .readable()
789                .await
790                .map_err(|e| Error(format!("await poller wakeup fd: {e}")))?;
791            // Readiness may be stale — `try_recv` drains the fd itself, behind
792            // the reactor's back — so clear it and re-check the channel; a real
793            // message always re-signals via a fresh edge.
794            guard.clear_ready();
795        }
796    }
797
798    /// Register the wakeup fd with the current Tokio reactor on first use.
799    fn ensure_async_fd(&mut self) -> Result<&AsyncFd<WakeupFd>> {
800        match self.async_fd {
801            Some(ref async_fd) => Ok(async_fd),
802            None => {
803                let async_fd = AsyncFd::new(WakeupFd(self.fd))
804                    .map_err(|e| Error(format!("register poller wakeup fd with tokio: {e}")))?;
805                Ok(self.async_fd.insert(async_fd))
806            }
807        }
808    }
809}
810
811impl Drop for Poller {
812    fn drop(&mut self) {
813        // Deregister the wakeup fd from the Tokio reactor *before* the C side
814        // closes it (mm_poller_destroy drops the poller's owned fd). Dropping
815        // the `AsyncFd` only removes it from the reactor; it does not close the
816        // borrowed fd.
817        self.async_fd = None;
818        // SAFETY: `self.ptr` is a live poller we own; this consumes it. The fd
819        // is owned by the poller, so we do not close it.
820        unsafe { ffi::mm_poller_destroy(self.ptr) };
821    }
822}
