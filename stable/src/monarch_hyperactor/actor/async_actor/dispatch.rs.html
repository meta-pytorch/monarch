9//! Event-loop-backed execution state for an async [`PythonActor`].
10
11use super::super::*;
12use super::inbox::Receiver;
13
14/// Async actor state retained between construction, initialization, and cleanup.
15#[derive(Debug)]
16pub(in super::super) struct State {
17    /// Taken once by `init` and split between the two loop tasks.
18    inbox_receiver: Option<Receiver>,
19    /// The task-local state for the actor's dedicated event loop thread.
20    task_locals: pyo3_async_runtimes::TaskLocals,
21}
22
23impl State {
24    pub(in super::super) fn new(py: Python<'_>, inbox_receiver: Receiver) -> Self {
25        Self {
26            inbox_receiver: Some(inbox_receiver),
27            task_locals: Python::detach(py, create_task_locals),
28        }
29    }
30
31    /// Start the message and callback tasks on the actor's event loop.
32    pub(in super::super) fn init(
33        &mut self,
34        py: Python<'_>,
35        actor_mesh_mod: &Bound<'_, PyModule>,
36        actor: &Py<PyAny>,
37        self_instance: &Py<PyInstance>,
38        callbacks_pending: Arc<AtomicUsize>,
39    ) -> anyhow::Result<()> {
40        let Receiver { messages, control } = self
41            .inbox_receiver
42            .take()
43            .expect("async inbox receiver already taken");
44        let loops = [
45            (
46                "message loop",
47                actor_mesh_mod.call_method(
48                    "_dispatch_loop",
49                    (
50                        actor.clone_ref(py),
51                        messages,
52                        self_instance.clone_ref(py),
53                        CallbacksPending(callbacks_pending),
54                    ),
55                    None,
56                )?,
57            ),
58            (
59                "callback loop",
60                actor_mesh_mod.call_method(
61                    "_callback_loop",
62                    (actor.clone_ref(py), control, self_instance.clone_ref(py)),
63                    None,
64                )?,
65            ),
66        ];
67        for (name, awaitable) in loops {
68            let future =
69                pyo3_async_runtimes::into_future_with_locals(&self.task_locals, awaitable)?;
70            tokio::spawn(async move {
71                if let Err(error) = future.await {
72                    tracing::error!("{} error: {}", name, error);
73                }
74            });
75        }
76        Ok(())
77    }
78
79    /// Run user cleanup on the actor's event loop, then cancel its remaining
80    /// tasks and stop the loop. Loop shutdown still runs when user cleanup
81    /// fails; the cleanup failure takes precedence when both fail.
82    pub(in super::super) async fn cleanup(
83        &self,
84        actor: &Py<PyAny>,
85        instance: Option<&Py<PyInstance>>,
86        cx: &Context<'_, PythonActor>,
87        this: &Instance<PythonActor>,
88        error: Option<String>,
89    ) -> anyhow::Result<()> {
90        let future = monarch_with_gil(GilSite::EndpointCleanup, |py| {
91            let py_cx = cleanup_context(py, cx, this, instance)?;
92            let actor = actor.bind(py);
93            // Some tests don't use the Actor base class, so add this check
94            // to be defensive.
95            match actor.hasattr("__cleanup__") {
96                Ok(false) | Err(_) => return Ok(None),
97                _ => {}
98            }
99            let awaitable = actor
100                .call_method("__cleanup__", (py_cx.bind(py), error), None)
101                .map_err(|error| anyhow::Error::from(SerializablePyErr::from(py, &error)))?;
102            if awaitable.is_none() {
103                Ok(None)
104            } else {
105                pyo3_async_runtimes::into_future_with_locals(&self.task_locals, awaitable)
106                    .map(Some)
107                    .map_err(anyhow::Error::from)
108            }
109        })
110        .await;
111        let cleanup_result = match future {
112            Ok(Some(future)) => future.await.map(|_| ()).map_err(anyhow::Error::from),
113            Ok(None) => Ok(()),
114            Err(error) => Err(error),
115        };
116        let loop_shutdown_result = shutdown_loop(&self.task_locals);
117        cleanup_result?;
118        loop_shutdown_result
119    }
120}

122/// Construct the context for async cleanup, preserving the inherited fallback
123/// for cleanup after partial initialization.
124fn cleanup_context(
125    py: Python<'_>,
126    cx: &Context<'_, PythonActor>,
127    this: &Instance<PythonActor>,
128    instance: Option<&Py<PyInstance>>,
129) -> anyhow::Result<Py<PyContext>> {
130    let instance = match instance {
131        Some(instance) => instance.clone_ref(py),
132        None => {
133            // Async cleanup may run after initialization failed before a
134            // cached PyInstance was installed. Derive one from the Rust actor
135            // instance so `__cleanup__` still receives a Context.
136            let instance: PyInstance = this.into();
137            Py::new(py, instance)?
138        }
139    };
140    Ok(Py::new(py, PyContext::new(cx, instance))?)
141}

143/// Cancel every task on the actor's event loop, wait for the loop to observe
144/// those cancellations, then ask it to stop.
145fn shutdown_loop_with_gil(
146    py: Python<'_>,
147    task_locals: &pyo3_async_runtimes::TaskLocals,
148) -> PyResult<()> {
149    let asyncio = py.import("asyncio")?;
150    let event_loop = task_locals.event_loop(py);
151    let tasks = asyncio.call_method1("all_tasks", (&event_loop,))?;
152    let mut has_tasks = false;
153    for task in tasks.try_iter()? {
154        let task = task?;
155        let cancel = task.getattr("cancel")?;
156        // The loop thread, not this Rust caller, must invoke `Task.cancel`.
157        event_loop.call_method1("call_soon_threadsafe", (cancel,))?;
158        has_tasks = true;
159    }
160    if has_tasks {
161        // This zero-delay coroutine is queued behind the cancellation
162        // callbacks. Waiting for its result gives the loop a turn to deliver
163        // cancellation before it is stopped.
164        asyncio
165            .call_method1(
166                "run_coroutine_threadsafe",
167                (asyncio.call_method1("sleep", (0,))?, &event_loop),
168            )?
169            .call_method0("result")?;
170    }
171    // `stop` must also run on the loop's owning thread.
172    let stop = event_loop.getattr("stop")?;
173    event_loop.call_method1("call_soon_threadsafe", (stop,))?;
174    Ok(())
175}

177/// Acquire the GIL for loop shutdown and translate a Python failure into the
178/// error returned by Rust actor cleanup.
179fn shutdown_loop(task_locals: &pyo3_async_runtimes::TaskLocals) -> anyhow::Result<()> {
180    monarch_with_gil_blocking(GilSite::Stop, |py| -> anyhow::Result<()> {
181        shutdown_loop_with_gil(py, task_locals)
182            .map_err(|error| anyhow::Error::from(SerializablePyErr::from(py, &error)))?;
183        Ok(())
184    })
185}

187/// Create TaskLocals backed by a new event loop on a dedicated Python thread.
188fn create_task_locals() -> pyo3_async_runtimes::TaskLocals {
189    monarch_with_gil_blocking(GilSite::TaskLocals, |py| {
190        let asyncio = Python::import(py, "asyncio").unwrap();
191        let event_loop = asyncio.call_method0("new_event_loop").unwrap();
192        let task_locals = pyo3_async_runtimes::TaskLocals::new(event_loop.clone())
193            .copy_context(py)
194            .unwrap();
195
196        let kwargs = PyDict::new(py);
197        let target = event_loop.getattr("run_forever").unwrap();
198        kwargs.set_item("target", target).unwrap();
199        // Need to make this a daemon thread, otherwise shutdown will hang.
200        kwargs.set_item("daemon", true).unwrap();
201        kwargs
202            .set_item("name", "monarch-actor-event-loop")
203            .expect("thread name should be accepted");
204        let thread = py
205            .import("threading")
206            .unwrap()
207            .call_method("Thread", (), Some(&kwargs))
208            .unwrap();
209        mark_actor_event_loop_thread(&thread, &event_loop)
210            .expect("actor event loop should attach to its owning thread");
211        thread.call_method0("start").unwrap();
212        task_locals
213    })
