127    /// Reserve `count` ordering seqs against the receiver actor's
128    /// `PythonMessage` handler port. Subsequent fire-and-forget
129    /// endpoint sends to `receiver` pick up at the post-reservation
130    /// seq, creating a deterministic gap visible through the
131    /// receiver's `OrderedSender::snapshot` and
132    /// `/v1/{receiver}.inbound_ordering`.
133    ///
134    /// Test/demo only. Underscore-prefixed; production code must not
135    /// use this. The port computation matches the one used by normal
136    /// Python `.broadcast()` / `.call_one()` sends, so the gap is
137    /// observable without any extra plumbing.
138    #[pyo3(name = "_debug_skip_next_ordering_seq")]
139    fn debug_skip_next_ordering_seq(&self, receiver: &PyActorAddr, count: u64) {
140        use crate::actor::PythonMessage;
141        let port_addr = receiver
142            .inner
143            .port_addr(hyperactor::Port::handler::<PythonMessage>());
144        self.inner.debug_skip_next_ordering_seq(&port_addr, count);
145    }
146
147    /// Producer write-side for the mesh `execution` field: `_Actor.handle`
148    /// calls these around the real user-method invocation. `_execution_start`
149    /// returns the in-flight token; `_execution_finish` ends it. When this
150    /// instance has no tracker (root-client / `From`-built), start returns
151    /// the `0` no-op sentinel and finish ignores it (PE-4).
152    fn _execution_start(&self, name: String) -> u64 {
153        match &self.execution_tracker {
154            Some(tracker) => tracker.start(name),
155            None => 0,
156        }
157    }
158
159    fn _execution_finish(&self, token: u64) {
160        if let Some(tracker) = &self.execution_tracker {
161            tracker.finish(token);
162        }
163    }
164}

166/// Stop an actor, wait for terminal status, and flush its proc's gateway.
167///
168/// The flush needs the proc's transports to still be serving, so host
169/// shutdown calls this for the root client after child drain and before
170/// transport teardown.
171pub(crate) async fn stop_instance_and_wait(
172    instance: &Instance<PythonActor>,
173    reason: String,
174    flush_timeout: Duration,
175) {
176    let actor_id = instance.self_addr().clone();
177    let proc = instance.proc().clone();
178    let mut status = instance.status();
179    if let Err(error) = instance.stop(&reason) {
180        tracing::warn!(%actor_id, %error, "stop_and_wait: stop failed");
181    } else {
182        let _ = status.wait_for(|s| s.is_terminal()).await;
183    }
184    match tokio::time::timeout(flush_timeout, proc.flush()).await {
185        Ok(Err(error)) => {
186            tracing::warn!(%actor_id, %error, "stop_and_wait: flush failed");
187        }
188        Err(_elapsed) => {
189            tracing::warn!(%actor_id, "stop_and_wait: flush timed out");
190        }
191        Ok(Ok(())) => {}
192    }
193}

224#[pyclass(name = "Context", module = "monarch._src.actor.actor_mesh")]
225pub struct PyContext {
226    instance: Py<PyInstance>,
227    rank: Point,
228    /// Cloneable handle to a span carrying the actor's recording key.
229    /// When entered, events emitted under this span are captured by
230    /// the per-actor flight recorder. `None` for bootstrap/client
231    /// contexts that are not actor handler execution paths.
232    recording_span: Option<tracing::Span>,
233}
234
235#[pymethods]
236impl PyContext {
237    #[getter]
238    fn actor_instance(&self) -> &Py<PyInstance> {
239        &self.instance
240    }
241
242    #[getter]
243    fn message_rank(&self) -> PyPoint {
244        self.rank.clone().into()
245    }
246
247    #[staticmethod]
248    fn _root_client_context(py: Python<'_>) -> PyResult<PyContext> {
249        let _guard = runtime::get_tokio_runtime().enter();
250        let instance: PyInstance = root_client_actor(py).into();
251        Ok(PyContext {
252            instance: instance.into_pyobject(py)?.into(),
253            rank: Extent::unity().point_of_rank(0).unwrap(),
254            recording_span: None,
255        })
256    }
257
258    /// Create a context from an existing instance.
259    /// This is used when the root client was bootstrapped via bootstrap_host()
260    /// instead of the default bootstrap_client().
261    #[staticmethod]
262    fn _from_instance(py: Python<'_>, instance: PyInstance) -> PyResult<PyContext> {
263        Ok(PyContext {
264            instance: instance.into_pyobject(py)?.into(),
265            rank: Extent::unity().point_of_rank(0).unwrap(),
266            recording_span: None,
267        })
268    }
269}

271impl PyContext {
272    pub(crate) fn new<T: hyperactor::actor::Actor>(
273        cx: &hyperactor::Context<T>,
274        instance: Py<PyInstance>,
275    ) -> PyContext {
276        Self::from_parts(instance, cx.cast_point(), Some(cx.recording_span()))
277    }
278
279    pub(crate) fn from_parts(
280        instance: Py<PyInstance>,
281        rank: Point,
282        recording_span: Option<tracing::Span>,
283    ) -> PyContext {
284        PyContext {
285            instance,
286            rank,
287            recording_span,
288        }
289    }
290
291    /// The actor's recording span, if this context is an actor handler
292    /// execution path. Used by `forward_to_tracing` to enter the
293    /// recording scope on the asyncio thread so that log events are
294    /// captured in the flight recorder.
295    pub(crate) fn recording_span(&self) -> Option<&tracing::Span> {
296        self.recording_span.as_ref()
297    }
298
299    /// Test-only: build a PyContext with a chosen recording span.
300    /// Uses the root client actor for the instance field (the test
301    /// only exercises the recording_span extraction path).
302    #[doc(hidden)]
303    pub fn for_test(py: Python<'_>, recording_span: Option<tracing::Span>) -> PyResult<PyContext> {
304        let mut ctx = Self::_root_client_context(py)?;
305        ctx.recording_span = recording_span;
306        Ok(ctx)
307    }
308}
