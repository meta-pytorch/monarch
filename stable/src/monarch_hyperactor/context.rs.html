129    /// Reserve `count` ordering seqs against the receiver actor's
130    /// `PythonMessage` handler port. Subsequent fire-and-forget
131    /// endpoint sends to `receiver` pick up at the post-reservation
132    /// seq, creating a deterministic gap visible through the
133    /// receiver's `OrderedSender::snapshot` and
134    /// `/v1/{receiver}.inbound_ordering`.
135    ///
136    /// Test/demo only. Underscore-prefixed; production code must not
137    /// use this. The port computation matches the one used by normal
138    /// Python `.broadcast()` / `.call_one()` sends, so the gap is
139    /// observable without any extra plumbing.
140    #[pyo3(name = "_debug_skip_next_ordering_seq")]
141    fn debug_skip_next_ordering_seq(&self, receiver: &PyActorAddr, count: u64) {
142        use crate::actor::PythonMessage;
143        let port_addr = receiver
144            .inner
145            .port_addr(hyperactor::Port::handler::<PythonMessage>());
146        self.inner.debug_skip_next_ordering_seq(&port_addr, count);
147    }
148
149    /// Producer write-side for the mesh `execution` field: `_Actor.handle`
150    /// calls these around the real user-method invocation. `_execution_start`
151    /// returns the in-flight token; `_execution_finish` ends it. When this
152    /// instance has no tracker (root-client / `From`-built), start returns
153    /// the `0` no-op sentinel and finish ignores it (PE-4).
154    fn _execution_start(&self, name: String) -> u64 {
155        match &self.execution_tracker {
156            Some(tracker) => tracker.start(name),
157            None => 0,
158        }
159    }
160
161    fn _execution_finish(&self, token: u64) {
162        if let Some(tracker) = &self.execution_tracker {
163            tracker.finish(token);
164        }
165    }

168/// Stop the root client, wait for terminal status, destroy its proc so the
169/// remaining actors there stop too, and flush the proc's gateway.
170///
171/// The flush needs the proc's transports to still be serving, so host
172/// shutdown calls this after child drain and before transport teardown.
173pub(crate) async fn shutdown_root_client(
174    instance: &Instance<PythonActor>,
175    reason: String,
176    flush_timeout: Duration,
177) {
178    let actor_id = instance.self_addr().clone();
179    let mut proc = instance.proc().clone();
180    let mut status = instance.status();
181    // The caller's budget bounds child cleanup, capped so exit stays prompt
182    // for callers with a generous flush timeout.
183    let child_timeout = flush_timeout.min(DEFAULT_STOP_CHILDREN_TIMEOUT);
184    instance.set_stop_children_timeout(child_timeout);
185    if let Err(error) = instance.stop(&reason) {
186        tracing::warn!(%actor_id, %error, "stop_and_wait: stop failed");
187    } else {
188        // The serving loop finishes its current work, publishes `Stopping`,
189        // stops children within `child_timeout`, and then publishes terminal.
190        // Each phase is bounded, and the flush runs either way because it is
191        // best-effort.
192        let child_stop_started =
193            |s: &ActorStatus| matches!(s, ActorStatus::Stopping(_)) || s.is_terminal();
194        let terminal_wait = child_timeout + Duration::from_millis(100);
195        if tokio::time::timeout(flush_timeout, status.wait_for(child_stop_started))
196            .await
197            .is_err()
198        {
199            tracing::warn!(%actor_id, ?flush_timeout, "stop_and_wait: timed out waiting for child stop to start");
200        } else if tokio::time::timeout(terminal_wait, status.wait_for(|s| s.is_terminal()))
201            .await
202            .is_err()
203        {
204            tracing::warn!(%actor_id, ?terminal_wait, "stop_and_wait: timed out waiting for terminal status");
205        }
206    }
207    // Actors co-located with the root client live on its proc, which host
208    // shutdown does not destroy in process mode. Stop them so they can run
209    // cleanup before the process exits. The outer bound matches the inner
210    // one, so if an actor is still cleaning up when it expires, the rest of
211    // `destroy_and_wait` is skipped: aborting that actor, draining the
212    // supervision coordinator and its own flush. The flush below still runs.
213    if tokio::time::timeout(child_timeout, proc.destroy_and_wait(child_timeout, &reason))
214        .await
215        .is_err()
216    {
217        tracing::warn!(%actor_id, ?child_timeout, "stop_and_wait: timed out stopping co-located actors");
218    }
219    // `destroy_and_wait` ends by flushing this same gateway. This second
220    // flush only matters when the `child_timeout` bound above expires first
221    // and cancels `destroy_and_wait` before that flush; otherwise there is
222    // typically nothing left to flush and it returns at once.
223    // TODO: Remove this flush, and the outer bound above, once
224    // `destroy_and_wait` keeps its whole teardown, flush included, within
225    // the timeout it is given.
226    match tokio::time::timeout(flush_timeout, proc.flush()).await {
227        Ok(Err(error)) => {
228            tracing::warn!(%actor_id, %error, "stop_and_wait: flush failed");
229        }
230        Err(_elapsed) => {
231            tracing::warn!(%actor_id, "stop_and_wait: flush timed out");
232        }
233        Ok(Ok(())) => {}
234    }
235}

266#[pyclass(name = "Context", module = "monarch._src.actor.actor_mesh")]
267pub struct PyContext {
268    instance: Py<PyInstance>,
269    rank: Point,
270    /// Cloneable handle to a span carrying the actor's recording key.
271    /// When entered, events emitted under this span are captured by
272    /// the per-actor flight recorder. `None` for bootstrap/client
273    /// contexts that are not actor handler execution paths.
274    recording_span: Option<tracing::Span>,
275}
276
277#[pymethods]
278impl PyContext {
279    #[getter]
280    fn actor_instance(&self) -> &Py<PyInstance> {
281        &self.instance
282    }
283
284    #[getter]
285    fn message_rank(&self) -> PyPoint {
286        self.rank.clone().into()
287    }
288
289    #[staticmethod]
290    fn _root_client_context(py: Python<'_>) -> PyResult<PyContext> {
291        let _guard = runtime::get_tokio_runtime().enter();
292        let instance: PyInstance = root_client_actor(py).into();
293        Ok(PyContext {
294            instance: instance.into_pyobject(py)?.into(),
295            rank: Extent::unity().point_of_rank(0).unwrap(),
296            recording_span: None,
297        })
298    }
299
300    /// Create a context from an existing instance.
301    /// This is used when the root client was bootstrapped via bootstrap_host()
302    /// instead of the default bootstrap_client().
303    #[staticmethod]
304    fn _from_instance(py: Python<'_>, instance: PyInstance) -> PyResult<PyContext> {
305        Ok(PyContext {
306            instance: instance.into_pyobject(py)?.into(),
307            rank: Extent::unity().point_of_rank(0).unwrap(),
308            recording_span: None,
309        })
310    }
311}

313impl PyContext {
314    pub(crate) fn new<T: hyperactor::actor::Actor>(
315        cx: &hyperactor::Context<T>,
316        instance: Py<PyInstance>,
317    ) -> PyContext {
318        Self::from_parts(instance, cx.cast_point(), Some(cx.recording_span()))
319    }
320
321    pub(crate) fn from_parts(
322        instance: Py<PyInstance>,
323        rank: Point,
324        recording_span: Option<tracing::Span>,
325    ) -> PyContext {
326        PyContext {
327            instance,
328            rank,
329            recording_span,
330        }
331    }
332
333    /// The actor's recording span, if this context is an actor handler
334    /// execution path. Used by `forward_to_tracing` to enter the
335    /// recording scope on the asyncio thread so that log events are
336    /// captured in the flight recorder.
337    pub(crate) fn recording_span(&self) -> Option<&tracing::Span> {
338        self.recording_span.as_ref()
339    }
340
341    /// Test-only: build a PyContext with a chosen recording span.
342    /// Uses the root client actor for the instance field (the test
343    /// only exercises the recording_span extraction path).
344    #[doc(hidden)]
345    pub fn for_test(py: Python<'_>, recording_span: Option<tracing::Span>) -> PyResult<PyContext> {
346        let mut ctx = Self::_root_client_context(py)?;
347        ctx.recording_span = recording_span;
348        Ok(ctx)
349    }
350}
