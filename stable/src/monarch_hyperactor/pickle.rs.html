9//! Pickling support for Monarch.
10//!
11//! This module pickles Python objects for actor messages, collecting tensor
12//! engine references and out-of-band mesh references during serialization so
13//! they are restored together on decode.
14//!
15//! ## Out-of-band mesh-reference invariants
16//!
17//! Mesh references (proc/host/actor meshes) are not pickled inline. They ride
18//! *out of band* in a `refs` table carried next to the payload bytes, leaving a
19//! `pop_mesh_reference` sentinel in the pickle stream. Two invariants keep the
20//! table and its payload from ever drifting apart:
21//!
22//! **REFS-1 (ref-aware decode).** Any payload that can carry out-of-band refs
23//! must be decoded only through a ref-aware path: `PicklingState::from_parts`
24//! and `PicklingState::unpickle`, as wrapped by `PythonMessage::decode` and
25//! `PythonResponseMessage::decode`. A bare `pickle.loads`/`cloudpickle.loads`
26//! on the payload bytes drops the table, so a sentinel later pops with no active
27//! state and raises "No active pickling state". The raw payload bytes are
28//! deliberately not exposed on `PythonMessage` (there is no `.message` getter),
29//! so a bare decode is not even expressible from Python.
30//!
31//! **REFS-2 (refs preserved through intermediates).** Any intermediate that
32//! relays a payload -- `PythonResponseMessage` and the `ValueOverlay` behind a
33//! `.call()` valuemesh -- must carry its `refs` beside the bytes, all the way to
34//! the decode site. Dropping refs at a relay boundary reintroduces the REFS-1
35//! failure downstream. Refs are therefore part of `PythonResponseMessage`
36//! equality: two runs with identical bytes but different refs must not coalesce.
37//!
38//! These invariants have one sound exception. A table entry and a
39//! `pop_mesh_reference` sentinel are emitted together during pickling (1:1),
40//! so an empty `refs` table means the payload carries no sentinels: a bare
41//! decode of it pops nothing and is sound. The `.call()` valuemesh collector
42//! (`collect_valuemesh` in `endpoint.rs`) relies on this to preserve
43//! D96180139's lazy unpickle: it decodes a ref-empty batch lazily via
44//! `PyValueMesh::build_from_parts` (`value_mesh.rs`), which resolves on access
45//! outside the ref-aware path, and only a ref-carrying batch eagerly via
46//! `build_from_objects`. That lazy build is the sole bare decode of a payload
47//! that *could* carry refs; see `collect_valuemesh` for the gate itself.

109// Thread-local storage for the active pickling state.
110// Set by pickle/unpickle operations so free functions used in __reduce__
111// implementations can access it.
112//
113// It is thread-local by design: two threads pickling at once (even the same
114// mesh ref) collect into independent `mesh_references` / `pending_mesh_fills`,
115// so there is nothing shared to race on. `pickle()` then moves the state out of
116// the thread-local into an owned `PicklingState` before the GIL-releasing
117// `PicklingState::resolve`, so the sender-side slot fill mutates a single-owner
118// value across its awaits, never this thread-local. Both properties (thread-
119// local, and moved out before resolve) are what make releasing the GIL safe.
120thread_local! {
121    static ACTIVE_PICKLING_STATE: RefCell<Option<ActivePicklingState>> = const { RefCell::new(None) };
122}

124/// Counters for tests. `PENDING_RESERVE_COUNT` bumps when a still-pending mesh
125/// reserves an out-of-band slot on the send side; it is pending-specific (a
126/// resolved mesh fills directly), so it proves the pending path ran.
127/// `MESH_POP_COUNT` bumps when any out-of-band mesh reference is reunited on the
128/// decode side; it is not pending-specific, since a resolved mesh also travels
129/// out-of-band, so it proves receive and reconstruct, not pending-ness.
130static PENDING_RESERVE_COUNT: AtomicUsize = AtomicUsize::new(0);
131static MESH_POP_COUNT: AtomicUsize = AtomicUsize::new(0);

133/// RAII guard that sets the thread-local `ACTIVE_PICKLING_STATE` on creation
134/// and restores the previous state (if any) on drop. This supports nesting:
135/// if a guard already exists, the new guard saves the old state and restores
136/// it when dropped, even on panic.
137struct ActivePicklingGuard {
138    previous: Option<ActivePicklingState>,
139}
140
141impl ActivePicklingGuard {
142    /// Set `state` as the active pickling state, saving any existing state.
143    fn enter(state: ActivePicklingState) -> Self {
144        let previous = ACTIVE_PICKLING_STATE.with(|cell| cell.borrow_mut().replace(state));
145        Self { previous }
146    }
147}
148
149impl Drop for ActivePicklingGuard {
150    fn drop(&mut self) {
151        ACTIVE_PICKLING_STATE.with(|cell| {
152            *cell.borrow_mut() = self.previous.take();
153        });
154    }
155}

157thread_local! {
158    /// The actor that is *receiving* the payload currently being decoded.
159    ///
160    /// A reply is decoded on a Tokio worker with no Monarch context, so
161    /// `Port._reconstruct_port` cannot recover the receiver from `context()`
162    /// without bootstrapping a client in a worker process. The eager decoders
163    /// install the caller instance they already hold here for the duration of
164    /// the decode.
165    ///
166    /// Deliberately a sibling of `ACTIVE_PICKLING_STATE` rather than a field
167    /// on it: the receiver is decode context, not pickling payload state, and
168    /// must never reach `PicklingStateInner` or the serialized bytes.
169    static RECEIVER_INSTANCE: RefCell<Option<Instance<PythonActor>>> =
170        const { RefCell::new(None) };
171}
172
173/// RAII guard installing the receiver for the current decode, restoring the
174/// previous value on drop -- on success, on error, and on panic.
175///
176/// Private: the only way to install a receiver is
177/// [`PicklingState::unpickle_with_receiver`], which cannot span an `await`.
178struct ReceiverInstanceGuard {
179    previous: Option<Instance<PythonActor>>,
180}
181
182impl ReceiverInstanceGuard {
183    /// Install `instance` as the receiver, saving any existing one.
184    ///
185    /// Takes a reference and clones internally so callers keep ownership of
186    /// the instance they are already holding across the collector.
187    fn enter(instance: &Instance<PythonActor>) -> Self {
188        let previous =
189            RECEIVER_INSTANCE.with(|cell| cell.borrow_mut().replace(instance.clone_for_py()));
190        Self { previous }
191    }
192}
193
194impl Drop for ReceiverInstanceGuard {
195    fn drop(&mut self) {
196        RECEIVER_INSTANCE.with(|cell| {
197            *cell.borrow_mut() = self.previous.take();
198        });
199    }
200}

202/// Clone of the receiver installed for the current decode, if any.
203///
204/// Clones rather than consumes: one payload may carry several `Port`s and each
205/// must reconstruct against the same receiver.
206fn current_receiver_instance_raw() -> Option<Instance<PythonActor>> {
207    RECEIVER_INSTANCE.with(|cell| cell.borrow().as_ref().map(|i| i.clone_for_py()))
208}

210/// The receiver for the decode in progress, or `None` outside one.
211///
212/// `Port._reconstruct_port` prefers this and falls back to `context()`, which

222/// State maintained during active pickling/unpickling operations.
223///
224/// This is the thread-local state used while cloudpickle is running.
225/// It collects tensor engine references and pending pickles during serialization.
226struct ActivePicklingState {
227    /// References to tensor engine objects that need special handling.
228    tensor_engine_references: VecDeque<Py<PyAny>>,
229    /// Mesh references collected out-of-band into the message's `refs` table.
230    /// A `None` is a slot reserved for a pending mesh, filled sender-side.
231    mesh_references: VecDeque<Option<MeshRef>>,
232    /// Pending mesh slots awaiting a sender-side fill: (index into
233    /// `mesh_references`, the handle whose resolved mesh supplies the ref).
234    pending_mesh_fills: Vec<(usize, Py<PyShared>)>,
235    /// Whether tensor engine references are allowed in this pickling context.
236    allow_tensor_engine_references: bool,
237    /// Whether mesh references are collected out-of-band in this context.
238    allow_mesh_references: bool,
239}
240
241impl ActivePicklingState {
242    /// Create a new ActivePicklingState.
243    fn new(allow_tensor_engine_references: bool, allow_mesh_references: bool) -> Self {
244        Self {
245            tensor_engine_references: VecDeque::new(),
246            mesh_references: VecDeque::new(),
247            pending_mesh_fills: Vec::new(),
248            allow_tensor_engine_references,
249            allow_mesh_references,
250        }
251    }
252
253    /// Convert this active state into a frozen PicklingState.
254    fn into_pickling_state(self, buffer: Part) -> PicklingStateInner {
255        PicklingStateInner {
256            buffer,
257            tensor_engine_references: self.tensor_engine_references,
258            mesh_references: self.mesh_references,
259            pending_mesh_fills: self.pending_mesh_fills,
260        }
261    }
262}

301impl PicklingState {
302    pub fn take_inner(&mut self) -> PyResult<PicklingStateInner> {
303        self.inner.take().ok_or_else(|| {
304            pyo3::exceptions::PyRuntimeError::new_err("PicklingState has already been consumed")
305        })

311        })

315        Ok(!self.inner_ref()?.pending_mesh_fills.is_empty())

318    /// Build a PicklingState directly from already-separated parts: the pickled

325        mesh_references: VecDeque<Option<MeshRef>>,

327        Self {
328            inner: Some(PicklingStateInner {

331                mesh_references,
332                pending_mesh_fills: Vec::new(),

334        }

338#[pymethods]
339impl PicklingState {
340    /// Create a new PicklingState from a buffer and optional tensor engine references.
341    ///
342    /// This is used for unpickling received messages that may contain tensor engine
343    /// references that need to be restored during deserialization.
344    #[new]
345    #[pyo3(signature = (buffer, tensor_engine_references=None, mesh_references=None))]
346    fn py_new(
347        py: Python<'_>,
348        buffer: PyRef<'_, crate::buffers::FrozenBuffer>,
349        tensor_engine_references: Option<&Bound<'_, PyList>>,
350        mesh_references: Option<Vec<Py<PyMeshRef>>>,
351    ) -> PyResult<Self> {
352        let refs: VecDeque<Py<PyAny>> = tensor_engine_references
353            .map(|list| list.iter().map(|item| item.unbind()).collect())
354            .unwrap_or_default();
355
356        // pyo3 type-checks each element as a `PyMeshRef` during extraction.
357        let mesh_refs: VecDeque<Option<MeshRef>> = mesh_references
358            .map(|list| {
359                list.into_iter()
360                    .map(|m| Some(m.borrow(py).inner.clone()))
361                    .collect()
362            })
363            .unwrap_or_default();
364
365        Ok(Self {
366            inner: Some(PicklingStateInner {
367                buffer: Part::from(buffer.inner.clone()),
368                tensor_engine_references: refs,
369                mesh_references: mesh_refs,
370                pending_mesh_fills: Vec::new(),
371            }),
372        })
373    }
374
375    /// Get a copy of all tensor engine references from this pickling state.
376    ///
377    /// Returns a Python list containing copies of the tensor engine references.
378    fn tensor_engine_references(&self, py: Python<'_>) -> PyResult<Py<PyList>> {
379        let inner = self.inner_ref()?;
380        let refs: Vec<Py<PyAny>> = inner
381            .tensor_engine_references
382            .iter()
383            .map(|r| r.clone_ref(py))
384            .collect();
385        Ok(PyList::new(py, refs)?.unbind())
386    }
387
388    /// Get the buffer from this pickling state.
389    ///
390    /// Returns a FrozenBuffer containing the pickled bytes.
391    /// This does not consume the PicklingState.
392    fn buffer(&self) -> PyResult<crate::buffers::FrozenBuffer> {
393        let inner = self.inner_ref()?;
394        Ok(crate::buffers::FrozenBuffer {
395            inner: inner.buffer.clone().into_bytes(),
396        })
397    }
398
399    /// Unpickle the buffer contents.
400    ///
401    /// This consumes the PicklingState. It will fail if there are any pending
402    /// pickles that haven't been resolved.
403    pub(crate) fn unpickle(&mut self, py: Python<'_>) -> PyResult<Py<PyAny>> {
404        let inner = self.take_inner()?;
405
406        // Set up an active state for unpickling (to handle pop calls).
407        // The guard restores any previous state on drop (including on panic).
408        let mut active = ActivePicklingState::new(false, false);
409        active.tensor_engine_references = inner.tensor_engine_references;
410        active.mesh_references = inner.mesh_references;
411        active.pending_mesh_fills = inner.pending_mesh_fills;
412
413        let _guard = ActivePicklingGuard::enter(active);
414
415        let frozen = crate::buffers::FrozenBuffer {
416            inner: inner.buffer.into_bytes(),
417        };
418
419        // Unpickle the object. If torch is loaded, use torch_loads which
420        // disables dispatch modes during unpickling.
421        let result = if maybe_torch_fn(py).call0()?.is_truthy()? {
422            torch_loads_fn(py).call1((frozen,))
423        } else {
424            cloudpickle(py).getattr("loads")?.call1((frozen,))
425        };
426
427        result.map(|obj| obj.unbind())
428    }
429}

431impl PicklingState {
432    /// [`unpickle`](Self::unpickle) with `instance` installed as the receiver.
433    ///
434    /// Rust-only and deliberately unexposed to Python: the guard must not
435    /// outlive the synchronous decode. Every caller is the body of a
436    /// `monarch_with_gil_blocking` closure, so the guard never spans an
437    /// `await` and the thread-local cannot leak to another task.
438    pub(crate) fn unpickle_with_receiver(
439        &mut self,
440        py: Python<'_>,
441        instance: &Instance<PythonActor>,
442    ) -> PyResult<Py<PyAny>> {
443        let _guard = ReceiverInstanceGuard::enter(instance);
444        self.unpickle(py)
445    }
446
447    /// Fill the reserved mesh slots from their pending handles, producing a
448    /// PicklingState whose out-of-band `refs` table is fully populated.
449    ///
450    /// Awaits each pending mesh's init task, extracts its `*MeshRef`, and writes
451    /// it into the reserved slot. The payload bytes are unchanged (no re-pickle).
452    pub async fn resolve(mut self) -> PyResult<PicklingState> {
453        // Take the pending mesh fills out: a plain move, no GIL and no
454        // `clone_ref`. Each is a slot index plus the handle whose resolved mesh
455        // supplies the ref.
456        let fills = std::mem::take(
457            &mut self
458                .inner
459                .as_mut()
460                .ok_or_else(|| {
461                    pyo3::exceptions::PyRuntimeError::new_err(
462                        "PicklingState has already been consumed",
463                    )
464                })?
465                .pending_mesh_fills,
466        );
467
468        for (index, handle) in fills {
469            // Await the mesh's init task (these run concurrently, so the wait
470            // is the slowest init, not the sum).
471            let mut task =
472                monarch_with_gil_blocking(GilSite::AwaitDrive, |py| handle.borrow(py).task())?;
473            task.take_task()?.await?;
474
475            // Extract the `*MeshRef` from the now-resolved mesh and fill the slot.
476            monarch_with_gil_blocking(GilSite::Convert, |py| -> PyResult<()> {
477                let value = handle.borrow(py).poll()?.ok_or_else(|| {
478                    pyo3::exceptions::PyRuntimeError::new_err("pending mesh handle did not resolve")
479                })?;
480                let mesh_ref = crate::actor::mesh_ref_from_pyobject(value.bind(py))?;
481                if let Some(inner) = self.inner.as_mut() {
482                    inner.mesh_references[index] = Some(mesh_ref);
483                }
484                Ok(())
485            })?;
486        }
487
488        Ok(self)
489    }
490}

492/// A message whose reserved mesh slots must be filled before it can be sent.
493///
494/// Contains a `PythonMessageKind` and a `PicklingState`. The `PicklingState` may
495/// hold mesh slots reserved for pending meshes, filled sender-side once their
496/// handles resolve, before the message can be converted into a `PythonMessage`.
497#[pyclass(module = "monarch._rust_bindings.monarch_hyperactor.pickle")]
498pub struct PendingMessage {
499    pub(crate) kind: PythonMessageKind,
500    state: PicklingState,
501}
502
503impl PendingMessage {
504    /// Create a new PendingMessage from a kind and pickling state.
505    pub fn new(kind: PythonMessageKind, state: PicklingState) -> Self {
506        Self { kind, state }
507    }
508
509    /// Take ownership of the inner state from a mutable reference.
510    ///
511    /// This is used by pyo3 pymethods that receive `&mut PendingMessage`
512    /// but need to pass ownership to the trait method.
513    pub fn take(&mut self) -> PyResult<PendingMessage> {
514        let inner = self.state.take_inner()?;
515        Ok(PendingMessage {
516            kind: std::mem::take(&mut self.kind),
517            state: PicklingState { inner: Some(inner) },
518        })
519    }
520
521    /// The size in bytes of the pickled payload. Out-of-band mesh references
522    /// are carried beside these bytes, so they do not count towards it.
523    pub(crate) fn payload_len(&self) -> usize {
524        self.state
525            .inner_ref()
526            .expect("should be called before the message is resolved")
527            .payload_len()
528    }
529
530    fn into_python_message(mut self) -> PyResult<PythonMessage> {
531        let inner = self.state.take_inner()?;
532        let refs: Vec<MeshRef> = inner
533            .mesh_references
534            .into_iter()
535            .map(|reference| {
536                reference.ok_or_else(|| {
537                    pyo3::exceptions::PyRuntimeError::new_err(
538                        "mesh reference slot was never filled",
539                    )
540                })
541            })
542            .collect::<PyResult<_>>()?;
543        Ok(PythonMessage::new_from_buf_with_refs(
544            self.kind,
545            inner.buffer,
546            refs,
547        ))
548    }
549
550    /// Convert this message synchronously when no mesh fills are pending.
551    ///
552    /// A pending message is returned unchanged in the inner `Err` so the
553    /// caller can use the existing asynchronous resolution path.
554    /// Every reserved `None` mesh slot is created together with one pending
555    /// fill entry, while resolved and reconstructed slots are always `Some`.
556    /// Therefore a fresh zero-fill message cannot fail assembly due to an
557    /// unfilled slot.
558    pub fn try_resolve_now(self) -> PyResult<Result<PythonMessage, PendingMessage>> {
559        if self.state.has_pending_mesh_fills()? {
560            return Ok(Err(self));
561        }
562
563        Ok(Ok(self.into_python_message()?))
564    }
565
566    /// Resolve any reserved mesh slots and convert this into a PythonMessage.
567    ///
568    /// This is an async method that:
569    /// 1. Fills the reserved mesh slots from their pending handles
570    /// 2. Builds a PythonMessage with the resolved bytes and `refs` table
571    pub async fn resolve(self) -> PyResult<PythonMessage> {
572        let Self { kind, state } = self;
573        let state = state.resolve().await?;
574        Self::new(kind, state).into_python_message()
575    }
576}

580    /// Create a new PendingMessage from a kind and pickling state.
581    #[new]
582    pub fn py_new(
583        kind: PythonMessageKind,
584        mut state: PyRefMut<'_, PicklingState>,
585    ) -> PyResult<Self> {
586        // Take the inner state from the PicklingState
587        let inner = state.take_inner()?;
588        Ok(Self {
589            kind,
590            state: PicklingState { inner: Some(inner) },
591        })
592    }
593
594    /// Get the message kind.
595    #[getter]
596    fn kind(&self) -> PythonMessageKind {
597        self.kind.clone()
598    }
599
600    /// Return a materialized message when no mesh fills are pending.
601    ///
602    /// A `None` result leaves this object unchanged for `resolve()`.
603    #[pyo3(name = "try_resolve_now")]
604    fn py_try_resolve_now(&mut self) -> PyResult<Option<PythonMessage>> {
605        if self.state.has_pending_mesh_fills()? {
606            return Ok(None);
607        }
608
609        self.take()?.into_python_message().map(Some)
610    }
611
612    /// Fill reserved mesh slots and return a fully materialized PythonMessage.
613    #[pyo3(name = "resolve")]
614    fn py_resolve(&mut self) -> PyResult<PyHandle> {
615        let message = self.take()?;

622/// This is called from Python during pickling when a tensor engine object
623/// is encountered that needs special handling.
624///
625/// Returns False if there is no active pickling state.
626/// Returns True if the reference was successfully pushed.
627/// Raises an error if tensor engine references are not allowed in the current pickling context.

630    ACTIVE_PICKLING_STATE.with(|cell| {

636                        "Tensor engine references are not allowed in the current pickling context",
637                    ));
638                }

641            }

671/// Python mesh wrapper.
672///
673/// Called from the unpickle stream wherever a mesh slot was emitted in
674/// place of inline bytes.
675#[pyfunction]
676fn pop_mesh_reference(py: Python<'_>) -> PyResult<Py<PyAny>> {
677    let mesh_ref = ACTIVE_PICKLING_STATE.with(|cell| {

714/// registering `handle` for a sender-side fill once the mesh resolves.

741/// Reduce a PyShared for pickling.
742///
743/// This function implements the pickle protocol for PyShared:
744/// 1. If the shared is already finished, return (Shared.from_value, (value,))
745/// 2. Otherwise, block on the shared and return (Shared.from_value, (value,))
746pub fn reduce_shared<'py>(
747    py: Python<'py>,
748    py_shared: &Bound<'py, PyShared>,
749) -> PyResult<(Bound<'py, PyAny>, Bound<'py, PyTuple>)> {
750    // First, check if the shared is already finished
751    if let Some(value) = py_shared.borrow().poll()? {
752        let from_value = shared_class(py).getattr("from_value")?;
753        let args = PyTuple::new(py, [value])?;
754        return Ok((from_value, args));
755    }
756
757    // Pending: block on the shared. Meshes ride the out-of-band table via their
758    // own type-specific reducers; this generic path is the fallback for any
759    // other pending `PyShared`.
760    let value = PyShared::block_on(py_shared.borrow(), py)?;
761    let from_value = shared_class(py).getattr("from_value")?;
762    let args = PyTuple::new(py, [value])?;
763    Ok((from_value, args))
764}

766/// Pickle a Python object into a [`Buffer`].
767///
768/// This is the shared pickling core. The caller is responsible for setting up
769/// the [`ActivePicklingGuard`] before calling this function.
770fn pickle_into_buffer(py: Python<'_>, obj: &Py<PyAny>, buffer: &Py<Buffer>) -> PyResult<()> {
771    // Ensure the cloudpickle monkeypatch for RemoteImportLoader is applied.
772    pickle_monkeypatch(py);
773
774    // If torch is loaded, use the torch-aware pickler that handles
775    // torch storage types via dispatch_table.
776    if maybe_torch_fn(py).call0()?.is_truthy()? {
777        torch_dump_fn(py).call1((obj, buffer.bind(py)))?;
778    } else {
779        let pickler = cloudpickle(py)
780            .getattr("Pickler")?
781            .call1((buffer.bind(py),))?;
782        pickler.call_method1("dump", (obj,))?;
783    }
784
785    Ok(())
786}

788/// Pickle a Python object and return the serialized data as a [`Part`].
789///
790/// This is a simplified variant of [`pickle`] that disallows pending pickles
791/// and tensor engine references, and returns the raw serialized bytes instead
792/// of a [`PicklingState`].
793pub fn pickle_to_part(py: Python<'_>, obj: &Py<PyAny>) -> PyResult<Part> {
794    let active = ActivePicklingState::new(false, false);
795    let buffer = Py::new(py, Buffer::default())?;
796    let _guard = ActivePicklingGuard::enter(active);
797
798    pickle_into_buffer(py, obj, &buffer)?;
799
800    Ok(buffer.borrow_mut(py).take_part())
801}

803/// Pickle an object with support for pending pickles and tensor engine references.
804///
805/// This function creates a PicklingState and calls cloudpickle.dumps with
806/// an active thread-local PicklingState, allowing __reduce__ implementations
807/// to push tensor engine references and pending pickles.
808///
809/// # Arguments
810/// * `obj` - The Python object to pickle
811/// * `allow_tensor_engine_references` - If true, allow tensor engine references to be registered
812///
813/// # Returns
814/// A PicklingState containing the pickled buffer and any registered references/pending pickles
815#[pyfunction]
816#[pyo3(signature = (obj, allow_tensor_engine_references=true, allow_mesh_references=false))]
817pub fn pickle(
818    py: Python<'_>,
819    obj: Py<PyAny>,
820    allow_tensor_engine_references: bool,
821    allow_mesh_references: bool,
822) -> PyResult<PicklingState> {
823    let active = ActivePicklingState::new(allow_tensor_engine_references, allow_mesh_references);
824    let buffer = Py::new(py, Buffer::default())?;
825    let _guard = ActivePicklingGuard::enter(active);
826
827    pickle_into_buffer(py, &obj, &buffer)?;
828
829    // Take the state (which may have been modified during pickling).
830    // The guard will restore the previous state on drop.
831    let active = ACTIVE_PICKLING_STATE
832        .with(|cell| cell.borrow_mut().take())
833        .expect("active pickling state should still be set");
834
835    // Take the Part (zero-copy fragmented buffer) directly.
836    let part = buffer.borrow_mut(py).take_part();
837    let inner = active.into_pickling_state(part);
838    Ok(PicklingState { inner: Some(inner) })
839}

1357    /// `py_resolve` starts waiting on a pending handle, and dropping its
1358    /// returned observer does not cancel that wait.
1359    ///
1360    /// The sender's receiver count is the witness: `Shared::task()` clones the
1361    /// watch receiver for the life of the waiter. The original `PyShared` stays
1362    /// alive to keep the baseline stable after the returned observer is
1363    /// dropped.
1364    #[tokio::test]
1365    async fn py_resolve_starts_resolution_and_survives_returned_handle_drop() {
1366        pyo3::Python::initialize();
1367
1368        let (sender, pending_handle, baseline, mut message) =
1369            monarch_with_gil_blocking(GilSite::Test, |py| {
1370                let (sender, pending) = PyShared::pending();
1371                let handle = Py::new(py, pending)?;
1372                let baseline = sender.receiver_count();
1373
1374                let mut message = PendingMessage::new(
1375                    PythonMessageKind::Result { rank: Some(7) },
1376                    pickling_state(vec![1, 2, 3], vec![None], vec![(0, handle.clone_ref(py))]),
1377                );
1378
1379                let observer = message.py_resolve()?;
1380                drop(observer);
1381                Ok::<_, PyErr>((sender, handle, baseline, message))
1382            })
1383            .expect("construction should succeed");
1384
1385        tokio::time::timeout(std::time::Duration::from_secs(2), async {
1386            while sender.receiver_count() == baseline {
1387                tokio::task::yield_now().await;
1388            }
1389        })
1390        .await
1391        .expect("resolution must start and retain a waiter after its observer is dropped");
1392        assert_eq!(
1393            sender.receiver_count(),
1394            baseline + 1,
1395            "one active resolver must retain exactly one additional receiver"
1396        );
1397
1398        let value = monarch_with_gil_blocking(GilSite::Test, |py| {
1399            Ok::<_, PyErr>(mesh_value(py, &proc_mesh_ref(44, "pending")))
1400        })
1401        .expect("the pending mesh value should construct");
1402        sender
1403            .send(Some(Ok(value)))
1404            .expect("the retained pending handle should receive its value");
1405
1406        tokio::time::timeout(std::time::Duration::from_secs(2), async {
1407            while sender.receiver_count() != baseline {
1408                tokio::task::yield_now().await;
1409            }
1410        })
1411        .await
1412        .expect("the detached resolver must finish and release its waiter");
1413
1414        monarch_with_gil_blocking(GilSite::Test, |py| {
1415            let second = expect_error(
1416                message.py_resolve(),
1417                "the source was consumed by the first call",
1418            );
1419            assert_eq!(
1420                second.value(py).to_string(),
1421                "PicklingState has already been consumed",
1422            );
1423        });
1424        drop(pending_handle);
1425    }
