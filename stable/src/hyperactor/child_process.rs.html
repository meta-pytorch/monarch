9//! Spawns child processes from a thread that lives as long as the process.
10//!
11//! On Linux, `prctl(PR_SET_PDEATHSIG)` fires when the *thread* that forked a
12//! child exits, not when the child's parent process exits. A child forked
13//! from a tokio worker therefore dies with that worker, and tokio can retire a
14//! worker at any time, for example after `block_in_place`. Children that arm
15//! the death signal must be forked here instead, by one thread that never
16//! exits, so the signal fires only when the parent process does.
17//!
18//! # Invariants
19//!
20//! - **CPS-1 (spawner-outlives-children):** every child returned by
21//!   [`spawn_child`] is forked by the child-spawner thread, which does not
22//!   exit while the process runs. Its request sender is never dropped, and a
23//!   panic while spawning a command or dropping it is contained inside the
24//!   thread's loop.
25//! - **CPS-2 (caller-runtime):** the returned [`Child`] is registered with the
26//!   runtime that called [`spawn_child`]: the caller's handle is taken before
27//!   the request is sent and entered on the spawner thread for the spawn.
28//! - **CPS-3 (no-silent-fallback):** [`spawn_child`] never forks on the
29//!   calling thread; it has no path that calls [`Command::spawn`] there.
30//! - **CPS-4 (caller-owns-child):** the child returned by a successful
31//!   [`spawn_child`] belongs to its caller from the moment the call returns:
32//!   the handoff is synchronous, so there is no cancellation point between the
33//!   fork and the return. A spawn that panics after forking, as tokio does for
34//!   a runtime without I/O, returns an error and leaves that child
35//!   unregistered.

50/// Name of the thread that forks every child spawned through [`spawn_child`].
51/// Linux keeps the first 15 bytes of a thread name.
52const SPAWNER_THREAD_NAME: &str = "child-spawner";
53
54/// A caller's spawn request to the child-spawner thread.
55struct Request {
56    /// The command to spawn on that thread, rather than on the caller's thread.
57    command: Command,
58
59    /// The caller's tokio runtime. The spawner enters it before spawning so
60    /// the child is registered with this runtime, not a runtime of its own.
61    runtime: Handle,
62
63    /// Sends the child or spawn error back to the caller that made this request.
64    reply: mpsc::Sender<io::Result<Child>>,
65}
66
67/// Request sender, initialized when the child-spawner thread starts. A failed
68/// start leaves it unset, so later callers can retry. Once stored, the sender
69/// is never dropped, keeping the thread's request loop alive.
70static SPAWNER: OnceLock<mpsc::Sender<Request>> = OnceLock::new();
71
72/// Spawns `command` on a thread that stays alive until this process exits.
73/// The returned child is registered with the caller's tokio runtime.
74///
75/// Use this when the child will call `prctl(PR_SET_PDEATHSIG, SIGKILL)`.
76/// Linux then kills the child when the thread that forked it exits, even
77/// if the parent process is still running. A tokio worker can be retired;
78/// the child-spawner thread stays alive, so retiring a worker cannot kill
79/// the child.
80///
81/// This helper does not set the death signal; the child sets it itself.
82///
83/// Blocks the caller until spawning succeeds or fails. Requests from all
84/// callers are handled one at a time.
85///
86/// Returns an error if the caller has no current tokio runtime, the
87/// child-spawner thread cannot be started, or spawning fails or panics.
88pub fn spawn_child(command: Command) -> io::Result<Child> {
89    // CPS-2: register the child with the caller's runtime, not the spawner's.
90    let runtime = Handle::try_current().map_err(io::Error::other)?;
91    let (reply, result) = mpsc::channel();
92    // CPS-3: on failure, return an error; never spawn on the caller's thread.
93    spawner()?
94        .send(Request {
95            command,
96            runtime,
97            reply,
98        })
99        .map_err(|_| io::Error::other("the child-spawner thread has exited"))?;
100    // CPS-4: if we awaited the reply, the caller's task could be cancelled
101    // after the child starts but before it receives the handle needed to
102    // wait for or stop that child. Wait synchronously so cancellation cannot
103    // interrupt this handoff.
104    result
105        .recv()
106        .map_err(|_| io::Error::other("the child-spawner thread dropped the request"))?
107}

109/// Returns the request sender for the OS thread used by [`spawn_child`].
110///
111/// On first use, creates a channel and starts one OS thread running
112/// [`serve`] to receive the requests and spawn children. [`SPAWNER`] keeps
113/// the sender, so later calls use the same thread rather than start another.
114///
115/// If starting the thread fails, returns the error and leaves [`SPAWNER`]
116/// uninitialized, allowing a later call to try again.
117fn spawner() -> io::Result<&'static mpsc::Sender<Request>> {
118    SPAWNER.get_or_try_init(|| {
119        let (sender, requests) = mpsc::channel();
120        thread::Builder::new()
121            .name(SPAWNER_THREAD_NAME.to_string())
122            .spawn(move || serve(requests))?;
123        Ok(sender)
124    })
125}

127/// Runs the request loop on the child-spawner OS thread.
128///
129/// Waits for a request, enters the caller's tokio runtime, spawns the
130/// child, and sends the result back to that caller. Requests are handled
131/// one at a time.
132///
133/// [`SPAWNER`] keeps the channel open for the process's lifetime. Panics
134/// while spawning or dropping a command are caught so this thread keeps
135/// running and children spawned earlier do not lose their parent thread.
136fn serve(requests: mpsc::Receiver<Request>) {
137    // CPS-1: `SPAWNER` keeps the channel open for the process's lifetime.
138    // When there are no requests, this OS thread blocks waiting for the
139    // next one rather than exiting.
140    for Request {
141        mut command,
142        runtime,
143        reply,
144    } in requests
145    {
146        // CPS-1: a panic must not end this thread and trigger the parent-death
147        // signals of children it spawned earlier. Catch it and return an error
148        // to this caller instead. For example, tokio can start the child and then
149        // panic because the caller's runtime has I/O disabled.
150        let result = catch_unwind(AssertUnwindSafe(|| {
151            // CPS-2: make the caller's tokio runtime current on this OS thread
152            // so `spawn` registers the child with it. The `_runtime` guard
153            // restores the previous runtime context when this block ends,
154            // including on panic.
155            let _runtime = runtime.enter();
156            command.spawn()
157        }))
158        .unwrap_or_else(|panic| {
159            // CPS-1: copy the panic's description into the error, then forget
160            // its payload without running its destructor. That destructor could
161            // panic again, ending this thread and killing earlier children.
162            Err(io::Error::other(describe_and_forget(
163                panic,
164                "spawning the child",
165            )))
166        });
167        // CPS-1: the command owns caller state, such as a `pre_exec` closure's
168        // captures, whose destructor could panic. Drop it inside the unwind
169        // boundary too, and forget any payload unread, since its destructor is
170        // caller code. Nothing is logged here: a `tracing` subscriber is caller
171        // code as well. CPS-4: a child that was spawned is still handed over.
172        if let Err(panic) = catch_unwind(AssertUnwindSafe(move || drop(command))) {
173            mem::forget(panic);
174        }
175        // A caller that has gone away no longer wants the result.
176        let _ = reply.send(result);
177    }
178}

180/// Describes a caught panic, then forgets its payload. CPS-1: the payload can
181/// be caller code whose destructor panics again, which would unwind this
182/// thread, so its destructor must never run here.
183fn describe_and_forget(panic: Box<dyn Any + Send>, what: &str) -> String {
184    let description = format!("{what} panicked: {}", panic_message(panic.as_ref()));
185    mem::forget(panic);
186    description
187}

189/// Borrows the message from a `&str` or `String` panic payload.
190/// For other payloads, returns a fixed description without invoking
191/// caller-supplied formatting code.
192fn panic_message(panic: &(dyn Any + Send)) -> &str {
193    if let Some(message) = panic.downcast_ref::<&str>() {
194        message
195    } else if let Some(message) = panic.downcast_ref::<String>() {
196        message
197    } else {
198        "a non-string panic payload"
199    }
200}

222    /// How long a child must keep running to count as having survived.
223    const SURVIVAL_WAIT: Duration = Duration::from_secs(1);
224    /// Upper bound on any wait for a child to exit.
225    const EXIT_WAIT: Duration = Duration::from_secs(10);

227    /// Builds a command to run `sleep <seconds>` with a parent-death signal.
228    ///
229    /// `pre_exec` runs in the child before starting `sleep` and asks Linux to
230    /// send SIGKILL when the forking thread exits. `kill_on_drop` is separate:
231    /// it kills the child if a failing test drops its `Child` handle. Tests
232    /// keep that handle alive while checking the parent-death signal.
233    fn sleep_with_death_signal(seconds: &str) -> Command {
234        let mut command = Command::new("sleep");
235        command.arg(seconds).kill_on_drop(true);
236        // SAFETY: the closure runs in the forked child before exec, where only
237        // async-signal-safe calls are allowed. It calls `prctl`, which is, with
238        // constant arguments, and `last_os_error`, which reads `errno` and does
239        // not allocate.
240        unsafe {
241            command.pre_exec(|| {
242                if libc::prctl(libc::PR_SET_PDEATHSIG, libc::SIGKILL) != 0 {
243                    return Err(io::Error::last_os_error());
244                }
245                Ok(())
246            });
247        }
248        command
249    }

251    /// Requests a child from a temporary OS thread and joins that thread
252    /// before returning the child handle to the test.
253    ///
254    /// `spawn` selects either [`spawn_child`] or ordinary `Command::spawn`,
255    /// allowing both paths to use the same setup. The temporary thread enters
256    /// the test's runtime so either path can register the child with tokio.
257    fn spawn_from_short_lived_thread(
258        command: Command,
259        spawn: fn(Command) -> io::Result<Child>,
260    ) -> Child {
261        let runtime = Handle::current();
262        thread::spawn(move || {
263            // A new OS thread has no current tokio runtime of its own.
264            // Use the test's runtime while making the spawn request.
265            let _runtime = runtime.enter();
266            spawn(command)
267        })
268        .join()
269        .expect("the spawning thread panicked")
270        .expect("spawn the child")
271    }

273    /// Watches `child` for `wait`, returning false if an exit is observed.
274    ///
275    /// A child being alive immediately after spawning is not enough: poll
276    /// throughout the observation period to allow a death signal to arrive.
277    async fn keeps_running(child: &mut Child, wait: Duration) -> bool {
278        let deadline = Instant::now() + wait;
279        while Instant::now() < deadline {
280            if child.try_wait().expect("poll the child").is_some() {
281                return false;
282            }
283            // Yield between polls so the test's runtime can keep running.
284            tokio::time::sleep(Duration::from_millis(50)).await;
285        }
286        true
287    }

289    /// Kills the test child and waits at most [`EXIT_WAIT`] to reap it.
290    /// Sending the kill alone would not collect the child's exit status or
291    /// remove its remaining OS process record.
292    async fn kill(mut child: Child) {
293        child.start_kill().expect("kill the child");
294        tokio::time::timeout(EXIT_WAIT, child.wait())
295            .await
296            .expect("the child exited")
297            .expect("reap the child");
298    }

300    /// Whether the test child's `/proc/<pid>` entry has disappeared.
301    /// An exited but unreaped child still has an entry, so checking this
302    /// distinguishes completed cleanup from merely stopping the process.
303    fn is_gone(pid: u32) -> bool {
304        fs::metadata(format!("/proc/{pid}")).is_err()
305    }

307    /// Builds a `true` command whose child writes its PID to a pipe before
308    /// exec. Returns the command and the pipe's read end for the test.
309    ///
310    /// The no-I/O test makes tokio panic after starting the child but before
311    /// returning a `Child` handle. This PID report lets the test reap that
312    /// exact child directly, without relying on tokio or scanning processes.
313    fn true_reporting_pid() -> (Command, File) {
314        // Both ends close at exec, so the executed program does not retain
315        // these descriptors. The PID report happens before that point.
316        let mut fds = [0; 2];
317        // SAFETY: `fds` is a writable array of two `c_int`s, as `pipe2`
318        // requires.
319        let rc = unsafe { libc::pipe2(fds.as_mut_ptr(), libc::O_CLOEXEC) };
320        assert_eq!(rc, 0, "pipe2: {}", io::Error::last_os_error());
321
322        // SAFETY: `pipe2` succeeded, so both descriptors are open and owned by
323        // nothing else.
324        let (read, write) = unsafe { (OwnedFd::from_raw_fd(fds[0]), OwnedFd::from_raw_fd(fds[1])) };
325        let mut command = Command::new("true");
326        // SAFETY: the closure runs in the forked child before exec, where only
327        // async-signal-safe calls are allowed. `getpid` and `write` are, and
328        // `last_os_error` only reads `errno`. The write end is open in the
329        // child because `O_CLOEXEC` closes it only at exec, and the closure
330        // keeps the `OwnedFd` alive in this process until the command drops.
331        unsafe {
332            command.pre_exec(move || {
333                let pid = libc::getpid().to_ne_bytes();
334                if libc::write(write.as_raw_fd(), pid.as_ptr().cast(), pid.len())
335                    != pid.len() as isize
336                {
337                    return Err(io::Error::last_os_error());
338                }
339                Ok(())
340            });
341        }
342        (command, File::from(read))
343    }

345    /// Reaps the unregistered test child `pid` within [`EXIT_WAIT`].
346    ///
347    /// There is no tokio `Child` handle for this process, so the test must
348    /// collect its exit directly. `WNOHANG` lets us poll with a deadline
349    /// rather than block indefinitely inside `waitpid`.
350    fn reap(pid: libc::pid_t) {
351        let deadline = Instant::now() + EXIT_WAIT;
352        loop {
353            // SAFETY: `waitpid` takes a PID, a status pointer that may be
354            // null, and flags. `pid` is this process's child, unregistered with
355            // tokio, so nothing else reaps it.
356            match unsafe { libc::waitpid(pid, ptr::null_mut(), libc::WNOHANG) } {
357                0 => {
358                    // The child has not exited yet. Retry only while the
359                    // cleanup deadline allows it.
360                    assert!(Instant::now() < deadline, "child {pid} was never reaped");
361                    thread::sleep(Duration::from_millis(10));
362                }
363                reaped if reaped == pid => return,
364                _ => panic!("waitpid({pid}): {}", io::Error::last_os_error()),
365            }
366        }
367    }

408    // CPS-1: a panicking spawn must not kill earlier children or prevent
409    // later requests from being completed.
410    #[tokio::test]
411    async fn panicking_request_leaves_the_spawner_running() {
412        // If the shared spawning thread exits, this child's parent-death
413        // signal will kill it.
414        let mut earlier = spawn_child(sleep_with_death_signal("60")).expect("spawn the child");
415
416        // Force a panic on the spawning thread: tokio starts `true`, but
417        // cannot register it with a runtime whose I/O driver is disabled.
418        // Build and drop that runtime on a plain OS thread, because tokio
419        // does not allow a runtime to be dropped inside an async task.
420        let (result, mut reported_pid) = thread::spawn(|| {
421            let without_io = tokio::runtime::Builder::new_current_thread()
422                .build()
423                .expect("build a runtime without I/O");
424            let _runtime = without_io.enter();
425            let (command, reported_pid) = true_reporting_pid();
426            (spawn_child(command), reported_pid)
427        })
428        .join()
429        .expect("the requesting thread panicked");
430
431        // The child started, but failed registration left us without a Child
432        // handle. Read the PID it sent through the pipe and reap it directly.
433        let mut pid = [0; size_of::<libc::pid_t>()];
434        reported_pid
435            .read_exact(&mut pid)
436            .expect("the unregistered child reported its PID");
437        reap(libc::pid_t::from_ne_bytes(pid));
438
439        // The spawning panic must have been caught and returned as an error.
440        let error = result.expect_err("spawning into a runtime without I/O fails");
441        assert!(
442            error.to_string().contains("panicked"),
443            "unexpected error: {error}"
444        );
445
446        // The earlier child must survive: the panic must not have ended
447        // the shared thread that forked it.
448        assert!(
449            keeps_running(&mut earlier, SURVIVAL_WAIT).await,
450            "an earlier child died after a request panicked"
451        );
452
453        // The spawning thread must also remain able to complete new requests.
454        let later = spawn_child(Command::new("true")).expect("spawn after the panic");
455        let status = tokio::time::timeout(EXIT_WAIT, later.wait_with_output())
456            .await
457            .expect("the later child exited")
458            .expect("reap the later child")
459            .status;
460        assert!(status.success());
461
462        // Kill and reap the long-running test child.
463        kill(earlier).await;
464    }

496        // The callback itself succeeds. Its captured value panics only

511        // command-drop panic. `true` should then exit successfully.
512        let child = spawn_child(command).expect("the child is handed over");

517            .status;

521        assert!(
522            keeps_running(&mut earlier, SURVIVAL_WAIT).await,

527        let later = spawn_child(Command::new("true")).expect("spawn after the panic");

604    /// Sets up the worker-retirement scenario from meta-pytorch/monarch#5248.
605    ///
606    /// A one-worker runtime requests a death-signal child using `spawn`.
607    /// `block_in_place` then moves the worker role to another OS thread,
608    /// letting the original thread retire from the blocking pool. The test
609    /// must observe that original thread's exit, not merely the role moving.
610    ///
611    /// Returns both the child and the still-live runtime. The calling test
612    /// can then check the child while ruling out runtime shutdown as the
613    /// cause of its death. `spawn` selects our helper or ordinary spawning.
614    fn spawn_then_retire_worker(spawn: fn(Command) -> io::Result<Child>) -> (Child, Runtime) {
615        // Keep the blocking pool's idle-thread lifetime short so the old
616        // worker can retire during the test. The runtime itself stays alive.
617        let runtime = tokio::runtime::Builder::new_multi_thread()
618            .worker_threads(1)
619            .thread_keep_alive(Duration::from_millis(100))
620            .enable_all()
621            .build()
622            .expect("build a one-worker runtime");
623
624        // `block_on` alone runs its main future on the calling thread.
625        // Use `tokio::spawn` to make the request from a runtime worker,
626        // and record that worker's OS thread ID before the request.
627        let (child, worker) = runtime.block_on(async {
628            tokio::spawn(async move {
629                let worker = gettid();
630                (
631                    spawn(sleep_with_death_signal("60")).expect("spawn the child"),
632                    worker,
633                )
634            })
635            .await
636            .expect("the spawning task panicked")
637        });
638
639        runtime.block_on(async {
640            // `block_in_place` lets another thread take over the worker's
641            // async work. A short call can reclaim that work before the
642            // replacement starts, so retry until a probe sees a new thread.
643            for _ in 0..100 {
644                tokio::spawn(async {
645                    tokio::task::block_in_place(|| thread::sleep(Duration::from_millis(10)));
646                })
647                .await
648                .expect("the blocking task panicked");
649
650                // Run the probe as a task too, so its ID belongs to the
651                // current worker rather than the thread driving `block_on`.
652                let current = tokio::spawn(async { gettid() })
653                    .await
654                    .expect("the probe task panicked");
655                if current != worker {
656                    return;
657                }
658            }
659            panic!("the worker role never left thread {worker}");
660        });
661
662        // The role has moved, but the old OS thread may still be alive in
663        // the blocking pool. Wait for its /proc entry to disappear before
664        // checking the child; fail if the thread never actually exits.
665        let deadline = Instant::now() + EXIT_WAIT;
666        while fs::metadata(format!("/proc/self/task/{worker}")).is_ok() {
667            assert!(Instant::now() < deadline, "thread {worker} never retired");
668            thread::sleep(Duration::from_millis(50));
669        }
670        (child, runtime)
671    }

673    // CPS-1: retiring the requesting tokio worker must not kill a child
674    // forked by the process-lifetime spawning thread.
675    #[test]
676    fn child_survives_its_worker_retiring_after_block_in_place() {
677        // Return only once the worker that requested the spawn has exited.
678        // `spawn_child` forks on the shared OS thread instead of that worker.
679        let (mut child, runtime) = spawn_then_retire_worker(spawn_child);
680        runtime.block_on(async {
681            // The runtime remains alive, but the original worker is gone.
682            // Its retirement must not trigger this child's death signal.
683            assert!(
684                keeps_running(&mut child, SURVIVAL_WAIT).await,
685                "the child died when its spawning worker retired"
686            );
687
688            // Kill and reap the test child while its runtime is still alive.
689            kill(child).await;
690        });
691    }

693    // CPS-1: comparison case. Direct spawning on a tokio worker must
694    // reproduce the bug: worker retirement kills the death-signal child.
695    #[test]
696    fn child_forked_on_a_worker_dies_when_it_retires() {
697        // Use ordinary `spawn`, making that worker the child's forking
698        // thread. The helper waits for the worker's OS thread to exit.
699        let (mut child, runtime) = spawn_then_retire_worker(|mut command| command.spawn());
700        runtime.block_on(async {
701            // The parent process and runtime are alive. The test passes
702            // only if the worker's exit nevertheless kills the child.
703            let status = tokio::time::timeout(EXIT_WAIT, child.wait())
704                .await
705                .expect("the death signal killed the child")
706                .expect("reap the child");
707            assert_eq!(status.signal(), Some(libc::SIGKILL));
708        });
