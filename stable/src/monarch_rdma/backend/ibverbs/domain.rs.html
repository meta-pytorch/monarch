36/// Manages RDMA resources including context and protection domain.
37///
38/// # Fields
39///
40/// * `device_info`: Metadata for the device this PD is allocated on.
41/// * `domain_impl`: The backend [`IbvDomainImpl`] strategy for this PD. It may
42///   own FFI resources allocated against the PD, so it is declared before `pd`
43///   and thus dropped first — releasing those resources while the PD is still
44///   alive.
45/// * `pd`: The protection domain (and, through [`IbvPd`], the device context).
46///   Held in an `Arc` because the resources built against it — memory regions
47///   and queue pairs — each keep their own clone (via [`Self::pd`]) to hold the
48///   PD open for as long as they need it, independent of this domain.
49///
50/// `I` is the backend [`IbvDomainImpl`] strategy parameterizing per-PD
51/// behavior.
52///
53/// `IbvDomain` is not `Clone`: it owns the backend strategy's FFI resources.
54/// Hand out [`Self::pd`] clones to share the protection domain.
55pub struct IbvDomain<I: IbvDomainImpl> {
56    pub device_info: IbvDeviceInfo,
57    domain_impl: I,
58    pd: Arc<IbvPd>,
59}

61impl<I: IbvDomainImpl> std::fmt::Debug for IbvDomain<I> {
62    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
63        f.debug_struct("IbvDomain")
64            .field("context", &format!("{:p}", self.pd.context().as_ptr()))
65            .field("pd", &format!("{:p}", self.pd.as_ptr()))
66            .field("device_info", &self.device_info)
67            .field("domain_impl", &self.domain_impl)
68            .finish()
69    }
70}

72impl<I: IbvDomainImpl> IbvDomain<I> {
73    /// Creates an `IbvDomain` over an already-opened device `context`.
74    ///
75    /// Builds the backend [`IbvDomainImpl`] strategy `I` from `config`, then
76    /// allocates the protection domain against `context`. The PD is allocated
77    /// only *after* the strategy is built, so a panicking
78    /// [`IbvDomainImpl::new`] never leaks a PD.
79    ///
80    /// Note:
81    /// Host memory registration may use implicit ODP when configured, which
82    /// exposes a large virtual address range to the NIC instead of pinning only
83    /// the allocation.
84    ///
85    /// We currently assume a trusted, single-environment and are not enforcing finer-grained memory isolation
86    /// at this layer. We plan to investigate mitigations - such as memory windows or tighter registration
87    /// boundaries in future follow-ups.
88    ///
89    /// # Safety
90    ///
91    /// `context` must wrap a valid, live `ibv_context`: the `Arc<IbvContext>`
92    /// keeps it open for this call and for the returned domain's lifetime (the
93    /// PD allocated here is freed against that context on `Drop`).
94    ///
95    /// # Panics
96    ///
97    /// Panics if `context` is null.
98    ///
99    /// # Errors
100    ///
101    /// Returns an error if protection-domain allocation fails.
102    pub unsafe fn new(
103        context: Arc<IbvContext>,
104        device_info: IbvDeviceInfo,
105        config: &IbvConfig,
106    ) -> Result<Self, anyhow::Error> {
107        assert!(
108            !context.as_ptr().is_null(),
109            "IbvDomain::new requires a non-null ibv_context"
110        );
111        // Build the strategy first; the PD below is allocated only if this
112        // returns, so a panicking `IbvDomainImpl::new` never leaks a PD.
113        // SAFETY: per this function's contract `context` wraps a valid, live
114        // `ibv_context`, which is what `I::new` requires to query the device.
115        let domain_impl = unsafe { I::new(&context, &device_info, config) };
116        // SAFETY: per this function's contract `context` wraps a valid, live
117        // `ibv_context`, which `IbvPd::create` allocates the PD against. The PD
118        // takes ownership of `context`; the domain reaches it via `pd.context()`.
119        let pd = Arc::new(unsafe { IbvPd::create(context) }?);
120        Ok(Self {
121            device_info,
122            domain_impl,
123            pd,
124        })
125    }

127    /// Test-only constructor assembling a domain from parts, so unit tests can
128    /// fabricate a domain (typically with a null `pd` whose `Drop` is a no-op).
129    ///
130    /// # Safety
131    ///
132    /// `pd` must satisfy [`IbvPd`]'s validity contract.
133    #[cfg(test)]
134    pub(super) unsafe fn for_test(
135        pd: Arc<IbvPd>,
136        device_info: IbvDeviceInfo,
137        domain_impl: I,
138    ) -> Self {
139        Self {
140            device_info,
141            domain_impl,
142            pd,
143        }
144    }

204/// Per-backend strategy for a protection domain: how memory regions are
205/// registered and how queue pairs are built against the PD.
206///
207/// One strategy is constructed per domain via [`Self::new`], which
208/// inspects the device behind the context to decide backend-specific behavior
209/// up front, and is then stored in the [`IbvDomain`] it drives. The per-op
210/// methods are associated functions taking `&IbvDomain<Self>` and reach the
211/// strategy itself through [`IbvDomain::domain_impl`].
212pub trait IbvDomainImpl: std::fmt::Debug + Send + Sync + 'static + Sized {
213    /// The concrete queue-pair type built against this domain's PD.
214    type QueuePair: IbvQueuePair;
215
216    /// Build the strategy for the device behind `context` (whose queried
217    /// metadata is `device_info`), using `config` for any setup it performs.
218    ///
219    /// # Safety
220    ///
221    /// If `context` is non-null it must wrap a valid, live `ibv_context`;
222    /// implementations may query the device behind it.
223    unsafe fn new(context: &IbvContext, device_info: &IbvDeviceInfo, config: &IbvConfig) -> Self;
224
225    /// Access flags used when registering memory regions and creating queue
226    /// pairs on this domain.
227    fn access_flags(&self) -> i32;
228
229    /// Register `mem` against `domain`'s PD and return a view of the
230    /// resulting memory region.
231    ///
232    /// The default implementation covers host memory (`ibv_reg_mr`) and
233    /// the device-memory dmabuf path (`ibv_reg_dmabuf_mr`); backends
234    /// override to add hardware-specific registration, falling back to
235    /// this default for the cases they do not special-case.
236    ///
237    /// # Safety
238    ///
239    /// `domain`'s PD (`domain.as_ptr()`) must be null or a live protection
240    /// domain; `mem`'s backing memory must stay valid for the returned MR's
241    /// lifetime.
242    unsafe fn register_mr(
243        domain: &IbvDomain<Self>,
244        mem: &KeepaliveLocalMemory,
245    ) -> anyhow::Result<IbvMemoryRegionView> {
246        // SAFETY: `domain.as_ptr()` is null or a live PD (per this method's
247        // contract; `register_host_or_dmabuf_mr` errors on null), and the caller
248        // keeps `mem`'s backing memory valid for the MR's lifetime.
249        unsafe { register_host_or_dmabuf_mr(domain, mem) }
250    }
251
252    /// Create a queue pair against `domain`, reporting every completion on
253    /// `cq`. The default builds [`Self::QueuePair`] directly; backends override
254    /// to construct their own queue-pair type.
255    ///
256    /// One queue serves both the send and receive sides, because nothing posts
257    /// receive work requests on these queue pairs -- a separate receive queue
258    /// would only ever stay empty.
259    ///
260    /// # Safety
261    ///
262    /// `domain` must hold a null or live PD (`domain.as_ptr()`) and, with it, a
263    /// null or live device context, and `cq` must hold a live `ibv_cq` created
264    /// on that context: the queue pair is built against that PD and reports its
265    /// completions on that queue. A null PD yields `Err` rather than reaching
266    /// the driver.
267    unsafe fn create_queue_pair(
268        domain: &IbvDomain<Self>,
269        config: &IbvConfig,
270        cq: Arc<IbvCq>,
271    ) -> anyhow::Result<Self::QueuePair> {
272        // Reject a null PD (e.g. a test domain) up front, so a domain that
273        // cannot back a queue pair says so rather than reaching the driver.
274        if domain.as_ptr().is_null() {
275            anyhow::bail!("cannot create a queue pair on a null protection domain");
276        }
277
278        // SAFETY: `domain` holds a live PD (null was rejected above) and `cq` a
279        // live queue on its context, per this method's contract, which is what
280        // `IbvQueuePair::new` requires.
281        unsafe { Self::QueuePair::new(domain, config.clone(), Arc::clone(&cq), cq) }
282    }
283}

285/// Register host memory as a standard MR via `ibv_reg_mr`.
286///
287/// # Safety
288///
289/// If `pd` is non-null it must be a live protection domain whose context
290/// outlives this call. For an ordinary registration, `[addr, addr + size)` must
291/// name a host mapping that stays valid for the lifetime of the returned MR.
292/// For implicit ODP, pass address zero, `usize::MAX`, and
293/// `IBV_ACCESS_ON_DEMAND`.
294pub(super) unsafe fn register_host_mr(
295    pd: &Arc<IbvPd>,
296    addr: usize,
297    size: usize,
298    access_flags: i32,
299) -> anyhow::Result<IbvMr> {
300    if pd.as_ptr().is_null() {
301        anyhow::bail!("register_host_mr called with a null protection domain");
302    }
303    // SAFETY: `pd.as_ptr()` is non-null (checked above) and, per this function's
304    // contract, a live protection domain; `[addr, addr + size)` is a
305    // caller-guaranteed valid mapping. `ibv_reg_mr` returns null on failure,
306    // which we check before wrapping the pointer.
307    let mr =
308        unsafe { rdmaxcel_sys::ibv_reg_mr(pd.as_ptr(), addr as *mut c_void, size, access_flags) };
309    if mr.is_null() {
310        anyhow::bail!("failed to register host MR");
311    }
312    // SAFETY: `mr` is non-null (checked above) and freshly registered against
313    // `pd`.
314    Ok(unsafe { IbvMr::from_raw(mr, pd.clone()) })
315}

317/// Register exactly `[addr, addr + size)` of device memory as a dmabuf MR via
318/// `ibv_reg_dmabuf_mr`, mapped at iova 0.
319///
320/// `cuMemGetHandleForAddressRange` requires both `addr` and `size` to be
321/// host-page aligned, so this errors if either is not.
322///
323/// # Safety
324///
325/// If `pd` is non-null it must be a live protection domain whose context
326/// outlives this call. `[addr, addr + size)` must name device memory that
327/// stays valid for the lifetime of the returned MR.
328pub(super) unsafe fn register_dmabuf_range(
329    pd: &Arc<IbvPd>,
330    addr: usize,
331    size: usize,
332    access_flags: i32,
333) -> anyhow::Result<IbvMr> {
334    if pd.as_ptr().is_null() {
335        anyhow::bail!("register_dmabuf_range called with a null protection domain");
336    }
337    // SAFETY: `sysconf` reads a process-global parameter and takes no pointers.
338    let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
339    anyhow::ensure!(
340        page_size > 0,
341        "sysconf(_SC_PAGESIZE) failed: {}",
342        Error::last_os_error()
343    );
344    let host_page_size = page_size as usize;
345    anyhow::ensure!(
346        addr.is_multiple_of(host_page_size) && size.is_multiple_of(host_page_size),
347        "dmabuf range is not host-page aligned (addr: 0x{addr:x}, size: {size}, page size: {host_page_size})"
348    );
349
350    let mut fd: i32 = -1;
351    // SAFETY: `rdmaxcel_cuMemGetHandleForAddressRange` writes the dmabuf fd for
352    // `[addr, addr + size)` into `fd` and touches no Rust memory beyond the
353    // `&mut fd` out-param; it reports failure via its return code, checked next.
354    let cu_err = unsafe {
355        rdmaxcel_sys::rdmaxcel_cuMemGetHandleForAddressRange(
356            &mut fd,
357            addr as rdmaxcel_sys::CUdeviceptr,
358            size,
359            rdmaxcel_sys::CU_MEM_RANGE_HANDLE_TYPE_DMA_BUF_FD,
360            0,
361        )
362    };
363    if cu_err != rdmaxcel_sys::CUDA_SUCCESS || fd < 0 {
364        anyhow::bail!(
365            "failed to get dmabuf handle for CUDA memory (addr: 0x{:x}, size: {}, cu_err: {}, fd: {})",
366            addr,
367            size,
368            cu_err,
369            fd
370        );
371    }
372    // SAFETY: `fd >= 0` (checked above) is a fresh dmabuf descriptor we
373    // exclusively own; wrapping it in `OwnedFd` closes it on drop and keeps it
374    // open across the registration below.
375    let fd = unsafe { OwnedFd::from_raw_fd(fd) };
376    // SAFETY: `pd.as_ptr()` is a non-null protection domain (checked above)
377    // belonging to a live context; `fd` is a valid dmabuf descriptor kept open
378    // by the `OwnedFd` across this call; `size` matches the range queried above.
379    // `ibv_reg_dmabuf_mr` returns null on failure, which we check.
380    let mr = unsafe {
381        rdmaxcel_sys::ibv_reg_dmabuf_mr(pd.as_ptr(), 0, size, 0, fd.as_raw_fd(), access_flags)
382    };
383    if mr.is_null() {
384        anyhow::bail!("failed to register dmabuf MR");
385    }
386    // SAFETY: `mr` is non-null (checked above) and freshly registered against
387    // `pd`.
388    Ok(unsafe { IbvMr::from_raw(mr, pd.clone()) })
389}

391/// Base address and size of the CUDA allocation containing `addr`, as reported
392/// by `cuMemGetAddressRange`.
393///
394/// `cuMemGetAddressRange` resolves `addr` in the current CUDA context, so this
395/// returns an error unless `addr` is a device pointer whose owning context is
396/// current (see [`crate::local_memory::set_ctx_for_ptr`]).
397fn cuda_alloc_range(addr: usize) -> anyhow::Result<(usize, usize)> {
398    let mut base: rdmaxcel_sys::CUdeviceptr = unsafe { std::mem::zeroed() };
399    let mut size: usize = 0;
400    // SAFETY: the out-params point to local variables and `addr` is passed by
401    // value as an opaque device address (never dereferenced); the call writes
402    // only through the out-params and reports a bad address or context via its
403    // return code, checked next.
404    let cu_err = unsafe {
405        rdmaxcel_sys::rdmaxcel_cuMemGetAddressRange(
406            &mut base,
407            &mut size,
408            addr as rdmaxcel_sys::CUdeviceptr,
409        )
410    };
411    if cu_err != rdmaxcel_sys::CUDA_SUCCESS {
412        anyhow::bail!(
413            "failed to get address range for CUDA memory (addr: 0x{:x}, cu_err: {})",
414            addr,
415            cu_err
416        );
417    }
418    Ok((base as usize, size))
419}

421/// Register the CUDA device memory backing `[addr, addr + size)` as a single
422/// dmabuf MR, mapped at iova 0. The returned `usize` is the offset of `addr`
423/// within that MR.
424///
425/// The MR covers, in full, every allocation the requested range touches: it
426/// spans from the base of the allocation containing `addr` to the end of the
427/// allocation containing the range's last byte. A range may extend past the
428/// allocation holding its first byte, so `cuMemGetAddressRange` — which reports
429/// the bounds of the allocation containing a given pointer — is queried at both
430/// ends, and the MR covers everything between them.
431///
432/// Allocation bounds are granularity-aligned, hence host-page aligned, so the
433/// base and size handed to [`register_dmabuf_range`] pass its alignment check.
434///
435/// # Safety
436///
437/// If `pd` is non-null it must be a live protection domain whose context
438/// outlives this call. `addr` must belong to CUDA device memory, and
439/// `[addr, addr + size)` must stay valid for the lifetime of the returned MR.
440pub(super) unsafe fn register_dmabuf_mr(
441    pd: &Arc<IbvPd>,
442    addr: usize,
443    size: usize,
444    access_flags: i32,
445) -> anyhow::Result<(IbvMr, usize)> {
446    anyhow::ensure!(
447        size > 0,
448        "register_dmabuf_mr called with a zero-size range (addr: 0x{addr:x})"
449    );
450
451    // `cuMemGetAddressRange` resolves the pointer in the *current* CUDA context,
452    // so make the pointer's own device context current first. Without this, in a
453    // multi-GPU process it fails with `CUDA_ERROR_NOT_FOUND` whenever the active
454    // context belongs to a different device than `addr`.
455    // SAFETY: this path is only taken for device memory (the
456    // `MemoryLocation::Gpu` arm in `register_host_or_dmabuf_mr`), so `addr` is a
457    // valid CUDA device pointer as `set_ctx_for_ptr` requires. The guard
458    // restores the prior context on drop.
459    let _ctx_guard = unsafe { crate::local_memory::set_ctx_for_ptr(addr)? };
460
461    let (base, base_size) = cuda_alloc_range(addr)?;
462    let last = addr.checked_add(size - 1).with_context(|| {
463        format!(
464            "final byte of range [{}, {} + {}) would overflow max usize",
465            addr, addr, size
466        )
467    })?;
468    // Extend to the end of the allocation holding the range's last byte. When
469    // the range lies within the first allocation, that allocation is the end;
470    // otherwise a second lookup resolves the allocation the last byte falls in.
471    let end = if last - base < base_size {
472        base + base_size
473    } else {
474        let (last_base, last_size) = cuda_alloc_range(last)?;
475        last_base + last_size
476    };
477
478    // SAFETY: forwards this function's contract; `register_dmabuf_range` checks
479    // that `base` and `end - base` are host-page aligned.
480    let mr = unsafe { register_dmabuf_range(pd, base, end - base, access_flags)? };
481    Ok((mr, addr - base))
482}

484/// Default MR registration: host memory via [`register_host_mr`], device
485/// memory via [`register_dmabuf_mr`]. Shared by the [`IbvDomainImpl`]
486/// default `register_mr` and by backends as the fallback for memory they
487/// do not special-case.
488///
489/// # Safety
490///
491/// `domain.as_ptr()` must be null or a live protection domain whose context
492/// outlives this call (a null PD yields an error). `mem`'s
493/// `[addr, addr + size)` must stay valid for the lifetime of the returned
494/// view's MR — the MR keepalive maintains the `ibv_mr` but does not keep the
495/// backing memory mapped.
496pub(super) unsafe fn register_host_or_dmabuf_mr<I: IbvDomainImpl>(
497    domain: &IbvDomain<I>,
498    mem: &KeepaliveLocalMemory,
499) -> anyhow::Result<IbvMemoryRegionView> {
500    let addr = mem.addr();
501    let size = mem.size();
502    let access_flags = domain.access_flags();
503    // `mr_offset` is the offset of `addr` within the MR. For device memory the
504    // MR covers every allocation the range spans, so the requested range may start
505    // partway in; for host memory the MR is the requested range itself, so the
506    // offset is 0.
507    // SAFETY: per this function's contract `domain.as_ptr()` is null or a live
508    // PD (the helpers error on null), and `[addr, addr + size)` stays valid for
509    // the returned MR's lifetime.
510    let (mr, mr_offset) = unsafe {
511        match mem.location() {
512            MemoryLocation::Gpu(_) => register_dmabuf_mr(domain.pd(), addr, size, access_flags)?,
513            MemoryLocation::Cpu(_) => (register_host_mr(domain.pd(), addr, size, access_flags)?, 0),
514        }
515    };
516
517    // SAFETY: `mr` wraps a non-null, freshly-registered `ibv_mr` —
518    // `register_dmabuf_mr`/`register_host_mr` only return `Ok` with one — so
519    // reading its `addr`/`lkey`/`rkey` here is sound.
520    let (mr_addr, lkey, rkey) = unsafe {
521        let p = mr.as_ptr();
522        ((*p).addr as usize, (*p).lkey, (*p).rkey)
523    };
524    // The view addresses the requested sub-range, which sits at `mr_offset`
525    // within the MR's zero-based address space.
526    let rdma_addr = mr_addr + mr_offset;
527    let device_name = domain.device_info().name().to_string();
528    // The `IbvMr` guard owns the MR (deregistered on its `Drop`) and anchors the
529    // PD past that deregistration; it coerces to `Arc<dyn IbvMemoryRegionKeepalive>`.
530    let guard = Arc::new(mr);
531    Ok(IbvMemoryRegionView::new(
532        addr,
533        rdma_addr,
534        size,
535        lkey,
536        rkey,
537        device_name,
538        guard,
539    ))
540}

558    /// Open an [`IbvDomain`] on the NIC mapped to CUDA `device`. These tests
559    /// require a GPU with a mapped RDMA NIC, so a missing device, missing NIC, or
560    /// open/creation failure panics. The returned domain owns its context, so it
561    /// (and its PD) stays valid after the local [`IbvDevice`] drops.
562    fn open_domain_for_cuda_device(device: i32) -> IbvDomain<MlxDomain> {
563        // Ranking NICs against a CUDA ordinal asks the driver for the GPU's PCI
564        // address, and `device_selection` deliberately never loads or initializes
565        // the driver itself. Do it here: nothing else in this module touches CUDA
566        // before this point, so the ranking below would otherwise fail with
567        // `CUDA_ERROR_NOT_INITIALIZED`.
568        assert!(
569            cuda_device_count() > device,
570            "CUDA device {device} is required by this test",
571        );
572        let ordinal_to_nics = get_cuda_device_to_ibv_devices::<MlxDevice>()
573            .expect("the CUDA driver is initialized above");
574        let nic = ordinal_to_nics
575            .get(device as usize)
576            .and_then(|nics| nics.first())
577            .expect("CUDA device should map to RDMA NIC")
578            .name()
579            .clone();
580        let mut config = IbvConfig::default();
581        MlxDevice::apply_config_defaults(&mut config);
582        let dev =
583            IbvDevice::<MlxDevice>::try_open(&nic, config.clone()).expect("mapped NIC should open");
584        // SAFETY: `dev.context()` wraps the live `ibv_context` opened above; the
585        // returned `Arc<IbvContext>` keeps it open for the new domain's lifetime.
586        unsafe { IbvDomain::new(dev.context(), dev.device_info().clone(), &config) }
587            .expect("domain creation should succeed")
588    }

590    /// A 4 MiB allocation, fully committed (reserved == committed) so it is
591    /// entirely mapped and `cuMemGetAddressRange` reports the whole extent.
592    fn committed_allocation() -> crate::backend::cuda_test_utils::CudaAllocation {
593        CudaAllocator::get().allocate(0, 4 * 1024 * 1024, 4 * 1024 * 1024)
594    }

596    /// `addr` (iova) and `length` of an `ibv_mr`.
597    ///
598    /// # Safety
599    ///
600    /// `mr` must be a live MR (e.g. owned by a not-yet-dropped [`IbvMr`]).
601    unsafe fn mr_extent(mr: *mut rdmaxcel_sys::ibv_mr) -> (usize, usize) {
602        // SAFETY: per this function's contract `mr` is a live MR.
603        unsafe { ((*mr).addr as usize, (*mr).length) }
604    }

606    // `register_dmabuf_mr` registers the whole enclosing allocation and reports
607    // the requested address's offset within it — even when that address is not
608    // host-page aligned, which is the case the bug fix targets.
609    #[test]
610    fn register_dmabuf_mr_covers_whole_allocation() {
611        let domain = open_domain_for_cuda_device(0);
612        let pd = domain.pd();
613        let access = domain.access_flags();
614        let alloc = committed_allocation();
615        let alloc_size = alloc.size();
616
617        // Requesting the whole allocation: offset 0, MR spans it at iova 0. `mr`
618        // deregisters on drop, after the assertions below.
619        // SAFETY: `pd` is a live PD; `alloc.ptr()` is a live CUDA allocation kept
620        // mapped by `alloc` for the MR's lifetime.
621        let (mr, offset) =
622            unsafe { register_dmabuf_mr(pd, alloc.ptr(), alloc_size, access) }.unwrap();
623        // SAFETY: `mr` owns a live MR and has not been dropped.
624        let (iova, length) = unsafe { mr_extent(mr.as_ptr()) };
625        assert_eq!(offset, 0, "base address sits at offset 0");
626        assert_eq!(iova, 0, "dmabuf MR is mapped at iova 0");
627        assert_eq!(length, alloc_size, "MR covers the whole allocation");
628
629        // At an unaligned interior address, with a range that stays within the
630        // allocation: the MR still spans the whole allocation and the offset
631        // locates the requested address.
632        let unaligned: usize = 257;
633        assert!(!unaligned.is_multiple_of(host_page_size()));
634        // SAFETY: as above; `alloc.ptr() + unaligned` is inside the allocation.
635        let (mr, offset) =
636            unsafe { register_dmabuf_mr(pd, alloc.ptr() + unaligned, host_page_size(), access) }
637                .unwrap();
638        // SAFETY: `mr` owns a live MR and has not been dropped.
639        let (iova, length) = unsafe { mr_extent(mr.as_ptr()) };
640        assert_eq!(offset, unaligned, "offset locates the requested address");
641        assert_eq!(iova, 0, "dmabuf MR is mapped at iova 0");
642        assert_eq!(length, alloc_size, "MR covers the whole allocation");
643    }

645    // A range crossing the mapping boundaries of an expandable segment yields a
646    // single MR covering every mapping it spans, in full — and no further. This
647    // is the case whole-allocation registration cannot handle: `addr` and the
648    // range's last byte live in different mappings.
649    #[test]
650    fn register_dmabuf_mr_spans_multiple_mappings() {
651        let domain = open_domain_for_cuda_device(0);
652        let pd = domain.pd();
653        let access = domain.access_flags();
654
655        // Reserve generously, commit one chunk, then expand three times so the
656        // segment is backed by four separate mappings at offsets 0, g, 2g, 3g.
657        let alloc = CudaAllocator::get().allocate(0, 128 * 1024 * 1024, 1);
658        let g = alloc.granularity();
659        for _ in 0..3 {
660            alloc
661                .expand(g)
662                .expect("expand stays within the reserved range");
663        }
664        assert_eq!(alloc.size(), 4 * g, "four mappings of one granularity each");
665
666        // Start at an unaligned interior address in the first mapping and run
667        // into the third: the range straddles the 0->1 and 1->2 boundaries but
668        // stops short of the fourth mapping.
669        let start_offset: usize = 257;
670        assert!(!start_offset.is_multiple_of(host_page_size()));
671        let size = 2 * g + 100;
672        let last_offset = start_offset + size - 1;
673        assert!(
674            (2 * g..3 * g).contains(&last_offset),
675            "range must end inside the third mapping"
676        );
677
678        // SAFETY: `pd` is a live PD; `alloc` keeps all four mappings resident for
679        // the MR's lifetime (it outlives `mr`, which drops first).
680        let (mr, offset) =
681            unsafe { register_dmabuf_mr(pd, alloc.ptr() + start_offset, size, access) }.unwrap();
682        // SAFETY: `mr` owns a live MR and has not been dropped.
683        let (iova, length) = unsafe { mr_extent(mr.as_ptr()) };
684        assert_eq!(offset, start_offset, "offset locates the requested address");
685        assert_eq!(iova, 0, "dmabuf MR is mapped at iova 0");
686        assert_eq!(
687            length,
688            3 * g,
689            "MR covers the three spanned mappings in full, not the fourth"
690        );
691    }

694    // page-aligned sub-range of the allocation here, not the whole thing.

767    // The full `register_host_or_dmabuf_mr` path over device memory covering the
768    // whole allocation: the view reports the allocation's address and size, and
769    // an `rdma_addr` of 0 (the request sits at the MR's base).
770    #[test]
771    fn register_host_or_dmabuf_mr_covers_whole_allocation() {
772        let domain = open_domain_for_cuda_device(0);
773        let alloc = committed_allocation();
774        let mem = alloc.keepalive_slice(0, alloc.size());
775
776        // SAFETY: `domain`'s PD is live; `mem` keeps the allocation mapped for
777        // the view's MR lifetime.
778        let view = unsafe { register_host_or_dmabuf_mr(&domain, &mem) }.unwrap();
779        // Tie the view's lifetime to the allocation's lifetime so that the safety contract
780        // above holds.
781        mem.install_mr::<MlxDevice>(view.clone())
782            .expect("the region has no registration on this device yet");
783        assert_eq!(view.virtual_addr, alloc.ptr());
784        assert_eq!(view.size, alloc.size());
785        assert_eq!(view.rdma_addr, 0, "whole allocation starts at MR offset 0");
786    }
