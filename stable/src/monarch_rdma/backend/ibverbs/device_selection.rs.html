34/// What an [`IbvConfig`](super::primitives::IbvConfig) targets: a memory
35/// location (whose best NIC is auto-selected) or an explicit NIC by name.
36#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
37pub enum IbvDeviceTarget {
38    /// Auto-select the best NIC for a CPU/GPU memory location.
39    MemoryLocation(MemoryLocation),
40    /// Use the NIC with this exact device name (e.g. `"mlx5_0"`).
41    Nic(String),
42}
43
44impl IbvDeviceTarget {
45    /// Target the best NIC for CPU memory on NUMA node `numa`.
46    pub fn cpu(numa: u32) -> Self {
47        Self::MemoryLocation(MemoryLocation::Cpu(Some(numa)))
48    }
49
50    /// Target the best NIC for GPU memory on CUDA ordinal `ordinal`.
51    pub fn gpu(ordinal: u32) -> Self {
52        Self::MemoryLocation(MemoryLocation::Gpu(Some(ordinal)))
53    }
54
55    /// Target the NIC with the given device name.
56    pub fn nic(name: impl Into<String>) -> Self {
57        Self::Nic(name.into())
58    }
59}

61impl FromStr for IbvDeviceTarget {
62    type Err = anyhow::Error;
63
64    fn from_str(spec: &str) -> Result<Self> {
65        let (kind, value) = spec.split_once(':').with_context(|| {
66            format!(
67                "ibverbs target {spec:?} must be `cpu:<numa>`, `gpu:<ordinal>`, or `nic:<name>`"
68            )
69        })?;
70
71        let parse_index = |what: &str| -> Result<u32> {
72            value
73                .parse()
74                .with_context(|| format!("{what} {value:?} must be an integer in 0..={}", u32::MAX))
75        };
76
77        match kind {
78            "cpu" => Ok(Self::cpu(parse_index("cpu target NUMA node")?)),
79            "gpu" => Ok(Self::gpu(parse_index("gpu target ordinal")?)),
80            "nic" => {
81                anyhow::ensure!(
82                    !value.is_empty(),
83                    "nic target must name a device, for example `nic:mlx5_0`"
84                );
85                Ok(Self::nic(value))
86            }
87            other => anyhow::bail!(
88                "unknown ibverbs target kind {other:?}; expected `cpu`, `gpu`, or `nic`"
89            ),
90        }
91    }
92}

111/// Which of a peer's NICs a local NIC is allowed to pair with for a transfer.
112///
113/// Two NICs both being up does not mean the fabric carries traffic between
114/// them. For example, some fabrics may be organized into planes, where NICs
115/// can only talk to other NICs in the same plane. Instead of trying to add
116/// complex, dynamic network topology detection into monarch, it's much easier
117/// to simply let the user tell us the topology of their network. This is
118/// configurable using the config attr.
119/// [`RDMA_PEER_DEVICE_AFFINITY`](crate::config::RDMA_PEER_DEVICE_AFFINITY).
120#[derive(Debug, Clone, PartialEq, Eq)]
121pub enum PeerDeviceAffinityPolicy {
122    /// Every local NIC reaches every peer NIC.
123    Any,
124    /// A local NIC reaches a peer NIC only if the two carry the same device
125    /// name.
126    MatchName,
127    /// A local NIC reaches a peer NIC only where some group names both. A NIC
128    /// no group names reaches nothing. There may be any number of groups, of
129    /// any size, and they are disjoint, so a NIC belongs to at most one.
130    Groups(Vec<BTreeSet<String>>),
131}
132
133impl PeerDeviceAffinityPolicy {
134    /// Pair each device in `local` with at most one device in `remote`.
135    ///
136    /// Entry `i` is the index in `remote` that `local[i]` should use, or `None`
137    /// when it has no partner: no NIC appears in two pairs, so the shorter list
138    /// bounds how many pairs there are, and the policy can rule out the rest.
139    ///
140    /// Same-named NICs pair first, no matter what the policy says.
141    /// Whatever is left over pairs subject to the policy, taking the
142    /// first NIC in `remote` it can still reach.
143    ///
144    /// The leftovers are matched greedily in order, so an earlier device can
145    /// take the only partner a later one could have used. Maximizing the number
146    /// of pairs is likely not worth a matching algorithm here.
147    pub fn pairs(&self, local: &[String], remote: &[String]) -> Vec<Option<usize>> {
148        let mut pairs: Vec<Option<usize>> = vec![None; local.len()];
149        let mut taken = vec![false; remote.len()];
150
151        for (i, local_device) in local.iter().enumerate() {
152            let same_name = remote
153                .iter()
154                .position(|remote_device| remote_device == local_device)
155                .filter(|j| !taken[*j]);
156            if let Some(j) = same_name {
157                pairs[i] = Some(j);
158                taken[j] = true;
159            }
160        }
161
162        for (i, local_device) in local.iter().enumerate() {
163            if pairs[i].is_some() {
164                continue;
165            }
166            let candidate = remote
167                .iter()
168                .enumerate()
169                .find(|(j, remote_device)| !taken[*j] && self.can_pair(local_device, remote_device))
170                .map(|(j, _)| j);
171            if let Some(j) = candidate {
172                pairs[i] = Some(j);
173                taken[j] = true;
174            }
175        }
176        pairs
177    }
178
179    /// The devices to serve one buffer from: at most `max` of `devices`, or all
180    /// of them when `max` is `None`.
181    ///
182    /// Which devices are dropped is decided by position, so the caller decides
183    /// what "first" means by how it orders `devices`.
184    ///
185    /// Selection logic depends on the policy:
186    ///
187    /// - [`Self::Any`]: any two NICs can talk to each other, so buffers should
188    ///   spread across tied devices rather than always funneling through
189    ///   the first. `choose` picks a random offset into `devices`, then
190    ///   takes up to `max` starting from that offset, wrapping around if
191    ///   necessary.
192    /// - [`Self::MatchName`]: takes the first `max` NICs in the order in which
193    ///   they appear in devices. Callers should pass `devices` in a known,
194    ///   consistent order so that different processes agree on the choice.
195    /// - [`Self::Groups`]: takes one device from every group `devices` touches
196    ///   before taking a second from any of them, so a peer served by any of
197    ///   those groups still finds a partner here. A device no group names comes
198    ///   last: it reaches no peer device, so it is worth registering only once
199    ///   nothing else fills `max`.
200    pub fn choose(&self, devices: &[String], max: Option<NonZeroUsize>) -> Vec<String> {
201        if devices.is_empty() {
202            return Vec::new();
203        }
204        let limit = max
205            .map_or(devices.len(), NonZeroUsize::get)
206            .min(devices.len());
207        match self {
208            Self::Any => devices
209                .iter()
210                .cycle()
211                .skip(rand::random_range(0..devices.len()))
212                .take(limit)
213                .cloned()
214                .collect(),
215            Self::MatchName => devices.iter().take(limit).cloned().collect(),
216            Self::Groups(groups) => Self::choose_across_groups(groups, devices, limit),
217        }
218    }
219
220    /// [`Self::Groups`]'s arm of [`Self::choose`].
221    fn choose_across_groups(
222        groups: &[BTreeSet<String>],
223        devices: &[String],
224        limit: usize,
225    ) -> Vec<String> {
226        // One bucket per group `devices` touches, created on first touch and
227        // filled in `devices` order.
228        let mut buckets: Vec<Vec<&String>> = Vec::new();
229        let mut bucket_of: Vec<Option<usize>> = vec![None; groups.len()];
230        let mut ungrouped: Vec<&String> = Vec::new();
231        for device in devices {
232            let Some(group) = groups.iter().position(|names| names.contains(device)) else {
233                ungrouped.push(device);
234                continue;
235            };
236            match bucket_of[group] {
237                Some(bucket) => buckets[bucket].push(device),
238                None => {
239                    bucket_of[group] = Some(buckets.len());
240                    buckets.push(vec![device]);
241                }
242            }
243        }
244        if buckets.len() > limit {
245            warn_uncovered_groups(limit, buckets.len(), &buckets[limit..]);
246        }
247
248        let rounds = buckets.iter().map(Vec::len).max().unwrap_or(0);
249        let mut ordered: Vec<&String> = Vec::with_capacity(devices.len());
250        for round in 0..rounds {
251            ordered.extend(
252                buckets
253                    .iter()
254                    .filter_map(|bucket| bucket.get(round))
255                    .copied(),
256            );
257        }
258        ordered.extend(ungrouped);
259        ordered.into_iter().take(limit).cloned().collect()
260    }
261
262    /// Whether a transfer may run between `local` and `remote`.
263    fn can_pair(&self, local: &str, remote: &str) -> bool {
264        match self {
265            Self::Any => true,
266            Self::MatchName => local == remote,
267            Self::Groups(groups) => groups
268                .iter()
269                .any(|group| group.contains(local) && group.contains(remote)),
270        }
271    }
272}

274impl FromStr for PeerDeviceAffinityPolicy {
275    type Err = anyhow::Error;
276
277    fn from_str(spec: &str) -> Result<Self> {
278        let spec = spec.trim();
279        match spec {
280            "" | "any" => Ok(Self::Any),
281            "match_name" => Ok(Self::MatchName),
282            _ => {
283                let groups = spec.strip_prefix("groups:").with_context(|| {
284                    format!(
285                        "peer device affinity {spec:?} must be `any`, `match_name`, or `groups:` \
286                         followed by `|`-separated groups of comma-separated device names"
287                    )
288                })?;
289                let groups = groups
290                    .split('|')
291                    .map(|group| {
292                        let names: BTreeSet<String> = group
293                            .split(',')
294                            .map(str::trim)
295                            .filter(|name| !name.is_empty())
296                            .map(str::to_owned)
297                            .collect();
298                        anyhow::ensure!(
299                            !names.is_empty(),
300                            "peer device affinity {spec:?} has a group naming no device",
301                        );
302                        Ok(names)
303                    })
304                    .collect::<Result<Vec<BTreeSet<String>>>>()?;
305
306                let mut claimed: BTreeSet<&String> = BTreeSet::new();
307                for name in groups.iter().flatten() {
308                    anyhow::ensure!(
309                        claimed.insert(name),
310                        "peer device affinity {spec:?} names {name:?} in more than one group; \
311                         groups must be disjoint",
312                    );
313                }
314                Ok(Self::Groups(groups))
315            }
316        }
317    }
318}

320/// Report, at most once per distinct set of left-out devices, that
321/// `rdma_max_nics_per_buffer` is too low for a buffer to reach every
322/// peer-affinity group that it could hypothetically reach.
323///
324/// `uncovered` holds the buckets [`PeerDeviceAffinityPolicy::choose`] had to
325/// drop, one per group it could not reach.
326fn warn_uncovered_groups(max: usize, groups: usize, uncovered: &[Vec<&String>]) {
327    static WARNED: LazyLock<DashSet<String>> = LazyLock::new(DashSet::new);
328    let mut left_out: Vec<&str> = uncovered
329        .iter()
330        .flatten()
331        .map(|device| device.as_str())
332        .collect();
333    left_out.sort();
334    // `insert` is true only for the first caller to claim this key, and it takes
335    // the shard lock, so exactly one of them warns.
336    if WARNED.insert(left_out.join(",")) {
337        tracing::warn!(
338            "rdma_max_nics_per_buffer is {max}, but this buffer's RDMA devices span {groups} \
339             peer-affinity groups; raise it to {groups} to also serve {left_out:?}, or else \
340             some peers may be unreachable.",
341        );
342    }
343}

358/// The PCI address of an RDMA NIC, resolved from its sysfs device link
359/// (`/sys/class/infiniband/<name>/device`).
360pub fn get_pci_address(device: &IbvDeviceInfo) -> Result<PCIAddress> {
361    let link = format!("/sys/class/infiniband/{}/device", device.name());
362    let resolved =
363        std::fs::canonicalize(&link).with_context(|| format!("resolving sysfs link {link}"))?;
364    resolved
365        .file_name()
366        .and_then(|name| name.to_str())
367        .and_then(PCIAddress::parse)
368        .with_context(|| format!("no PCI address in resolved path {resolved:?}"))
369}

371/// The NIC(s) of backend `I` with the best path to `location`, ranked by
372/// [`PciPath::is_better_than`] (most local, then highest port-capped
373/// bandwidth) and returning all that tie for best.
374///
375/// A NIC that cannot be ranked -- no ACTIVE port, unresolvable PCI address -- is
376/// logged and skipped, so a host with no usable NIC yields an empty list rather
377/// than an error. Errors are reserved for a memory location that cannot be
378/// resolved at all, which no NIC could have compensated for.
379///
380/// A NIC's path bandwidth is the lesser of its PCIe-chain bottleneck and its
381/// RDMA port speed. Results are cached per `(backend, location)`, empty ones
382/// included, since the PCI/NUMA topology is fixed for the process lifetime. A
383/// host whose links are all down at the first call therefore keeps an empty
384/// answer for the rest of the process.
385///
386/// The NICs come back in lexicographic order by name, per
387/// [`compute_optimal_ibv_devices`]; the cache hands back the same order.
388/// Callers rely on it to agree with their peers on which of several tied NICs to
389/// use without exchanging anything.
390pub fn select_optimal_ibv_devices<I: IbvDeviceImpl>(
391    location: MemoryLocation,
392) -> Result<Vec<IbvDeviceInfo>> {
393    static CACHE: LazyLock<DashMap<(&'static str, MemoryLocation), Vec<IbvDeviceInfo>>> =
394        LazyLock::new(DashMap::new);
395    let key = (I::typename(), location);
396    if let Some(cached) = CACHE.get(&key) {
397        return Ok(cached.value().clone());
398    }
399    let result = compute_optimal_ibv_devices::<I>(location)?;
400    CACHE.insert(key, result.clone());
401    Ok(result)
402}
403
404/// Uncached core of [`select_optimal_ibv_devices`], returning the tied-best NICs
405/// in lexicographic order by device name.
406///
407/// That order is part of the contract rather than an artifact of how the devices
408/// were enumerated: two processes ranking the same [`MemoryLocation`] over the
409/// same NICs produce the same sequence, so each can pick the same element — the
410/// first, say — and know the other picked its counterpart.
411fn compute_optimal_ibv_devices<I: IbvDeviceImpl>(
412    location: MemoryLocation,
413) -> Result<Vec<IbvDeviceInfo>> {
414    // Resolve the GPU side once, before ranking NICs. An unresolvable GPU
415    // location is a fact about the request, not a reason to skip each NIC in
416    // turn.
417    let gpus = match location {
418        MemoryLocation::Cpu(_) => Vec::new(),
419        MemoryLocation::Gpu(ordinal) => gpu_pci_addresses(ordinal).with_context(|| {
420            format!(
421                "cannot rank RDMA NICs for {location:?}; if the CUDA driver is not initialized \
422                 in this process, initialize it first -- allocate a tensor on the device, call \
423                 torch.cuda.init(), or use your method of choice."
424            )
425        })?,
426    };
427
428    let mut best: Option<PciPath> = None;
429    let mut devices: Vec<IbvDeviceInfo> = Vec::new();
430    for nic in IbvDevice::<I>::list() {
431        // A NIC that cannot be ranked is not a failure of the request; drop it
432        // and say why, rather than failing the whole ranking or skipping in
433        // silence.
434        let path = match nic_path(&nic, location, &gpus) {
435            Ok(path) => path,
436            Err(error) => {
437                tracing::warn!("excluding RDMA device {}: {error:#}", nic.name());
438                continue;
439            }
440        };
441        match best {
442            Some(current) if current.is_better_than(&path) => continue,
443            Some(current) if path.is_better_than(&current) => devices.clear(),
444            _ => {}
445        }
446        best = Some(path);
447        devices.push(nic);
448    }
449    devices.sort_by(|a, b| a.name().cmp(b.name()));
450    Ok(devices)
451}

453/// The PCI addresses a [`MemoryLocation::Gpu`] names: just the one for a specific
454/// ordinal, or every visible device in ordinal order when unspecified.
455fn gpu_pci_addresses(ordinal: Option<u32>) -> Result<Vec<PCIAddress>> {
456    match ordinal {
457        Some(ordinal) => Ok(vec![cuda_pci_address(ordinal)?]),
458        None => (0..cuda_device_count()? as u32)
459            .map(cuda_pci_address)
460            .collect(),
461    }
462}

464/// The best [`PciPath`] from `location` to `nic`, capped by the NIC's RDMA port
465/// speed. Errors when `nic` cannot be ranked, naming the reason so the caller
466/// can report which device it dropped and why.
467///
468/// `gpus` holds the already-resolved GPU addresses for a
469/// [`MemoryLocation::Gpu`] location, and is empty for a CPU location.
470fn nic_path(nic: &IbvDeviceInfo, location: MemoryLocation, gpus: &[PCIAddress]) -> Result<PciPath> {
471    // A NIC with no ACTIVE port cannot carry traffic, so drop it from the
472    // candidate set. Leaving it in with a zero bandwidth is not equivalent:
473    // `is_better_than` compares `PathType` first, so a down NIC that happens to
474    // be PCIe-closer would outrank a working but more distant one.
475    let port_speed_mbytes_per_sec =
476        NonZeroU32::new(nic.port_speed_mbytes_per_sec()).context("no ACTIVE port")?;
477    let nic_addr = get_pci_address(nic)?;
478    let base = match location {
479        MemoryLocation::Cpu(numa) => cpu_path(&nic_addr, numa),
480        MemoryLocation::Gpu(_) => gpus
481            .iter()
482            .map(|gpu_addr| pci_path(gpu_addr, &nic_addr))
483            .reduce(|a, b| if b.is_better_than(&a) { b } else { a })
484            .context("no visible CUDA device to measure a path from")?,
485    };
486    Ok(cap_by_port_speed(base, port_speed_mbytes_per_sec))
487}
488
489/// Caps `path`'s bottleneck at the NIC's RDMA port speed, so a wide PCIe path
490/// to a slow NIC is not ranked as though the NIC could saturate it.
491fn cap_by_port_speed(path: PciPath, port_speed_mbytes_per_sec: NonZeroU32) -> PciPath {
492    PciPath {
493        bottleneck_mbytes_per_sec: path
494            .bottleneck_mbytes_per_sec
495            .min(port_speed_mbytes_per_sec.get()),
496        ..path
497    }
498}

500/// Resolves an [`IbvDeviceTarget`] to a single NIC of backend `I`: the
501/// named device for [`IbvDeviceTarget::Nic`], or the best NIC for a memory
502/// location. Both arms are scoped to backend `I`, so a name belonging to a
503/// different backend resolves to `Ok(None)`.
504///
505/// Where several NICs tie for a memory location, this takes the first, which by
506/// [`select_optimal_ibv_devices`]'s ordering is the lexicographically smallest
507/// name. Two processes pinning the same location over the same NICs therefore
508/// resolve to the same device.
509///
510/// `Ok(None)` means "no such device"; an error means the target could not be
511/// evaluated at all, which for a GPU location includes CUDA not being
512/// initialized in this process.
513pub fn resolve_target<I: IbvDeviceImpl>(target: &IbvDeviceTarget) -> Result<Option<IbvDeviceInfo>> {
514    match target {
515        IbvDeviceTarget::Nic(name) => Ok(IbvDevice::<I>::list()
516            .into_iter()
517            .find(|device| device.name() == name)),
518        IbvDeviceTarget::MemoryLocation(location) => {
519            Ok(select_optimal_ibv_devices::<I>(*location)?
520                .into_iter()
521                .next())
522        }
523    }
524}

526/// The NICs of backend `I` for each CUDA runtime ordinal, indexed by ordinal.
527///
528/// Each entry holds every NIC tied for the best path to that ordinal, in
529/// lexicographic order by name per [`select_optimal_ibv_devices`], so an entry
530/// is empty when no NIC could be ranked against its ordinal. Errors only when
531/// the ordinals themselves cannot be enumerated, which in practice means the
532/// CUDA driver is not initialized in this process.
533pub fn get_cuda_device_to_ibv_devices<I: IbvDeviceImpl>() -> Result<Vec<Vec<IbvDeviceInfo>>> {
534    (0..cuda_device_count()?)
535        .map(|ordinal| select_optimal_ibv_devices::<I>(MemoryLocation::Gpu(Some(ordinal as u32))))
536        .collect()
537}

618    #[test]
619    fn same_names_pair_before_anything_else() {
620        // `Any` permits every pairing, so only the name-first pass decides this
621        // one. Both sides lead with a NIC the other does not have, and the
622        // shared `mlx5_5` sits at a different index on each side: pairing by
623        // position would hand `mlx5_2` the peer's `mlx5_5` and leave the local
624        // `mlx5_5` with nothing.
625        let local = devices(["mlx5_0", "mlx5_2", "mlx5_5"]);
626        let remote = devices(["mlx5_3", "mlx5_5"]);
627        let pairs = PeerDeviceAffinityPolicy::Any.pairs(&local, &remote);
628        assert_eq!(
629            pairs,
630            vec![
631                // mlx5_0 takes what the name match left: the peer's mlx5_3.
632                Some(0),
633                // Nothing is left for mlx5_2.
634                None,
635                // The name match, claimed before either leftover was placed.
636                Some(1),
637            ],
638        );
639        assert_one_pair_per_device(&pairs, remote.len());
640    }

718    #[test]
719    fn each_group_is_reached_before_any_is_doubled_up() {
720        let policy: PeerDeviceAffinityPolicy =
721            "groups:mlx5_0,mlx5_1|mlx5_2,mlx5_3|mlx5_4,mlx5_5|mlx5_6,mlx5_7"
722                .parse()
723                .expect("group spec should parse");
724        let local = devices([
725            "mlx5_0", "mlx5_1", "mlx5_2", "mlx5_3", "mlx5_4", "mlx5_5", "mlx5_6", "mlx5_7",
726        ]);
727        // One device out of each group. A prefix of the ranking would have taken
728        // mlx5_0 through mlx5_3 and left the last two groups unreachable.
729        assert_eq!(
730            chosen(&policy, &local, 4),
731            devices(["mlx5_0", "mlx5_2", "mlx5_4", "mlx5_6"]),
732        );
733        // Once every group is covered the budget doubles up within them.
734        assert_eq!(
735            chosen(&policy, &local, 6),
736            ["mlx5_0", "mlx5_2", "mlx5_4", "mlx5_6", "mlx5_1", "mlx5_3"].map(str::to_owned),
737        );
738        // The best-ranked device is still taken first.
739        assert_eq!(chosen(&policy, &local, 1), devices(["mlx5_0"]));
740    }

782    fn any_starts_at_a_random_device_and_wraps() {
783        let policy = PeerDeviceAffinityPolicy::Any;
784        let ranked = devices(["mlx5_0", "mlx5_1", "mlx5_2", "mlx5_3"]);
785        // Every pair of devices adjacent in the ranking, wrapping around.
786        let windows: BTreeSet<Vec<String>> = [
787            ["mlx5_0", "mlx5_1"].map(str::to_owned).to_vec(),
788            ["mlx5_1", "mlx5_2"].map(str::to_owned).to_vec(),
789            ["mlx5_2", "mlx5_3"].map(str::to_owned).to_vec(),
790            ["mlx5_3", "mlx5_0"].map(str::to_owned).to_vec(),
791        ]
792        .into();
793        let mut seen: BTreeSet<Vec<String>> = BTreeSet::new();
794        for _ in 0..64 {
795            let picked = chosen(&policy, &ranked, 2);
796            assert!(
797                windows.contains(&picked),
798                "{picked:?} does not start somewhere in {ranked:?} and run on from there",
799            );
800            seen.insert(picked);
801        }
802        // 64 draws that all miss one of four starts would be a one-in-billions
803        // coincidence.
804        assert_eq!(seen, windows);
