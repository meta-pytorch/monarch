9//! A device's pool of completion queues, and the leases queue pairs hold on
10//! them.
11//!
12//! One completion queue serves several queue pairs, so that one poller can reap
13//! for all of them. How many share a completion queue is fixed by
14//! [`crate::config::RDMA_QPS_PER_CQ`], and each completion queue is sized to
15//! hold every sharer's work requests at once:
16//!
17//! ```text
18//! cq_entries = queue_pairs_per_cq * max_send_wr
19//! ```
20//!
21//! That is what lets completions be reaped without any runtime credit
22//! accounting: a queue pair caps its own outstanding work requests at
23//! `max_send_wr`, so it can never leave more than `max_send_wr` completion-queue
24//! entries unreaped, and the completion queue is sized for exactly that much per
25//! queue pair leasing it. Overflow -- which puts a completion queue into
26//! an unrecoverable error state -- is therefore structurally impossible rather
27//! than merely unlikely.
28//!
29//! One assumption holds that reasoning up, and breaking it means this sizing has
30//! to be revisited: nothing posts receive work requests on these queue pairs. A
31//! receive work request yields an entry too, but is bounded by `max_recv_wr`
32//! rather than `max_send_wr`, so it would consume capacity this sizing does not
33//! account for.
34

43/// Entries a completion queue needs in order to serve `queue_pairs` queue pairs,
44/// each able to have `max_send_wr` work requests outstanding.
45///
46/// Fails when the device cannot hold a completion queue that large.
47///
48/// `max_cqe` of 0 means the device's limit was never queried -- true only of a
49/// synthetic [`IbvDeviceInfo`] in a test -- and imposes no limit.
50pub(super) fn cq_entries_for(
51    queue_pairs: u32,
52    max_send_wr: u32,
53    max_cqe: i32,
54) -> anyhow::Result<i32> {
55    let needed = u64::from(queue_pairs) * u64::from(max_send_wr);
56    anyhow::ensure!(
57        needed > 0,
58        "a completion queue serving {queue_pairs} queue pairs of {max_send_wr} \
59         work requests each would hold no entries",
60    );
61    if max_cqe > 0 {
62        anyhow::ensure!(
63            needed <= max_cqe as u64,
64            "{} queue pairs of {} work requests each need a completion queue of {} entries, \
65             but this device holds at most {}; lower rdma_qps_per_cq",
66            queue_pairs,
67            max_send_wr,
68            needed,
69            max_cqe,
70        );
71    }
72    i32::try_from(needed)
73        .map_err(|_| anyhow::anyhow!("a completion queue of {needed} entries does not fit an i32"))
74}

76/// One completion queue in a [`CqPool`], with the number of live leases on it.
77#[derive(Debug)]
78struct CqEntry {
79    cq: Arc<IbvCq>,
80    /// How many queue pairs have claimed a lease on this CQ.
81    /// Increments when a new `CqLease` is minted and decrements
82    /// when that `CqLease` drops.
83    leases: Arc<AtomicU32>,
84}
85
86/// One queue pair's lease on a completion queue.
87///
88/// Holding a lease is what entitles a queue pair to post: the completion queue
89/// has room for its full `max_send_wr` of work requests. Dropping the lease
90/// returns that room to the pool, so a lease must outlive every completion its
91/// queue pair can still produce -- including the flush entries a teardown
92/// generates.
93#[derive(Debug)]
94pub(crate) struct CqLease {
95    cq: Arc<IbvCq>,
96    leases: Arc<AtomicU32>,
97}
98
99impl CqLease {
100    /// The completion queue this lease is on, shareable as a keepalive.
101    pub(super) fn cq(&self) -> &Arc<IbvCq> {
102        &self.cq
103    }
104
105    /// A lease on no completion queue, for tests that build a queue pair without
106    /// a device.
107    #[cfg(test)]
108    pub(super) fn for_test() -> Self {
109        Self::for_test_counted(Arc::new(AtomicU32::new(0)))
110    }
111
112    /// A lease on no completion queue that counts against `leases`, for tests
113    /// that observe when it is released.
114    #[cfg(test)]
115    pub(super) fn for_test_counted(leases: Arc<AtomicU32>) -> Self {
116        leases.fetch_add(1, Ordering::Relaxed);
117        Self {
118            cq: Arc::new(IbvCq::null()),
119            leases,
120        }
121    }
122}
123
124impl Drop for CqLease {
125    fn drop(&mut self) {
126        self.leases.fetch_sub(1, Ordering::Relaxed);
127    }
128}

130/// A device's completion queues, leased to its queue pairs.
131///
132/// Completion queues are created on demand and never destroyed: one outlives the
133/// queue pairs leasing it, and its slots are reused as queue pairs come and go.
134#[derive(Debug)]
135pub(super) struct CqPool {
136    context: Arc<IbvContext>,
137    /// Live leases each completion queue admits.
138    queue_pairs_per_cq: u32,
139    /// Entries each completion queue is created with.
140    cq_entries: i32,
141    completion_queues: Vec<CqEntry>,
142}
143
144impl CqPool {
145    /// Builds an empty pool for queue pairs on `context`, sized from
146    /// [`crate::config::RDMA_QPS_PER_CQ`], `max_send_wr` and the device's own
147    /// per-completion-queue entry limit.
148    ///
149    /// Fails when the device cannot hold a completion queue that large; see
150    /// [`cq_entries_for`].
151    pub(super) fn new(
152        context: Arc<IbvContext>,
153        device_info: &IbvDeviceInfo,
154        max_send_wr: u32,
155    ) -> anyhow::Result<Self> {
156        let configured: usize =
157            hyperactor_config::global::get(crate::config::RDMA_QPS_PER_CQ).into();
158        let queue_pairs_per_cq = u32::try_from(configured)
159            .map_err(|_| anyhow::anyhow!("rdma_qps_per_cq {configured} does not fit a u32"))?;
160        let cq_entries = cq_entries_for(queue_pairs_per_cq, max_send_wr, device_info.max_cqe())?;
161        Ok(Self {
162            context,
163            queue_pairs_per_cq,
164            cq_entries,
165            completion_queues: Vec::new(),
166        })
167    }
168
169    /// A pool that admits `queue_pairs_per_cq` leases per completion queue,
170    /// bypassing the config.
171    #[cfg(test)]
172    pub(super) fn for_test(
173        context: Arc<IbvContext>,
174        queue_pairs_per_cq: u32,
175        max_send_wr: u32,
176        max_cqe: i32,
177    ) -> anyhow::Result<Self> {
178        Ok(Self {
179            context,
180            queue_pairs_per_cq,
181            cq_entries: cq_entries_for(queue_pairs_per_cq, max_send_wr, max_cqe)?,
182            completion_queues: Vec::new(),
183        })
184    }
185
186    /// Leases capacity for one queue pair, on a completion queue that has room
187    /// or on a newly created one.
188    pub(super) fn acquire_one(&mut self) -> anyhow::Result<CqLease> {
189        // Claim a slot with a compare-exchange rather than a load followed by an
190        // increment: a lease is released from whichever thread drops it, so the
191        // count can fall between reading it and acting on it. `Relaxed` suffices
192        // because nothing is published through the count, and read-modify-writes
193        // on one atomic are totally ordered regardless.
194        let limit = self.queue_pairs_per_cq;
195        let claimed = self.completion_queues.iter().find(|cq| {
196            cq.leases
197                .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |live| {
198                    (live < limit).then_some(live + 1)
199                })
200                .is_ok()
201        });
202        if let Some(cq) = claimed {
203            return Ok(CqLease {
204                cq: Arc::clone(&cq.cq),
205                leases: Arc::clone(&cq.leases),
206            });
207        }
208
209        // SAFETY: an `IbvContext` holds a null-or-live `ibv_context` for its own
210        // lifetime per its construction contract, and `IbvCq::create` rejects a
211        // null one.
212        let cq = Arc::new(unsafe { IbvCq::create(Arc::clone(&self.context), self.cq_entries) }?);
213        let leases = Arc::new(AtomicU32::new(1));
214        self.completion_queues.push(CqEntry {
215            cq: Arc::clone(&cq),
216            leases: Arc::clone(&leases),
217        });
218        tracing::debug!(
219            completion_queues = self.completion_queues.len(),
220            cq_entries = self.cq_entries,
221            queue_pairs_per_cq = self.queue_pairs_per_cq,
222            "created a completion queue"
223        );
224        Ok(CqLease { cq, leases })
225    }
226
227    /// A completion queue for a queue pair that never posts.
228    ///
229    /// Such a queue pair still needs a completion queue to be created against,
230    /// but produces no entries, so it consumes no capacity and takes no lease:
231    /// it shares whichever queue is already there rather than pinning one of its
232    /// own.
233    pub(super) fn cq_without_lease(&mut self) -> anyhow::Result<Arc<IbvCq>> {
234        if let Some(entry) = self.completion_queues.first() {
235            return Ok(Arc::clone(&entry.cq));
236        }
237        // SAFETY: an `IbvContext` holds a null-or-live `ibv_context` for its own
238        // lifetime per its construction contract, and `IbvCq::create` rejects a
239        // null one.
240        let cq = Arc::new(unsafe { IbvCq::create(Arc::clone(&self.context), self.cq_entries) }?);
241        self.completion_queues.push(CqEntry {
242            cq: Arc::clone(&cq),
243            leases: Arc::new(AtomicU32::new(0)),
244        });
245        Ok(cq)
246    }
247
248    /// Completion queues this pool has created.
249    #[cfg(test)]
250    pub(super) fn len(&self) -> usize {
251        self.completion_queues.len()
252    }
253}

269    fn entries_fail_when_the_device_cannot_hold_them() {
270        let err = cq_entries_for(32, 512, 1000)
271            .expect_err("a device holding 1000 entries cannot serve 32 x 512");
272        let message = err.to_string();

275            "the error should name what was needed and what the device holds: {message}",

279            "the error should name the knob to lower: {message}",

300    /// A lease releases its slot on drop, so a later queue pair reuses it rather
301    /// than forcing another completion queue.
302    #[test]
303    fn dropping_a_lease_frees_its_slot() {
304        let leases = Arc::new(AtomicU32::new(1));
305        let lease = CqLease {
306            cq: Arc::new(IbvCq::null()),
307            leases: Arc::clone(&leases),
308        };
309        assert_eq!(leases.load(Ordering::Relaxed), 1);
310        drop(lease);
311        assert_eq!(leases.load(Ordering::Relaxed), 0);
312    }

315    fn pool_accepts_multiple_configured_queue_pairs_per_cq() {
316        let info =
317            IbvDeviceInfo::first_available().expect("test runs on machines with RDMA devices");
318        let lock = hyperactor_config::global::lock();
319        let _guard = lock.override_key(

332    /// Queue pairs share a completion queue up to `queue_pairs_per_cq`, the next
333    /// one gets a new completion queue, and a released lease frees its slot for
334    /// reuse.
335    #[test]
336    fn pool_fills_a_completion_queue_before_creating_another() {
337        let info =
338            IbvDeviceInfo::first_available().expect("test runs on machines with RDMA devices");
339        let device = IbvDevice::<MlxDevice>::try_open(info.name(), IbvConfig::default())
340            .expect("the first available device should open");
341        let mut pool = CqPool::for_test(device.context(), 2, 512, device.device_info().max_cqe())
342            .expect("this device holds two queue pairs' worth of entries");
343
344        let first = pool.acquire_one().expect("first lease");
345        assert_eq!(pool.len(), 1);
346
347        let second = pool.acquire_one().expect("second lease");
348        assert_eq!(
349            pool.len(),
350            1,
351            "a second queue pair shares the first completion queue",
352        );
353        assert!(Arc::ptr_eq(first.cq(), second.cq()));
354
355        let third = pool.acquire_one().expect("third lease");
356        assert_eq!(
357            pool.len(),
358            2,
359            "a third queue pair needs a second completion queue",
360        );
361        assert!(!Arc::ptr_eq(first.cq(), third.cq()));
362
363        drop(second);
364        let fourth = pool.acquire_one().expect("fourth lease");
365        assert_eq!(pool.len(), 2, "the released slot is reused, not grown into");
366        assert!(Arc::ptr_eq(first.cq(), fourth.cq()));
367    }
