23/// Tear `connection` down by emitting a `Severed` to the command loop. Shared by
24/// every transport (the QUIC reader, its heartbeat coroutines, …) so the
25/// connection-failure signal is constructed in exactly one place.
26pub(crate) fn sever(
27    loop_tx: &mpsc::UnboundedSender<Command>,
28    connection: ConnectionRef,
29    reason: Vec<u8>,
30) {
31    let _ = loop_tx.send(Command::ConnectionAction {
32        connection,
33        action: ConnectionCommand::Severed { reason },
34    });
35}

37pub(crate) struct ConnectRequest {
38    pub(crate) role: Role,
39    pub(crate) name_for_other: Option<MsgPart>,
40    pub(crate) hello_prefix: Vec<MsgPart>,
41    pub(crate) failure_prefix: Vec<MsgPart>,
42}

44#[derive(Clone, Copy)]
45pub(crate) enum ConnectionRef {
46    ParentConnection {
47        ofactor: Key,
48    },
49    ChildConnection {
50        ofactor: Key,
51        slot: ChildConnectionKey,
52    },
53}
54
55impl ConnectionRef {
56    pub(crate) fn owning_actor(self) -> Key {
57        match self {
58            Self::ParentConnection { ofactor } | Self::ChildConnection { ofactor, .. } => ofactor,
59        }
60    }
61
62    pub(crate) fn role(self) -> Role {
63        match self {
64            Self::ParentConnection { .. } => Role::Child,
65            Self::ChildConnection { .. } => Role::Parent,
66        }
67    }
68}

70/// A parent/child link. State only moves forward:
71///
72/// ```text
73/// Unestablished → Connecting → Established → Failed
74/// ```
75///
76/// - `Unestablished`: attached, but the transport has not come up yet. Outgoing
77///   commands are buffered.
78/// - `Connecting`: the transport is up (we can send), our own `Establish` has
79///   gone out over it, and we are waiting for the peer's `Establish` to learn the
80///   peer's identity. Outgoing commands are still buffered for hello-ordering.
81/// - `Established`: the peer's identity is known; buffered commands are flushed
82///   and routing/hello proceed.
83pub(crate) enum Connection {
84    Unestablished {
85        name_for_other: Option<Vec<u8>>,
86        hello_prefix: Vec<MsgPart>,
87        failure_prefix: Vec<MsgPart>,
88        queued_commands: VecDeque<ConnectionCommand>,
89    },
90    Connecting {
91        transport: Box<dyn ConnectionTransport>,
92        // The name we assigned the peer (if any). Kept — even though we already
93        // sent it in our own Establish — so we can resolve the peer's ident from
94        // it should the peer still have been unnamed when it sent its Establish.
95        name_for_other: Option<Vec<u8>>,
96        hello_prefix: Vec<MsgPart>,
97        failure_prefix: Vec<MsgPart>,
98        queued_commands: VecDeque<ConnectionCommand>,
99    },
100    Established {
101        transport: Box<dyn ConnectionTransport>,
102        failure_prefix: Vec<MsgPart>,
103        peer_ident: Vec<u8>,
104        /// Every ident that has been *live* through this connection (the peer and,
105        /// for a child connection, the live routes published up through it). On
106        /// failure these are exactly the idents to mark dead — no route-table scan
107        /// needed. Idents that arrived already dead are not tracked: they are dead
108        /// regardless of this connection's fate.
109        routed_idents: HashSet<Vec<u8>>,
110    },
111    Failed,
112}

114/// The send half of a connection's pipe. Transports are pure plumbing: `send`
115/// ferries a [`ConnectionCommand`] to the peer, and **dropping** the transport is
116/// the universal "pipe closed" signal — the peer observes it and is severed.
117/// (For UNIX this falls out of closing the socket; the inproc transport pushes a
118/// `Severed` from its `Drop`.)
119pub(crate) trait ConnectionTransport: Send {
120    /// Send a command to the peer. Each transport decides what to do with each
121    /// command — in particular, quic silently drops
122    /// [`ConnectionCommand::GatewayState`] (shared memory is machine-local), while
123    /// unix and inproc forward it.
124    fn send(&self, action: ConnectionCommand) -> bool;
125}

127/// What a [`ConnectionCommand::SendMessage`] delivers once it reaches the actor
128/// it is addressed to. Both kinds route identically (destination-driven, the same
129/// table walk); they differ only in what the destination does on arrival, so a
130/// single routing pathway carries them.
131pub(crate) enum SendPayload {
132    /// An ordinary actor message: opaque part bytes handed to the destination's
133    /// poller.
134    ActorMessage(Vec<MsgPart>),
135    /// A monitor firing: the dead-or-absent target ident, from which the
136    /// destination (the monitoring actor) reconstructs each local monitor's
137    /// failure message. `is_timeout` distinguishes a non-existence timeout (reason
138    /// `"actor does not exist"`) from an actual death (`"actor died"`); both route
139    /// identically. A fire is only ever armed at an ancestor that already has a
140    /// route to the monitoring actor, so it routes strictly *downward* and never
141    /// buffers at a gateway.
142    FireMonitor {
143        to_monitor: Vec<u8>,
144        is_timeout: bool,
145    },
146}
147
148/// A monitor subscription change, carried both up the local hierarchy (in
149/// [`ConnectionCommand::UpdateMonitorSubscription`]) and across a gateway
150/// side-channel (in [`SideChannelAction::UpdateRemoteMonitorState`]). It crosses
151/// both wire formats, so it is `Serialize`/`Deserialize`.
152#[derive(Clone, Copy, Serialize, Deserialize)]
153pub(crate) enum MonitorOp {
154    /// Register a monitor. `timeout_ms` (0 = none) is the "must exist" timeout:
155    /// for a local target it arms a non-existence timer; for a remote one it also
156    /// bounds how long the owning gateway has to acknowledge the registration.
157    Subscribe { timeout_ms: u64 },
158    /// Cancel a previously registered monitor.
159    Unsubscribe,
160}
161
162/// A self-addressing message crossing a gateway-to-gateway side-channel.
163/// `gateway_for_actor` names the actor whose gateway should receive it, and the
164/// side-channel is dialed at `gateway_tag(gateway_for_actor)` — there is no
165/// separate destination tag. Routing never inspects the action; the receiving
166/// gateway resolves the gateway once and dispatches on [`SideChannelAction`].
167pub(crate) struct SideChannelMessage {
168    /// Ident of the actor whose gateway gets this message. For a `Send` it is also
169    /// the destination actor; for an ack it is the `listener`. The side-channel is
170    /// dialed at `gateway_tag(this)`.
171    pub(crate) gateway_for_actor: Vec<u8>,
172    pub(crate) action: SideChannelAction,
173}
174
175/// What a [`SideChannelMessage`] does at the owning gateway.
176pub(crate) enum SideChannelAction {
177    /// Deliver this payload to `gateway_for_actor` (an ordinary cross-gateway
178    /// message, or a `FireMonitor` headed back to a listener).
179    Send(SendPayload),
180    /// Register/cancel a monitor on `gateway_for_actor` (the target) firing to
181    /// `listener`. The owning gateway applies it and, for a subscribe, replies with
182    /// [`SideChannelAction::AckRemoteMonitor`].
183    UpdateRemoteMonitorState { listener: Vec<u8>, op: MonitorOp },
184    /// Confirm a registration. The carrying message's `gateway_for_actor` is the
185    /// `listener`; `monitoring` identifies the target (hence the owning gateway, by
186    /// its `@tag`).
187    AckRemoteMonitor { monitoring: Vec<u8> },
188}

190pub(crate) enum ConnectionCommand {
191    SendMessage {
192        destination_ident: Vec<u8>,
193        payload: SendPayload,
194    },
195    /// The sender announcing itself to the peer: its role, ident, the name it
196    /// assigns the peer, and whether it is still alive. Flows over the transport
197    /// like any other command. A live sender drives the receiver to `Established`;
198    /// a sender that announces `alive: false` (its actor died before the pipe came
199    /// up) drives the receiver to sever instead — but still carries the ident, so
200    /// the failure names the connection that died.
201    Establish {
202        role: Role,
203        ident: Option<Vec<u8>>,
204        name_for_other: Option<Vec<u8>>,
205        alive: bool,
206    },
207    Severed {
208        reason: Vec<u8>,
209    },
210    /// Carry route state up the ancestry. `live` idents become
211    /// [`Route::Connection`](crate::actor::Route::Connection) toward this child;
212    /// `dead` idents become [`Route::Dead`](crate::actor::Route::Dead) and fire any
213    /// monitors watching them. Death travels in the *same* command as life so a
214    /// route's state is never lost on the way up (a dead actor is never mistaken
215    /// for one that does not exist yet). This is the only way dead routes propagate.
216    PublishRoutes {
217        live: Vec<Vec<u8>>,
218        dead: Vec<Vec<u8>>,
219    },
220    /// Climb a monitor subscription change up the *local* hierarchy toward the
221    /// actor that handles `target` (the common ancestor that routes it, or the
222    /// gateway at the top of the domain). One per monitoring actor per target — no
223    /// per-monitor id, since a fire is identified by the dead ident. The
224    /// [`MonitorOp`] selects register vs. cancel. Acks never travel this link —
225    /// they only ever cross side-channels as
226    /// [`SideChannelAction::AckRemoteMonitor`].
227    UpdateMonitorSubscription {
228        listener: Vec<u8>,
229        target: Vec<u8>,
230        op: MonitorOp,
231    },
232    /// The parent handing the child its gateway's [`ShmClient`] (the slab + dgram
233    /// request socket fds), so the child can move large parts through the same
234    /// slab and re-propagate the state to its own machine-local children. Flows
235    /// down unix and inproc edges only (never quic).
236    GatewayState {
237        client: ShmClient,
238    },
239    /// Carry newly-reachable gateway tags up the ancestry. Each hop records them
240    /// against the child connection they arrived on (in
241    /// [`gateway_routes`](crate::actor::ActorEntry::gateway_routes)) and forwards
242    /// them up — *without* stopping at gateway boundaries, so the whole ancestry up
243    /// to the root learns where each gateway lives. This builds the tree a gateway
244    /// death is later broadcast over. Crosses every transport (gateway routing is
245    /// inherently cross-machine).
246    PublishGatewayRoutes {
247        live: Vec<Vec<u8>>,
248    },
249    /// Announce that one or more gateways have died. The direction is inferred from
250    /// the connection it arrives on: arriving from a child it travels *up* toward
251    /// the root; the root turns it around and it travels *down* every gateway-route
252    /// child connection, reaching every gateway so each records the death in its
253    /// gateway state (firing any monitors it held for the now-dead gateway).
254    GatewayDied {
255        dead: Vec<Vec<u8>>,
256    },
257}

259impl Connection {
260    pub(crate) fn new_unestablished(request: ConnectRequest) -> Self {
261        let name_for_other = request
262            .name_for_other
263            .as_ref()
264            .map(|name| name.as_bytes().to_vec());
265        Self::Unestablished {
266            name_for_other,
267            hello_prefix: request.hello_prefix,
268            failure_prefix: request.failure_prefix,
269            queued_commands: VecDeque::new(),
270        }
271    }
272
273    pub(crate) fn send(&mut self, command: ConnectionCommand) -> bool {
274        match self {
275            Self::Established { transport, .. } => transport.send(command),
276            // Before establishment outgoing commands are buffered (even once the
277            // transport is up, so the peer's hello precedes any message).
278            Self::Unestablished {
279                queued_commands, ..
280            }
281            | Self::Connecting {
282                queued_commands, ..
283            } => {
284                queued_commands.push_back(command);
285                true
286            }
287            Self::Failed => false,
288        }
289    }
290
291    /// Consume a connection that is being torn down, yielding the failure-message
292    /// prefix, the peer ident to report, and every ident that was reachable
293    /// through it (to mark dead). `None` if it had already failed.
294    pub(crate) fn into_failure_report(self) -> Option<FailureReport> {
295        match self {
296            Self::Established {
297                failure_prefix,
298                peer_ident,
299                routed_idents,
300                ..
301            } => Some((failure_prefix, peer_ident, routed_idents)),
302            Self::Unestablished {
303                name_for_other,
304                failure_prefix,
305                ..
306            } => Some((
307                failure_prefix,
308                name_for_other.unwrap_or_default(),
309                HashSet::new(),
310            )),
311            // Connecting has not yet learned the peer's ident; fall back to the
312            // name we assigned it, if any.
313            Self::Connecting {
314                name_for_other,
315                failure_prefix,
316                ..
317            } => Some((
318                failure_prefix,
319                name_for_other.unwrap_or_default(),
320                HashSet::new(),
321            )),
322            Self::Failed => None,
323        }
324    }
325}

327/// What [`Connection::into_failure_report`] yields: the failure-message prefix,
328/// the peer ident to report, and every ident that routed through the connection.
329pub(crate) type FailureReport = (Vec<MsgPart>, Vec<u8>, HashSet<Vec<u8>>);
