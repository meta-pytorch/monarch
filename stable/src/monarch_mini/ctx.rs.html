75/// Cumulative UDP-over-IPv6 datagram counters from `/proc/net/snmp6`, as
76/// `(InDatagrams, InErrors, RcvbufErrors)`. `RcvbufErrors` counts datagrams the
77/// kernel dropped because a receiving socket's buffer was full — the direct signal
78/// for root-side ingress overflow (packets arriving faster than the single-threaded
79/// driver can drain them). System-wide, but on the root host our traffic dominates.
80/// Reading this proc pseudo-file is an instant, non-blocking memory read, so
81/// `std::fs` is fine here (debug instrumentation on the once-per-second log path).
82fn read_udp6_stats() -> Option<(u64, u64, u64)> {
83    let text = std::fs::read_to_string("/proc/net/snmp6").ok()?;
84    let (mut indg, mut inerr, mut rcv) = (None, None, None);
85    for line in text.lines() {
86        let mut it = line.split_whitespace();
87        let (Some(name), Some(val)) = (it.next(), it.next()) else {
88            continue;
89        };
90        match name {
91            "Udp6InDatagrams" => indg = val.parse().ok(),
92            "Udp6InErrors" => inerr = val.parse().ok(),
93            "Udp6RcvbufErrors" => rcv = val.parse().ok(),
94            _ => {}
95        }
96    }
97    Some((indg?, inerr?, rcv?))
98}

100/// Current wall-clock time as `HH:MM:SS.mmm` (UTC), for debug log lines. Kept
101/// dependency-free (no chrono): derived straight from the UNIX epoch, so it lines
102/// up with the smoke test's Python `log()` timestamps when hosts run in UTC.
103pub(crate) fn wall_clock_hms() -> String {
104    let now = std::time::SystemTime::now()
105        .duration_since(std::time::UNIX_EPOCH)
106        .unwrap_or_default();
107    let secs = now.as_secs();
108    format!(
109        "{:02}:{:02}:{:02}.{:03}",
110        (secs / 3600) % 24,
111        (secs / 60) % 60,
112        secs % 60,
113        now.subsec_millis(),
114    )
115}

117/// Whether `url`'s scheme is one we have a transport for. Kept separate from
118/// [`Ctx::transport_for`] so the scheme can be validated — before the connection
119/// is attached, while the request can still report failure — without a `&mut self`
120/// borrow.
121fn valid_scheme(url: &str) -> bool {
122    url.starts_with("inproc://")
123        || url.starts_with("unix://")
124        || url.starts_with("quic://")
125        || url.starts_with("tcp://")
126}

128/// Whether `url`'s scheme connects across machines (a network transport). A
129/// gateway may only gain a parent over such a link — a gateway joined to a
130/// unix/inproc parent is rejected, since it would no longer be the entry point
131/// for its process group.
132fn is_network_scheme(url: &str) -> bool {
133    url.starts_with("quic://") || url.starts_with("tcp://")
134}

136/// The gateway specifier (the `@endpoint` suffix) of an ident, or an empty slice
137/// if it has none. Actor idents are formatted `name@gateway_specifier`; an ident
138/// with no `@` belongs to the root gateway (specifier-less). The specifier is the
139/// address a gateway dials to reach the ident's owning gateway directly, so it is
140/// also a gateway's own domain tag (parsed from its own ident).
141fn gateway_tag(ident: &[u8]) -> &[u8] {
142    match ident.iter().rposition(|&b| b == b'@') {
143        Some(pos) => &ident[pos + 1..],
144        None => &[],
145    }
146}

154/// Grace period before a target's debounced `Unsubscribe` actually goes out. A
155/// monitor recreated on the same target within this window reuses the still-live
156/// upstream subscription, so churn (e.g. per-message monitor/cancel) sends no
157/// subscribe/unsubscribe traffic.
158const MONITOR_DEBOUNCE: Duration = Duration::from_millis(25);
159
160/// Wall-clock cost of fanning one gateway-death broadcast out to every gateway at
161/// max scale (~6s for thousands of machines). This single number drives the whole
162/// gateway-death coalescing policy (see `spawn_gateway_death_coroutine`): the root
163/// must never issue broadcasts faster than it can drain them, so it is the spacing
164/// floor between broadcasts — and every other window is derived from it. Overridable
165/// via `MM_GATEWAY_BROADCAST_COST_MS` (mainly so tests can shrink it).
166fn gateway_broadcast_cost() -> Duration {
167    std::env::var("MM_GATEWAY_BROADCAST_COST_MS")
168        .ok()
169        .and_then(|v| v.parse::<u64>().ok())
170        .map(Duration::from_millis)
171        .unwrap_or(Duration::from_secs(6))
172}

286struct Ctx {
287    actors: SlotMap<Key, ActorEntry>,
288    pollers: SlotMap<PollerKey, PollerEntry>,
289    // The transports. Each owns its own pairing/socket state and coroutines plus
290    // a clone of the loop sender; the command loop only forwards serves/joins to
291    // them and handles the generic TransportConnected/ConnectionAction they
292    // emit.
293    inproc: InprocTransport,
294    unix: UnixTransport,
295    quic: QuicTransport,
296    // TCP is the cross-region fallback (quic's UDP is packet-filtered across
297    // regions). An actor is reached over tcp iff its `@endpoint` tag carries the
298    // `tcp://` scheme; a bare or `quic://` tag routes over quic (see
299    // `send_to_gateway` and `valid_scheme`).
300    tcp: TcpTransport,
301    // The context-global shared-memory address-space manager. Always present
302    // (idle if shm is unused); handed to the unix and quic transports, and unmaps
303    // everything when the context — and thus this last reference — is dropped.
304    #[expect(
305        dead_code,
306        reason = "held so the mapper outlives the context; used via the unix and quic transports"
307    )]
308    mapper: MapperHandle,
309    // The context-global shared-memory server, shared by every gateway in this
310    // context (created lazily when the first gateway appears). One slab + client
311    // per serving process — not per gateway actor — so multiple independent actor
312    // trees in the same process share it. Dropping it (with the context) releases
313    // the slab.
314    shm_server: Option<ShmServer>,
315    // The single shm client for this context: the slab handed to gateways and to
316    // the quic listener coroutines. Empty until the first gateway creates the
317    // server. A clone of this slot lives in the quic transport.
318    shm_client: ShmClientSlot,
319    // Side-channel messages whose `gateway_for_actor` no gateway in this context can
320    // resolve yet, keyed by that ident. The owning gateway is not known up front
321    // (several gateways may serve the same endpoint url, and the actor may not exist
322    // yet), so the *whole* self-addressing message is held context-wide — regardless
323    // of its action — and replayed through `deliver_side_channel` once a route to the
324    // ident is learned (see `populate_routes`).
325    pending_side_channel: HashMap<Vec<u8>, Vec<SideChannelMessage>>,
326    // Every gateway actor ever created in this context, so resolving a side-channel
327    // destination scans only gateways (a handful) rather than every actor. A
328    // *dead* gateway is kept: its routing table still tells us whether a destination
329    // was reachable through it, so we can drop a side-channel message for a dead
330    // subtree rather than hold it forever.
331    gateways: Vec<Key>,
332    // A clone of the loop sender, used to schedule debounced monitor unsubscribes
333    // back onto the event loop after a grace period.
334    loop_tx: mpsc::UnboundedSender<Command>,
335    thread: Option<JoinHandle<()>>,
336    _not_send: PhantomData<*const ()>,
337}

361    /// Insert a new actor, tracking it in `gateways` if it is one (so side-channel
362    /// delivery scans only gateways, never every actor).
363    fn insert_actor(&mut self, ident: Option<Vec<u8>>, gateway: bool) -> Key {
364        let key = self.actors.insert(ActorEntry::new(ident, gateway));
365        if gateway {
366            self.gateways.push(key);
367        }
368        key

371    fn run_command(&mut self, command: Command) {
372        match command {
373            Command::Init { thread } => {
374                self.thread = Some(thread);
375            }
376            Command::CreateActor {
377                ident,
378                gateway,
379                done,
380            } => {
381                let key =
382                    self.insert_actor(ident.as_ref().map(|part| part.as_bytes().to_vec()), gateway);
383                drop(ident);
384                // A gateway shares the context's slab (created lazily on the first
385                // gateway): seed its client slot from it, so it can shm-ify large
386                // sends to its children and hand the state down to them. Best-effort
387                // — on failure the gateway just streams inline.
388                if gateway {
389                    self.init_gateway_shm(key);
390                }
391                let _ = done.send(key);
392            }
393            Command::DestroyActor { key } => {
394                let subscription = self.actor(key).poller_subscription();
395                self.die_actor(key, b"actor destroyed".to_vec());
396                if let Some((poller, index)) = subscription {
397                    if let Some(poller) = self.pollers.get_mut(poller) {
398                        let _ = poller.remove(index);
399                    }
400                    self.actor_mut(key).unsubscribe();
401                }
402            }
403            Command::CreatePoller { tx, event_fd, done } => {
404                let key = self.pollers.insert(PollerEntry::new(tx, event_fd));
405                let _ = done.send(key);
406            }
407            Command::DestroyPoller { poller } => {
408                if let Some(poller) = self.pollers.remove(poller) {
409                    for subscription in poller.subscriptions {
410                        self.actor_mut(subscription.actor).unsubscribe();
411                    }
412                }
413            }
414            Command::Subscribe {
415                poller,
416                index,
417                actor,
418                done,
419            } => {
420                if !self.pollers.contains_key(poller) {
421                    let _ = done.send(Err(anyhow::anyhow!("poller does not exist")));
422                } else if self.pollers[poller].has_index(index) {
423                    let _ = done.send(Err(anyhow::anyhow!("poller index is already subscribed")));
424                } else {
425                    // Move the actor onto the poller and flush any messages that were
426                    // buffered while it had no poller attached.
427                    let result = (|| -> anyhow::Result<()> {
428                        let buffered = self.actor_mut(actor).subscribe(poller, index)?;
429                        let poller_entry =
430                            self.pollers.get_mut(poller).expect("poller should exist");
431                        poller_entry.insert(index, actor);
432                        for msg in buffered {
433                            poller_entry.deliver(index, msg);
434                        }
435                        Ok(())
436                    })();
437                    let _ = done.send(result);
438                }
439            }
440            Command::Unsubscribe { poller, index } => {
441                if let Some(poller) = self.pollers.get_mut(poller) {
442                    if let Some(actor) = poller.remove(index) {
443                        self.actor_mut(actor).unsubscribe();
444                    }
445                }
446            }
447            Command::ArmPoller { poller, wake_after } => {
448                if let Some(poller) = self.pollers.get_mut(poller) {
449                    poller.arm(wake_after);
450                }
451            }
452            Command::Send {
453                sender,
454                destination_ident,
455                parts,
456            } => {
457                self.route_message(
458                    sender,
459                    destination_ident.as_bytes().to_vec(),
460                    SendPayload::ActorMessage(parts),
461                );
462            }
463            Command::Serve {
464                actor,
465                url,
466                request,
467            } => {
468                self.serve(actor, url, request);
469            }
470            Command::Join {
471                actor,
472                url,
473                request,
474            } => {
475                self.join(actor, url, request);
476            }
477            Command::ConnectionAction { connection, action } => {
478                if !matches!(self.connection(connection), Connection::Failed) {
479                    self.run_connection_command(connection, action);
480                }
481            }
482            Command::TransportConnected {
483                connection,
484                transport,
485            } => {
486                self.transport_connected(connection, transport);
487            }
488            Command::SideChannelDeliver(message) => {
489                self.deliver_side_channel(message);
490            }
491            Command::Die { actor, reason } => {
492                let reason = reason.as_bytes().to_vec();
493                self.die_actor(actor, reason);
494            }
495            Command::Monitor {
496                actor,
497                id,
498                to_monitor,
499                failure_prefix,
500                timeout_ms,
501            } => {
502                let to_monitor = to_monitor.as_bytes().to_vec();
503                self.monitor_add(actor, id, to_monitor, failure_prefix, timeout_ms);
504            }
505            Command::CancelMonitor { actor, id } => {
506                self.monitor_remove(actor, id);
507            }
508            Command::CheckMonitorTimeout {
509                at,
510                listener,
511                target,
512            } => {
513                self.check_monitor_timeout(at, listener, target);
514            }
515            Command::UnsubscribeMonitor { actor, to_monitor } => {
516                self.unsubscribe_monitor(actor, to_monitor);
517            }
518            Command::PropagateGatewayDeaths { at, dead } => {
519                // The coroutine fires onto the loop, so the root may have been
520                // destroyed since the batch was buffered; nothing to broadcast if so.
521                if self.actors.contains_key(at) {
522                    self.broadcast_gateway_deaths(at, dead);
523                }
524            }
525            Command::Shutdown { .. } => {
526                // Handled directly by the event loop (see CtxHandle::new) so it
527                // can await the UNIX writer flush before tearing the loop down.
528                unreachable!("Shutdown is intercepted by the event loop");
529            }
530        }
531    }

573    /// The single destination-driven routing pathway. Both an ordinary actor
574    /// message and a monitor firing travel through here, differing only in what
575    /// `deliver_payload` does once the destination is reached. A monitor fire is
576    /// armed only at an ancestor that already routes to its destination, so it
577    /// always finds a concrete route down (or a dead one) and never reaches the
578    /// gateway-buffer branch.
579    fn route_message(&mut self, sender: Key, destination_ident: Vec<u8>, payload: SendPayload) {
580        let entry = self.actor(sender);
581        if !entry.alive {
582            return;
583        }
584        if entry.name() == Some(destination_ident.as_slice()) {
585            self.deliver_payload(sender, payload);
586            return;
587        }
588
589        match entry.routes.get(destination_ident.as_slice()) {
590            // A known child route: hand the payload down toward the destination.
591            Some(Route::Connection { child, .. }) => {
592                let child_key = *child;
593                let connection = self
594                    .actor_mut(sender)
595                    .children
596                    .get_mut(child_key)
597                    .expect("route child connection should exist");
598                let _ = connection.send(ConnectionCommand::SendMessage {
599                    destination_ident,
600                    payload,
601                });
602                return;
603            }
604            // The destination is known to be dead: drop the payload rather than
605            // forwarding it up (where it would buffer forever at the gateway).
606            Some(Route::Dead) => return,
607            Some(Route::Unknown { .. }) | None => {}
608        }
609
610        // No local route. A gateway is the boundary of its routing domain, so a
611        // destination on a *different* gateway is reached by a direct side-channel
612        // rather than by climbing toward the root. Destinations in the root domain
613        // (no specifier) still climb the parent chain; destinations in our own
614        // domain that we simply do not know yet are buffered locally — never sent
615        // up, which would only bounce them back to us.
616        if self.actor(sender).is_gateway() {
617            let own_tag = gateway_tag(self.actor(sender).name().unwrap_or_default()).to_vec();
618            let dest_tag = gateway_tag(&destination_ident).to_vec();
619            if !dest_tag.is_empty() && dest_tag != own_tag {
620                self.send_to_gateway(SideChannelMessage {
621                    gateway_for_actor: destination_ident,
622                    action: SideChannelAction::Send(payload),
623                });
624                return;
625            }
626            if !own_tag.is_empty() && dest_tag == own_tag {
627                self.actor_mut(sender)
628                    .buffer_unrouted(destination_ident, payload);
629                return;
630            }
631        }
632
633        // No known route here. Forward the payload — intact — toward the gateway;
634        // the gateway itself (no parent) buffers it until the destination's route
635        // is published, since the destination may not even have been created yet. A
636        // `Route::Unknown` entry only exists at a gateway, so it also lands here.
637        if let Some(parent_connection) = self.actor_mut(sender).parent.as_mut() {
638            let _ = parent_connection.send(ConnectionCommand::SendMessage {
639                destination_ident,
640                payload,
641            });
642        } else {
643            self.actor_mut(sender)
644                .buffer_unrouted(destination_ident, payload);
645        }
646    }

648    /// Hand a routed payload to the actor it is addressed to: an actor message
649    /// goes to the poller, a monitor fire fans out to every local monitor on the
650    /// dead target.
651    fn deliver_payload(&mut self, actor: Key, payload: SendPayload) {
652        match payload {
653            SendPayload::ActorMessage(parts) => self.deliver_to_actor(actor, parts),
654            SendPayload::FireMonitor {
655                to_monitor,
656                is_timeout,
657            } => {
658                let reason: &[u8] = if is_timeout {
659                    b"actor does not exist"
660                } else {
661                    b"actor died"
662                };
663                self.fire_local_monitors(actor, to_monitor, reason);
664            }
665        }
666    }

668    /// Select the transport for a url's scheme. Only called once the scheme is
669    /// known valid (see [`valid_scheme`]), so a missing transport is a bug.
670    fn transport_for(&mut self, url: &str) -> &mut dyn Transport {
671        if url.starts_with("inproc://") {
672            &mut self.inproc
673        } else if url.starts_with("unix://") {
674            &mut self.unix
675        } else if url.starts_with("quic://") {
676            &mut self.quic
677        } else if url.starts_with("tcp://") {
678            &mut self.tcp
679        } else {
680            unreachable!("scheme already validated by valid_scheme");
681        }
682    }

684    fn serve(&mut self, actor: Key, url: String, request: ConnectRequest) {
685        // Shared setup runs first regardless of transport: validate the scheme,
686        // then attach the unestablished connection. Only then do we hand the
687        // connection to the transport, which brings up its pipe and announces it
688        // back via TransportConnected. (The scheme is validated before attach so a
689        // bad scheme can still report failure with the as-yet-unconsumed request.)
690        if !valid_scheme(&url) {
691            self.deliver_connect_failure(actor, request, b"unsupported url scheme".to_vec());
692            return;
693        }
694        if self.gateway_parent_rejected(actor, request.role, &url) {
695            self.deliver_connect_failure(
696                actor,
697                request,
698                b"gateway must have no parent or a network parent".to_vec(),
699            );
700            return;
701        }
702        let Some(connection) = self.attach_connection(actor, request) else {
703            return;
704        };
705        let shm_client = self.actor(actor).shm_client.clone();
706        self.transport_for(&url).serve(url, connection, shm_client);
707    }

709    fn join(&mut self, actor: Key, url: String, request: ConnectRequest) {
710        if !valid_scheme(&url) {
711            self.deliver_connect_failure(actor, request, b"unsupported url scheme".to_vec());
712            return;
713        }
714        if self.gateway_parent_rejected(actor, request.role, &url) {
715            self.deliver_connect_failure(
716                actor,
717                request,
718                b"gateway must have no parent or a network parent".to_vec(),

722        let Some(connection) = self.attach_connection(actor, request) else {

729    /// Whether attaching a parent over `url` to `actor` must be rejected because
730    /// `actor` is a gateway gaining a non-network parent. A gateway is the entry
731    /// point for its process group, so it may only have no parent or a network
732    /// (quic/tcp) parent; a unix/inproc parent would demote it. Only a `Child`
733    /// role gains a parent, so a `Parent` role never trips this.
734    fn gateway_parent_rejected(&self, actor: Key, role: Role, url: &str) -> bool {
735        role == Role::Child && self.actor(actor).is_gateway() && !is_network_scheme(url)
736    }

738    fn attach_connection(&mut self, actor: Key, request: ConnectRequest) -> Option<ConnectionRef> {
739        let role = request.role;
740        let connection = Connection::new_unestablished(request);
741
742        let connection = match role {
743            Role::Parent => {
744                // An actor may adopt children before it has a parent (it buffers
745                // routes and hands them upward once it gains one — see
746                // publish_routes_after_established). The gateway flag is fixed at
747                // creation and no longer gates this.
748                let slot = self.actor_mut(actor).children.insert(connection);
749                // A new child is where we (re)generate gateway state from what we
750                // already hold: send it down the fresh connection. It buffers until
751                // the connection establishes, and the transport decides whether to
752                // actually forward it (quic drops it).
753                if let Some(client) = self.shm_client_of(actor) {
754                    if let Some(connection) = self.actor_mut(actor).children.get_mut(slot) {
755                        let _ = connection.send(ConnectionCommand::GatewayState { client });
756                    }
757                }
758                ConnectionRef::ChildConnection {
759                    ofactor: actor,
760                    slot,
761                }
762            }
763            Role::Child => {
764                let actor_entry = self.actor_mut(actor);
765                // An actor may have at most one parent. It is free to have already
766                // served children or buffered messages; gaining a parent hands all
767                // of that upward (see publish_routes_after_established).
768                if actor_entry.parent.is_some() {
769                    let Connection::Unestablished {
770                        mut failure_prefix, ..
771                    } = connection
772                    else {
773                        unreachable!("new connection should be unestablished");
774                    };
775                    failure_prefix.push(MsgPart::from_bytes(Vec::new()));
776                    failure_prefix.push(MsgPart::from_bytes(
777                        b"invalid parent-child topology".to_vec(),
778                    ));
779                    self.deliver_to_actor(actor, failure_prefix);
780                    return None;
781                }
782                actor_entry.parent = Some(connection);
783                ConnectionRef::ParentConnection { ofactor: actor }
784            }
785        };
786
787        Some(connection)
788    }

790    /// A transport's pipe is up. Announce ourselves to the peer along it via an
791    /// `Establish`, then hold the transport on the connection until the peer's
792    /// `Establish` lands (`Unestablished` → `Connecting`).
793    ///
794    /// If the owning actor already died, the connection is `Failed`: we still send
795    /// an `Establish`, carrying our (dead) ident but `alive: false`, so the peer
796    /// severs *and names us* in its failure. We then drop the transport without
797    /// transitioning (its drop-severance is the generic fallback, skipped once the
798    /// peer is already failing).
799    fn transport_connected(
800        &mut self,
801        connection: ConnectionRef,
802        transport: Box<dyn ConnectionTransport>,
803    ) {
804        let role = connection.role();
805        let ident = self
806            .actor(connection.owning_actor())
807            .name()
808            .map(|name| name.to_vec());
809        let alive = matches!(
810            self.connection(connection),
811            Connection::Unestablished { .. }
812        );
813        let name_for_other = match self.connection(connection) {
814            Connection::Unestablished { name_for_other, .. } => name_for_other.clone(),
815            _ => None,
816        };
817        transport.send(ConnectionCommand::Establish {
818            role,
819            ident,
820            name_for_other: name_for_other.clone(),
821            alive,
822        });
823        if !alive {
824            return;
825        }
826
827        let conn = self.connection_mut(connection);
828        let Connection::Unestablished {
829            hello_prefix,
830            failure_prefix,
831            queued_commands,
832            ..
833        } = std::mem::replace(conn, Connection::Failed)
834        else {
835            unreachable!("alive implies Unestablished");
836        };
837        *conn = Connection::Connecting {
838            transport,
839            name_for_other,
840            hello_prefix,
841            failure_prefix,
842            queued_commands,
843        };
844    }

846    fn connection_topology_is_valid(&self, connection: ConnectionRef) -> bool {
847        // The gateway/parent topology is enforced up front at serve/join time (a
848        // gateway only gains a network parent; an actor has at most one parent),
849        // so by establishment the only thing that can have changed is the actor
850        // dying. Both roles are valid as long as the owning actor is still alive.
851        self.actor(connection.owning_actor()).alive
852    }

854    /// Record routes arriving over `child_connection`: `live` idents become
855    /// concrete routes to that child (and are remembered against it, so they can be
856    /// marked dead if it later fails), `dead` idents are marked [`Route::Dead`].
857    /// Both kinds are carried further up.
858    fn populate_routes(
859        &mut self,
860        parent: Key,
861        child_connection: ChildConnectionKey,
862        live: Vec<Vec<u8>>,
863        dead: Vec<Vec<u8>>,
864    ) {
865        for ident in &live {
866            // Take any existing entry so we can carry its monitors onto the now-known
867            // route and flush its buffered messages. Monitors stay here: this actor is
868            // the responsible ancestor for the destination (it is reachable below it).
869            let previous = self.actor_mut(parent).routes.remove(ident.as_slice());
870            let (monitors, buffered) = match previous {
871                Some(Route::Unknown { messages, monitors }) => (monitors, messages),
872                Some(Route::Connection { monitors, .. }) => (monitors, Vec::new()),
873                _ => (Vec::new(), Vec::new()),
874            };
875            self.actor_mut(parent).routes.insert(
876                ident.clone(),
877                Route::Connection {
878                    child: child_connection,
879                    monitors,
880                },
881            );
882            // Remember the ident is carried by this child so we can mark exactly
883            // these dead (no table scan) if the connection later fails.
884            self.actor_mut(parent)
885                .record_routed_via_child(child_connection, ident);
886
887            // Re-route every payload buffered against the now-resolved Unknown route;
888            // route_message sends them down the child connection we just recorded.
889            for payload in buffered {
890                self.route_message(parent, ident.clone(), payload);
891            }
892            // Self-addressing side-channel messages held context-wide for this ident
893            // can now be resolved, so replay each through deliver_side_channel — the
894            // one place a previously-unresolvable side-channel ident gains a route.
895            if let Some(pending) = self.pending_side_channel.remove(ident.as_slice()) {
896                for message in pending {
897                    self.deliver_side_channel(message);
898                }
899            }
900        }
901
902        // Mark the dead idents dead (they are not tracked against the child — they
903        // are already dead, so a later failure of this connection has nothing to
904        // re-kill there). Keep only those that newly transitioned so propagation
905        // can't loop, firing the monitors that were waiting on each.
906        let mut newly_dead = Vec::new();
907        // Each fire is the dead ident paired with a subscriber's `dest`.
908        let mut to_fire: Vec<(Vec<u8>, Vec<u8>)> = Vec::new();
909        for ident in dead {
910            let routes = &mut self.actor_mut(parent).routes;
911            match routes.get_mut(&ident) {
912                Some(Route::Dead) => continue, // already dead
913                Some(route) => {
914                    let subs = route.monitors_mut().map(std::mem::take).unwrap_or_default();
915                    to_fire.extend(subs.into_iter().map(|sub| (sub.dest, ident.clone())));
916                    *route = Route::Dead;
917                }
918                None => {
919                    routes.insert(ident.clone(), Route::Dead);
920                }
921            }
922            newly_dead.push(ident);
923        }
924        for (dest, to_monitor) in to_fire {
925            self.route_message(
926                parent,
927                dest,
928                SendPayload::FireMonitor {
929                    to_monitor,
930                    is_timeout: false,
931                },
932            );
933        }
934
935        // Forward both the live and the newly-dead idents up to the parent.
936        self.publish_routes_to_parent(parent, live, newly_dead);
937    }

939    fn publish_routes_to_parent(&mut self, actor: Key, live: Vec<Vec<u8>>, dead: Vec<Vec<u8>>) {
940        if live.is_empty() && dead.is_empty() {
941            return;
942        }
943        // A gateway is the top of its routing domain: it does not leak its
944        // subtree's routes up to its (network) parent. This is what keeps actors
945        // on one gateway out of another gateway's routing tables — cross-gateway
946        // delivery uses direct side-channels (see route_message) instead. (Message
947        // forwarding up the parent chain is unaffected; only route publication is.)
948        if self.actor(actor).is_gateway() {
949            return;
950        }
951        let actor = self.actor_mut(actor);
952        let Some(parent) = actor.parent.as_mut() else {
953            return;
954        };
955        let _ = parent.send(ConnectionCommand::PublishRoutes { live, dead });
956    }

958    fn publish_routes_after_established(
959        &mut self,
960        connection: ConnectionRef,
961        local_ident: Vec<u8>,
962        peer_ident: Vec<u8>,
963    ) {
964        match connection {
965            ConnectionRef::ChildConnection { ofactor, slot } => {
966                self.populate_routes(ofactor, slot, vec![peer_ident], Vec::new());
967            }
968            ConnectionRef::ParentConnection { ofactor } => {
969                // We just gained a parent. Walk the whole routing table once: concrete
970                // routes are advertised up (live as live, dead as dead — a dead route
971                // must travel up *as dead*, else the parent treats a known-dead actor
972                // as alive), while buffered Unknown routes — which we could not place
973                // while parentless — are dropped and their held messages and monitor
974                // subscriptions re-routed up (route_message / route_monitor_change now
975                // forward, since there is a parent and no local route).
976                let mut live = vec![local_ident];
977                let mut dead = Vec::new();
978                let mut kept = HashMap::new();
979                for (ident, route) in std::mem::take(&mut self.actor_mut(ofactor).routes) {
980                    match route {
981                        Route::Connection { .. } => {
982                            live.push(ident.clone());
983                            kept.insert(ident, route);
984                        }
985                        Route::Dead => {
986                            dead.push(ident.clone());
987                            kept.insert(ident, route);
988                        }
989                        Route::Unknown { messages, monitors } => {
990                            for payload in messages {
991                                self.route_message(ofactor, ident.clone(), payload);
992                            }
993                            for sub in monitors {
994                                // Forward each subscription up *with its own
995                                // timeout*, so the new ancestor re-arms a fresh
996                                // timer where the subscription now lives. The old
997                                // ancestor's route for this target is now `None`,
998                                // so its still-pending timer no-ops when it fires.
999                                self.route_monitor_change(
1000                                    ofactor,
1001                                    sub.dest,
1002                                    ident.clone(),
1003                                    MonitorOp::Subscribe {
1004                                        timeout_ms: sub.timeout_ms,
1005                                    },
1006                                );
1007                            }
1008                        }
1009                    }
1010                }
1011                self.actor_mut(ofactor).routes = kept;
1012                self.publish_routes_to_parent(ofactor, live, dead);
1013
1014                // Gaining a parent is also where gateway routes climb the next hop.
1015                // Re-publish every gateway tag reachable below us — and, if we are a
1016                // gateway, our own tag — up the new link, so the ancestry up to the
1017                // root can route a gateway-death broadcast back down to us. Unlike
1018                // actor routes, this does not stop at gateway boundaries.
1019                let mut tags: Vec<Vec<u8>> = self
1020                    .actor(ofactor)
1021                    .gateway_routes
1022                    .values()
1023                    .flatten()
1024                    .cloned()
1025                    .collect();
1026                if self.actor(ofactor).is_gateway() {
1027                    if let Some(own) = self
1028                        .actor(ofactor)
1029                        .name()
1030                        .map(|name| gateway_tag(name).to_vec())
1031                        .filter(|own| !own.is_empty())
1032                    {
1033                        tags.push(own);
1034                    }
1035                }
1036                self.publish_gateway_routes_to_parent(ofactor, tags);
1037            }
1038        }
1039    }

1041    fn run_connection_command(&mut self, connection: ConnectionRef, action: ConnectionCommand) {
1042        match action {
1043            ConnectionCommand::SendMessage {
1044                destination_ident,
1045                payload,
1046            } => self.route_message(connection.owning_actor(), destination_ident, payload),
1047            ConnectionCommand::Establish {
1048                role,
1049                ident,
1050                name_for_other,
1051                alive,
1052            } => {
1053                // The peer announced itself; finalize our side using the transport we
1054                // stashed when our pipe connected. On failure establish_connection
1055                // parks the connection carrying its failure info, so fail_connection
1056                // tears it down the same way a sever does.
1057                if let Err(reason) =
1058                    self.establish_connection(connection, role, ident, name_for_other, alive)
1059                {
1060                    self.fail_connection(connection, reason);
1061                }
1062            }
1063            ConnectionCommand::Severed { reason } => {
1064                // The peer (or our own death cascade) tore this connection down.
1065                self.fail_connection(connection, reason);
1066            }
1067            ConnectionCommand::PublishRoutes { live, dead } => {
1068                assert!(
1069                    matches!(self.connection(connection), Connection::Established { .. }),
1070                    "published routes should arrive on an established connection"
1071                );
1072                // Published routes always arrive over a child connection: record them
1073                // (live and dead) as routes to that child, forwarding both up the tree.
1074                let ConnectionRef::ChildConnection { ofactor, slot } = connection else {
1075                    panic!("published routes should arrive on a child connection");
1076                };
1077                self.populate_routes(ofactor, slot, live, dead);
1078            }
1079            ConnectionCommand::UpdateMonitorSubscription {
1080                listener,
1081                target,
1082                op,
1083            } => {
1084                self.route_monitor_change(connection.owning_actor(), listener, target, op);
1085            }
1086            ConnectionCommand::GatewayState { client } => {
1087                // Our parent handed us its gateway's client: adopt it and pass it
1088                // on to our own machine-local children.
1089                self.receive_gateway_state(connection.owning_actor(), client);
1090            }
1091            ConnectionCommand::PublishGatewayRoutes { live } => {
1092                // Gateway routes always arrive over a child connection (they climb
1093                // toward the root). Record each against that child, then forward the
1094                // ones we had not already recorded further up — never stopping at a
1095                // gateway boundary, so they accumulate all the way to the root.
1096                let ConnectionRef::ChildConnection { ofactor, slot } = connection else {
1097                    panic!("published gateway routes should arrive on a child connection");
1098                };
1099                let entry = self.actor_mut(ofactor);
1100                let mut forward = Vec::new();
1101                for tag in live {
1102                    // A gateway that died never returns under the same tag (a
1103                    // recovered host reuses its address with a fresh pseudo-port), so
1104                    // a tag already known dead is stale — drop it rather than
1105                    // resurrect a route to it. (Only a gateway holds a dead-set; a
1106                    // non-gateway relay never filters here.)
1107                    let known_dead = match &entry.gateway {
1108                        GatewayState::Gateway { gateway_state } => {
1109                            matches!(gateway_state.get(&tag), Some(GatewayMonitors::Dead))
1110                        }
1111                        GatewayState::NotAGateway => false,
1112                    };
1113                    if known_dead {
1114                        continue;
1115                    }
1116                    if entry
1117                        .gateway_routes
1118                        .entry(slot)
1119                        .or_default()
1120                        .insert(tag.clone())
1121                    {
1122                        forward.push(tag);
1123                    }
1124                }
1125                self.publish_gateway_routes_to_parent(ofactor, forward);
1126            }
1127            ConnectionCommand::GatewayDied { dead } => {
1128                // The direction is the connection's role: a death arriving over a
1129                // child connection is still climbing toward the root; one arriving
1130                // over the parent connection is fanning back down.
1131                match connection {
1132                    ConnectionRef::ChildConnection { ofactor, .. } => {
1133                        self.gateway_died(ofactor, dead, false)
1134                    }
1135                    ConnectionRef::ParentConnection { ofactor } => {
1136                        self.gateway_died(ofactor, dead, true)
1137                    }
1138                }
1139            }
1140        }
1141    }

1143    /// Tear `connection` down: mark it [`Connection::Failed`] and derive everything
1144    /// to report from the connection it replaced — the failure prefix, the peer
1145    /// ident, and the idents it routed. Deliver the failure (prefix, peer ident,
1146    /// reason), and — since a child cannot exist without its parent — tear the actor
1147    /// down if this was its parent connection. An already-failed connection yields
1148    /// nothing (a duplicate severance), so this is safe to call more than once.
1149    fn fail_connection(&mut self, connection: ConnectionRef, reason: Vec<u8>) {
1150        let Some((mut failure_prefix, peer_ident, routed_idents)) = self
1151            .connection_set(connection, Connection::Failed)
1152            .into_failure_report()
1153        else {
1154            return;
1155        };
1156        let actor = connection.owning_actor();
1157        failure_prefix.push(MsgPart::from_bytes(peer_ident));
1158        failure_prefix.push(MsgPart::from_bytes(reason.clone()));
1159        self.deliver_to_actor(actor, failure_prefix);
1160        match connection.role() {
1161            // The failed connection was this actor's parent: the actor cannot
1162            // outlive its parent, so it dies (which severs its own children).
1163            Role::Child => self.die_actor(actor, reason),
1164            // The failed connection was a child: everything that was live through it
1165            // is now dead. Publish exactly that — as if those idents had arrived dead
1166            // over the connection — which marks them dead, fires monitors, and
1167            // propagates up. (`slot` goes unused: there are no live idents to record.)
1168            Role::Parent => {
1169                let ConnectionRef::ChildConnection { slot, .. } = connection else {
1170                    unreachable!("a Role::Parent connection is a child connection");
1171                };
1172                self.populate_routes(actor, slot, Vec::new(), routed_idents.into_iter().collect());
1173                // Any gateways reachable only through this now-failed connection are
1174                // implicitly dead — including ones nested below us under a
1175                // non-gateway child. Begin the gateway-death propagation for them
1176                // (climbing toward the root, hence `downward: false`). A
1177                // gateway-route entry is always non-empty, so there is nothing to
1178                // guard against. (This also covers a parent actor's own death: its
1179                // death severs its parent link, and *that* actor's parent detects the
1180                // failure here, holding the aggregated gateway routes for the whole
1181                // lost subtree.)
1182                if let Some(tags) = self.actor_mut(actor).gateway_routes.remove(&slot) {
1183                    self.gateway_died(actor, tags.into_iter().collect(), false);
1184                }
1185            }
1186        }
1187    }

1189    /// Finalize a connecting connection from the peer's `Establish`. On any failure
1190    /// (peer dead, roles disagree, name conflict, ...) the connection is parked as
1191    /// `Established` carrying only its failure info — prefix and the peer ident to
1192    /// blame — and `Err(reason)` is returned so the caller's `fail_connection` tears
1193    /// it down uniformly. Routes are empty in that case (it never really came up).
1194    fn establish_connection(
1195        &mut self,
1196        connection: ConnectionRef,
1197        peer_role: Role,
1198        peer_name: Option<Vec<u8>>,
1199        requested_name: Option<Vec<u8>>,
1200        peer_alive: bool,
1201    ) -> Result<(), Vec<u8>> {
1202        // Take the connecting connection apart; establishing it consumes the
1203        // hello/failure prefixes, any commands queued before it was ready, and the
1204        // transport stashed when the pipe connected.
1205        let local_status = std::mem::replace(self.connection_mut(connection), Connection::Failed);
1206        let Connection::Connecting {
1207            transport,
1208            name_for_other,
1209            hello_prefix,
1210            failure_prefix,
1211            queued_commands,
1212        } = local_status
1213        else {
1214            unreachable!("peer Establish should arrive only on a connecting connection");
1215        };
1216
1217        let actor = connection.owning_actor();
1218        // Peer ident to blame in a failure: the peer's own ident, else whatever we
1219        // named it. Resolved up front so it is available on the error path.
1220        let failure_peer_ident = peer_name
1221            .clone()
1222            .or_else(|| name_for_other.clone())
1223            .unwrap_or_default();
1224
1225        // Validate and resolve the two idents we need, without mutating actor state.
1226        let resolved = 'resolve: {
1227            if !peer_alive {
1228                break 'resolve Err(b"peer actor died".to_vec());
1229            }
1230            if connection.role() == peer_role {
1231                break 'resolve Err(b"connection roles do not agree".to_vec());
1232            }
1233            if !self.connection_topology_is_valid(connection) {
1234                break 'resolve Err(b"invalid parent-child topology".to_vec());
1235            }
1236            // Resolve the name we will end up with (erroring on a conflict), without
1237            // mutating yet — the actual adopt-and-flush happens via assign_name once
1238            // the connection is established and deferred monitors can route.
1239            let local_ident = if let Some(requested_name) = requested_name {
1240                match self.actor(actor).name() {
1241                    Some(existing) if existing != requested_name.as_slice() => {
1242                        break 'resolve Err(b"actor name conflict".to_vec());
1243                    }
1244                    Some(existing) => existing.to_vec(),
1245                    None => requested_name,
1246                }
1247            } else {
1248                match self.actor(actor).name() {
1249                    Some(name) => name.to_vec(),
1250                    None => break 'resolve Err(b"actor has no ident".to_vec()),
1251                }
1252            };
1253            match peer_name.or(name_for_other) {
1254                Some(peer_ident) => Ok((local_ident, peer_ident)),
1255                None => Err(b"peer actor has no ident".to_vec()),
1256            }
1257        };
1258
1259        let (local_ident, peer_ident) = match resolved {
1260            Ok(idents) => idents,
1261            Err(reason) => {
1262                // Park the connection carrying just its failure info so
1263                // fail_connection can report it from the connection like any sever.
1264                self.connection_set(
1265                    connection,
1266                    Connection::Established {
1267                        transport,
1268                        failure_prefix,
1269                        peer_ident: failure_peer_ident,
1270                        routed_idents: HashSet::new(),
1271                    },
1272                );
1273                return Err(reason);
1274            }
1275        };
1276
1277        self.connection_set(
1278            connection,
1279            Connection::Established {
1280                transport,
1281                failure_prefix,
1282                peer_ident: peer_ident.clone(),
1283                routed_idents: HashSet::new(),
1284            },
1285        );
1286        let connection_entry = self.connection_mut(connection);
1287        let mut queued_commands = queued_commands;
1288        while let Some(command) = queued_commands.pop_front() {
1289            let _ = connection_entry.send(command);
1290        }
1291
1292        // Adopt the name now that the connection is established: this also replays
1293        // any monitor subscriptions deferred while we were unnamed, which can now
1294        // route up through this freshly-established link.
1295        self.assign_name(actor, local_ident.clone());
1296
1297        self.publish_routes_after_established(connection, local_ident.clone(), peer_ident.clone());
1298
1299        let mut msg = hello_prefix;
1300        msg.push(MsgPart::from_bytes(local_ident));
1301        msg.push(MsgPart::from_bytes(peer_ident));
1302        self.deliver_to_actor(actor, msg);
1303        Ok(())
1304    }

1359    /// Propagate a gateway death through `at`. First forget any routes to the dead
1360    /// gateways here. Then, unless `downward`, forward the announcement up the
1361    /// parent toward the root (which is what carries it there); `downward`
1362    /// masks off that link, as does simply having no parent (the root). Once the
1363    /// announcement is heading down it is broadcast — but with a difference between
1364    /// the two ways down: a wave arriving *from above* (`downward`) is relayed
1365    /// immediately, while the root turn-around (`!downward`, no parent) instead
1366    /// *coalesces* — see [`Self::enqueue_root_gateway_deaths`] — so a storm of
1367    /// deaths detected at (or climbing to) the root collapses into one fan-out per
1368    /// window rather than one per death.
1369    fn gateway_died(&mut self, at: Key, dead: Vec<Vec<u8>>, downward: bool) {
1370        // Forget routes to the dead gateways: they no longer route anywhere, and a
1371        // child entry left empty is dropped. Idempotent — an absent tag is a no-op.
1372        let entry = self.actor_mut(at);
1373        for tags in entry.gateway_routes.values_mut() {
1374            for tag in &dead {
1375                tags.remove(tag);
1376            }
1377        }
1378        entry.gateway_routes.retain(|_, tags| !tags.is_empty());
1379
1380        if !downward {
1381            // Still climbing toward the root: forward up and stop here.
1382            if let Some(parent) = self.actor_mut(at).parent.as_mut() {
1383                let _ = parent.send(ConnectionCommand::GatewayDied { dead });
1384                return;
1385            }
1386            // Root turn-around: coalesce instead of broadcasting immediately.
1387            self.enqueue_root_gateway_deaths(at, dead);
1388            return;
1389        }
1390
1391        // Arrived from above: relay this wave down immediately.
1392        self.broadcast_gateway_deaths(at, dead);
1393    }

1395    /// Hand `dead` to this root's coalescing coroutine, spawning it on the first
1396    /// turn-around. The coroutine (see [`Self::spawn_gateway_death_coroutine`]) owns
1397    /// all the timing: it batches every death within a window and emits a single
1398    /// [`Command::PropagateGatewayDeaths`], bounding a storm to one broadcast per
1399    /// window instead of one per death. Without a tokio runtime (unit tests) there
1400    /// is no coroutine, so broadcast immediately.
1401    fn enqueue_root_gateway_deaths(&mut self, at: Key, dead: Vec<Vec<u8>>) {
1402        if tokio::runtime::Handle::try_current().is_err() {
1403            self.broadcast_gateway_deaths(at, dead);
1404            return;
1405        }
1406        let loop_tx = self.loop_tx.clone();
1407        let tx = self
1408            .actor_mut(at)
1409            .gateway_death_tx
1410            .get_or_insert_with(|| Self::spawn_gateway_death_coroutine(loop_tx, at));
1411        let _ = tx.send(dead);
1412    }

1414    /// Spawn the per-root coalescing coroutine and return the channel that feeds it
1415    /// dead-tag packets. It owns the whole coalescing policy — an adaptive window
1416    /// that grows with evidence of a storm, plus a spacing floor that rate-limits
1417    /// broadcasts — all derived from one number, [`gateway_broadcast_cost`] (`C`),
1418    /// the wall-clock cost of fanning one broadcast out to every gateway. Each round
1419    /// collects deaths until a deadline, broadcasts, and repeats; the deadline is
1420    ///
1421    /// ```text
1422    /// deadline = max(last_broadcast + C, batch_start + wait)
1423    /// ```
1424    ///
1425    /// where the two terms serve different jobs:
1426    ///
1427    /// - `batch_start + wait` is the **adaptive window** shaping the leading edge:
1428    ///   `wait` starts at `C/16` (**initial wait** — a lone death, the common case,
1429    ///   fires this fast) and doubles per genuinely-new tag up to `C` (**max wait**),
1430    ///   anchored at the batch's first death so it never slides past `+ C`. The more
1431    ///   deaths arrive, the longer we wait to batch them.
1432    /// - `last_broadcast + C` is the **spacing floor**: the root cannot fan out
1433    ///   faster than `C`, so the next broadcast is held to at least `C` after the
1434    ///   previous one regardless of arrival pattern (deaths that land during the gap
1435    ///   are collected into the batch as they arrive). This closes the moderate
1436    ///   rolling-failure regime that the adaptive window alone would miss — one
1437    ///   fan-out per death spaced under `C`. `last_broadcast` is `None` until the
1438    ///   first broadcast, so the leading edge is never gated by it.
1439    ///
1440    /// Cross-broadcast duplicates need no handling here: `broadcast_gateway_deaths`
1441    /// dedups them against the root's gateway state. The coroutine ends when the
1442    /// channel closes (the actor — and its sender — is torn down) or the loop dies.
1443    fn spawn_gateway_death_coroutine(
1444        loop_tx: mpsc::UnboundedSender<Command>,
1445        at: Key,
1446    ) -> mpsc::UnboundedSender<Vec<Vec<u8>>> {
1447        let (tx, mut rx) = mpsc::unbounded_channel::<Vec<Vec<u8>>>();
1448        let broadcast_time = gateway_broadcast_cost();
1449        let max_wait = broadcast_time;
1450        let initial_wait = broadcast_time / 16;
1451        // The collection deadline: the adaptive window, but never sooner than
1452        // `broadcast_time` after the previous broadcast.
1453        let deadline_for = move |batch_start: tokio::time::Instant,
1454                                 last_broadcast: Option<tokio::time::Instant>,
1455                                 wait: Duration| {
1456            let adaptive = batch_start + wait;
1457            match last_broadcast {
1458                Some(last) => adaptive.max(last + broadcast_time),
1459                None => adaptive,
1460            }
1461        };
1462        tokio::task::spawn_local(async move {
1463            let mut last_broadcast: Option<tokio::time::Instant> = None;
1464            loop {
1465                // Idle: block for the first death of a batch.
1466                let mut batch: HashSet<Vec<u8>> = match rx.recv().await {
1467                    Some(first) => first.into_iter().collect(),
1468                    None => return, // channel closed: actor torn down
1469                };
1470
1471                // Collect until the deadline, growing `wait` on each new tag and
1472                // recomputing the deadline (which also enforces the spacing floor).
1473                let batch_start = tokio::time::Instant::now();
1474                let mut wait = initial_wait;
1475                let deadline =
1476                    tokio::time::sleep_until(deadline_for(batch_start, last_broadcast, wait));
1477                tokio::pin!(deadline);
1478                loop {
1479                    tokio::select! {
1480                        _ = &mut deadline => break,
1481                        packet = rx.recv() => match packet {
1482                            None => return,
1483                            Some(more) => {
1484                                let before = batch.len();
1485                                batch.extend(more);
1486                                if batch.len() != before {
1487                                    wait = (wait * 2).min(max_wait);
1488                                    deadline
1489                                        .as_mut()
1490                                        .reset(deadline_for(batch_start, last_broadcast, wait));
1491                                }
1492                            }
1493                        },
1494                    }
1495                }
1496
1497                // Broadcast the batch and record when, so the next round is spaced.
1498                let dead: Vec<Vec<u8>> = batch.into_iter().collect();
1499                if loop_tx
1500                    .send(Command::PropagateGatewayDeaths { at, dead })
1501                    .is_err()
1502                {
1503                    return; // loop gone
1504                }
1505                last_broadcast = Some(tokio::time::Instant::now());
1506            }
1507        });
1508        tx
1509    }

1511    /// Record the deaths at `at` and fan them out down every gateway-route child, so
1512    /// every gateway below is reached. A gateway records each death in its
1513    /// `gateway_state` (deduplicating against an already-`Dead` entry so repeated
1514    /// waves die out) and, at the moment of transition, fires every cross-gateway
1515    /// monitor it held for that gateway. A non-gateway holds no state and merely
1516    /// relays. Recording here (on the way down, never up) is what lets the broadcast
1517    /// returning from the root still reach the detector's other gateway children.
1518    fn broadcast_gateway_deaths(&mut self, at: Key, dead: Vec<Vec<u8>>) {
1519        let mut to_fire: Vec<(Vec<u8>, Vec<u8>)> = Vec::new(); // (listener, monitoring)
1520        let newly: Vec<Vec<u8>> = match &mut self.actor_mut(at).gateway {
1521            GatewayState::Gateway { gateway_state } => {
1522                let mut newly = Vec::new();
1523                for tag in dead {
1524                    if matches!(gateway_state.get(&tag), Some(GatewayMonitors::Dead)) {
1525                        continue; // already dead; broadcast already fanned out
1526                    }
1527                    if let Some(GatewayMonitors::Subscribed(subs)) =
1528                        gateway_state.insert(tag.clone(), GatewayMonitors::Dead)
1529                    {
1530                        to_fire.extend(subs.into_iter().map(|m| (m.listener, m.monitoring)));
1531                    }
1532                    newly.push(tag);
1533                }
1534                newly
1535            }
1536            GatewayState::NotAGateway => dead,
1537        };
1538        for (listener, monitoring) in to_fire {
1539            self.route_message(
1540                at,
1541                listener,
1542                SendPayload::FireMonitor {
1543                    to_monitor: monitoring,
1544                    is_timeout: false,
1545                },
1546            );
1547        }
1548        if newly.is_empty() {
1549            return;
1550        }
1551        let slots: Vec<ChildConnectionKey> =
1552            self.actor(at).gateway_routes.keys().copied().collect();
1553        for slot in slots {
1554            if let Some(connection) = self.actor_mut(at).children.get_mut(slot) {
1555                let _ = connection.send(ConnectionCommand::GatewayDied {
1556                    dead: newly.clone(),
1557                });
1558            }
1559        }
1560    }

1633    /// Handle a message that arrived over a gateway side-channel. The owning gateway
1634    /// is resolved *once* from `gateway_for_actor` (see [`Self::gateway_for`]),
1635    /// uniformly for every action. If it cannot be resolved yet, the whole
1636    /// self-addressing message is held in `pending_side_channel` and replayed when a
1637    /// route to `gateway_for_actor` is learned (see `populate_routes`) — there is no
1638    /// per-action resolution or dropping. Once resolved, the action is dispatched:
1639    ///
1640    /// - `Send`: hand off to `route_message`, which delivers a live route, drops a
1641    ///   dead one, and buffers an `Unknown` entry.
1642    /// - `UpdateRemoteMonitorState`: register/cancel the monitor at the owning
1643    ///   gateway via the *same* `route_monitor_change` (where the target is now
1644    ///   same-domain, so it becomes an ordinary local `MonitorSub`); a subscribe is
1645    ///   then confirmed back to the listener's gateway with `AckRemoteMonitor`.
1646    /// - `AckRemoteMonitor`: mark the matching held `MonitorToFire` acked so its
1647    ///   "must exist" timer stops being able to declare the gateway dead.
1648    fn deliver_side_channel(&mut self, message: SideChannelMessage) {
1649        let Some(gw) = self.gateway_for(&message.gateway_for_actor) else {
1650            self.pending_side_channel
1651                .entry(message.gateway_for_actor.clone())
1652                .or_default()
1653                .push(message);
1654            return;
1655        };
1656        let SideChannelMessage {
1657            gateway_for_actor,
1658            action,
1659        } = message;
1660        match action {
1661            SideChannelAction::Send(payload) => self.route_message(gw, gateway_for_actor, payload),
1662            SideChannelAction::UpdateRemoteMonitorState { listener, op } => {
1663                let is_subscribe = matches!(op, MonitorOp::Subscribe { .. });
1664                self.route_monitor_change(gw, listener.clone(), gateway_for_actor.clone(), op);
1665                if is_subscribe {
1666                    // Confirm the registration back to the listener's gateway.
1667                    self.send_to_gateway(SideChannelMessage {
1668                        gateway_for_actor: listener,
1669                        action: SideChannelAction::AckRemoteMonitor {
1670                            monitoring: gateway_for_actor,
1671                        },
1672                    });
1673                }
1674            }
1675            SideChannelAction::AckRemoteMonitor { monitoring } => {
1676                let tag = gateway_tag(&monitoring).to_vec();
1677                if let GatewayState::Gateway { gateway_state } = &mut self.actor_mut(gw).gateway {
1678                    if let Some(GatewayMonitors::Subscribed(subs)) = gateway_state.get_mut(&tag) {
1679                        for m in subs.iter_mut() {
1680                            if m.listener == gateway_for_actor && m.monitoring == monitoring {
1681                                m.acked = true;
1682                            }
1683                        }
1684                    }
1685                }
1686            }
1687        }
1688    }

1691    /// routes or names it, else (for a target not yet created) the gateway whose own
1692    /// `@tag` serves `ident`'s `@tag`. An exact route/name match always wins over the
1693    /// tag fallback. `None` if no gateway owns it yet (the caller holds or drops the
1694    /// message accordingly).
1695    fn gateway_for(&self, ident: &[u8]) -> Option<Key> {
1696        if let Some(key) = self.gateways.iter().copied().find(|&key| {
1697            let actor = &self.actors[key];
1698            actor.name() == Some(ident) || actor.routes.contains_key(ident)
1699        }) {
1700            return Some(key);
1701        }
1702        let tag = gateway_tag(ident);
1703        if tag.is_empty() {
1704            return None;
1705        }
1706        self.gateways
1707            .iter()
1708            .copied()
1709            .find(|&key| gateway_tag(self.actors[key].name().unwrap_or_default()) == tag)
1710    }

1712    /// Ship a self-addressing side-channel message to the gateway that owns
1713    /// `msg.gateway_for_actor`, dialing the side-channel at that actor's `@tag`.
1714    /// Without a tokio runtime (the sync unit-test harness) this is a no-op; those
1715    /// tests drive the receive side (`deliver_side_channel`) directly.
1716    fn send_to_gateway(&mut self, msg: SideChannelMessage) {
1717        if tokio::runtime::Handle::try_current().is_err() {
1718            return;
1719        }
1720        let tag = gateway_tag(&msg.gateway_for_actor);
1721        if tag.is_empty() {
1722            tracing::warn!("side-channel message has no gateway specifier; dropping");
1723            return;
1724        }
1725        match std::str::from_utf8(tag) {
1726            Ok(tag) => {
1727                let tag = tag.to_owned();
1728                // Pick the transport from the tag's scheme: a `tcp://` gateway is
1729                // reached over tcp, everything else (a bare `@addr`, for backward
1730                // compat, or an explicit `quic://`) over quic. The tag is passed
1731                // through verbatim; each transport's `parse_addr` strips its own
1732                // scheme prefix.
1733                if tag.starts_with("tcp://") {
1734                    self.tcp.send_to_gateway(tag, msg);
1735                } else {
1736                    self.quic.send_to_gateway(tag, msg);
1737                }
1738            }
1739            Err(_) => {
1740                tracing::warn!("gateway destination has non-utf8 specifier; dropping message");
1741            }
1742        }
1743    }

1747    /// Assign `name` to `actor` (a no-op if already named) and send any target
1748    /// subscriptions that were deferred while it was unnamed — now they have a
1749    /// fire-back address. The connection that carried the name must already be
1750    /// established so the deferred subscriptions can route up.
1751    fn assign_name(&mut self, actor: Key, name: Vec<u8>) {
1752        let entry = self.actor_mut(actor);
1753        if matches!(entry.name, ActorName::Named(_)) {
1754            return; // already named; nothing was deferred
1755        }
1756        entry.name = ActorName::Named(name.clone());
1757        // Everything watched while unnamed has its subscription held back; send
1758        // them all now, each carrying the timeout stored on its record. An unnamed
1759        // actor only ever holds `Active` entries (an emptied target is removed,
1760        // never left `PendingUnsubscribe` — that path only exists for named
1761        // actors), so every one needs exactly one upstream `Subscribe`.
1762        let deferred: Vec<(Vec<u8>, u64)> = entry
1763            .monitored
1764            .iter()
1765            .filter_map(|(target, tm)| match tm {
1766                TargetMonitors::Active { timeout_ms, .. } => Some((target.clone(), *timeout_ms)),
1767                TargetMonitors::PendingUnsubscribe(_) => None,
1768            })
1769            .collect();
1770        for (to_monitor, timeout_ms) in deferred {
1771            self.route_monitor_change(
1772                actor,
1773                name.clone(),
1774                to_monitor,
1775                MonitorOp::Subscribe { timeout_ms },
1776            );
1777        }
1778    }

1780    /// Add a local monitor (`id`, `failure_prefix`) on `to_monitor`. The first
1781    /// monitor for a target sends one upstream `Subscribe` (deferred until named).
1782    /// Joining an `Active` target, or re-monitoring a `PendingUnsubscribe` one
1783    /// (whose subscription is still live — we abort its pending unsubscribe),
1784    /// reuses the existing subscription, so churn sends no extra traffic.
1785    fn monitor_add(
1786        &mut self,
1787        actor: Key,
1788        id: u64,
1789        to_monitor: Vec<u8>,
1790        failure_prefix: Vec<MsgPart>,
1791        timeout_ms: u64,
1792    ) {
1793        let entry = self.actor_mut(actor);
1794        match entry.monitored.get_mut(&to_monitor) {
1795            Some(TargetMonitors::Active { entries, .. }) => {
1796                // Only the first monitor on a target arms a timeout; later ones
1797                // join the existing subscription and ignore their `timeout_ms`.
1798                entries.insert(id, failure_prefix);
1799                return; // subscription already live (or deferred); nothing to send
1800            }
1801            Some(TargetMonitors::PendingUnsubscribe(_)) => {
1802                // Re-monitor within the grace window: cancel the pending unsubscribe
1803                // and reactivate. The upstream subscription was never torn down, so
1804                // we do not re-subscribe (and do not re-arm a timeout).
1805                if let Some(TargetMonitors::PendingUnsubscribe(timer)) =
1806                    entry.monitored.remove(&to_monitor)
1807                {
1808                    timer.abort();
1809                }
1810                entry.monitored.insert(
1811                    to_monitor,
1812                    TargetMonitors::Active {
1813                        entries: HashMap::from([(id, failure_prefix)]),
1814                        timeout_ms: 0,
1815                    },
1816                );
1817                return;
1818            }
1819            None => {}
1820        }
1821        // Brand-new target: record it (storing the timeout), then subscribe now if
1822        // named, else defer to `assign_name` (the local record still allows cancel
1823        // meanwhile and carries the timeout until naming).
1824        entry.monitored.insert(
1825            to_monitor.clone(),
1826            TargetMonitors::Active {
1827                entries: HashMap::from([(id, failure_prefix)]),
1828                timeout_ms,
1829            },
1830        );
1831        let dest = match &entry.name {
1832            ActorName::Named(name) => name.clone(),
1833            ActorName::Unknown { .. } => return,
1834        };
1835        self.route_monitor_change(actor, dest, to_monitor, MonitorOp::Subscribe { timeout_ms });
1836    }

1838    /// Remove local monitor `id` (found by scanning its target). Dropping the last
1839    /// monitor on a target does *not* unsubscribe immediately: it flips the target
1840    /// to `PendingUnsubscribe` and schedules a debounced unsubscribe, so churn does
1841    /// not hammer the hierarchy. An in-flight fire is dropped because no local
1842    /// monitors remain.
1843    fn monitor_remove(&mut self, actor: Key, id: u64) {
1844        // Cloned up front so the spawned timer can own it without conflicting with
1845        // the `entry` borrow below.
1846        let loop_tx = self.loop_tx.clone();
1847        let entry = self.actor_mut(actor);
1848        let Some(to_monitor) = entry
1849            .monitored
1850            .iter()
1851            .find(|(_, tm)| match tm {
1852                TargetMonitors::Active { entries, .. } => entries.contains_key(&id),
1853                TargetMonitors::PendingUnsubscribe(_) => false,
1854            })
1855            .map(|(target, _)| target.clone())
1856        else {
1857            return;
1858        };
1859        let now_empty = match entry.monitored.get_mut(&to_monitor) {
1860            Some(TargetMonitors::Active { entries, .. }) => {
1861                entries.remove(&id);
1862                entries.is_empty()
1863            }
1864            _ => return,
1865        };
1866        if !now_empty {
1867            return; // other monitors still watch this target
1868        }
1869        if !matches!(entry.name, ActorName::Named(_)) {
1870            // Never sent upstream (deferred while unnamed); just drop it locally.
1871            entry.monitored.remove(&to_monitor);
1872            return;
1873        }
1874        match tokio::runtime::Handle::try_current() {
1875            Ok(_) => {
1876                // Schedule the debounced unsubscribe and park its abort handle in
1877                // the entry, which now becomes `PendingUnsubscribe`.
1878                let target = to_monitor.clone();
1879                let timer = tokio::task::spawn_local(async move {
1880                    tokio::time::sleep(MONITOR_DEBOUNCE).await;
1881                    let _ = loop_tx.send(Command::UnsubscribeMonitor {
1882                        actor,
1883                        to_monitor: target,
1884                    });
1885                });
1886                entry.monitored.insert(
1887                    to_monitor,
1888                    TargetMonitors::PendingUnsubscribe(timer.abort_handle()),
1889                );
1890            }
1891            Err(_) => {
1892                // No runtime (unit-test harness): unsubscribe synchronously.
1893                entry.monitored.remove(&to_monitor);
1894                self.unsubscribe_now(actor, to_monitor);
1895            }
1896        }
1897    }

1899    /// Debounce timer fired. Act only if the target is still `PendingUnsubscribe`
1900    /// — i.e. it was not re-monitored (which would have flipped it back to
1901    /// `Active` and aborted this task) nor torn down by a fire. The variant itself
1902    /// is the truth, so there is no generation to check.
1903    fn unsubscribe_monitor(&mut self, actor: Key, to_monitor: Vec<u8>) {

1905            return;

1908            entry.monitored.get(&to_monitor),
1909            Some(TargetMonitors::PendingUnsubscribe(_))

1912        }

1926    /// Apply a monitor subscribe/cancel, sending it to the actor responsible for
1927    /// `target`. (Replaces the base's `route_ancestor`.) The responsible actor is
1928    /// chosen by the target's domain, relative to ours:
1929    ///
1930    /// - **Root domain** (blank tag): the common ancestor that routes the target —
1931    ///   climbing transparently *past* gateways, up to the root. Held locally there.
1932    /// - **Our own domain**: the common ancestor, or — failing that — our gateway, the
1933    ///   top of the domain, which holds it until the target appears. Held locally.
1934    /// - **Another gateway's domain**: our gateway, which mirrors the monitor onto the
1935    ///   gateway that owns the target over a side-channel.
1936    ///
1937    /// Until we reach that actor we forward the change one hop up (see
1938    /// [`Self::forward_monitor_up`]).
1939    fn route_monitor_change(&mut self, at: Key, listener: Vec<u8>, target: Vec<u8>, op: MonitorOp) {
1940        let target_domain = gateway_tag(&target);
1941        let my_domain = gateway_tag(self.actor(at).name().unwrap_or_default());
1942        let routes_target = self.actor(at).routes.contains_key(target.as_slice());
1943        let is_gateway = self.actor(at).is_gateway();
1944
1945        if target_domain.is_empty() {
1946            // Root domain: only the common ancestor that routes the target handles it;
1947            // gateways are transparent, so an unrouted target keeps climbing to the root.
1948            if routes_target {
1949                self.establish_local_monitor(at, listener, target, op);
1950            } else {
1951                self.forward_monitor_up(at, listener, target, op);
1952            }
1953        } else if target_domain == my_domain {
1954            // Our own domain: the common ancestor, or our gateway (the top of the
1955            // domain), holds it; anyone below them forwards up.
1956            if routes_target || is_gateway {
1957                self.establish_local_monitor(at, listener, target, op);
1958            } else {
1959                self.forward_monitor_up(at, listener, target, op);
1960            }
1961        } else {
1962            // Another gateway's domain: our gateway mirrors it onto the owning gateway;
1963            // anyone below the gateway forwards up to it.
1964            if is_gateway {
1965                self.mirror_monitor_to_gateway(at, listener, target, op);
1966            } else {
1967                self.forward_monitor_up(at, listener, target, op);
1968            }
1969        }
1970    }

1987    /// Establish a *local* monitor on `target`'s route at the responsible actor `at`
1988    /// (the target is in `at`'s own domain). A subscribe fires immediately if the

1991    /// while the route is `Unknown` — on a `Connection` route the target exists, so a

2025                    to_monitor: target,

2034                dest: listener.clone(),

2046    /// Mirror a cross-gateway monitor onto the gateway that owns `target`, from the
2047    /// subscribing gateway `at`. A subscribe records an (unacked) `MonitorToFire`
2048    /// against the owning gateway's tag — so a later death of that gateway fires the
2049    /// listener — unless the gateway is already known dead, in which case it fires
2050    /// straight back and holds nothing. A cancel drops the matching `MonitorToFire`,
2051    /// removing the tag entry if it empties so a stale timer cannot later declare the
2052    /// gateway dead. Either way the op is relayed to the owning gateway (which applies
2053    /// it as an ordinary local monitor on `target`); a subscribe additionally arms the
2054    /// "must exist" timer that declares the gateway dead if it never acknowledges.
2055    fn mirror_monitor_to_gateway(
2056        &mut self,
2057        at: Key,
2058        listener: Vec<u8>,
2059        target: Vec<u8>,
2060        op: MonitorOp,
2061    ) {
2062        let tag = gateway_tag(&target).to_vec();
2063        match op {
2064            MonitorOp::Subscribe { timeout_ms } => {
2065                let already_dead = matches!(
2066                    &self.actor(at).gateway,
2067                    GatewayState::Gateway { gateway_state }
2068                        if matches!(gateway_state.get(&tag), Some(GatewayMonitors::Dead))
2069                );
2070                if already_dead {
2071                    self.route_message(
2072                        at,
2073                        listener,
2074                        SendPayload::FireMonitor {
2075                            to_monitor: target,
2076                            is_timeout: false,
2077                        },
2078                    );
2079                    return;
2080                }
2081                let GatewayState::Gateway { gateway_state } = &mut self.actor_mut(at).gateway
2082                else {
2083                    unreachable!("a cross-gateway target is handled only at a gateway");
2084                };
2085                let GatewayMonitors::Subscribed(subs) = gateway_state
2086                    .entry(tag)
2087                    .or_insert_with(|| GatewayMonitors::Subscribed(Vec::new()))
2088                else {
2089                    unreachable!("a known-dead gateway fired above and returned");
2090                };
2091                subs.push(MonitorToFire {
2092                    acked: false,
2093                    listener: listener.clone(),
2094                    monitoring: target.clone(),
2095                });
2096                self.relay_to_owning_gateway(&listener, &target, op);
2097                self.arm_must_exist_timer(at, listener, target, timeout_ms);
2098            }
2099            MonitorOp::Unsubscribe => {
2100                if let GatewayState::Gateway { gateway_state } = &mut self.actor_mut(at).gateway {
2101                    if let Some(GatewayMonitors::Subscribed(subs)) = gateway_state.get_mut(&tag) {
2102                        subs.retain(|m| !(m.listener == listener && m.monitoring == target));
2103                        if subs.is_empty() {
2104                            gateway_state.remove(&tag);
2105                        }
2106                    }
2107                }
2108                self.relay_to_owning_gateway(&listener, &target, op);
2109            }
2110        }
2111    }

2147    /// A "must exist" timer armed at `at` expired. Remote and local cases are
2148    /// mutually exclusive (only a cross-gateway target has a `gateway_state` entry;
2149    /// only a same-domain one has a local route):
2150    /// - Remote, still unacknowledged: the owning gateway never answered the
2151    ///   registration, so treat it as unreachable — declare it dead, which fires the
2152    ///   monitor as the death propagates back (Trigger 2).
2153    /// - Local, never appeared: the route is still `Unknown`, so the target never
2154    ///   existed — fire the non-existence timeout. Any other route state (it exists,
2155    ///   it died, or the subscription migrated up leaving no route) is a no-op.
2156    fn check_monitor_timeout(&mut self, at: Key, listener: Vec<u8>, target: Vec<u8>) {
2157        let tag = gateway_tag(&target).to_vec();
2158        let unacked_remote = match self.actors.get(at).map(|actor| &actor.gateway) {
2159            Some(GatewayState::Gateway { gateway_state }) => matches!(
2160                gateway_state.get(&tag),
2161                Some(GatewayMonitors::Subscribed(subs))
2162                    if subs.iter().any(|m| m.listener == listener && m.monitoring == target && !m.acked)
2163            ),
2164            _ => false,
2165        };
2166        if unacked_remote {
2167            self.gateway_died(at, vec![tag], false);
2168            return;
2169        }
2170        let still_unknown = matches!(
2171            self.actors
2172                .get(at)
2173                .map(|actor| actor.routes.get(target.as_slice())),
2174            Some(Some(Route::Unknown { .. }))
2175        );
2176        if still_unknown {
2177            self.route_message(
2178                at,
2179                listener,
2180                SendPayload::FireMonitor {
2181                    to_monitor: target,
2182                    is_timeout: true,
2183                },
2184            );
2185        }
2186    }

2188    /// A monitor firing (death or non-existence timeout) reached the monitoring
2189    /// actor. Fan it out to *every* local monitor on that target, reconstructing
2190    /// each one's failure message from its own prefix and the given `reason`, then
2191    /// drop the whole entry (the monitors are consumed). Because the entry is
2192    /// removed, the monitor cannot fire twice: a later real death routes a fire to
2193    /// an actor that no longer has the entry → no-op. A fire for a target with no
2194    /// entries — already cancelled-and-debounced-away, or already fired — is
2195    /// dropped.
2196    fn fire_local_monitors(&mut self, actor: Key, to_monitor: Vec<u8>, reason: &[u8]) {
2197        let entries = match self.actor_mut(actor).monitored.remove(&to_monitor) {
2198            Some(TargetMonitors::Active { entries, .. }) => entries,
2199            // All monitors were cancelled (sub still live during the grace); the
2200            // pending unsubscribe task is now moot — drop it and fire nothing.
2201            Some(TargetMonitors::PendingUnsubscribe(timer)) => {
2202                timer.abort();
2203                return;
2204            }
2205            None => return,
2206        };
2207        for failure_prefix in entries.into_values() {
2208            let mut msg = failure_prefix;
2209            msg.push(MsgPart::from_bytes(to_monitor.clone()));
2210            msg.push(MsgPart::from_bytes(reason.to_vec()));
2211            self.deliver_to_actor(actor, msg);
2212        }
2213    }
2214
2215    fn deliver_to_actor(&mut self, actor: Key, msg: Vec<MsgPart>) {
2216        let actor = self.actor_mut(actor);
2217        if !actor.alive {
2218            return;
2219        }
2220        match &mut actor.delivery {
2221            Delivery::NoPoller { buffered } => {
2222                buffered.push_back(msg);
2223            }
2224            Delivery::Poller { poller_key, index } => {
2225                let poller_key = *poller_key;
2226                let index = *index;
2227                let _ = actor;
2228                if let Some(poller) = self.pollers.get_mut(poller_key) {
2229                    poller.deliver(index, msg);
2230                }
2231            }
2232        }
2233    }

2350                                last_log = std::time::Instant::now();
2351                            }
2352                        }
2353                    }
2354                });
2355            })?;
2356
2357        let handle = Self { tx };
2358        handle.send_command(Command::Init { thread })?;
2359        Ok(handle)
2360    }
2361
2362    pub(crate) fn send_command(&self, command: Command) -> anyhow::Result<()> {
2363        self.tx
2364            .send(command)
2365            .map_err(|_| anyhow::anyhow!("context runtime stopped"))
2366    }
