9//! Protocol-independent connection transport, generic over a [`Net`].
10//!
11//! A connection's coroutines bring up the connection's streams, produce a
12//! [`ConnectionTransport`] for the data stream, and hand it to the command loop via
13//! `Command::TransportConnected`; the data reader forwards every decoded frame back
14//! as a `ConnectionAction`. Establishment policy (identity exchange, hello, liveness
15//! reporting) lives in the command loop and is identical across transports. Wire
16//! framing is shared (see [`crate::framing`]).
17//!
18//! ## Two streams per connection
19//!
20//! Every connection carries a data/control stream (index 0) and a heartbeat stream
21//! (index 1). A single reliable, in-order stream head-of-line-blocks everything queued
22//! behind a large message — including heartbeats, whose whole job is to keep flowing
23//! while data does. So we put heartbeats on their **own** stream, separate from the
24//! data/control stream; the heartbeat stream is given a higher send priority
25//! ([`PRIORITY_HEARTBEAT`]), so a beat is packed ahead of queued data even under a full
26//! congestion window.
27//!
28//! The heartbeat stream is materialized lazily (see [`crate::net`]): a message-only
29//! gateway side channel never opens it. A parent/child link always heartbeats, so it
30//! opens the stream off the establishment path — the first beat never delays either
31//! establishment or serve-pairing.
32//!
33//! ## Liveness
34//!
35//! A protocol like quic runs in userspace over an unreliable datagram, so there is
36//! no file-descriptor close to signal a lost peer. Instead an application-level
37//! **bidirectional heartbeat** runs on the heartbeat stream (see
38//! [`crate::heartbeat`]); a clean drop still finishes both streams so the peer
39//! also observes EOF — the heartbeat is the backstop for the unclean cases.

89/// Stream index of the data/control stream: the preamble and every

100/// Connect-retry backoff bounds. A join may precede its serve, and the handshake
101/// fails until the server is bound, so the connector polls — fast at first, backing
102/// off to a steady poll.
103const CONNECT_RETRY_MIN: Duration = Duration::from_millis(5);
104const CONNECT_RETRY_MAX: Duration = Duration::from_millis(1000);

106/// On graceful shutdown each writer sends an explicit `Severed{"context shutdown"}`
107/// frame (reliable stream data — retransmitted by the transport, unlike an abrupt
108/// connection close), then keeps the connection open until the peer *responds* (its
109/// own `Severed`/EOF, observed by our reader) before closing — so we close as soon
110/// as we know the peer got it, rather than after a fixed delay. This is the upper
111/// bound on that wait, for a peer that never responds (e.g. already dead). Tunable
112/// via `MM_QUIC_SHUTDOWN_ACK_TIMEOUT_MS`.
113const DEFAULT_SHUTDOWN_ACK_TIMEOUT: Duration = Duration::from_secs(10);
114
115fn shutdown_ack_timeout() -> Duration {
116    std::env::var("MM_QUIC_SHUTDOWN_ACK_TIMEOUT_MS")
117        .ok()
118        .and_then(|v| v.parse::<u64>().ok())
119        .map(Duration::from_millis)
120        .unwrap_or(DEFAULT_SHUTDOWN_ACK_TIMEOUT)
121}

123/// Optional cap on how many client connect *attempts* may be in flight at once
124/// across the whole context. A root that joins tens of thousands of peers spawns
125/// that many connector tasks, each driving a handshake; run all at once on the
126/// single-threaded runtime they compete for CPU and reorder connection setup badly.
127/// Bounds the simultaneous attempts (each connector still retries independently, and
128/// the permit is released while it backs off).
129///
130/// `MM_QUIC_MAX_CONCURRENT_CONNECTS` overrides it: a positive value caps to that;
131/// `0` means unlimited. When the env var is unset (or unparseable) the protocol's
132/// [`Net::default_connect_concurrency`] applies — 1024 for quic, a much smaller cap for
133/// tcp, which otherwise starves stragglers under a large connect storm.
134fn max_concurrent_connects<N: Net>() -> Option<usize> {
135    match std::env::var("MM_QUIC_MAX_CONCURRENT_CONNECTS") {
136        Ok(v) => match v.parse::<usize>() {
137            Ok(0) => None,                              // explicit opt-out ⇒ unlimited
138            Ok(n) => Some(n),                           // explicit cap
139            Err(_) => N::default_connect_concurrency(), // unparseable ⇒ protocol default
140        },
141        Err(_) => N::default_connect_concurrency(), // unset ⇒ protocol default
142    }
143}

145/// Build the optional multi-threaded runtime that the per-connection data
146/// coroutines (the message writer/reader) run on, off the single-threaded command
147/// loop. `MM_NET_DATA_THREADS` selects it: unset defaults to the striping width
148/// ([`crate::dataio::n_data_streams`]), `0` explicitly keeps every coroutine on the
149/// command-loop thread (no data parallelism); otherwise an `N`-worker runtime runs the
150/// data-stream framing/crypto/copy for many connections across cores while accounting
151/// stays serialized on one core. The worker count is floored at the stream width —
152/// striping is inert without at least one core per parallel stream to spread the
153/// per-stream crypto/copy across, so a bare `MM_NET_N_DATA_STREAMS` gets a runtime wide
154/// enough to actually parallelize it.
155///
156/// Only the two *data* coroutines move; the connection handle, heartbeat stream,
157/// side channels, and pairing all stay on the command-loop thread. The data stream
158/// halves were materialized on the command-loop runtime's I/O driver, so readiness is
159/// still driven there — this offloads the per-frame CPU work (notably kTLS-over-TCP's
160/// in-task syscalls/copy), not the epoll wakeups. Fully decoupling the reactor would
161/// require materializing the streams on this runtime, a later step.
162fn data_runtime() -> Option<tokio::runtime::Runtime> {
163    let requested = std::env::var("MM_NET_DATA_THREADS")
164        .ok()
165        .and_then(|v| v.parse::<usize>().ok());
166    let streams = crate::dataio::n_data_streams();
167    let n = match requested {
168        None => streams,           // unset ⇒ default to the stream width
169        Some(0) => 0,              // explicit 0 ⇒ force single-threaded (off)
170        Some(r) => r.max(streams), // explicit ⇒ honor it, but never below the width
171    };
172    if n == 0 {
173        return None;
174    }
175    match tokio::runtime::Builder::new_multi_thread()
176        .worker_threads(n)
177        .thread_name("mm-net-data")
178        .enable_all()
179        .build()
180    {
181        Ok(rt) => {
182            eprintln!("MM_NET_DATA_THREADS: net data coroutines on a {n}-worker runtime");
183            Some(rt)
184        }
185        Err(err) => {
186            tracing::warn!("MM_NET_DATA_THREADS set but runtime build failed: {err:#}");
187            None
188        }
189    }
190}

192/// The gateway dial tag of `ident` (the substring after the last `@`), as a
193/// `String`, or `None` if it has none or is non-utf8. This is the address a beat is
194/// dialed at — the recipient's own gateway.
195fn gateway_tag_str(ident: &[u8]) -> Option<String> {
196    let pos = ident.iter().rposition(|&b| b == b'@')?;
197    let tag = &ident[pos + 1..];
198    if tag.is_empty() {
199        return None;
200    }
201    std::str::from_utf8(tag).ok().map(str::to_owned)
202}

204/// What a connection's reader needs for shared memory: the context-global mapper and
205/// the owning actor's gateway-client slot. Only the *reader* uses it — to read a
206/// large incoming part straight into a slab block (every actor on a cross-machine
207/// link is a gateway, so it has a client). The writer needs nothing: a `Shm` part
208/// carries its own mapper.
209#[derive(Clone)]
210struct ShmCtx {
211    mapper: MapperHandle,
212    client: ShmClientSlot,
213}
214
215impl ShmCtx {
216    /// The owning actor's gateway-client slot, shared with a striping reader/decoder so
217    /// it can snapshot the client (`None` until learned; a gateway seeds its own at
218    /// creation, before any frame can arrive) and land striped/inline parts in the slab
219    /// as soon as the gateway is known.
220    fn client_slot(&self) -> ShmClientSlot {
221        self.client.clone()
222    }
223}

225/// What a gateway side-channel writer carries: either a routable
226/// [`SideChannelMessage`] (from ctx) or a transport-internal delegated-heartbeat
227/// beat/ack (from the heartbeat coroutines). Both ride the same per-gateway writer,
228/// so a beat reuses the exact connection an ordinary message would.
229enum SideChannelOut {
230    Message(SideChannelMessage),
231    Heartbeat {
232        recipient: Vec<u8>,
233        from: Vec<u8>,
234        conn_id: ConnectionId,
235        kind: BeatKind,
236    },
237}

239/// The client-side dialing + gateway side-channel state, shared (`Rc<RefCell>`)
240/// between the command-loop-facing [`NetTransport`] and the heartbeat coroutines.
241/// The transport owns every side-channel path end to end — opening, reusing, and
242/// writing it — so a delegated-heartbeat beat is sent by the heartbeat coroutine
243/// directly through here, never routed through ctx. Single-threaded (LocalSet),
244/// hence `Rc<RefCell>`. The protocol state ([`Net`]) is built lazily on first use.
245struct SideChannels<N: Net> {
246    shutdown_tx: watch::Sender<bool>,
247    /// The shared per-context networking state, built lazily on the first serve/join.
248    net: Option<N>,
249    /// One side-channel writer per remote gateway, keyed by its dial address (the
250    /// `@specifier` tag). Each owns a task that lazily connects (retrying until the
251    /// gateway is live), streams frames, and reconnects if the connection drops.
252    /// Cached across messages/beats and shared by ordinary messages and heartbeats.
253    channels: HashMap<String, mpsc::UnboundedSender<SideChannelOut>>,
254    /// Liveness-token issuer: each connection's/side-channel's writer holds a clone
255    /// for its lifetime. Teardown drops this issuing copy so `alive_rx` closes once
256    /// every writer has flushed and exited.
257    alive_tx: Option<mpsc::UnboundedSender<()>>,
258    /// The data runtime handle, handed to [`Net::create`] so a protocol whose work is
259    /// in a background driver (quic) builds its endpoints there. `None` ⇒ single core.
260    data_rt: Option<Handle>,
261}

263/// Owns all transport state and coroutines. The command loop holds one and forwards
264/// serves/joins to it; it never sees streams or pairing state.
265pub(crate) struct NetTransport<N: Net> {
266    loop_tx: mpsc::UnboundedSender<Command>,
267    // The context-global address-space mapper, captured once at construction and
268    // handed to every connection's reader so a large incoming part can be read
269    // straight into a slab block.
270    mapper: MapperHandle,
271    // The context's single shm client slot, used *only* by side-channel readers,
272    // which are not tied to any actor. A join/serve connection reads into its owning
273    // actor's own slot instead (passed through serve/join).
274    context_shm: ShmClientSlot,
275    // One listener coroutine per url; serve connections are forwarded to it and it
276    // owns the serve/accept pairing.
277    listeners: HashMap<String, mpsc::UnboundedSender<(ConnectionRef, ShmCtx)>>,
278    // Bounds simultaneous client connect attempts (see `max_concurrent_connects`).
279    // Shared by every connector task; `None` ⇒ unlimited.
280    connect_sem: Option<Arc<Semaphore>>,
281    // Closed once every writer has exited (see `SideChannels::alive_tx`).
282    alive_rx: mpsc::UnboundedReceiver<()>,
283    // Delegated-heartbeat state, shared by every connection's heartbeat coroutine.
284    heartbeat: Heartbeats,
285    // Client dialing + side-channel writers, shared with the heartbeat coroutines so
286    // the transport drives every side-channel path without ctx involvement.
287    side_channels: Rc<RefCell<SideChannels<N>>>,
288    // Optional multi-threaded runtime the per-connection data coroutines run on (see
289    // `data_runtime`). `None` ⇒ they run on the command-loop thread like everything
290    // else. Owned here for its lifetime; torn down (non-blocking) in `shutdown`.
291    data_rt: Option<tokio::runtime::Runtime>,
292}

294impl<N: Net> NetTransport<N> {
295    pub(crate) fn new(
296        loop_tx: mpsc::UnboundedSender<Command>,
297        mapper: MapperHandle,
298        context_shm: ShmClientSlot,
299    ) -> Self {
300        let (shutdown_tx, _) = watch::channel(false);
301        let (alive_tx, alive_rx) = mpsc::unbounded_channel();
302        if crate::ctx::connection_debug() {
303            if let Some(n) = max_concurrent_connects::<N>() {
304                eprintln!("MM_QUIC connect concurrency capped at {n} simultaneous attempts");
305            }
306        }
307        // The data runtime is owned here for its lifetime; its handle is shared with
308        // the side channels (→ `Net::create`, for quic's endpoint drivers) and with
309        // `spawn_connection` via `data_handle` (for the tcp data coroutines).
310        let data_rt = data_runtime();
311        let data_rt_handle = data_rt.as_ref().map(|rt| rt.handle().clone());
312        Self {
313            loop_tx,
314            mapper,
315            context_shm,
316            listeners: HashMap::new(),
317            connect_sem: max_concurrent_connects::<N>().map(|n| Arc::new(Semaphore::new(n))),
318            alive_rx,
319            heartbeat: Heartbeats::new(),
320            side_channels: Rc::new(RefCell::new(SideChannels {
321                shutdown_tx,
322                net: None,
323                channels: HashMap::new(),
324                alive_tx: Some(alive_tx),
325                data_rt: data_rt_handle,
326            })),
327            data_rt,
328        }
329    }
330
331    /// A handle to the data-coroutine runtime, or `None` to run them on the
332    /// command-loop thread. Cloned per connection and passed to [`spawn_connection`].
333    fn data_handle(&self) -> Option<Handle> {
334        self.data_rt.as_ref().map(|rt| rt.handle().clone())
335    }
336
337    /// A connection's shared-memory context: the context mapper paired with the given
338    /// client slot. A join/serve connection passes its owning actor's slot; a
339    /// side-channel passes [`Self::context_shm`].
340    fn shm_ctx(&self, client: ShmClientSlot) -> ShmCtx {
341        ShmCtx {
342            mapper: self.mapper.clone(),
343            client,
344        }
345    }
346
347    /// Send a self-addressing [`SideChannelMessage`] to a remote gateway over a
348    /// direct side-channel. The caller (the command loop) has already derived `tag`
349    /// from the message's `gateway_for_actor`.
350    pub(crate) fn send_to_gateway(&mut self, tag: String, message: SideChannelMessage) {
351        self.side_channels.borrow_mut().send_message(tag, message);
352    }
353
354    /// Signal every writer to flush and exit, then wait until they all have.
355    ///
356    /// Each writer, on this signal, sends an explicit `Severed{"context shutdown"}`
357    /// frame and holds its connection open until the peer *responds* so the frame is
358    /// reliably delivered before the connection is dropped. So by the time every
359    /// writer has exited (`alive_rx` closed) the peers have been told — we do not
360    /// rely on a lossy connection-close being noticed.
361    pub(crate) async fn shutdown(&mut self) {
362        self.side_channels.borrow_mut().begin_shutdown();
363        let _ = self.alive_rx.recv().await;
364        // Tear the data runtime down without blocking: we are inside the command
365        // loop's async context, where dropping a `Runtime` (which blocks on its worker
366        // threads) would panic. The writers above have already flushed and the peers
367        // acknowledged, so its tasks have nothing left to do.
368        if let Some(rt) = self.data_rt.take() {
369            rt.shutdown_background();
370        }
371    }
372}

374impl<N: Net> SideChannels<N> {
375    /// The shared networking state, built on first use.
376    fn net(&mut self) -> anyhow::Result<&mut N> {
377        if self.net.is_none() {
378            self.net = Some(N::create(self.data_rt.clone())?);
379        }
380        Ok(self.net.as_mut().expect("net just created"))
381    }
382
383    /// A dialing handle for `addr`, from the (lazily built) client pool.
384    fn dialer(&mut self, addr: SocketAddr) -> anyhow::Result<N::Dialer> {
385        self.net()?.dialer(addr)
386    }
387
388    /// Bind a server listener on `addr`. The stream count is not passed here: tcp learns
389    /// it from each connecting side's pairing prefix, and quic multiplexes on one
390    /// connection (see [`Net::bind`]).
391    fn bind(&mut self, addr: SocketAddr) -> anyhow::Result<N::Listener> {
392        self.net()?.bind(addr)
393    }
394
395    fn alive_token(&self) -> mpsc::UnboundedSender<()> {
396        self.alive_tx
397            .as_ref()
398            .expect("alive-token issuer present before shutdown")
399            .clone()
400    }
401
402    fn subscribe_shutdown(&self) -> watch::Receiver<bool> {
403        self.shutdown_tx.subscribe()
404    }
405
406    /// The writer for the gateway at `tag`, opening (and caching) it on first use.
407    /// `None` if the tag is unparseable or the dialer can't be built.
408    fn writer_for(&mut self, tag: String) -> Option<&mpsc::UnboundedSender<SideChannelOut>> {
409        if !self.channels.contains_key(&tag) {
410            let addr = match N::parse_addr(&tag) {
411                Ok(addr) => addr,
412                Err(err) => {
413                    tracing::warn!("side channel address: {err:#}");
414                    return None;
415                }
416            };
417            let dialer = match self.dialer(addr) {
418                Ok(dialer) => dialer,
419                Err(err) => {
420                    tracing::warn!("side channel dialer: {err:#}");
421                    return None;
422                }
423            };
424            let (tx, rx) = mpsc::unbounded_channel();
425            tokio::task::spawn_local(side_channel_writer_task::<N>(
426                dialer,
427                addr,
428                rx,
429                self.shutdown_tx.subscribe(),
430                self.alive_token(),
431            ));
432            self.channels.insert(tag.clone(), tx);
433        }
434        self.channels.get(&tag)
435    }
436
437    /// Send a self-addressing [`SideChannelMessage`] to the gateway at `tag`, opening
438    /// (and caching) the connection on first use. Best-effort and not heartbeated; a
439    /// message enqueued while the remote gateway is not yet live waits for the
440    /// connection to come up.
441    fn send_message(&mut self, tag: String, message: SideChannelMessage) {
442        if let Some(tx) = self.writer_for(tag) {
443            let _ = tx.send(SideChannelOut::Message(message));
444        }
445    }
446
447    /// Send a delegated-heartbeat beat/ack to `recipient`, dialing the side channel
448    /// at `recipient`'s own gateway `@tag` and reusing the same per-gateway side
449    /// channel an ordinary message would. Dropped if `recipient` has no gateway tag
450    /// to dial. Called by the heartbeat coroutines directly — never via ctx.
451    fn send_heartbeat(
452        &mut self,
453        recipient: Vec<u8>,
454        from: Vec<u8>,
455        conn_id: ConnectionId,
456        kind: BeatKind,
457    ) {
458        let Some(tag) = gateway_tag_str(&recipient) else {
459            return;
460        };
461        if let Some(tx) = self.writer_for(tag) {
462            let _ = tx.send(SideChannelOut::Heartbeat {
463                recipient,
464                from,
465                conn_id,
466                kind,
467            });
468        }
469    }
470
471    /// Signal every writer to flush and exit, and drop the issuing alive token.
472    fn begin_shutdown(&mut self) {
473        let _ = self.shutdown_tx.send(true);
474        self.alive_tx = None;
475    }
476}

478impl<N: Net> Transport for NetTransport<N> {
479    fn serve(&mut self, url: String, connection: ConnectionRef, shm_client: ShmClientSlot) {
480        // The first serve on a url spawns its listener coroutine; later serves just
481        // forward another connection to it. The listener carries the *context* shm
482        // context for side-channel readers (which have no owning actor); each Join
483        // connection instead uses the owning actor's slot, forwarded alongside it.
484        if !self.listeners.contains_key(&url) {
485            let addr = match N::parse_addr(&url) {
486                Ok(addr) => addr,
487                Err(err) => {
488                    sever(&self.loop_tx, connection, format!("{err:#}").into_bytes());
489                    return;
490                }
491            };
492            let (tx, rx) = mpsc::unbounded_channel();
493            let context_shm = self.shm_ctx(self.context_shm.clone());
494            let shutdown = self.side_channels.borrow().subscribe_shutdown();
495            let alive = self.side_channels.borrow().alive_token();
496            let data_rt = self.data_handle();
497            // Bind synchronously so a port-in-use failure is known now; a failed bind
498            // spawns a task that severs every serve on this url until teardown rather
499            // than respawn-retrying.
500            match self.side_channels.borrow_mut().bind(addr) {
501                Ok(listener) => {
502                    tokio::task::spawn_local(listener_task::<N>(
503                        listener,
504                        rx,
505                        self.loop_tx.clone(),
506                        shutdown,
507                        alive,
508                        context_shm,
509                        self.heartbeat.clone(),
510                        self.side_channels.clone(),
511                        data_rt,
512                    ));
513                }
514                Err(err) => {
515                    let reason = format!("quic bind failed: {err:#}").into_bytes();
516                    tokio::task::spawn_local(dead_listener_task(
517                        rx,
518                        self.loop_tx.clone(),
519                        shutdown,
520                        reason,
521                    ));
522                }
523            }
524            self.listeners.insert(url.clone(), tx);
525        }
526        let shm = self.shm_ctx(shm_client);
527        let _ = self
528            .listeners
529            .get(&url)
530            .expect("listener just inserted")
531            .send((connection, shm));
532    }
533
534    fn join(&mut self, url: String, connection: ConnectionRef, shm_client: ShmClientSlot) {
535        let addr = match N::parse_addr(&url) {
536            Ok(addr) => addr,
537            Err(err) => {
538                sever(&self.loop_tx, connection, format!("{err:#}").into_bytes());
539                return;
540            }
541        };
542        let dialer = match self.side_channels.borrow_mut().dialer(addr) {
543            Ok(dialer) => dialer,
544            Err(err) => {
545                sever(
546                    &self.loop_tx,
547                    connection,
548                    format!("quic client dialer: {err:#}").into_bytes(),
549                );
550                return;
551            }
552        };
553        let shutdown = self.side_channels.borrow().subscribe_shutdown();
554        let alive = self.side_channels.borrow().alive_token();
555        let data_rt = self.data_handle();
556        tokio::task::spawn_local(connector_task::<N>(
557            dialer,
558            addr,
559            connection,
560            self.loop_tx.clone(),
561            shutdown,
562            alive,
563            self.shm_ctx(shm_client),
564            self.connect_sem.clone(),
565            self.heartbeat.clone(),
566            self.side_channels.clone(),
567            data_rt,
568        ));
569    }
570}

572/// A connection accepted by the listener, classified by the joiner's data-stream
573/// preamble. A `Join` is paired with a serve and driven by the command loop; a
574/// `SideChannel` is a gateway-to-gateway link read directly into the home gateway's
575/// routing (the joiner only sends, so we keep just the message recv).
576enum Accepted<N: Net> {
577    Join(JoinStreams<N>),
578    SideChannel {
579        conn: N::Conn,
580        msg_recv: ConnRecv<N>,
581    },
582}

584/// The connection plus the data-stream halves of a join link, as paired by the
585/// listener's [`Matcher`] and handed to [`spawn_connection`]. The heartbeat stream is
586/// not here — it is acquired later off the establishment path.
587struct JoinStreams<N: Net> {
588    conn: N::Conn,
589    data_send: ConnSend<N>,
590    data_recv: ConnRecv<N>,
591}

593/// Fail every serve forwarded to a url whose bind failed, until teardown. The url is
594/// dead for serving; we stay alive and sever each serve rather than respawn-retrying.
595async fn dead_listener_task(
596    mut serves: mpsc::UnboundedReceiver<(ConnectionRef, ShmCtx)>,
597    loop_tx: mpsc::UnboundedSender<Command>,
598    mut shutdown: watch::Receiver<bool>,
599    reason: Vec<u8>,
600) {
601    loop {
602        tokio::select! {
603            serve = serves.recv() => match serve {
604                Some((connection, _shm)) => sever(&loop_tx, connection, reason.clone()),
605                None => return,
606            },
607            _ = shutdown.changed() => return,
608        }
609    }
610}

612/// Dispatch each accepted connection by its preamble: a `Join` is paired with the
613/// next queued serve (either may arrive first); a `SideChannel` is read straight into
614/// the context's gateway routing. Stops on teardown or when the command loop drops
615/// the serve sender. `side_channel_shm` is the *context* shm context, used only for
616/// side-channel readers (which have no owning actor); each Join connection instead
617/// uses the per-serve shm forwarded with it.
618#[expect(
619    clippy::too_many_arguments,
620    reason = "each argument is a distinct piece of listener state; bundling adds indirection without clarifying anything"
621)]
622async fn listener_task<N: Net>(
623    listener: N::Listener,
624    mut serves: mpsc::UnboundedReceiver<(ConnectionRef, ShmCtx)>,
625    loop_tx: mpsc::UnboundedSender<Command>,
626    mut shutdown: watch::Receiver<bool>,
627    alive: mpsc::UnboundedSender<()>,
628    side_channel_shm: ShmCtx,
629    heartbeat: Heartbeats,
630    side_channels: Rc<RefCell<SideChannels<N>>>,
631    data_rt: Option<Handle>,
632) {
633    // Accepting a connection, its data stream, and its preamble is a multi-step
634    // handshake; run it off the pairing loop so a slow handshake doesn't stall
635    // matching. Each classified connection lands on `accepted_rx`.
636    let (accepted_tx, mut accepted_rx) = mpsc::unbounded_channel();
637    tokio::task::spawn_local(acceptor_task::<N>(listener, accepted_tx, shutdown.clone()));
638
639    let mut matcher: Matcher<(ConnectionRef, ShmCtx), JoinStreams<N>> = Matcher::new();
640    loop {
641        tokio::select! {
642            serve = serves.recv() => {
643                let Some((connection, shm)) = serve else { return; };
644                if let Some((left, right)) = matcher.push_left((connection, shm), |l, r| (l, r)) {
645                    spawn_join(left, right, &loop_tx, &shutdown, &alive, &heartbeat, &side_channels, &data_rt);
646                }
647            }
648            accepted = accepted_rx.recv() => {
649                match accepted {
650                    None => return,
651                    Some(Accepted::Join(streams)) => {
652                        if let Some((left, right)) = matcher.push_right(streams, |l, r| (l, r)) {
653                            spawn_join(left, right, &loop_tx, &shutdown, &alive, &heartbeat, &side_channels, &data_rt);
654                        }
655                    }
656                    Some(Accepted::SideChannel { conn, msg_recv }) => {
657                        // Messages and delegated beats arrive on separate streams, so
658                        // read each in its own task. The message recv keeps the
659                        // connection alive; the heartbeat reader materializes its own
660                        // stream, which may open late (on the first beat) or never. The
661                        // message reader gets a connection clone too, so it can open the
662                        // striped data streams a large message arrives on.
663                        tokio::task::spawn_local(side_channel_reader_task::<N>(
664                            conn.clone(), msg_recv, loop_tx.clone(), side_channel_shm.clone(),
665                        ));
666                        tokio::task::spawn_local(side_channel_heartbeat_reader_task::<N>(
667                            conn, heartbeat.clone(),
668                        ));
669                    }
670                }
671            }
672            _ = shutdown.changed() => return,
673        }
674    }
675}

677/// Pair a queued serve with an accepted join and wire up the connection. Both the
678/// serve-first (`push_left`) and accept-first (`push_right`) paths land here, so a
679/// pairing spawns identically: this is the serve/acceptor side, so `log_heartbeats =
680/// true` and `dialed = false` (the peer dialed us). The data-stream halves were already
681/// accepted (`conn.stream(STREAM_DATA, …)`) and the preamble read by the acceptor.
682#[expect(
683    clippy::too_many_arguments,
684    reason = "each argument is a distinct piece of per-connection state forwarded to spawn_connection; bundling adds indirection without clarifying anything"
685)]
686fn spawn_join<N: Net>(
687    serve: (ConnectionRef, ShmCtx),
688    streams: JoinStreams<N>,
689    loop_tx: &mpsc::UnboundedSender<Command>,
690    shutdown: &watch::Receiver<bool>,
691    alive: &mpsc::UnboundedSender<()>,
692    heartbeat: &Heartbeats,
693    side_channels: &Rc<RefCell<SideChannels<N>>>,
694    data_rt: &Option<Handle>,
695) {
696    let (connection, shm) = serve;
697    let JoinStreams {
698        conn,
699        data_send,
700        data_recv,
701    } = streams;
702    spawn_connection::<N>(
703        data_send,
704        data_recv,
705        connection,
706        loop_tx.clone(),
707        shutdown.clone(),
708        alive.clone(),
709        conn,
710        shm,
711        true,
712        false,
713        heartbeat.clone(),
714        side_channels.clone(),
715        data_rt.clone(),
716    );
717}

719/// Accept connections on `listener`, accept each one's data stream (index 0), read its
720/// preamble, and forward the classified result. Each handshake runs in its own task so
721/// one slow peer doesn't block others.
722async fn acceptor_task<N: Net>(
723    listener: N::Listener,
724    accepted_tx: mpsc::UnboundedSender<Accepted<N>>,
725    mut shutdown: watch::Receiver<bool>,
726) {
727    loop {
728        tokio::select! {
729            conn = N::accept(&listener) => {
730                let Some(conn) = conn else { return; }; // listener closed
731                let accepted_tx = accepted_tx.clone();
732                tokio::task::spawn_local(async move {
733                    // The data stream (index 0) carries the preamble; accept it and
734                    // classify. The heartbeat stream (index 1) is materialized later
735                    // (by the heartbeat task, or the side-channel heartbeat reader),
736                    // so establishment/pairing never waits on the first beat.
737                    let Ok((data_send, mut data_recv)) =
738                        conn.stream(STREAM_DATA, PRIORITY_DATA).await else { return; };
739                    let accepted = match framing::read_preamble(&mut data_recv).await {
740                        Ok(Preamble::Join) => Accepted::Join(JoinStreams {
741                            conn,
742                            data_send,
743                            data_recv,
744                        }),
745                        // A side-channel is unidirectional (the joiner only sends), so
746                        // keep just the message recv and drop the send half.
747                        Ok(Preamble::SideChannel) => Accepted::SideChannel {
748                            conn,
749                            msg_recv: data_recv,
750                        },
751                        // A decode error: the peer spoke a bad dialect — drop it.
752                        Err(_) => return,
753                    };
754                    let _ = accepted_tx.send(accepted);
755                });
756            }
757            _ = shutdown.changed() => return,
758        }
759    }
760}

762/// Read routable messages off a gateway side-channel's message stream and forward
763/// each to the command loop (which resolves the owning gateway). There is no
764/// establishment: an EOF or error just ends the reader (the sending gateway
765/// reconnects when it next has something to send). `recv` keeps the connection alive;
766/// `conn` lets the striping reader open the data streams a large message arrives on.
767/// Delegated beats travel on the companion heartbeat stream (see
768/// [`side_channel_heartbeat_reader_task`]). Large messages are striped and assembled
769/// in `seq` order exactly like the connection reader ([`reader_task`]); a dead data
770/// stream just ends this reader (the sending gateway reconnects).
771async fn side_channel_reader_task<N: Net>(
772    conn: N::Conn,
773    recv: ConnRecv<N>,
774    loop_tx: mpsc::UnboundedSender<Command>,
775    shm: ShmCtx,
776) {
777    // The accepting end of a side channel — it never dialed, so `dialed = false`.
778    let part_reader = PartReader::<N>::new(conn, false, shm.mapper.clone(), shm.client_slot());
779
780    // A side channel has no establishment or failure signalling of its own: whatever
781    // ends the loop (control-stream EOF/error, a dead data stream, or a lost command
782    // loop) just ends the reader — the sending gateway reconnects when it next has
783    // something to send.
784    let _ = SideChannelReader { loop_tx }
785        .read_messages(recv, part_reader)
786        .await;
787}

789/// Delivers side-channel messages to the command loop — the [`MessageReader`] for the
790/// one-directional gateway side-channel pathway.
791struct SideChannelReader {
792    loop_tx: mpsc::UnboundedSender<Command>,
793}
794
795impl<N: Net> MessageReader<N> for SideChannelReader {
796    type Item = SideChannelMessage;
797
798    fn read_frame(
799        control: &mut ConnRecv<N>,
800        parts: &mut PartReader<N>,
801    ) -> impl std::future::Future<Output = std::io::Result<SideChannelMessage>> + Send {
802        framing::read_side_channel(control, parts)
803    }
804
805    fn on_message(&mut self, message: SideChannelMessage) -> bool {
806        self.loop_tx
807            .send(Command::SideChannelDeliver(message))
808            .is_ok()
809    }

812/// Read delegated-heartbeat beats/acks/releases off a gateway side-channel's
813/// heartbeat stream and route them *directly* into the heartbeat subsystem — they
814/// never reach ctx. Kept on its own stream so a large routed message can never delay
815/// a beat. This side is the acceptor, so awaiting the heartbeat stream (index 1) never
816/// opens it: it materializes only if the dialing peer opens it (on its first beat), and
817/// never for a channel that only carries messages, so this await never blocks the
818/// message reader.
819async fn side_channel_heartbeat_reader_task<N: Net>(conn: N::Conn, heartbeat: Heartbeats) {
820    let Ok((_send, mut recv)) = conn.stream(STREAM_HEARTBEAT, PRIORITY_HEARTBEAT).await else {
821        return; // connection closed before any heartbeat stream opened
822    };
823    loop {
824        match framing::read_side_channel_heartbeat(&mut recv).await {
825            Ok(SideChannelHeartbeat {
826                recipient,
827                from,
828                conn_id,
829                kind,
830            }) => {
831                heartbeat.deliver(&recipient, from, conn_id, kind);
832            }
833            Err(_) => return,
834        }
835    }
836}

838/// Connect to `addr`, retrying until the server binds and the data stream (index 0)
839/// opens (so a join may precede its serve), then announce ourselves as the joiner via
840/// the preamble and wire it up. This is the dialer side, so `conn.stream` opens each
841/// stream.
842#[expect(
843    clippy::too_many_arguments,
844    reason = "each argument is a distinct piece of connector state; bundling adds indirection without clarifying anything"
845)]
846async fn connector_task<N: Net>(
847    dialer: N::Dialer,
848    addr: SocketAddr,
849    connection: ConnectionRef,
850    loop_tx: mpsc::UnboundedSender<Command>,
851    mut shutdown: watch::Receiver<bool>,
852    alive: mpsc::UnboundedSender<()>,
853    shm: ShmCtx,
854    // Bounds how many connect attempts run at once across the context (see
855    // `max_concurrent_connects`). `None` ⇒ unlimited. A permit is held only for the
856    // duration of one attempt and released before backing off, so a waiting task can
857    // attempt while this one sleeps.
858    connect_sem: Option<Arc<Semaphore>>,
859    heartbeat: Heartbeats,
860    side_channels: Rc<RefCell<SideChannels<N>>>,
861    data_rt: Option<Handle>,
862) {
863    let mut retry = CONNECT_RETRY_MIN;
864    // Retry until both the connect and the *first* stream open succeed. For quic the
865    // handshake is in `connect`; for tcp `connect` is cheap and the data stream (index
866    // 0) is where the socket is actually dialed — so either failing means the server is
867    // not up yet (a join may precede its serve), and we back off and retry.
868    let (mut data_send, data_recv, conn) = loop {
869        if *shutdown.borrow() {
870            return;
871        }
872        // Take a connect-attempt permit before doing any handshake work, so a root
873        // joining tens of thousands of peers drives at most N handshakes at once. Held
874        // across the data-stream dial (the throttled work) and released before backoff.
875        let permit = match &connect_sem {
876            Some(sem) => Some(
877                Arc::clone(sem)
878                    .acquire_owned()
879                    .await
880                    .expect("connect semaphore never closed"),
881            ),
882            None => None,
883        };
884        if let Some(conn) = N::connect(&dialer, addr).await {
885            if let Ok((data_send, data_recv)) = conn.stream(STREAM_DATA, PRIORITY_DATA).await {
886                drop(permit); // connection is up; free the attempt slot before handing off
887                break (data_send, data_recv, conn);
888            }
889        }
890        // Attempt failed; release the slot so another task can try while we back off.
891        drop(permit);
892        tokio::select! {
893            _ = tokio::time::sleep(retry) => {}
894            _ = shutdown.changed() => return,
895        }
896        retry = (retry * 2).min(CONNECT_RETRY_MAX);
897    };
898    // Tell the acceptor this is an ordinary join (not a side-channel) before the command
899    // loop drives establishment over the stream.
900    if framing::write_preamble(&mut data_send, Preamble::Join)
901        .await
902        .is_err()
903    {
904        sever(&loop_tx, connection, b"quic open stream failed".to_vec());
905        return;
906    }
907    // Joiner side (the root has tens of thousands of writers): do not log heartbeat sends
908    // here even under debug — it would flood. `dialed = true`: this is one half of the
909    // delegation guard (only a link the parent dialed may be delegated).
910    spawn_connection::<N>(
911        data_send,
912        data_recv,
913        connection,
914        loop_tx,
915        shutdown,
916        alive,
917        conn,
918        shm,
919        false,
920        true,
921        heartbeat,
922        side_channels,
923        data_rt,
924    );
925}

927/// Drive a gateway side-channel writer: drain queued outbound items — routable
928/// [`SideChannelMessage`]s and transport-internal heartbeat beats/acks alike —
929/// writing each as a frame to the remote gateway. The connection is established
930/// lazily on the first item (retrying until the gateway is live) and reconnected if
931/// it drops. The **heartbeat stream is opened only when the first beat is dequeued**,
932/// so a message-only channel never establishes it. An item in flight when the
933/// connection drops may be lost; the channel is best-effort by design.
934async fn side_channel_writer_task<N: Net>(
935    dialer: N::Dialer,
936    addr: SocketAddr,
937    mut rx: mpsc::UnboundedReceiver<SideChannelOut>,
938    mut shutdown: watch::Receiver<bool>,
939    _alive: mpsc::UnboundedSender<()>,
940) {
941    // The current connection, if up: the connection handle (held to keep it alive and
942    // to open the heartbeat stream lazily), the message send stream, and the
943    // heartbeat send stream once a beat has been sent. `None` means we must
944    // (re)connect before the next write.
945    let mut live: Option<LiveSideChannel<N>> = None;
946    // The part writer for the current connection's data streams, rebuilt on each
947    // (re)connect. A side channel is one-directional, so it only ever writes; large
948    // messages are striped, small ones stay inline (a disabled writer writes all parts
949    // inline).
950    let mut part_writer: Option<PartWriter<N>> = None;
951    loop {
952        let item = tokio::select! {
953            item = rx.recv() => match item {
954                Some(item) => item,
955                None => break, // sender dropped (gateway gone)
956            },
957            _ = shutdown.changed() => break,
958        };
959        if live.is_none() {
960            // `connect_side_channel` opens the message stream (index 0) and writes the
961            // preamble as part of its retry, so it returns a ready-to-write send half.
962            // The heartbeat stream (index 1) is opened lazily below, so a message-only
963            // channel never opens it.
964            let Some((conn, msg_send)) =
965                connect_side_channel::<N>(&dialer, addr, &mut shutdown).await
966            else {
967                break; // shutting down before the gateway came up
968            };
969            // The dialing end of a side channel — `dialed = true`.
970            part_writer = Some(PartWriter::<N>::new(conn.clone(), true));
971            live = Some((conn, msg_send, None));
972        }
973        let (conn, msg_send, hb_send) = live.as_mut().expect("connected");
974        let active_writer = part_writer
975            .as_mut()
976            .expect("part writer set with live connection");
977        let wrote = match item {
978            SideChannelOut::Message(message) => {
979                framing::write_side_channel(msg_send, message, active_writer).await
980            }
981            SideChannelOut::Heartbeat {
982                recipient,
983                from,
984                conn_id,
985                kind,
986            } => {
987                // Open the heartbeat stream on the first beat only. It rides a
988                // higher-priority stream (index > the message stream), so a beat jumps
989                // ahead of queued message bytes under a full congestion window.
990                if hb_send.is_none() {
991                    match conn.stream(STREAM_HEARTBEAT, PRIORITY_HEARTBEAT).await {
992                        Ok((send, _recv)) => *hb_send = Some(send),
993                        Err(_) => {
994                            live = None; // reconnect on the next item
995                            continue;
996                        }
997                    }
998                }
999                let hb_send = hb_send.as_mut().expect("heartbeat stream opened");
1000                framing::write_side_channel_heartbeat(hb_send, recipient, from, conn_id, kind).await
1001            }
1002        };
1003        if wrote.is_err() {
1004            live = None; // dropped; reconnect on the next item
1005            part_writer = None; // its data streams belong to the dropped connection
1006        }
1007    }
1008    if let Some((_conn, mut msg_send, hb_send)) = live {
1009        let _ = msg_send.shutdown().await;
1010        if let Some(mut hb_send) = hb_send {
1011            let _ = hb_send.shutdown().await;
1012        }
1013    }
1014}

1016/// Connect a side-channel to `addr` and open its message stream, retrying with backoff
1017/// until the gateway binds and the handshake succeeds, or until teardown (then `None`).
1018/// Returns the connection and the message send half, with the `SideChannel` preamble
1019/// already written. Folding the first stream open into the retry gives tcp — whose
1020/// `connect` is cheap and whose reachability is proven only when the socket is dialed —
1021/// proper backoff. Mirrors the join connector so a side-channel may be opened before its
1022/// target gateway is live.
1023async fn connect_side_channel<N: Net>(
1024    dialer: &N::Dialer,
1025    addr: SocketAddr,
1026    shutdown: &mut watch::Receiver<bool>,
1027) -> Option<(N::Conn, ConnSend<N>)> {
1028    let mut retry = CONNECT_RETRY_MIN;
1029    loop {
1030        if *shutdown.borrow() {
1031            return None;
1032        }
1033        if let Some(conn) = N::connect(dialer, addr).await {
1034            if let Ok((mut msg_send, _msg_recv)) = conn.stream(STREAM_DATA, PRIORITY_DATA).await {
1035                if framing::write_preamble(&mut msg_send, Preamble::SideChannel)
1036                    .await
1037                    .is_ok()
1038                {
1039                    return Some((conn, msg_send));
1040                }
1041            }
1042        }
1043        tokio::select! {
1044            _ = tokio::time::sleep(retry) => {}
1045            _ = shutdown.changed() => return None,
1046        }
1047        retry = (retry * 2).min(CONNECT_RETRY_MAX);
1048    }
1049}

1051/// Wire up an established connection: build its [`StreamConnectionTransport`] over the
1052/// data stream, announce it to the command loop, and spawn three tasks — a data
1053/// writer (commands → frames) and data reader (frames → `ConnectionAction`) on the
1054/// data stream, plus a heartbeat management task that materializes the heartbeat
1055/// stream and runs a beat writer + reader on it. Splitting the streams keeps a large
1056/// message from delaying a beat. Crucially, only the two data-stream halves are
1057/// passed in: the heartbeat stream is materialized *inside* the management task, so
1058/// neither establishment nor serve-pairing ever waits on the first heartbeat.
1059#[expect(
1060    clippy::too_many_arguments,
1061    reason = "each argument is a distinct piece of per-connection state handed off to the writer or reader; bundling them adds indirection without clarifying anything"
1062)]
1063fn spawn_connection<N: Net>(
1064    data_send: ConnSend<N>,
1065    data_recv: ConnRecv<N>,
1066    connection: ConnectionRef,
1067    loop_tx: mpsc::UnboundedSender<Command>,
1068    shutdown: watch::Receiver<bool>,
1069    alive: mpsc::UnboundedSender<()>,
1070    conn: N::Conn,
1071    shm: ShmCtx,
1072    // Forwarded to the heartbeat writer: log heartbeat sends (gated to the
1073    // serve/acceptor side by the call sites, and further gated by MM_QUIC_DEBUG).
1074    log_heartbeats: bool,
1075    // Whether *we* dialed this connection (join). One half of the delegation guard
1076    // (only a link the parent dialed may be delegated).
1077    dialed: bool,
1078    heartbeat: Heartbeats,
1079    side_channels: Rc<RefCell<SideChannels<N>>>,
1080    // When `Some`, the data writer/reader run on this multi-threaded runtime instead
1081    // of the command-loop thread (see `data_runtime`). The heartbeat stream, side
1082    // channels, and pairing always stay on the command-loop thread.
1083    data_rt: Option<Handle>,
1084) {
1085    let (writer_tx, writer_rx) = mpsc::unbounded_channel();
1086    // Reader → writer signal: the peer responded to our shutdown, so the writer may
1087    // close. Unbounded but only ever carries a single `()`.
1088    let (peer_responded_tx, peer_responded_rx) = mpsc::unbounded_channel();
1089    // heartbeat coroutine → heartbeat writer: beats to write out.
1090    let (beats_tx, beats_rx) = mpsc::unbounded_channel();
1091    let transport = Box::new(StreamConnectionTransport { tx: writer_tx });
1092    let _ = loop_tx.send(Command::TransportConnected {
1093        connection,
1094        transport,
1095    });
1096    // The heartbeat coroutine sends its side-channel beats through this closure — it
1097    // never sees the transport's side-channel types, keeping `heartbeat`
1098    // decoupled from the transport. The closure reuses the existing gateway side
1099    // channels.
1100    let send_beat =
1101        move |recipient: Vec<u8>, from: Vec<u8>, conn_id: ConnectionId, kind: BeatKind| {
1102            side_channels
1103                .borrow_mut()
1104                .send_heartbeat(recipient, from, conn_id, kind);
1105        };
1106    // Spawn the coroutine; the returned sender is how the readers/writer feed it
1107    // inbound beats, Establish snoops, and reader-closed.
1108    let hb_event_tx = heartbeat.spawn(connection, dialed, beats_tx, send_beat, loop_tx.clone());
1109    // Data stream: command writer and frame reader. Built here (moving in their state)
1110    // then spawned either on the data runtime (parallel across connections) or, by
1111    // default, on the command-loop thread. Only these two coroutines may move off the
1112    // command-loop thread — their captured state is `Send` (the `Send`-bounded stream
1113    // halves, channel handles, `Copy` connection ref, and `Arc`-backed shm); the
1114    // heartbeat/side-channel machinery below stays local.
1115    // The writer and reader each own a part writer/reader over this side's data streams
1116    // (outbound for the writer, inbound for the reader); `dialed` lets each pick its
1117    // stream indices. Both hold a clone of the connection handle to open those streams.
1118    let writer = writer_task::<N>(
1119        data_send,
1120        writer_rx,
1121        shutdown.clone(),
1122        alive,
1123        peer_responded_rx,
1124        hb_event_tx.clone(),
1125        PartWriter::new(conn.clone(), dialed),
1126    );
1127    let reader = reader_task::<N>(
1128        data_recv,
1129        connection,
1130        loop_tx,
1131        peer_responded_tx,
1132        hb_event_tx.clone(),
1133        PartReader::new(conn.clone(), dialed, shm.mapper.clone(), shm.client_slot()),
1134    );
1135    // The data coroutines move to the data runtime for cross-connection parallelism; the
1136    // per-stream shard tasks they spawn are `tokio::spawn`ed there too (the halves are
1137    // `Send`), so striping composes with the data runtime.
1138    match &data_rt {
1139        Some(handle) => {
1140            handle.spawn(writer);
1141            handle.spawn(reader);
1142        }
1143        None => {
1144            tokio::task::spawn_local(writer);
1145            tokio::task::spawn_local(reader);
1146        }
1147    }
1148    // Heartbeat stream (index 1): materialized off the establishment path, then a beat
1149    // writer and reader run on it. Direction is a property of the connection now — the
1150    // dialer opens it, the acceptor awaits it (see [`crate::net`]) — so no per-call flag.
1151    // The dialer opens it proactively here (a parent/child link always heartbeats), and
1152    // its index-prefix write is what resolves the acceptor's await; whoever is the
1153    // heartbeat **Child** then beats first over the established stream.
1154    tokio::task::spawn_local(heartbeat_stream_task::<N>(
1155        conn,
1156        beats_rx,
1157        hb_event_tx,
1158        shutdown,
1159        log_heartbeats,
1160    ));
1161}

1163/// Materialize the connection's heartbeat stream (index 1) and run the beat writer +
1164/// reader on it. Materialization happens here, in a task spawned *after* the connection
1165/// is established, so the first beat never delays establishment; the dialer opens the
1166/// stream and the acceptor awaits it. If the connection dies before the stream comes up,
1167/// this just ends.
1168async fn heartbeat_stream_task<N: Net>(
1169    conn: N::Conn,
1170    beats_rx: mpsc::UnboundedReceiver<Heartbeat>,
1171    hb_events: mpsc::UnboundedSender<HeartbeatEvent>,
1172    shutdown: watch::Receiver<bool>,
1173    log_heartbeats: bool,
1174) {
1175    let Ok((hb_send, hb_recv)) = conn.stream(STREAM_HEARTBEAT, PRIORITY_HEARTBEAT).await else {
1176        return; // connection died before the heartbeat stream came up
1177    };
1178    // Reader in its own task; writer runs inline (it owns `conn` to keep it alive and
1179    // to name it in the send log).
1180    tokio::task::spawn_local(heartbeat_reader_task::<N>(hb_recv, hb_events));
1181    heartbeat_writer_task::<N>(hb_send, beats_rx, shutdown, conn, log_heartbeats).await;
1182}

1184/// Transport for one end of a stream: `send` hands a command to the writer task.
1185/// Dropping it ends the writer, which finishes the stream — the peer's reader then
1186/// sees EOF.
1187struct StreamConnectionTransport {
1188    tx: mpsc::UnboundedSender<ConnectionCommand>,
1189}
1190
1191impl ConnectionTransport for StreamConnectionTransport {
1192    fn send(&self, action: ConnectionCommand) -> bool {
1193        self.tx.send(action).is_ok()
1194    }
1195}
1196
1197/// Write each queued command as a frame onto the data stream. Beats travel on the
1198/// separate heartbeat stream (see [`heartbeat_writer_task`]), so this task only
1199/// handles data/control frames. Snoops its own outgoing `Establish` and forwards the
1200/// local ident to the heartbeat task. On teardown, drains queued frames first, then
1201/// finishes the stream. The send half keeps the connection alive for the writer's
1202/// lifetime — including the graceful-shutdown wait below — so dropping it (together
1203/// with the reader's and heartbeat's halves) closes the connection.
1204async fn writer_task<N: Net>(
1205    mut send: ConnSend<N>,
1206    mut rx: mpsc::UnboundedReceiver<ConnectionCommand>,
1207    mut shutdown: watch::Receiver<bool>,
1208    _alive: mpsc::UnboundedSender<()>,
1209    // Signalled by this connection's reader once the peer responds to our shutdown
1210    // (its own Severed frame, or the connection ending) — our cue that the peer
1211    // received the shutdown notice and we can close.
1212    mut peer_responded: mpsc::UnboundedReceiver<()>,
1213    // To forward our own outgoing Establish's ident to the heartbeat task.
1214    hb_events: mpsc::UnboundedSender<HeartbeatEvent>,
1215    // Writes each command's parts inline or striped across this side's data streams
1216    // (framing routes actor messages through it).
1217    mut part_writer: PartWriter<N>,
1218) {
1219    let mut graceful = false;
1220    loop {
1221        tokio::select! {
1222            command = rx.recv() => {
1223                let Some(command) = command else {
1224                    break; // transport dropped
1225                };
1226                // Snoop our own Establish so the heartbeat task learns our ident (its
1227                // side-channel address / delegate-eligibility key) without any new
1228                // threading through ctx.
1229                if let ConnectionCommand::Establish { ident: Some(ident), .. } = &command {
1230                    let _ = hb_events.send(HeartbeatEvent::EstablishLocal {
1231                        local_ident: ident.clone(),
1232                    });
1233                }
1234                if framing::write_command(&mut send, command, &mut part_writer).await.is_err() {
1235                    return;
1236                }
1237            }
1238            _ = shutdown.changed() => {
1239                // Graceful teardown. Flush whatever is already queued, then send an
1240                // explicit shutdown notice. A Severed frame is stream data, which the
1241                // transport retransmits until delivered, so the peer learns of
1242                // teardown directly rather than by inferring a dropped socket.
1243                while let Ok(command) = rx.try_recv() {
1244                    if framing::write_command(&mut send, command, &mut part_writer).await.is_err() {
1245                        return;
1246                    }
1247                }
1248                let _ = framing::write_command(
1249                    &mut send,
1250                    ConnectionCommand::Severed {
1251                        reason: b"context shutdown".to_vec(),
1252                    },
1253                    &mut part_writer,
1254                )
1255                .await;
1256                graceful = true;
1257                break;
1258            }
1259        }
1260    }
1261    // Finish the stream so the peer's reader sees EOF promptly (the fast path).
1262    let _ = send.shutdown().await;
1263    // On graceful shutdown, hold the connection (via `send`) open until the peer
1264    // responds to our shutdown notice — keeping it open lets the transport retransmit
1265    // the notice if the first packet was lost, and we close as soon as the peer's
1266    // reply arrives rather than after a fixed delay. Bounded so a peer that never
1267    // replies (e.g. already dead) can't hang teardown.
1268    if graceful {
1269        let _ = tokio::time::timeout(shutdown_ack_timeout(), peer_responded.recv()).await;
1270    }
1271    // `send` is dropped here, releasing this task's hold on the connection.
1272    drop(send);
1273}
1274
1275/// Write each beat this connection's heartbeat task produces onto the dedicated
1276/// heartbeat stream. The stream's send priority (set when it was materialized) packs
1277/// beats ahead of queued data-stream bytes under a full congestion window. Holds the
1278/// connection so it stays alive while beats flow, and finishes the stream on
1279/// shutdown. A write error just ends the task — the data reader is the one that
1280/// severs on connection loss.
1281async fn heartbeat_writer_task<N: Net>(
1282    mut send: ConnSend<N>,
1283    mut beats: mpsc::UnboundedReceiver<Heartbeat>,
1284    mut shutdown: watch::Receiver<bool>,
1285    // Held to keep the connection alive while beats flow and to name it in the log.
1286    conn: N::Conn,
1287    // When set (acceptor/serve side only, so the joiner root's tens of thousands of
1288    // connections don't flood), and MM_QUIC_DEBUG is on, log each heartbeat sent — so
1289    // a connection the peer later reaps can be proven to have kept sending.
1290    log_heartbeats: bool,
1291) {
1292    let log_heartbeats = log_heartbeats && crate::ctx::connection_debug();
1293    let mut heartbeats_sent: u64 = 0;
1294    loop {
1295        tokio::select! {
1296            beat = beats.recv() => {
1297                let Some(heartbeat) = beat else {
1298                    break; // heartbeat task gone
1299                };
1300                if framing::write_heartbeat(&mut send, heartbeat).await.is_err() {
1301                    return;
1302                }
1303                // Flushed to the stream; log it from the *sender* so loss vs. a
1304                // stalled sender can be told apart at the far (receiving) end.
1305                if log_heartbeats {
1306                    heartbeats_sent += 1;
1307                    eprintln!(
1308                        "{} MM_HB pid={} sent={} on {:?}",
1309                        crate::ctx::wall_clock_hms(),
1310                        std::process::id(),
1311                        heartbeats_sent,
1312                        conn,
1313                    );
1314                }
1315            }
1316            _ = shutdown.changed() => break,
1317        }
1318    }
1319    let _ = send.shutdown().await;
1320    // `conn` is dropped here, releasing this task's hold on the connection.
1321    drop(conn);
1322}

1324/// Read the peer's beats off the dedicated heartbeat stream and forward each to this
1325/// connection's heartbeat task. On EOF/error it ends quietly: liveness loss is caught
1326/// either by the heartbeat task's own beat timeout or by the data reader's `Severed`
1327/// on connection close.
1328async fn heartbeat_reader_task<N: Net>(
1329    mut recv: ConnRecv<N>,
1330    hb_events: mpsc::UnboundedSender<HeartbeatEvent>,
1331) {
1332    loop {
1333        match framing::read_heartbeat(&mut recv).await {
1334            Ok(heartbeat) => {
1335                if hb_events
1336                    .send(HeartbeatEvent::ReceivedHeartbeat(heartbeat))
1337                    .is_err()
1338                {
1339                    return;
1340                }
1341            }
1342            Err(_) => return,
1343        }
1344    }
1345}

1347/// Decode each frame off the data stream and forward it to the command loop. Beats
1348/// arrive on the separate heartbeat stream (see [`heartbeat_reader_task`]), so this
1349/// reader is pure data plumbing. An error/EOF is a hard transport close: it emits
1350/// `Severed` to the loop and `ReaderClosed` to the heartbeat task — this is the path
1351/// that detects an unclean peer loss even for a link whose beats are silent (a
1352/// delegated child, or a live-but-idle connection).
1353async fn reader_task<N: Net>(
1354    recv: ConnRecv<N>,
1355    connection: ConnectionRef,
1356    loop_tx: mpsc::UnboundedSender<Command>,
1357    // Signals this connection's writer that the peer has responded to our shutdown
1358    // (it sent its own Severed, or the connection ending) so the writer can close.
1359    peer_responded: mpsc::UnboundedSender<()>,
1360    // Forwards inbound heartbeats and liveness/close signals to the heartbeat task.
1361    hb_events: mpsc::UnboundedSender<HeartbeatEvent>,
1362    // Reads/assembles each message's parts off this side's inbound data streams.
1363    part_reader: PartReader<N>,
1364) {
1365    // Per-connection diagnostics, folded into the failure reason under MM_QUIC_DEBUG:
1366    // did we ever read the peer's Establish, and how many commands arrived? The reader
1367    // tracks them per delivered message; only formatted when debug is on.
1368    let debug = crate::ctx::connection_debug();
1369    let start = std::time::Instant::now();
1370
1371    // Deliver each in-order message to the command loop (see [`ConnectionReader`]).
1372    let mut reader = ConnectionReader {
1373        connection,
1374        loop_tx: loop_tx.clone(),
1375        peer_responded: peer_responded.clone(),
1376        hb_events: hb_events.clone(),
1377        established: false,
1378        commands: 0,
1379    };
1380    let stop = reader.read_messages(recv, part_reader).await;
1381
1382    // The reader is done: tell the writer (so a graceful-shutdown wait can end) and the
1383    // heartbeat task (its backstop for an unclean loss), then sever with a reason —
1384    // except when the command loop itself is already gone.
1385    let _ = peer_responded.send(());
1386    let _ = hb_events.send(HeartbeatEvent::ReaderClosed);
1387    let reason = match stop {
1388        ReadStop::Delivered => return, // command loop gone; nothing to sever to
1389        ReadStop::DataStreamDied => b"quic data stream closed".to_vec(),
1390        ReadStop::Ended(_) if !debug => b"quic connection closed".to_vec(),
1391        ReadStop::Ended(err) => {
1392            // Walk the io::Error source chain to recover the concrete underlying error —
1393            // the top-level Display is often just a bare "connection lost", dropping the
1394            // real cause ("timed out" vs "reset by peer" vs "closed by peer").
1395            let mut detail = err.to_string();
1396            let mut src = std::error::Error::source(&err);
1397            while let Some(e) = src {
1398                detail.push_str(": ");
1399                detail.push_str(&e.to_string());
1400                src = e.source();
1401            }
1402            format!(
1403                "quic connection closed: {detail} (established={}, commands={}, age={:.1}s)",
1404                reader.established,
1405                reader.commands,
1406                start.elapsed().as_secs_f64(),
1407            )
1408            .into_bytes()
1409        }
1410    };
1411    sever(&loop_tx, connection, reason);
1412}

1414/// Delivers connection messages to the command loop — the [`MessageReader`] for the
1415/// join/serve pathway. Snoops the peer's `Establish` (forwarding its ident to the
1416/// heartbeat task) and `Severed` (waking the writer to close), and counts what it
1417/// delivered for the failure diagnostics.
1418struct ConnectionReader {
1419    connection: ConnectionRef,
1420    loop_tx: mpsc::UnboundedSender<Command>,
1421    peer_responded: mpsc::UnboundedSender<()>,
1422    hb_events: mpsc::UnboundedSender<HeartbeatEvent>,
1423    established: bool,
1424    commands: u64,
1425}
1426
1427impl<N: Net> MessageReader<N> for ConnectionReader {
1428    type Item = ConnectionCommand;
1429
1430    fn read_frame(
1431        control: &mut ConnRecv<N>,
1432        parts: &mut PartReader<N>,
1433    ) -> impl std::future::Future<Output = std::io::Result<ConnectionCommand>> + Send {
1434        framing::read_frame(control, parts)
1435    }
1436
1437    fn on_message(&mut self, action: ConnectionCommand) -> bool {
1438        self.commands += 1;
1439        match &action {
1440            // The peer's Establish is how we learn its identity — forward it to the
1441            // heartbeat task (one-shot, at setup).
1442            ConnectionCommand::Establish { ident, .. } => {
1443                self.established = true;
1444                if let Some(ident) = ident {
1445                    let _ = self.hb_events.send(HeartbeatEvent::EstablishPeer {
1446                        peer_ident: ident.clone(),
1447                    });
1448                }
1449            }
1450            // A Severed is the peer's response to our shutdown — wake the writer.
1451            ConnectionCommand::Severed { .. } => {
1452                let _ = self.peer_responded.send(());
1453            }
1454            _ => {}
1455        }
1456        self.loop_tx
1457            .send(Command::ConnectionAction {
1458                connection: self.connection,
1459                action,
1460            })
1461            .is_ok()
1462    }
1463}
